   366→function seedHistory(mean: number, sd: number, n: number): number[] {
   367→  const out: number[] = [];
   368→  let last = mean;
   369→  for (let i = 0; i < n; i++) {
   370→    // simple mean-reverting walk
   371→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   372→    last = last + (mean - last) * 0.1 + shock;
   373→    out.push(Math.max(0.01, last));
   374→  }
   375→  return out;
   376→}
   377→
   378→// --- §14.1 Constitutional Separation helpers ---------------------------------
   379→// Maintain the §14.1 invariant: paxg = indexPaxg + reservePaxg (and same for
   380→// xaut). Index gold is LOCKED (only changes via commitIndexGold); reserve gold
   381→// is the buffer MARP rebalances. These two helpers are the only sanctioned
   382→// ways to keep the splits consistent after any mutation to total gold:
   383→
   384→/** Re-derive reserve gold from total − index (call after the legacy path or
   385→ *  concentration optimizer mutates `s.paxg` / `s.xaut` directly). Index gold
   386→ *  stays locked. */
   387→export function syncReserveFromTotal(s: ReserveState): void {
   388→  s.reservePaxg = Math.max(0, s.paxg - s.indexPaxg);
   389→  s.reserveXaut = Math.max(0, s.xaut - s.indexXaut);
   390→}
   391→
   392→/** Re-derive total gold from index + reserve (call after the MARP path mutates
   393→ *  `s.reservePaxg` / `s.reserveXaut`). Index gold stays locked. */
   394→export function syncTotalFromReserve(s: ReserveState): void {
   395→  s.paxg = s.indexPaxg + Math.max(0, s.reservePaxg);
   396→  s.xaut = s.indexXaut + Math.max(0, s.reserveXaut);
   397→}

   399→/** §14.1 commitIndexGold — the keeper-role equivalent in the TS engine.
   400→ *  Sets the index gold holdings (PAXG + XAUT) that back the Strategic Prior
   401→ *  Gold weight (26%). Only this function may move gold between the index
   402→ *  vault and the reserve buffer. Recomputes the totals via the §14.1 invariant
   403→ *  (reserve = total − index). Emits an IndexGoldCommitted event-equivalent
   404→ *  (console log — production should emit an on-chain event). */
   405→export function commitIndexGold(
   406→  s: ReserveState,
   407→  indexPaxg: number,
   408→  indexXaut: number,
   409→): { indexPaxg: number; indexXaut: number; reservePaxg: number; reserveXaut: number; totalPaxg: number; totalXaut: number } {
   410→  const safeIndexPaxg = Math.max(0, indexPaxg);
   411→  const safeIndexXaut = Math.max(0, indexXaut);
   412→  s.indexPaxg = safeIndexPaxg;
   413→  s.indexXaut = safeIndexXaut;
   414→  // Reserve = max(0, total − index) — never let reserve go negative even if
   415→  // the keeper over-commits (log a warning in that case).
   416→  if (s.paxg < safeIndexPaxg) {
   417→    console.warn(`tq-engine] commitIndexGold: indexPaxg ${safeIndexPaxg.toFixed(4)} > total paxg ${s.paxg.toFixed(4)} — reserve went to 0; total bumped to match index.`);
   418→    s.paxg = safeIndexPaxg;
   419→  }
   420→  if (s.xaut < safeIndexXaut) {
   421→    console.warn(`tq-engine] commitIndexGold: indexXaut ${safeIndexXaut.toFixed(4)} > total xaut ${s.xaut.toFixed(4)} — reserve went to 0; total bumped to match index.`);
   422→    s.xaut = safeIndexXaut;
   423→  }
   424→  syncReserveFromTotal(s);
   425→  s.updatedAt = Date.now();
   426→  console.log(`tq-engine] IndexGoldCommitted: indexPaxg=${s.indexPaxg.toFixed(4)}, indexXaut=${s.indexXaut.toFixed(4)}, reservePaxg=${s.reservePaxg.toFixed(4)}, reserveXaut=${s.reserveXaut.toFixed(4)}`);
   427→  return {
   428→    indexPaxg: s.indexPaxg,
   429→    indexXaut: s.indexXaut,
   430→    reservePaxg: s.reservePaxg,

   437→// --- §2 GFB Index (v1.0 — 7-component chain-linked) -------------------------
   438→// GFB_t = Σ_i W^Prior_i × P_{i,t}  (then normalised by GFB_BASE_DENOMINATOR so
   439→// GFB = 1.00 exactly at the base date). The 7 components are the Strategic
   440→// Prior: USD, EUR, JPY, GBP, CNY, CHF, Gold. Gold is now a first-class index
   441→// component (P_Gold = XAU_USD).
   442→//
   443→// DEPRECATED (P0-FIX-1): this is the LEGACY fixed-base Laspeyres index. It
   444→// weights each component by its raw USD notional share (gold contributes
   445→// 99.9% of the index by notional despite a 26% Strategic Prior weight), which
   446→// creates a structural short-gold exposure that crashed the §23 validation
   447→// program (S5: 0% survival, RR crashed to 0.83 in 100% of gold +50% runs).
   448→//
   449→// The CANONICAL chain-linked index lives in `src/lib/mtq/chain-index.ts`. The
   450→// engine's `computeSnapshot` now uses `getMTQPrice(s.chainIndex)` (the
   451→// chain-linked form) for both `gfbIndex` and `mtqPrice`. This legacy
   452→// `computeGfbIndex` is RETAINED for backward compatibility with the
   453→// standalone audit-stress.ts simulations (which need a pure function of fx,
   454→// not the persisted chain index state).
   455→export function computeGfbIndex(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): number {
   456→  // Raw USD value of the basket at time t. USD is the unit of account (price 1.0);
   457→  // every other component contributes its USD-equivalent via its FX rate.
   458→  const numerator =
   459→    STRATEGIC_PRIOR.USD  * 1.0 +
   460→    STRATEGIC_PRIOR.EUR  * fx.EUR_USD +
   461→    STRATEGIC_PRIOR.JPY  * fx.JPY_USD +
   462→    STRATEGIC_PRIOR.GBP  * fx.GBP_USD +
   463→    STRATEGIC_PRIOR.CNY  * fx.CNY_USD +
   464→    STRATEGIC_PRIOR.CHF  * fx.CHF_USD +
   465→    STRATEGIC_PRIOR.Gold * fx.XAU_USD;
   466→  // Normalised: GFB_t = numerator / GFB_BASE_DENOMINATOR  (=1.0 at base date).
   467→  // GFB_BASE_DENOMINATOR is the v1.0 chain-linked denominator (includes Gold
   468→  // and CHF at base fixings — see blueprint.ts).
   469→  return numerator / GFB_BASE_DENOMINATOR;
   470→}

   476→export function computeMtqPrice(gfb: number): number {

   519→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   520→// USD value of each holding (gross), then apply haircuts for NAV.
   521→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   522→  const usdGross = (s.usdc + s.usdp + s.usdt) * 1.0; // USDC + USDP + USDT combined
   523→  const eurGross = s.eurc * fx.EUR_USD;
   524→  const gbpGross = s.gbp * fx.GBP_USD;
   525→  const jpyGross = s.jpy * fx.JPY_USD;
   526→  const cnyGross = s.cny * fx.CNY_USD;
   527→  const chfGross = s.chf * fx.CHF_USD; // NEW v1.0 — CHF is a first-class reserve component
   528→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross + chfGross;
   529→
   530→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   531→  const goldRef = fx.XAU_USD;
   532→  const goldExec = fx.XAU_USD * 0.995;
   533→  const goldPrice = Math.min(goldRef, goldExec);
   534→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   535→
   536→  // Net values (after haircuts)
   537→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   538→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   539→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   540→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   541→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   542→  const chfNet = chfGross * (1 - HAIRCUTS.CHF); // NEW v1.0
   543→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   544→
   545→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet + goldNet;
   546→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet;
   547→
   548→  // Per-issuer breakdown (for concentration visualization)
   549→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   550→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   551→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   552→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);
   553→  const xautUsd = s.xaut * goldPrice * (1 - HAIRCUTS.XAU);
   554→
   555→  // §14.1 — index vs reserve gold split (USD net of haircut).
   556→  // Index gold is locked (backs the 26% Strategic Prior Gold weight); reserve
   557→  // gold is the buffer MARP rebalances. The total gold (= index + reserve) is
   558→  // what serves the index weight for MASE/MARP observed-weight purposes.
   559→  const indexPaxgUsd = s.indexPaxg * goldPrice * (1 - HAIRCUTS.XAU);
   560→  const indexXautUsd = s.indexXaut * goldPrice * (1 - HAIRCUTS.XAU);
   561→  const reservePaxgUsd = (s.reservePaxg >= 0 ? s.reservePaxg : Math.max(0, s.paxg - s.indexPaxg)) * goldPrice * (1 - HAIRCUTS.XAU);
   562→  const reserveXautUsd = (s.reserveXaut >= 0 ? s.reserveXaut : Math.max(0, s.xaut - s.indexXaut)) * goldPrice * (1 - HAIRCUTS.XAU);
   563→  const indexGoldNet = indexPaxgUsd + indexXautUsd;
   564→  const reserveGoldNet = reservePaxgUsd + reserveXautUsd;
   565→
   566→  return {
   567→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, chfGross, fiatGross,
   568→    goldGross, goldPrice,
   569→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, chfNet, goldNet,
   570→    nav, fiatNet,
   571→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   572→    // §14.1 gold split
   573→    indexPaxgUsd, indexXautUsd, reservePaxgUsd, reserveXautUsd,
   574→    indexGoldNet, reserveGoldNet,
   575→  };
   576→}

   591→export function computeLcr(s: ReserveState, vals: { fiatNet: number }, price: number): number {
   592→  const liquidAssets = vals.fiatNet; // stablecoins = 100% liquid; gold = 0%
