    73→              <tr>
    74→                {headers.map((h, i) => (
    75→                  <th
    76→                    key={i}

    82→              </tr>

    89→                >

    91→                    <td
    92→                      key={ci}

   110→  if (severity === "fixed") {
   111→    return (
   112→      <Pill tone="emerald">
   113→        <GlowDot color="emerald" size="h-1.5 w-1.5" />
   114→        fixed
   115→      </Pill>
   116→    );
   117→  }
   118→  if (severity === "outstanding") {
   119→    return (
   120→      <Pill tone="amber">
   121→        <GlowDot color="amber" size="h-1.5 w-1.5" />
   122→        outstanding
   123→      </Pill>
   124→    );
   125→  }
   126→  return (
   127→    <Pill tone="muted">
   128→      <GlowDot color="gold" size="h-1.5 w-1.5" />
   129→      informational
   130→    </Pill>
   131→  );
   132→}
