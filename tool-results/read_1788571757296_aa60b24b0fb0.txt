   192→function seedHistory(mean: number, sd: number, n: number): number[] {
   193→  const out: number[] = [];
   194→  let last = mean;
   195→  for (let i = 0; i < n; i++) {
   196→    // simple mean-reverting walk
   197→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   198→    last = last + (mean - last) * 0.1 + shock;
   199→    out.push(Math.max(0.01, last));
   200→  }
   201→  return out;
   202→}

   227→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   228→// USD value of each holding (gross), then apply haircuts for NAV.
   229→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   230→  const usdGross = s.usdc * 1.0;
   231→  const eurGross = s.eurc * fx.EUR_USD;
   232→  const gbpGross = s.gbp * fx.GBP_USD;
   233→  const jpyGross = s.jpy * fx.JPY_USD;
   234→  const cnyGross = s.cny * fx.CNY_USD;
   235→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross;
   236→
   237→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   238→  // In the pilot we treat the live XAU_USD as the executable price and apply
   239→  // a small conservative discount (0.5%) to model the reference/executable spread.
   240→  const goldRef = fx.XAU_USD;
   241→  const goldExec = fx.XAU_USD * 0.995;
   242→  const goldPrice = Math.min(goldRef, goldExec);
   243→  const goldGross = s.paxg * goldPrice;
   244→
   245→  // Net values (after haircuts)
   246→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   247→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   248→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   249→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   250→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   251→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   252→
   253→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + goldNet;
   254→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet;
   255→
   256→  return {
   257→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, fiatGross,
   258→    goldGross, goldPrice,
   259→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, goldNet,
   260→    nav, fiatNet,
   261→  };
   262→}

   294→// --- §6 Adaptive Macro Engine -----------------------------------------------
   295→function rollingStats(hist: number[]): { mean: number; sd: number } {
   296→  const filled = hist.filter((x) => x > 0);
   297→  if (filled.length === 0) return { mean: 0, sd: 0 };
   298→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   299→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   300→  return { mean, sd: Math.sqrt(variance) };
   301→}
   302→
   303→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   304→  const v = rollingStats(s.vixHistory);
   305→  const d = rollingStats(s.dxyHistory);
   306→  const eps = 1e-9;
   307→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   308→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   309→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   310→}

   312→export function computeRawTargetTheta(zVix: number, zDxy: number): number {
   313→  const theta = ALPHA * zVix + BETA * zDxy;
   314→  return clampSym(theta, -THETA_MAX, THETA_MAX);

   322→export function smoothGoldWeight(prev: number, target: number): number {
   323→  if (prev <= 0) return target;
   324→  return SMOOTHING_LAMBDA * target + (1 - SMOOTHING_LAMBDA) * prev;

   334→export function currentBufferGoldRatio(s: ReserveState): number {
   335→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   336→  if (s.rampStartAt === 0) {
   337→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   338→  }
   339→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   340→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   341→  const progress = elapsedH / RAMP_DURATION_HOURS;
   342→  const from = s.rampFromRatio;
   343→  const to = s.rampToRatio;
   344→  return from + (to - from) * progress;
   345→}

   347→export function bufferBaseFromRatio(rr: number): "BASE" | "STRESS" | "EMERGENCY" {
   348→  if (rr >= RR_TARGET) return "BASE";
   349→  if (rr >= RR_STRESS) return "STRESS";
   350→  return "EMERGENCY";
   351→}

   353→// Unified target gold weight (tokenomics reconciliation):
   354→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   355→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   356→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   357→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   358→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   359→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   360→}

   380→export function evaluateRebalance(
   381→  s: ReserveState,
   382→  vals: { nav: number; goldNet: number; fiatNet: number },
   383→  rr: number,
   384→): RebalanceDecision {
   385→  const nav = vals.nav;
   386→  if (nav <= 0) {
   387→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   388→  }
   389→  const observed = vals.goldNet / nav;
   390→  const target = computeTargetGoldWeight(s, rr);
   391→  const deviation = observed - target;
   392→
   393→  if (Math.abs(deviation) < 0.005) {
   394→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   395→  }
   396→
   397→  // Direction lock (overridden in stress)
   398→  const inStress = rr < RR_STRESS;
   399→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   400→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   401→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   402→    }
   403→  }
   404→
   405→  // Trade sizing
   406→  const surplusUsd = Math.abs(deviation) * nav;
   407→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   408→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   409→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   410→  maybeResetDailyTurnover(s);
   411→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   412→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   413→
   414→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   415→  const estCost = tradeUsd * 0.001 + 50;
   416→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   417→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   418→
   419→  if (benefit <= estCost) {
   420→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   421→  }
   422→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   423→}

   425→// Apply a rebalance trade in USD terms (mutates state).
   426→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   427→  if (!decision.shouldRebalance) return;
   428→  const usd = decision.tradeUsd;
   429→  if (decision.direction === -1) {
   430→    // sell gold → buy USDC (simplified: convert to USDC)
   431→    const goldUnits = usd / goldPrice;
   432→    s.paxg = Math.max(0, s.paxg - goldUnits);
   433→    s.usdc += usd;
   434→  } else {
   435→    // buy gold → spend USDC
   436→    s.usdc = Math.max(0, s.usdc - usd);
   437→    const goldUnits = usd / goldPrice;
   438→    s.paxg += goldUnits;
   439→  }
   440→  s.lastTradeDir = decision.direction;
   441→  s.lastTradeAt = Date.now();
   442→  s.dailyTurnoverUsd += usd;
   443→  s.updatedAt = Date.now();
   444→}

   446→// --- §11 Geopolitical Eject -------------------------------------------------
   447→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   448→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   449→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   450→  const drift = (cur: number, amp: number) => {
   451→    const shock = (Math.random() - 0.5) * 2 * amp;
   452→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   453→  };
   454→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   455→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   456→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   457→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   458→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   459→
   460→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   461→  for (const c of currencies) {
   462→    const p = s.pegHealth[c];
   463→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   464→    if (outside) {
   465→      s.depegHours[c] += dtHours;
   466→    } else {
   467→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   468→    }
   469→    // Staged ladder
   470→    const h = s.depegHours[c];
   471→    let stage = 0;
   472→    if (h > 48) stage = 3;
   473→    else if (h > 24) stage = 2;
   474→    else if (h > 12) stage = 1;
   475→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   476→    s.ejectStage[c] = stage;
   477→  }
   478→}

   494→export function applyMint(
   495→  s: ReserveState,

   504→  }

   507→  }

   509→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);

   514→  s.usdc += netUsd + feeUsd; // fee stays in reserve (operational surplus)
   515→  s.totalSupply += minted;

   518→  const vals = reserveAssetValues(s, fx);

   523→    feeUsd,
   524→    netUsd,
   525→    mtqPrice: price,
   526→    mtqMinted: minted,
   527→    throttleFactor: throttle,
   528→    newCirculatingSupply: circulatingSupply(s),

At RR>100% the §12.2 literal pays redeemers MORE than §3.4.2, draining the buffer surplus via arbitrage. We settle on the arbitrage-safe §3.4.2 value; the §12.2 figure is shown for audit only.";
   579→  if (inputMtq > circ) {
   580→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)), auditNavPerToken, auditGrossUsdNav, auditDeltaUsd: auditGrossUsdNav, auditNote };
   581→  }
   582→  // Fee by status (§14.1): NORMAL/CAUTION 0.15%, DEFENSIVE 0.5%, EMERGENCY 2%
   583→  let feeBps = REDEEM_FEE_BPS;
   584→  if (status === "DEFENSIVE") feeBps = 50;
   585→  if (status === "EMERGENCY") feeBps = 200;
   586→  const grossUsd = inputMtq * price; // redemption is priced against the GFB Index (§3.4.2)
   587→  const feeUsd = grossUsd * (feeBps / 10_000);
   588→  const netUsd = grossUsd - feeUsd;
   589→
   590→  // Release actual reserve composition proportionally (§12.2):
   591→  // gold portion = netUsd * W_target ; fiat portion split per GFB weights.
   592→  const vals = reserveAssetValues(s, fx);
   593→  const rr = computeReserveRatio(vals.nav, computeLiability(s, price));
   594→  const wGold = computeTargetGoldWeight(s, rr);
   595→  const goldUsd = netUsd * wGold;

   598→  const goldPaxg = goldUsd / goldPrice;
   599→
   600→  const basketFiatTotal = Q_USD + Q_EUR + Q_GBP + Q_JPY + Q_CNY; // = 1.0

   603→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * (Q_EUR / basketFiatTotal), nativeAmount: (fiatUsd * (Q_EUR / basketFiatTotal)) / fx.EUR_USD, weight: Q_EUR },
   604→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * (Q_GBP / basketFiatTotal), nativeAmount: (fiatUsd * (Q_GBP / basketFiatTotal)) / fx.GBP_USD, weight: Q_GBP },
   605→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * (Q_JPY / basketFiatTotal), nativeAmount: (fiatUsd * (Q_JPY / basketFiatTotal)) / fx.JPY_USD, weight: Q_JPY },
   606→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * (Q_CNY / basketFiatTotal), nativeAmount: (fiatUsd * (Q_CNY / basketFiatTotal)) / fx.CNY_USD, weight: Q_CNY },
   607→  ];

   610→  s.totalSupply -= inputMtq;
   611→  s.usdc = Math.max(0, s.usdc - basket[0].nativeAmount - (goldUsd > 0 && wGold < 1 ? 0 : 0));
   612→  s.usdc = Math.max(0, s.usdc - basket[0].nativeAmount);
   613→  s.eurc = Math.max(0, s.eurc - basket[1].nativeAmount);
   614→  s.gbp = Math.max(0, s.gbp - basket[2].nativeAmount);
   615→  s.jpy = Math.max(0, s.jpy - basket[3].nativeAmount);
   616→  s.cny = Math.max(0, s.cny - basket[4].nativeAmount);
   617→  s.paxg = Math.max(0, s.paxg - goldPaxg);
   618→  s.updatedAt = Date.now();
   619→
   620→  const newVals = reserveAssetValues(s, fx);
   621→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
   622→  return {
   623→    ok: true,
   624→    inputMtq,
   625→    mtqPrice: price,
   626→    grossUsd,
   627→    feeBps,
   628→    feeUsd,

   636→    auditGrossUsdNav,
   637→    auditDeltaUsd: auditGrossUsdNav - grossUsd,
   638→    auditNote,
   639→  };

   802→// --- §11.3 Reintegration Score (Anti-Gaming) --------------------------------
   803→// R_score = w1·TimeInBand(48h) + w2·LiquidityDepth + w3·OracleAgreement + w4·(1−Volatility)
   804→// All inputs normalised to [0,1]; threshold >0.80 → staged re-purchase 25%→50%→100%.
   805→export function updateReintegration(
   806→  s: ReserveState,
   807→  board: OracleBoard | null,
   808→  dtHours: number,
   809→) {
   810→  const currencies: ("USD" | "EUR" | "GBP" | "JPY" | "CNY")[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   811→  for (const c of currencies) {
   812→    const r = s.reintegration[c];
   813→    // Time-in-band: increases when peg is inside [0.98,1.02]; decays when outside.
   814→    const inBand = s.pegHealth[c] >= EJECT_DEPEG_BAND_LOWER && s.pegHealth[c] <= EJECT_DEPEG_BAND_UPPER;
   815→    // dtHours is simulated elapsed; we approximate 48h window fraction.
   816→    if (inBand) {
   817→      r.timeInBand48h = Math.min(1, r.timeInBand48h + dtHours / 48);
   818→    } else {
   819→      r.timeInBand48h = Math.max(0, r.timeInBand48h - dtHours / 48);
   820→    }
   821→    // Liquidity depth: randomised around per-currency baseline (honest: simulated).
   822→    const base = c === "USD" ? 1.0 : c === "EUR" ? 0.95 : c === "CNY" ? 0.4 : 0.5;
   823→    r.liquidityDepth = clamp(base + (Math.random() - 0.5) * 0.1, 0, 1);
   824→    // Oracle agreement: valid feeds / 3 (from oracle board if present, else last known)
   825→    if (board) {
   826→      const pairMap: Record<string, string> = { USD: "—", EUR: "EUR/USD", GBP: "GBP/USD", JPY: "JPY/USD", CNY: "CNY/USD" };
   827→      const cons = board.pairs.find((p) => p.pair === pairMap[c]);
   828→      if (cons) {
   829→        r.oracleAgreement = cons.validCount / 3;
   830→        // Volatility from oracle spread (bps → fraction)
   831→        r.volatility = clamp(cons.spreadBps / 100, 0, 1);
   832→      }
   833→    }
   834→    // R_score
   835→    r.score =
   836→      REINTEGRATION_WEIGHTS.w1_TimeInBand * r.timeInBand48h +
   837→      REINTEGRATION_WEIGHTS.w2_LiquidityDepth * r.liquidityDepth +
   838→      REINTEGRATION_WEIGHTS.w3_OracleAgreement * r.oracleAgreement +
   839→      REINTEGRATION_WEIGHTS.w4_OneMinusVolatility * (1 - r.volatility);
   840→    // Repurchase staging: only for currencies that were previously ejected
   841→    if (s.ejectStage[c] > 0 && r.score > REINTEGRATION_THRESHOLD) {
   842→      r.repurchaseStage = Math.min(3, r.repurchaseStage + 1);
   843→      // As repurchase progresses, reduce eject stage
   844→      if (r.repurchaseStage >= 3) {
   845→        s.ejectStage[c] = Math.max(0, s.ejectStage[c] - 1);
   846→        if (s.ejectStage[c] === 0) r.repurchaseStage = 0;
   847→      }
   848→    } else if (r.score < REINTEGRATION_THRESHOLD * 0.6) {
   849→      r.repurchaseStage = 0;
   850→    }
   851→  }
   852→}

   854→// --- §13.2 Treasury Sweep ---------------------------------------------------
   855→// If hot-wallet surplus stablecoins > $10,000 → sweep excess to 4/7 Cold Treasury.
   856→export function maybeTreasurySweep(s: ReserveState, feeRevenueAddedUsd = 0) {
   857→  // Accumulate fee revenue into the hot wallet (operational surplus)
   858→  s.treasury.hotWalletUsd += feeRevenueAddedUsd;
   859→  if (s.treasury.hotWalletUsd > TREASURY_SWEEP_THRESHOLD_USD) {
   860→    const excess = s.treasury.hotWalletUsd - 5_000; // keep $5k operating float
   861→    if (excess > 0) {
   862→      s.treasury.hotWalletUsd -= excess;
   863→      s.treasury.coldTreasuryUsd += excess;
   864→      s.treasury.lastSweepAt = Date.now();
   865→      s.treasury.lastSweepAmount = excess;
   866→      s.treasury.totalSwept += excess;
   867→    }
   868→  }
   869→}

   871→// --- §8.5 First-Loss Waterfall ----------------------------------------------
   872→// 5 layers: (1) Operational Surplus → (2) Buffer Fiat → (3) Buffer Gold →
   873→// (4) Core Fiat → (5) Core Gold. Apply a loss; track consumed per layer.
   874→export function applyLoss(s: ReserveState, vals: { fiatNet: number; goldNet: number }, lossUsd: number) {
   875→  let remaining = lossUsd;
   876→  s.waterfall.consumed.surplus = 0; // reset for this loss event tracking is cumulative; we add
   877→  const layers = [
   878→    { name: "surplus", cap: () => s.treasury.hotWalletUsd + s.waterfall.operationalSurplus, consume: (amt: number) => {
   879→      const fromSurplus = Math.min(s.treasury.hotWalletUsd, amt);
   880→      s.treasury.hotWalletUsd -= fromSurplus;
   881→      s.waterfall.operationalSurplus -= Math.min(s.waterfall.operationalSurplus, amt - fromSurplus);
   882→      s.waterfall.consumed.surplus += amt;
   883→    } },
   884→    { name: "bufferFiat", cap: () => vals.fiatNet * 0.10 * (1 - currentBufferGoldRatio(s)), consume: (amt: number) => { s.waterfall.consumed.bufferFiat += amt; } },
   885→    { name: "bufferGold", cap: () => vals.goldNet * 0.10 * currentBufferGoldRatio(s), consume: (amt: number) => { s.waterfall.consumed.bufferGold += amt; } },
   886→    { name: "coreFiat", cap: () => vals.fiatNet * 0.80, consume: (amt: number) => { s.waterfall.consumed.coreFiat += amt; } },
   887→    { name: "coreGold", cap: () => vals.goldNet * 0.20, consume: (amt: number) => { s.waterfall.consumed.coreGold += amt; } },
   888→  ] as const;
   889→  for (let i = 0; i < layers.length && remaining > 0; i++) {
   890→    const layer = layers[i];
   891→    const cap = layer.cap();
   892→    const consumed = Math.min(cap, remaining);
   893→    layer.consume(consumed);
   894→    remaining -= consumed;
   895→    if (remaining > 0) s.waterfall.currentLayer = (i + 2) as 1 | 2 | 3 | 4 | 5;
   896→  }
   897→  if (remaining <= 0) s.waterfall.currentLayer = 1;
   898→}

   900→// --- §3.6 PriceUpdated events (0.5% change threshold) -----------------------
   901→export function maybePriceEvent(s: ReserveState, newPrice: number) {
   902→  if (s.lastLoggedPrice <= 0) { s.lastLoggedPrice = newPrice; return; }
   903→  const diff = Math.abs(newPrice - s.lastLoggedPrice);
   904→  const changePct = diff / s.lastLoggedPrice;
   905→  if (changePct >= PRICE_EVENT_THRESHOLD) {
   906→    s.priceEvents.unshift({ ts: Date.now(), oldPrice: s.lastLoggedPrice, newPrice, changePct });
   907→    if (s.priceEvents.length > 20) s.priceEvents.pop();
   908→    s.lastLoggedPrice = newPrice;
   909→  }
   910→}

   912→// --- §10.1 Quote-based execution (route quote simulation) -------------------
   913→export interface ExecQuote {
   914→  provider: "1inch" | "Paraswap";
   915→  oracleRate: number;
   916→  quoteRate: number;
   917→  bound: number;        // oracleRate × (1∓τ)
   918→  withinBound: boolean;
   919→  effectiveSlippageBps: number;
   920→  amountUsd: number;
   921→}
   922→export function simulateQuote(amountUsd: number, isBuyGold: boolean, oracleGoldPrice: number): ExecQuote {
   923→  const provider: "1inch" | "Paraswap" = Math.random() < 0.5 ? "1inch" : "Paraswap";
   924→  // Realistic small slippage ~0.05%–0.4%
   925→  const slip = 0.0005 + Math.random() * 0.0035;
   926→  const oracleRate = oracleGoldPrice;
   927→  const quoteRate = isBuyGold ? oracleRate * (1 + slip) : oracleRate * (1 - slip);
   928→  const bound = isBuyGold ? oracleRate * (1 + SLIPPAGE_TOLERANCE) : oracleRate * (1 - SLIPPAGE_TOLERANCE);
   929→  const withinBound = isBuyGold ? quoteRate <= bound : quoteRate >= bound;
   930→  return {
   931→    provider, oracleRate, quoteRate, bound, withinBound,
   932→    effectiveSlippageBps: Math.round(slip * 10_000),
   933→    amountUsd,
   934→  };
   935→}
