   112→        fixed

   114→    );
   115→  }
   116→  if (severity === "outstanding") {
   117→    return (

   120→        outstanding

   125→    <Pill tone="muted">
   126→      <GlowDot color="gold" size="h-1.5 w-1.5" />
   127→      informational

   129→  );
   130→}
