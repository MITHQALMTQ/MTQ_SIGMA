    89→export function initReserveState(goldPrice: number): ReserveState {
    90→  const initialUsdDeposit = 1_100_000;
    91→  const fiatSpend = 880_000; // basket
    92→  const goldSpend = 220_000; // PAXG
    93→
    94→  // Split fiatSpend across GFB weights → token holdings
    95→  const usdc = fiatSpend * (Q_USD / (Q_USD + Q_EUR + Q_GBP + Q_JPY + Q_CNY)); // ≈ 0.389
    96→  const eurcUsd = fiatSpend * (Q_EUR / 1.0);
    97→  const gbpUsd = fiatSpend * (Q_GBP / 1.0);
    98→  const jpyUsd = fiatSpend * (Q_JPY / 1.0);
    99→  const cnyUsd = fiatSpend * (Q_CNY / 1.0);
   100→  // Convert USD-equivalent fiat holdings into native token units
   101→  // (at base rates, so native unit count = usdValue / baseRate)
   102→  const eurc = eurcUsd / 1.05; // € per $1.05
   103→  const gbp = gbpUsd / 1.25;
   104→  const jpy = jpyUsd / 0.0067;
   105→  const cny = cnyUsd / 0.14;
   106→  const paxg = goldSpend / goldPrice; // troy oz
   107→
   108→  return {
   109→    usdc,
   110→    eurc, gbp, jpy, cny,
   111→    paxg,
   112→    totalSupply: 1_000_000, // Genesis supply
   113→    genesisReserve: 1_000_000, // locked
   114→    vixHistory: seedHistory(18.5, 2.0, ROLLING_WINDOW_DAYS),
   115→    dxyHistory: seedHistory(104.2, 2.5, ROLLING_WINDOW_DAYS),
   116→    smoothedGoldWeight: BASE_GOLD_WEIGHT,

   134→function seedHistory(mean: number, sd: number, n: number): number[] {
   135→  const out: number[] = [];
   136→  let last = mean;
   137→  for (let i = 0; i < n; i++) {
   138→    // simple mean-reverting walk
   139→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   140→    last = last + (mean - last) * 0.1 + shock;
   141→    out.push(Math.max(0.01, last));
   142→  }
   143→  return out;
   144→}

   169→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   170→// USD value of each holding (gross), then apply haircuts for NAV.
   171→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   172→  const usdGross = s.usdc * 1.0;
   173→  const eurGross = s.eurc * fx.EUR_USD;
   174→  const gbpGross = s.gbp * fx.GBP_USD;
   175→  const jpyGross = s.jpy * fx.JPY_USD;
   176→  const cnyGross = s.cny * fx.CNY_USD;
   177→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross;
   178→
   179→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   180→  // In the pilot we treat the live XAU_USD as the executable price and apply
   181→  // a small conservative discount (0.5%) to model the reference/executable spread.
   182→  const goldRef = fx.XAU_USD;
   183→  const goldExec = fx.XAU_USD * 0.995;
   184→  const goldPrice = Math.min(goldRef, goldExec);
   185→  const goldGross = s.paxg * goldPrice;
   186→
   187→  // Net values (after haircuts)
   188→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   189→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   190→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   191→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   192→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   193→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   194→
   195→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + goldNet;
   196→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet;
   197→
   198→  return {
   199→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, fiatGross,
   200→    goldGross, goldPrice,
   201→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, goldNet,
   202→    nav, fiatNet,
   203→  };
   204→}

   236→// --- §6 Adaptive Macro Engine -----------------------------------------------
   237→function rollingStats(hist: number[]): { mean: number; sd: number } {
   238→  const filled = hist.filter((x) => x > 0);
   239→  if (filled.length === 0) return { mean: 0, sd: 0 };
   240→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   241→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   242→  return { mean, sd: Math.sqrt(variance) };
   243→}
   244→
   245→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   246→  const v = rollingStats(s.vixHistory);
   247→  const d = rollingStats(s.dxyHistory);
   248→  const eps = 1e-9;
   249→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   250→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   251→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   252→}

   254→export function computeRawTargetTheta(zVix: number, zDxy: number): number {
   255→  const theta = ALPHA * zVix + BETA * zDxy;
   256→  return clampSym(theta, -THETA_MAX, THETA_MAX);
   257→}

   261→  return clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);

   264→export function smoothGoldWeight(prev: number, target: number): number {
   265→  if (prev <= 0) return target;
   266→  return SMOOTHING_LAMBDA * target + (1 - SMOOTHING_LAMBDA) * prev;
   267→}

   270→export function bufferBaseGoldRatio(rr: number): number {
   271→  if (rr >= RR_TARGET) return BUFFER_GOLD_BASE; // 62.5% → total 26.25%
   272→  if (rr >= RR_STRESS) return BUFFER_GOLD_STRESS; // 85% → total 28.5%

   276→export function currentBufferGoldRatio(s: ReserveState): number {
   277→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   278→  if (s.rampStartAt === 0) {
   279→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   280→  }
   281→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   282→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   283→  const progress = elapsedH / RAMP_DURATION_HOURS;
   284→  const from = s.rampFromRatio;
   285→  const to = s.rampToRatio;
   286→  return from + (to - from) * progress;
   287→}

   289→export function bufferBaseFromRatio(rr: number): "BASE" | "STRESS" | "EMERGENCY" {
   290→  if (rr >= RR_TARGET) return "BASE";
   291→  if (rr >= RR_STRESS) return "STRESS";
   292→  return "EMERGENCY";
   293→}

   295→// Unified target gold weight (tokenomics reconciliation):
   296→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   297→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   298→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   299→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   300→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   301→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   302→}

   322→export function evaluateRebalance(
   323→  s: ReserveState,
   324→  vals: { nav: number; goldNet: number; fiatNet: number },
   325→  rr: number,
   326→): RebalanceDecision {
   327→  const nav = vals.nav;
   328→  if (nav <= 0) {
   329→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   330→  }
   331→  const observed = vals.goldNet / nav;
   332→  const target = computeTargetGoldWeight(s, rr);
   333→  const deviation = observed - target;
   334→
   335→  if (Math.abs(deviation) < 0.005) {
   336→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   337→  }
   338→
   339→  // Direction lock (overridden in stress)
   340→  const inStress = rr < RR_STRESS;
   341→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   342→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   343→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   344→    }
   345→  }
   346→
   347→  // Trade sizing
   348→  const surplusUsd = Math.abs(deviation) * nav;
   349→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   350→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   351→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   352→  maybeResetDailyTurnover(s);
   353→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   354→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   355→
   356→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   357→  const estCost = tradeUsd * 0.001 + 50;
   358→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   359→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   360→
   361→  if (benefit <= estCost) {
   362→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   363→  }
   364→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   365→}

   367→// Apply a rebalance trade in USD terms (mutates state).
   368→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   369→  if (!decision.shouldRebalance) return;
   370→  const usd = decision.tradeUsd;
   371→  if (decision.direction === -1) {
   372→    // sell gold → buy USDC (simplified: convert to USDC)
   373→    const goldUnits = usd / goldPrice;
   374→    s.paxg = Math.max(0, s.paxg - goldUnits);
   375→    s.usdc += usd;
   376→  } else {
   377→    // buy gold → spend USDC
   378→    s.usdc = Math.max(0, s.usdc - usd);
   379→    const goldUnits = usd / goldPrice;
   380→    s.paxg += goldUnits;
   381→  }
   382→  s.lastTradeDir = decision.direction;
   383→  s.lastTradeAt = Date.now();
   384→  s.dailyTurnoverUsd += usd;
   385→  s.updatedAt = Date.now();
   386→}

   388→// --- §11 Geopolitical Eject -------------------------------------------------
   389→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   390→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   391→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   392→  const drift = (cur: number, amp: number) => {
   393→    const shock = (Math.random() - 0.5) * 2 * amp;
   394→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   395→  };
   396→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   397→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   398→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   399→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   400→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   401→
   402→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   403→  for (const c of currencies) {
   404→    const p = s.pegHealth[c];
   405→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   406→    if (outside) {
   407→      s.depegHours[c] += dtHours;
   408→    } else {
   409→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   410→    }
   411→    // Staged ladder
   412→    const h = s.depegHours[c];
   413→    let stage = 0;
   414→    if (h > 48) stage = 3;
   415→    else if (h > 24) stage = 2;
   416→    else if (h > 12) stage = 1;
   417→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   418→    s.ejectStage[c] = stage;
   419→  }
   420→}

   436→export function applyMint(
   437→  s: ReserveState,
   438→  fx: FxRates,
   439→  status: ProtocolStatus,
   440→  inputUsd: number,
   441→): MintResult {
   442→  const gfb = computeGfbIndex(fx);
   443→  const price = computeMtqPrice(gfb);
   444→  if (!priceInSafetyBand(price)) {
   445→    return { ok: false, reason: "MTQ price outside safety band (0.50–2.00 USD); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   446→  }
   447→  if (status === "DEFENSIVE" || status === "EMERGENCY") {
   448→    return { ok: false, reason: `${status}: minting paused by Risk State Machine.`, inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   449→  }
   450→  const throttle = status === "CAUTION" ? 0.5 : status === "RECOVERY" ? 0.25 : 1.0;
   451→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
   452→  const netUsd = inputUsd - feeUsd;
   453→  const minted = (netUsd / price) * throttle;
   454→
   455→  // Mutate reserve: deposit USDC, mint MTQ to user (circulating supply rises)
   456→  s.usdc += netUsd + feeUsd; // fee stays in reserve (operational surplus)
   457→  s.totalSupply += minted;
   458→  s.updatedAt = Date.now();
   459→
   460→  const vals = reserveAssetValues(s, fx);
   461→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
   462→  return {
   463→    ok: true,
   464→    inputUsd,
   465→    feeUsd,
   466→    netUsd,
   467→    mtqPrice: price,
   468→    mtqMinted: minted,
   469→    throttleFactor: throttle,
   470→    newCirculatingSupply: circulatingSupply(s),
   471→    newReserveRatio: newRr,
   472→  };
   473→}

   483→export interface RedeemResult {
   484→  ok: boolean;
   485→  reason?: string;
   486→  inputMtq: number;
   487→  mtqPrice: number;
   488→  grossUsd: number;
   489→  feeBps: number;
   490→  feeUsd: number;
   491→  netUsd: number;
   492→  goldUsd: number;
   493→  goldPaxg: number;
   494→  basket: RedeemBasket[];
   495→  newCirculatingSupply: number;
   496→  newReserveRatio: number;
   497→}

   499→export function applyRedeem(
   500→  s: ReserveState,
   501→  fx: FxRates,
   502→  status: ProtocolStatus,
   503→  inputMtq: number,
   504→): RedeemResult {
   505→  const gfb = computeGfbIndex(fx);
   506→  const price = computeMtqPrice(gfb);
   507→  const circ = circulatingSupply(s);
   508→  if (inputMtq > circ) {
   509→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   510→  }
   511→  // Fee by status (§14.1): NORMAL/CAUTION 0.15%, DEFENSIVE 0.5%, EMERGENCY 2%
   512→  let feeBps = REDEEM_FEE_BPS;
   513→  if (status === "DEFENSIVE") feeBps = 50;
   514→  if (status === "EMERGENCY") feeBps = 200;
   515→  const grossUsd = inputMtq * price; // redemption is priced against the GFB Index (§3.4.2)
   516→  const feeUsd = grossUsd * (feeBps / 10_000);
   517→  const netUsd = grossUsd - feeUsd;
   518→
   519→  // Release actual reserve composition proportionally (§12.2):
   520→  // gold portion = netUsd * W_target ; fiat portion split per GFB weights.
   521→  const vals = reserveAssetValues(s, fx);
   522→  const rr = computeReserveRatio(vals.nav, computeLiability(s, price));
   523→  const wGold = computeTargetGoldWeight(s, rr);
   524→  const goldUsd = netUsd * wGold;
   525→  const fiatUsd = netUsd * (1 - wGold);
   526→  const goldPrice = vals.goldPrice;
   527→  const goldPaxg = goldUsd / goldPrice;
   528→
   529→  const basketFiatTotal = Q_USD + Q_EUR + Q_GBP + Q_JPY + Q_CNY; // = 1.0
   530→  const basket: RedeemBasket[] = [
   531→    { currency: "USD", token: "USDC", usdValue: fiatUsd * (Q_USD / basketFiatTotal), nativeAmount: fiatUsd * (Q_USD / basketFiatTotal), weight: Q_USD },
   532→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * (Q_EUR / basketFiatTotal), nativeAmount: (fiatUsd * (Q_EUR / basketFiatTotal)) / fx.EUR_USD, weight: Q_EUR },
   533→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * (Q_GBP / basketFiatTotal), nativeAmount: (fiatUsd * (Q_GBP / basketFiatTotal)) / fx.GBP_USD, weight: Q_GBP },
   534→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * (Q_JPY / basketFiatTotal), nativeAmount: (fiatUsd * (Q_JPY / basketFiatTotal)) / fx.JPY_USD, weight: Q_JPY },
   535→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * (Q_CNY / basketFiatTotal), nativeAmount: (fiatUsd * (Q_CNY / basketFiatTotal)) / fx.CNY_USD, weight: Q_CNY },
   536→  ];
   537→
   538→  // Mutate reserve: burn MTQ from circulating (reduce totalSupply), release assets
   539→  s.totalSupply -= inputMtq;
   540→  s.usdc = Math.max(0, s.usdc - basket[0].nativeAmount - (goldUsd > 0 && wGold < 1 ? 0 : 0));
   541→  s.usdc = Math.max(0, s.usdc - basket[0].nativeAmount);
   542→  s.eurc = Math.max(0, s.eurc - basket[1].nativeAmount);
   543→  s.gbp = Math.max(0, s.gbp - basket[2].nativeAmount);
   544→  s.jpy = Math.max(0, s.jpy - basket[3].nativeAmount);
   545→  s.cny = Math.max(0, s.cny - basket[4].nativeAmount);
   546→  s.paxg = Math.max(0, s.paxg - goldPaxg);
   547→  s.updatedAt = Date.now();
   548→
   549→  const newVals = reserveAssetValues(s, fx);
   550→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
   551→  return {
   552→    ok: true,
   553→    inputMtq,
   554→    mtqPrice: price,
   555→    grossUsd,
   556→    feeBps,
   557→    feeUsd,
   558→    netUsd,
   559→    goldUsd,
   560→    goldPaxg,
   561→    basket,
   562→    newCirculatingSupply: circulatingSupply(s),
   563→    newReserveRatio: newRr,
   564→  };
   565→}

   660→// Advance macro history (call ~ every tick; dtHours is simulated elapsed)
   661→export function advanceMacro(s: ReserveState, vix: number, dxy: number, dtHours: number) {
   662→  const v = clamp(vix, VIX_MIN, VIX_MAX);
   663→  const d = clamp(dxy, DXY_MIN, DXY_MAX);
   664→  s.lastVix = v;
   665→  s.lastDxy = d;
   666→  s.vixHistory.push(v);
   667→  s.dxyHistory.push(d);
   668→  if (s.vixHistory.length > ROLLING_WINDOW_DAYS) s.vixHistory.shift();
   669→  if (s.dxyHistory.length > ROLLING_WINDOW_DAYS) s.dxyHistory.shift();
   670→  const vStats = rollingStats(s.vixHistory);
   671→  const dStats = rollingStats(s.dxyHistory);
   672→  const zVix = vStats.sd > 1e-9 ? (v - vStats.mean) / (vStats.sd + 1e-9) : 0;
   673→  const zDxy = dStats.sd > 1e-9 ? (d - dStats.mean) / (dStats.sd + 1e-9) : 0;
   674→  const rawTarget = computeRawTargetGoldWeight(zVix, zDxy);
   675→  s.smoothedGoldWeight = smoothGoldWeight(s.smoothedGoldWeight, rawTarget);
   676→}

   678→// Stochastic macro walk (honest: simulated pilot macro signals)
   679→export function stepMacroSignals(prev: { vix: number; dxy: number }): { vix: number; dxy: number } {
   680→  const vix = clamp(prev.vix + (Math.random() - 0.5) * 1.2 + (18 - prev.vix) * 0.02, VIX_MIN, VIX_MAX);
   681→  const dxy = clamp(prev.dxy + (Math.random() - 0.5) * 0.8 + (104 - prev.dxy) * 0.02, DXY_MIN, DXY_MAX);
   682→  return { vix, dxy };

   685→// Update buffer state + ramp on RR change (call after metrics recompute)
   686→export function updateBufferState(s: ReserveState, rr: number) {
   687→  const newState = bufferBaseFromRatio(rr);
   688→  if (newState === s.bufferState) return;
   689→  if (newState === "EMERGENCY") {
   690→    s.bufferState = "EMERGENCY";
   691→    s.rampStartAt = 0;
   692→  } else {
   693→    s.rampStartAt = Date.now();
   694→    s.rampFromRatio = currentBufferGoldRatio(s);
   695→    s.rampToRatio = newState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   696→    s.bufferState = newState;
   697→  }
