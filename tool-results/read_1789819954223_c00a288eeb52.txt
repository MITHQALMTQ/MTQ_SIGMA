   387→function seedHistory(mean: number, sd: number, n: number): number[] {
   388→  const out: number[] = [];
   389→  let last = mean;
   390→  for (let i = 0; i < n; i++) {
   391→    // simple mean-reverting walk
   392→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   393→    last = last + (mean - last) * 0.1 + shock;
   394→    out.push(Math.max(0.01, last));
   395→  }
   396→  return out;
   397→}

   399→// --- §14.1 Constitutional Separation helpers ---------------------------------
   400→// Maintain the §14.1 invariant: paxg = indexPaxg + reservePaxg (and same for
   401→// xaut). Index gold is LOCKED (only changes via commitIndexGold); reserve gold
   402→// is the buffer MARP rebalances. These two helpers are the only sanctioned
   403→// ways to keep the splits consistent after any mutation to total gold:
   404→
   405→/** Re-derive reserve gold from total − index (call after the legacy path or
   406→ *  concentration optimizer mutates `s.paxg` / `s.xaut` directly). Index gold
   407→ *  stays locked. */
   408→export function syncReserveFromTotal(s: ReserveState): void {
   409→  s.reservePaxg = Math.max(0, s.paxg - s.indexPaxg);
   410→  s.reserveXaut = Math.max(0, s.xaut - s.indexXaut);
   411→}
   412→
   413→/** Re-derive total gold from index + reserve (call after the MARP path mutates
   414→ *  `s.reservePaxg` / `s.reserveXaut`). Index gold stays locked. */

   416→  s.paxg = s.indexPaxg + Math.max(0, s.reservePaxg);

   458→// --- §2 GFB Index (v1.0 — 7-component chain-linked) -------------------------
   459→// GFB_t = Σ_i W^Prior_i × P_{i,t}  (then normalised by GFB_BASE_DENOMINATOR so
   460→// GFB = 1.00 exactly at the base date). The 7 components are the Strategic
   461→// Prior: USD, EUR, JPY, GBP, CNY, CHF, Gold. Gold is now a first-class index
   462→// component (P_Gold = XAU_USD).
   463→//
   464→// DEPRECATED (P0-FIX-1): this is the LEGACY fixed-base Laspeyres index. It
   465→// weights each component by its raw USD notional share (gold contributes
   466→// 99.9% of the index by notional despite a 26% Strategic Prior weight), which
   467→// creates a structural short-gold exposure that crashed the §23 validation
   468→// program (S5: 0% survival, RR crashed to 0.83 in 100% of gold +50% runs).
   469→//
   470→// The CANONICAL chain-linked index lives in `src/lib/mtq/chain-index.ts`. The
   471→// engine's `computeSnapshot` now uses `getMTQPrice(s.chainIndex)` (the
   472→// chain-linked form) for both `gfbIndex` and `mtqPrice`. This legacy
   473→// `computeGfbIndex` is RETAINED for backward compatibility with the
   474→// standalone audit-stress.ts simulations (which need a pure function of fx,
   475→// not the persisted chain index state).
   476→export function computeGfbIndex(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): number {
   477→  // Raw USD value of the basket at time t. USD is the unit of account (price 1.0);
   478→  // every other component contributes its USD-equivalent via its FX rate.
   479→  const numerator =
   480→    STRATEGIC_PRIOR.USD  * 1.0 +
   481→    STRATEGIC_PRIOR.EUR  * fx.EUR_USD +
   482→    STRATEGIC_PRIOR.JPY  * fx.JPY_USD +
   483→    STRATEGIC_PRIOR.GBP  * fx.GBP_USD +
   484→    STRATEGIC_PRIOR.CNY  * fx.CNY_USD +
   485→    STRATEGIC_PRIOR.CHF  * fx.CHF_USD +
   486→    STRATEGIC_PRIOR.Gold * fx.XAU_USD;
   487→  // Normalised: GFB_t = numerator / GFB_BASE_DENOMINATOR  (=1.0 at base date).
   488→  // GFB_BASE_DENOMINATOR is the v1.0 chain-linked denominator (includes Gold
   489→  // and CHF at base fixings — see blueprint.ts).
   490→  return numerator / GFB_BASE_DENOMINATOR;
   491→}

   540→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   541→// USD value of each holding (gross), then apply haircuts for NAV.
   542→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   543→  const usdGross = (s.usdc + s.usdp + s.usdt) * 1.0; // USDC + USDP + USDT combined
   544→  const eurGross = s.eurc * fx.EUR_USD;
   545→  const gbpGross = s.gbp * fx.GBP_USD;
   546→  const jpyGross = s.jpy * fx.JPY_USD;
   547→  const cnyGross = s.cny * fx.CNY_USD;
   548→  const chfGross = s.chf * fx.CHF_USD; // NEW v1.0 — CHF is a first-class reserve component
   549→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross + chfGross;
   550→
   551→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   552→  const goldRef = fx.XAU_USD;
   553→  const goldExec = fx.XAU_USD * 0.995;
   554→  const goldPrice = Math.min(goldRef, goldExec);
   555→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   556→
   557→  // Net values (after haircuts)
   558→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   559→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   560→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   561→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   562→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   563→  const chfNet = chfGross * (1 - HAIRCUTS.CHF); // NEW v1.0
   564→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   565→
   566→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet + goldNet;
   567→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet;
   568→
   569→  // Per-issuer breakdown (for concentration visualization)
   570→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   571→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   572→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   573→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);
   574→  const xautUsd = s.xaut * goldPrice * (1 - HAIRCUTS.XAU);
   575→
   576→  // §14.1 — index vs reserve gold split (USD net of haircut).
   577→  // Index gold is locked (backs the 26% Strategic Prior Gold weight); reserve
   578→  // gold is the buffer MARP rebalances. The total gold (= index + reserve) is
   579→  // what serves the index weight for MASE/MARP observed-weight purposes.
   580→  const indexPaxgUsd = s.indexPaxg * goldPrice * (1 - HAIRCUTS.XAU);
   581→  const indexXautUsd = s.indexXaut * goldPrice * (1 - HAIRCUTS.XAU);
   582→  const reservePaxgUsd = (s.reservePaxg >= 0 ? s.reservePaxg : Math.max(0, s.paxg - s.indexPaxg)) * goldPrice * (1 - HAIRCUTS.XAU);
   583→  const reserveXautUsd = (s.reserveXaut >= 0 ? s.reserveXaut : Math.max(0, s.xaut - s.indexXaut)) * goldPrice * (1 - HAIRCUTS.XAU);
   584→  const indexGoldNet = indexPaxgUsd + indexXautUsd;
   585→  const reserveGoldNet = reservePaxgUsd + reserveXautUsd;
   586→
   587→  return {
   588→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, chfGross, fiatGross,
   589→    goldGross, goldPrice,
   590→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, chfNet, goldNet,
   591→    nav, fiatNet,
   592→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   593→    // §14.1 gold split
   594→    indexPaxgUsd, indexXautUsd, reservePaxgUsd, reserveXautUsd,
   595→    indexGoldNet, reserveGoldNet,
   596→  };
   597→}

   623→// engine's persisted riskState and returns just the state string. The full

   632→export function determineStatus(rr: number, lcr: number): ProtocolStatus {

   637→  // condition-binds rule exactly.

   639→  const cs = determineState(rr, lcr, prev, Date.now(), Date.now());
   640→  return cs.state;

   682→// --- §6 Adaptive Macro Engine -----------------------------------------------
   683→function rollingStats(hist: number[]): { mean: number; sd: number } {
   684→  const filled = hist.filter((x) => x > 0);
   685→  if (filled.length === 0) return { mean: 0, sd: 0 };
   686→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   687→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   688→  return { mean, sd: Math.sqrt(variance) };
   689→}
   690→
   691→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   692→  const v = rollingStats(s.vixHistory);
   693→  const d = rollingStats(s.dxyHistory);
   694→  const eps = 1e-9;
   695→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   696→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   697→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   698→}

   722→export function currentBufferGoldRatio(s: ReserveState): number {
   723→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   724→  if (s.rampStartAt === 0) {
   725→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   726→  }
   727→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   728→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   729→  const progress = elapsedH / RAMP_DURATION_HOURS;
   730→  const from = s.rampFromRatio;
   731→  const to = s.rampToRatio;
   732→  return from + (to - from) * progress;
   733→}

   744→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp

   761→export function maybeResetDailyTurnover(s: ReserveState): void {
   762→  if (Date.now() - s.lastTurnoverResetAt >= 86_400_000) {
   763→    s.dailyTurnoverUsd = 0;
   764→    s.lastTurnoverResetAt = Date.now();

   768→export function evaluateRebalance(
   769→  s: ReserveState,
   770→  vals: { nav: number; goldNet: number; fiatNet: number },
   771→  rr: number,
   772→): RebalanceDecision {
   773→  const nav = vals.nav;
   774→  if (nav <= 0) {
   775→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   776→  }
   777→  const observed = vals.goldNet / nav;
   778→  const target = computeTargetGoldWeight(s, rr);
   779→  const deviation = observed - target;
   780→
   781→  if (Math.abs(deviation) < 0.005) {
   782→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   783→  }
   784→
   785→  // Direction lock (overridden in stress)
   786→  const inStress = rr < RR_STRESS;
   787→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   788→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   789→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   790→    }
   791→  }
   792→
   793→  // Trade sizing
   794→  const surplusUsd = Math.abs(deviation) * nav;
   795→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   796→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   797→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   798→  maybeResetDailyTurnover(s);
   799→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   800→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   801→
   802→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   803→  const estCost = tradeUsd * 0.001 + 50;
   804→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   805→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   806→
   807→  if (benefit <= estCost) {
   808→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   809→  }
   810→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   811→}

   813→// --- B9-FIX — conservation-of-value subtraction helper -----------------------
   814→// Replaces `Math.max(0, holding - amount)` on every reserve mutation path.
   815→// When the subtraction would push the holding below 0, the deficit (in native
   816→// units) is converted to USD via `usdPerUnit` and accumulated into
   817→// `s.deficitUsd`. The holding is then clamped to 0 (the trade still executes)
   818→// — but the deficit is now visible on the books instead of silently
   819→// disappearing. This preserves the conservation invariant
   820→//   assets + deficitUsd = liabilities + injectedValue
   821→// across every rebalance/marp trade.
   822→//
   823→// `usdPerUnit` is the spot USD value of one native unit:
   824→//   - USD stablecoins (USDC/USDP/USDT): 1
   825→//   - EURC: fx.EUR_USD · JPY: fx.JPY_USD · GBP: fx.GBP_USD
   826→//   - CNY: fx.CNY_USD · CHF: fx.CHF_USD
   827→//   - PAXG/XAUT: goldPrice (XAU_USD)
   828→//
   829→// `component` is a human-readable label for the warn log (e.g. "paxg", "usdc").
   830→function subtractReserve(
   831→  s: ReserveState,
   832→  current: number,
   833→  amount: number,
   834→  usdPerUnit: number,
   835→  component: string,
   836→): number {
   837→  const newValue = current - amount;
   838→  if (newValue < 0) {
   839→    const deficitNative = -newValue;
   840→    const deficitUsd = deficitNative * usdPerUnit;
   841→    console.warn('[engine] B9: negative holding detected', {
   842→      component,
   843→      old: current,
   844→      subtracted: amount,
   845→      deficitNative,
   846→      deficitUsd,
   847→      cumulativeDeficitUsd: (s.deficitUsd ?? 0) + deficitUsd,
   848→    });
   849→    s.deficitUsd = (s.deficitUsd ?? 0) + deficitUsd;
   850→    return 0;
   851→  }
   852→  return newValue;
   853→}

   855→// Apply a rebalance trade in USD terms (mutates state).
   856→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   857→  if (!decision.shouldRebalance) return;
   858→  const usd = decision.tradeUsd;
   859→  if (decision.direction === -1) {
   860→    // sell gold → buy USDC (simplified: convert to USDC)
   861→    const goldUnits = usd / goldPrice;
   862→    // B9-FIX: track any deficit instead of silently zeroing.
   863→    s.paxg = subtractReserve(s, s.paxg, goldUnits, goldPrice, 'paxg');
   864→    // §14.1 — the sold gold comes from the RESERVE buffer (index gold is locked).
   865→    // Maintain the invariant paxg = indexPaxg + reservePaxg.
   866→    s.reservePaxg = subtractReserve(s, s.reservePaxg, goldUnits, goldPrice, 'reservePaxg');
   867→    s.usdc += usd;
   868→  } else {
   869→    // buy gold → spend USDC
   870→    // B9-FIX: track any deficit instead of silently zeroing.
   871→    s.usdc = subtractReserve(s, s.usdc, usd, 1, 'usdc');
   872→    const goldUnits = usd / goldPrice;
   873→    s.paxg += goldUnits;
   874→    // §14.1 — the bought gold goes to the RESERVE buffer (index gold is locked).
   875→    s.reservePaxg += goldUnits;
   876→  }
   877→  s.lastTradeDir = decision.direction;
   878→  s.lastTradeAt = Date.now();
   879→  s.dailyTurnoverUsd += usd;
   880→  s.updatedAt = Date.now();
   881→}

   883→// --- §14.1 + §10 MARP per-component rebalance execution ---------------------
   884→// applyMarpRebalance() is the v1.0 production-target rebalance path. Unlike the
   885→// legacy §7 single-direction `applyRebalanceTrade` (which only trades gold ↔
   886→// USDC), this takes the per-component MarpDecision[] (output of marpDecision())
   887→// and applies each "would-execute" trade to the RESERVE holdings — index gold
   888→// is LOCKED per §14.1 and never touched here.
   889→//
   890→// Trade execution model (pilot):
   891→//   - For non-USD components (EUR/JPY/GBP/CNY/CHF/Gold): each trade pairs with
   892→//     USD as the unit of account. A "sell" of component X converts X → USD
   893→//     (split 1/3 across USDC/USDP/USDT, respecting the §5.6 concentration
   894→//     target). A "buy" of component X converts USD → X.
   895→//   - For USD: each trade pairs with Gold (the other deep pool). A "sell" of
   896→//     USD converts USD → Gold (split 50/50 PAXG/XAUT in the RESERVE buffer).
   897→//     A "buy" of USD converts Gold → USD.
   898→//   - For Gold: a "sell" reduces reservePaxg + reserveXaut (50/50); a "buy"
   899→//     increases them. Index gold is never touched.
   900→//
   901→// Constraints enforced (§10):
   902→//   - MAX_DAILY_TURNOVER (5% of NAV) — trades that would breach are skipped.
   903→//   - DIRECTION_LOCK_HOURS (24h) — trades that reverse the last direction
   904→//     within 24h are skipped (whipsaw guard).
   905→//   - Only decisions with shouldTrade=true AND level>=6 are executed
   906→//     (level 1-5 are advisory: no-trade zone / low urgency / cost-benefit
   907→//     fail / turnover cap).
   908→//
   909→// Mutates: s.usdc/usdp/usdt, s.eurc/gbp/jpy/cny/chf, s.reservePaxg/reserveXaut
   910→//          (and re-syncs s.paxg/xaut totals via the §14.1 invariant),
   911→//          s.lastTradeDir/lastTradeAt, s.dailyTurnoverUsd, s.updatedAt.
   912→// Index gold (s.indexPaxg, s.indexXaut) is NEVER touched.
   913→export interface MarpExecDecision {
   914→  shouldTrade: boolean;
   915→  component: string;
   916→  direction: string;
   917→  tradeUsd: number;
   918→  level: number;
   919→}
   920→
   921→export interface MarpExecResult {
   922→  appliedCount: number;
   923→  skippedCount: number;
   924→  totalTradeUsd: number;
   925→  path: 'legacy' | 'marp';
   926→  // Per-decision trace: which were applied vs skipped (and why).
   927→  traces: { component: string; direction: string; tradeUsd: number; level: number; applied: boolean; skipReason?: string }[];
   928→}
   929→
   930→export function applyMarpRebalance(
   931→  s: ReserveState,
   932→  fx: FxRates,
   933→  decisions: MarpExecDecision[],
   934→): MarpExecResult {
   935→  let appliedCount = 0;
   936→  let skippedCount = 0;
   937→  let totalTradeUsd = 0;
   938→  const traces: MarpExecResult['traces'] = [];
   939→
   940→  maybeResetDailyTurnover(s);
   941→  const vals = reserveAssetValues(s, fx);
   942→  const nav = vals.nav;
   943→  if (nav <= 0) {
   944→    // Nothing to do — surface a single skip trace for visibility.
   945→    for (const d of decisions) {
   946→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'NAV ≤ 0' });
   947→    }
   948→    return { appliedCount: 0, skippedCount: decisions.length, totalTradeUsd: 0, path: 'marp', traces };
   949→  }
   950→  const goldPrice = vals.goldPrice;
   951→
   952→  for (const d of decisions) {
   953→    // Level gate: only execute shouldTrade && level >= 6 decisions.
   954→    if (!d.shouldTrade || d.level < 6) {
   955→      skippedCount++;
   956→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: `level ${d.level} (advisory)` });
   957→      continue;
   958→    }
   959→    let tradeUsd = d.tradeUsd;
   960→    // Enforce MAX_DAILY_TURNOVER (5% of NAV)
   961→    const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   962→    if (tradeUsd > remainingDaily) {
   963→      tradeUsd = Math.max(0, remainingDaily);
   964→    }
   965→    if (tradeUsd <= 0) {
   966→      skippedCount++;
   967→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'daily turnover cap' });
   968→      continue;
   969→    }
   970→    // Direction (+1 buy / -1 sell / 0 hold)
   971→    const dir: 0 | 1 | -1 = d.direction === 'buy' ? 1 : d.direction === 'sell' ? -1 : 0;
   972→    if (dir === 0) {
   973→      skippedCount++;
   974→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'hold' });
   975→      continue;
   976→    }
   977→    // Enforce DIRECTION_LOCK_HOURS (24h whipsaw guard)
   978→    if (Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   979→      if ((dir === 1 && s.lastTradeDir === -1) || (dir === -1 && s.lastTradeDir === 1)) {
   980→        skippedCount++;
   981→        traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'direction lock (24h whipsaw guard)' });
   982→        continue;
   983→      }
   984→    }
   985→
   986→    // Apply the trade to the RESERVE holdings.
   987→    if (d.component === 'Gold') {
   988→      // Gold ↔ USD. Split 50/50 across reservePaxg + reserveXaut.
   989→      const goldUnits = tradeUsd / goldPrice;
   990→      const half = goldUnits / 2;
   991→      if (dir === -1) {
   992→        // sell gold → USD
   993→        // B9-FIX: track any deficit instead of silently zeroing.
   994→        s.reservePaxg = subtractReserve(s, s.reservePaxg, half, goldPrice, 'reservePaxg');
   995→        s.reserveXaut = subtractReserve(s, s.reserveXaut, half, goldPrice, 'reserveXaut');
   996→        const usdThird = tradeUsd / 3;
   997→        s.usdc += usdThird;
   998→        s.usdp += usdThird;
   999→        s.usdt += usdThird;
  1000→      } else {
  1001→        // buy gold ← USD
  1002→        const usdThird = tradeUsd / 3;
  1003→        // B9-FIX: track any deficit instead of silently zeroing.
  1004→        s.usdc = subtractReserve(s, s.usdc, usdThird, 1, 'usdc');
  1005→        s.usdp = subtractReserve(s, s.usdp, usdThird, 1, 'usdp');
  1006→        s.usdt = subtractReserve(s, s.usdt, usdThird, 1, 'usdt');
  1007→        s.reservePaxg += half;
  1008→        s.reserveXaut += half;
  1009→      }
  1010→      // §14.1 — re-sync total paxg/xaut = index + reserve
  1011→      syncTotalFromReserve(s);
  1012→    } else if (d.component === 'USD') {
  1013→      // USD ↔ Gold. Split 50/50 across reservePaxg + reserveXaut.
  1014→      const goldUnits = tradeUsd / goldPrice;
  1015→      const half = goldUnits / 2;
  1016→      if (dir === -1) {
  1017→        // sell USD → gold
  1018→        const usdThird = tradeUsd / 3;
  1019→        // B9-FIX: track any deficit instead of silently zeroing.
  1020→        s.usdc = subtractReserve(s, s.usdc, usdThird, 1, 'usdc');
  1021→        s.usdp = subtractReserve(s, s.usdp, usdThird, 1, 'usdp');
  1022→        s.usdt = subtractReserve(s, s.usdt, usdThird, 1, 'usdt');
  1023→        s.reservePaxg += half;
  1024→        s.reserveXaut += half;
  1025→      } else {
  1026→        // buy USD ← gold
  1027→        // B9-FIX: track any deficit instead of silently zeroing.
  1028→        s.reservePaxg = subtractReserve(s, s.reservePaxg, half, goldPrice, 'reservePaxg');
  1029→        s.reserveXaut = subtractReserve(s, s.reserveXaut, half, goldPrice, 'reserveXaut');
  1030→        const usdThird = tradeUsd / 3;
  1031→        s.usdc += usdThird;
  1032→        s.usdp += usdThird;
  1033→        s.usdt += usdThird;
  1034→      }
  1035→      // §14.1 — re-sync total paxg/xaut = index + reserve
  1036→      syncTotalFromReserve(s);
  1037→    } else {
  1038→      // Non-USD fiat component (EUR/JPY/GBP/CNY/CHF) ↔ USD.
  1039→      // Convert tradeUsd to native units via the FX rate.
  1040→      const fxRate =
  1041→        d.component === 'EUR' ? fx.EUR_USD :
  1042→        d.component === 'JPY' ? fx.JPY_USD :
  1043→        d.component === 'GBP' ? fx.GBP_USD :
  1044→        d.component === 'CNY' ? fx.CNY_USD :
  1045→        d.component === 'CHF' ? fx.CHF_USD : 1.0;
  1046→      const nativeUnits = tradeUsd / fxRate;
  1047→      const usdThird = tradeUsd / 3;
  1048→      if (dir === -1) {
  1049→        // sell component → USD
  1050→        // B9-FIX: track any deficit instead of silently zeroing.
  1051→        if (d.component === 'EUR') s.eurc = subtractReserve(s, s.eurc, nativeUnits, fx.EUR_USD, 'eurc');
  1052→        else if (d.component === 'JPY') s.jpy = subtractReserve(s, s.jpy, nativeUnits, fx.JPY_USD, 'jpy');
  1053→        else if (d.component === 'GBP') s.gbp = subtractReserve(s, s.gbp, nativeUnits, fx.GBP_USD, 'gbp');
  1054→        else if (d.component === 'CNY') s.cny = subtractReserve(s, s.cny, nativeUnits, fx.CNY_USD, 'cny');
  1055→        else if (d.component === 'CHF') s.chf = subtractReserve(s, s.chf, nativeUnits, fx.CHF_USD, 'chf');
  1056→        s.usdc += usdThird;
  1057→        s.usdp += usdThird;
  1058→        s.usdt += usdThird;
  1059→      } else {
  1060→        // buy component ← USD
  1061→        // B9-FIX: track any deficit instead of silently zeroing.
  1062→        s.usdc = subtractReserve(s, s.usdc, usdThird, 1, 'usdc');
  1063→        s.usdp = subtractReserve(s, s.usdp, usdThird, 1, 'usdp');
  1064→        s.usdt = subtractReserve(s, s.usdt, usdThird, 1, 'usdt');
  1065→        if (d.component === 'EUR') s.eurc += nativeUnits;
  1066→        else if (d.component === 'JPY') s.jpy += nativeUnits;
  1067→        else if (d.component === 'GBP') s.gbp += nativeUnits;
  1068→        else if (d.component === 'CNY') s.cny += nativeUnits;
  1069→        else if (d.component === 'CHF') s.chf += nativeUnits;
  1070→      }
  1071→    }
  1072→
  1073→    s.lastTradeDir = dir;
  1074→    s.lastTradeAt = Date.now();
  1075→    s.dailyTurnoverUsd += tradeUsd;
  1076→    totalTradeUsd += tradeUsd;
  1077→    appliedCount++;
  1078→    traces.push({ component: d.component, direction: d.direction, tradeUsd, level: d.level, applied: true });
  1079→  }
  1080→
  1081→  s.updatedAt = Date.now();
  1082→  return { appliedCount, skippedCount, totalTradeUsd, path: 'marp', traces };
  1083→}

  1085→// --- §11 Geopolitical Eject -------------------------------------------------
  1086→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
  1087→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
  1088→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
  1089→  const drift = (cur: number, amp: number) => {
  1090→    const shock = (Math.random() - 0.5) * 2 * amp;
  1091→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
  1092→  };
  1093→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
  1094→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
  1095→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
  1096→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
  1097→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
  1098→
  1099→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
  1100→  for (const c of currencies) {
  1101→    const p = s.pegHealth[c];
  1102→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
  1103→    if (outside) {
  1104→      s.depegHours[c] += dtHours;
  1105→    } else {
  1106→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
  1107→    }
  1108→    // Staged ladder
  1109→    const h = s.depegHours[c];
  1110→    let stage = 0;
  1111→    if (h > 48) stage = 3;
  1112→    else if (h > 24) stage = 2;
  1113→    else if (h > 12) stage = 1;
  1114→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
  1115→    s.ejectStage[c] = stage;
  1116→  }
  1117→}

  1133→export function applyMint(
  1134→  s: ReserveState,
  1135→  fx: FxRates,
  1136→  status: ProtocolStatus,
  1137→  inputUsd: number,
  1138→): MintResult {
  1139→  // P0-FIX-1: use the canonical chain-linked index for the MTQ price (the
  1140→  // legacy Laspeyres form had a structural short-gold bug — see chain-index.ts).
  1141→  const price = getMtqPriceFromState(s);
  1142→  if (!priceInSafetyBand(price)) {
  1143→    return { ok: false, reason: "Reference Value outside safety band (0.50–2.00 USD reporting); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
  1144→  }
  1145→  // P0-FIX-3: use the canonical mint throttle / minting-allowed from
  1146→  // state-machine.ts. STRESS now also pauses minting (previously only
  1147→  // DEFENSIVE/EMERGENCY did). NORMAL=1.0, CAUTION=0.5, STRESS/DEFENSIVE/EMERGENCY=0 (paused), RECOVERY=0.25.
  1148→  if (!mintingAllowed(status)) {

  1150→  }
  1151→  const throttle = mintThrottle(status);
  1152→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
  1153→  const netUsd = inputUsd - feeUsd;
  1154→  const minted = (netUsd / price) * throttle;
  1155→
  1156→  // Mutate reserve: deposit USDC (split across 3 USD issuers per concentration policy),
  1157→  // mint MTQ to user (circulating supply rises). The concentration rebalancer in the
  1158→  // tick loop will re-optimize the split if any issuer exceeds 25%.
  1159→  const usdThird = netUsd / 3;
  1160→  s.usdc += usdThird;
  1161→  s.usdp += usdThird;
  1162→  s.usdt += usdThird;
  1163→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet (§13.2 sweep target)
  1164→  s.totalSupply += minted;
  1165→  s.updatedAt = Date.now();
  1166→
  1167→  const vals = reserveAssetValues(s, fx);
  1168→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
  1169→  return {
  1170→    ok: true,
  1171→    inputUsd,
  1172→    feeUsd,
  1173→    netUsd,
  1174→    mtqPrice: price,
  1175→    mtqMinted: minted,
  1176→    throttleFactor: throttle,
  1177→    newCirculatingSupply: circulatingSupply(s),
  1178→    newReserveRatio: newRr,
  1179→  };
  1180→}

  1239→  // Genesis fallback: at genesis (or any time circulatingSupply = 0), NAV_t
  1240→  // is undefined (division by zero). We fall back to P_MTQ in this case:
  1241→  //   - At genesis, there is no circulating MTQ to redeem (the inputMtq > circ
  1242→  //     guard below catches this case explicitly).
  1243→  //   - In practice the fallback is exercised only when the protocol has not
  1244→  //     yet minted any circulating supply — the redemption would fail with
  1245→  //     "insufficient supply" before reaching the NAV computation.
  1246→  //   - After the first mint, circulating supply > 0 and NAV_t = V_net / S_circ
  1247→  //     will be very high (e.g. $42/MTQ when RR = 2400%) — this is CORRECT:
  1248→  //     redeeming 1 MTQ returns $42 worth of reserve, which IS the actual
  1249→  //     value of the token. The Index/NAV divergence analysis (§12) surfaces
  1250→  //     when this matters (e.g. when NAV_per_MTQ diverges from P_MTQ by >5%
  1251→  //     → "monitor" status; >10% → "stress").
  1252→  const navPerMtq = circ > 0 ? vals0.nav / circ : price;
  1253→  // Honest audit fields (computed pre-settlement so they exist on the error path too).
  1254→  // P0-FIX-2: the primary settlement now uses NAV (§12.2 — formerly audit-only).
  1255→  // The legacy §3.4.2 (index-priced) is now the AUDIT-ONLY alternative.
  1256→  const auditNavPerToken = navPerMtq;
  1257→  const auditGrossUsdNav = inputMtq * navPerMtq;
  1258→  const auditGrossUsdIndex = inputMtq * price;
  1259→  const auditNote =
  1260→    "P0-FIX-2 (Master §19.3.2, Invariant I6): redemption is now priced against " +
  1261→    "NAV_t = V_net / circulatingSupply (the §12.2 form, formerly audit-only). " +
  1262→    "The legacy §3.4.2 (redeem at P_MTQ) is now the AUDIT-ONLY alternative " +
  1263→    "(auditGrossUsdIndex). At RR>100% the NAV pays redeemers MORE than the index " +
  1264→    "price (auditDeltaUsd > 0), which IS the intent: the redeemer receives the " +
  1265→    "actual book value of their token, not just the index-tracked price. The " +
  1266→    "§12 Index/NAV divergence analysis surfaces when this gap becomes material " +
  1267→    "(>5% monitor, >10% stress).";
  1268→  if (inputMtq > circ) {
  1269→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)), navPerMtq, auditNavPerToken, auditGrossUsdNav, auditGrossUsdIndex, auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex, auditNote, deficitUsd: 0 };
  1270→  }
  1271→  // P0-FIX-3: §21.4 — redemption is PAUSED in EMERGENCY (the reconciliation

  1275→  }
  1276→  // P0-FIX-3: state-dependent fee from the canonical 6-state risk machine.
  1277→  // NORMAL/CAUTION 0.15% (15 bps), STRESS 0.50% (50 bps), DEFENSIVE 1.00%

  1323→  // of value and creating phantom surplus over time. Now we revert the entire

  1325→  // into `s.deficitUsd` and the RedeemResult so the books stay closed.

  1327→  const usdThirdRelease = usdRelease / 3;
  1328→  const goldHalf = goldPaxg / 2;
  1329→

  1331→  // that is short. This is recorded into s.deficitUsd and the RedeemResult so

  1333→  // is REJECTED (ok:false) — no state mutation occurs. Each of the three

  1338→    Math.max(0, usdThirdRelease - s.usdc) +
  1339→    Math.max(0, usdThirdRelease - s.usdp) +
  1340→    Math.max(0, usdThirdRelease - s.usdt); // 1:1 USD stablecoins

  1342→    Math.max(0, basket[1].nativeAmount - s.eurc) * fx.EUR_USD +

  1344→    Math.max(0, basket[3].nativeAmount - s.gbp)  * fx.GBP_USD +

  1348→    (Math.max(0, goldHalf - s.paxg) + Math.max(0, goldHalf - s.xaut)) * goldPrice;

  1369→    };
  1370→  }
  1371→  if (s.eurc < basket[1].nativeAmount || s.jpy < basket[2].nativeAmount ||

  1411→  // B9: All holdings verified sufficient above — subtract directly (no Math.max needed).
  1412→  s.usdc -= usdThirdRelease;
  1413→  s.usdp -= usdThirdRelease;
  1414→  s.usdt -= usdThirdRelease;

  1757→// --- §6/§7/§8.4/§10 v1.0 MASE + 4-state weights + MARP ---------------------
  1758→// The Multi-Asset Stochastic Ensemble (§6/§7) runs 6 candidate models, blends
  1759→// them into a single target weight vector, clamps to per-component
  1760→// admissibility envelopes (§8.1), and EMA-smooths toward the constrained target
  1761→// (§8.4). MARP (§10) then decides per-component whether the deviation between
  1762→// the smoothed target and the observed execution weight justifies a trade
  1763→// (urgency test → no-trade zone → cost-benefit gate → partial correction →
  1764→// daily-turnover cap → execute).
  1765→//
  1766→// `advanceMase(s, fx)` is called ONCE per tick from the pilot-state tick loop.
  1767→// It is the only function that mutates `s.maseSmoothed`. `buildMaseSnapshot()`
  1768→// is a pure helper used by `computeSnapshot()` — it recomputes target/smoothed/
  1769→// MARP from the current fx + vals using the persisted prev-smoothed as the EMA
  1770→// prior, so multiple computeSnapshot calls within a single tick return the
  1771→// same value (idempotent).
  1772→
  1773→interface MaseSnapshotData {
  1774→  mase: NonNullable<MetricsSnapshot["mase"]>;
  1775→  weightStates: NonNullable<MetricsSnapshot["weightStates"]>;
  1776→  marp: NonNullable<MetricsSnapshot["marp"]>;
  1777→  // §14.1 — read-only projection of what MARP WOULD do this tick.
  1778→  marpExecution: NonNullable<MetricsSnapshot["marpExecution"]>;
  1779→  envelopes: MetricsSnapshot["envelopes"];
  1780→}

  1782→/** Estimate per-component annualized volatility from the engine's macro state.
  1783→ *  Pilot approximation: scale baseline vols by VIX / 18.5 (the "normal" VIX
  1784→ *  baseline), so a 30 VIX roughly doubles estimated vols and a 12 VIX halves
  1785→ *  them. Production should use a rolling covariance from real price history. */
  1786→function estimateVolsFromState(s: ReserveState): VolatilityData {
  1787→  const vix = s.lastVix > 0 ? s.lastVix : 18.5;
  1788→  const f = vix / 18.5;
  1789→  return {
  1790→    USD: 0.05 * f,
  1791→    EUR: 0.10 * f,
  1792→    JPY: 0.12 * f,
  1793→    GBP: 0.11 * f,
  1794→    CNY: 0.09 * f,
  1795→    CHF: 0.10 * f,
  1796→    Gold: 0.15 * f,
  1797→  };
  1798→}
  1799→
  1800→/** Detect the market regime from VIX (primary) + DXY (secondary).
  1801→ *  0 = calm, 1 = normal, 2 = stress, 3 = crisis.
  1802→ *  VIX thresholds: <15 calm, <22 normal, <30 stress, ≥30 crisis.
  1803→ *  DXY extremes (<88 or >115) bump the regime up to ≥ stress. */
  1804→function detectRegime(vix: number, dxy: number, goldVol: number): MarketRegime {
  1805→  let r: 0 | 1 | 2 | 3 = 1;
  1806→  if (vix >= 30) r = 3;
  1807→  else if (vix >= 22) r = 2;
  1808→  else if (vix >= 15) r = 1;
  1809→  else r = 0;
  1810→  if ((dxy >= 115 || dxy <= 88) && r < 2) r = 2;
  1811→  return { regime: r, vix, dxy, goldVol };
  1812→}

  1814→/** Build PriceData (P_{i,t} / P_{i,0}) from current FX rates + base fixings. */
  1815→function buildPriceData(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): PriceData {
  1816→  return {
  1817→    USD: 1.0,
  1818→    EUR: fx.EUR_USD / BASE_FIXINGS.EUR_USD,
  1819→    JPY: fx.JPY_USD / BASE_FIXINGS.JPY_USD,
  1820→    GBP: fx.GBP_USD / BASE_FIXINGS.GBP_USD,
  1821→    CNY: fx.CNY_USD / BASE_FIXINGS.CNY_USD,
  1822→    CHF: fx.CHF_USD / BASE_FIXINGS.CHF_USD,
  1823→    Gold: fx.XAU_USD / BASE_FIXINGS.XAU_USD,
  1824→  };
  1825→}
  1826→
  1827→/** Build the observed execution-weight vector from the actual reserve composition.
  1828→ *  §14.1 note: the Gold component's observed weight uses the TOTAL gold
  1829→ *  (= indexPaxg + indexXaut + reservePaxg + reserveXaut, USD-net). Both the
  1830→ *  index gold (locked) and the reserve gold (MARP-rebalanced) serve the
  1831→ *  Strategic Prior Gold weight (26%) — but only the RESERVE gold is available
  1832→ *  for MARP to rebalance. The index gold is locked and only changes when
  1833→ *  commitIndexGold() is called (keeper-role equivalent). */
  1834→function buildObservedWeights(
  1835→  vals: ReturnType<typeof reserveAssetValues>,
  1836→  nav: number,
  1837→): WeightVector {
  1838→  if (nav <= 0) {
  1839→    return { USD: 0, EUR: 0, JPY: 0, GBP: 0, CNY: 0, CHF: 0, Gold: 0 };
  1840→  }
  1841→  return {
  1842→    USD: vals.usdNet / nav,
  1843→    EUR: vals.eurNet / nav,
  1844→    JPY: vals.jpyNet / nav,
  1845→    GBP: vals.gbpNet / nav,
  1846→    CNY: vals.cnyNet / nav,
  1847→    CHF: vals.chfNet / nav,
  1848→    // §14.1 — gold weight = TOTAL gold (index + reserve) / NAV. Both pools
  1849→    // back the 26% Strategic Prior Gold weight; only reserve is MARP-rebalanced.
  1850→    Gold: vals.goldNet / nav,
  1851→  };
  1852→}

  1854→/** Pure helper — computes the MASE + 4-state + MARP + envelope snapshot data.
  1855→ *  Reads `s.maseSmoothed` (does NOT mutate). */
  1856→function buildMaseSnapshot(
  1857→  s: ReserveState,
  1858→  fx: FxSnapshot,
  1859→  vals: ReturnType<typeof reserveAssetValues>,
  1860→  nav: number,
  1861→  rr: number,
  1862→): MaseSnapshotData {
  1863→  const prices = buildPriceData(fx);
  1864→  const vols = estimateVolsFromState(s);
  1865→  const regime = detectRegime(s.lastVix > 0 ? s.lastVix : fx.VIX, s.lastDxy > 0 ? s.lastDxy : fx.DXY, vols.Gold);
  1866→
  1867→  // §6/§7 — MASE ensemble → 6 candidate models + target weight vector
  1868→  const mase = maseEnsemble(vols, prices, null, regime);
  1869→  // §8.1 — Constrain target to per-component admissibility envelopes
  1870→  const constrained = applyEnvelopes(mase.target);
  1871→  // §8.4 — EMA-smooth toward the constrained target (uses prev persisted as prior)
  1872→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1873→  const smoothed = smoothWeights(prevSmoothed, constrained);
  1874→  // §2.3 — Execution = actual observed reserve composition
  1875→  const observed = buildObservedWeights(vals, nav);
  1876→
  1877→  // §10 — MARP per-component rebalancing decision
  1878→  const volsRecord: Record<Component, number> = {
  1879→    USD: vols.USD, EUR: vols.EUR, JPY: vols.JPY, GBP: vols.GBP,
  1880→    CNY: vols.CNY, CHF: vols.CHF, Gold: vols.Gold,
  1881→  };
  1882→  const marpDecisions: MarpDecision[] = marpDecision(observed, smoothed, nav, rr, volsRecord);
  1883→  const totalTradeUsd = marpDecisions
  1884→    .filter((d) => d.shouldTrade)
  1885→    .reduce((a, d) => a + d.tradeUsd, 0);
  1886→
  1887→  // §14.1 — per-component MARP execution summary (READ-ONLY projection).
  1888→  // appliedCount = # of decisions with shouldTrade && level >= 6 (would-execute).
  1889→  // skippedCount  = # of decisions held back (no-trade zone / low urgency /
  1890→  //                 cost-benefit fail / turnover cap / sub-level-6).
  1891→  // totalTradeUsd = Σ tradeUsd for the would-execute decisions.
  1892→  // path = which execution path is currently mutating state (mirrors s.rebalancePath).
  1893→  // NOTE: this is what MARP WOULD do based on the current decisions. The actual
  1894→  // mutation happens in applyMarpRebalance() only when USE_MARP_EXECUTION=true
  1895→  // in pilot-state.ts. applyMarpRebalance may further skip decisions that
  1896→  // breach MAX_DAILY_TURNOVER or DIRECTION_LOCK_HOURS at execution time.
  1897→  const wouldExecute = marpDecisions.filter((d) => d.shouldTrade && d.level >= 6);
  1898→  const marpExecution = {
  1899→    appliedCount: wouldExecute.length,
  1900→    skippedCount: marpDecisions.length - wouldExecute.length,
  1901→    totalTradeUsd: wouldExecute.reduce((a, d) => a + d.tradeUsd, 0),
  1902→    path: s.rebalancePath,
  1903→  };
  1904→
  1905→  // §8.1 — Admissibility envelope status per component
  1906→  // "warn" = within 10% of the band edge; "breach" = outside the envelope.
  1907→  const envelopes = COMPONENTS.map((c) => {
  1908→    const env = ADMISSIBILITY_ENVELOPES[c];
  1909→    const current = observed[c];
  1910→    const lower = env.lower;
  1911→    const upper = env.upper;
  1912→    const margin = (upper - lower) * 0.1;
  1913→    let status: "ok" | "warn" | "breach" = "ok";
  1914→    if (current < lower || current > upper) status = "breach";
  1915→    else if (current < lower + margin || current > upper - margin) status = "warn";
  1916→    return { component: c, lower, upper, current, status };
  1917→  });
  1918→
  1919→  return {
  1920→    mase: {
  1921→      models: mase.models.map((m) => ({
  1922→        id: m.id,
  1923→        name: m.name,
  1924→        weights: m.weights as Record<string, number>,
  1925→      })),
  1926→      ensembleTarget: mase.target as Record<string, number>,
  1927→    },
  1928→    weightStates: {
  1929→      prior: STRATEGIC_PRIOR as Record<string, number>,
  1930→      target: constrained as Record<string, number>,
  1931→      smoothed: smoothed as Record<string, number>,
  1932→      execution: observed as Record<string, number>,
  1933→    },
  1934→    marp: {
  1935→      decisions: marpDecisions.map((d) => ({
  1936→        shouldTrade: d.shouldTrade,
  1937→        component: d.component,
  1938→        direction: d.direction,
  1939→        tradeUsd: d.tradeUsd,
  1940→        urgency: d.urgency,
  1941→        reason: d.reason,
  1942→        level: d.level,
  1943→      })),
  1944→      totalTradeUsd,
  1945→    },
  1946→    // §14.1 — MARP execution summary (read-only projection; mirrors s.rebalancePath).
  1947→    marpExecution,
  1948→    envelopes,
  1949→  };
  1950→}
  1951→
  1952→/** Advance MASE state once per tick. Mutates `s.maseSmoothed` (EMA prior for
  1953→ *  the next tick) and `s.chainIndex` (commits the new weights to preserve
  1954→ *  zero-artificial-return continuity per Master §9.2 COO-16). Called from
  1955→ *  the pilot-state tick loop, AFTER advanceMacro (so lastVix/lastDxy are
  1956→ *  fresh) and BEFORE the first computeSnapshot of the tick (so the snapshot
  1957→ *  reads the freshly-persisted smoothed weights).
  1958→ *
  1959→ *  P0-FIX-1 (chain-linking): the new smoothed weights are committed to the
  1960→ *    chain index via commitChainIndexWeights(). This updates G_t (the
  1961→ *    cumulative chain-link factor) and the prevWeights / prevPrices used by
  1962→ *    the next advanceIndex() call. The divisor D_t = B_t^- / B_t^+ ensures
  1963→ *    the rebalance creates ZERO artificial index return.
  1964→ *
  1965→ *  P0-FIX-3 + §8.4/§7.4 (velocity + stress-adaptive smoothing): the EMA
  1966→ *    smoothing now uses smoothWeightsAdaptive() with the canonical risk
  1967→ *    state's stress level and the target velocity (max component-wise |Δ|
  1968→ *    between the previous smoothed and the new constrained target). In
  1969→ *    stress states (STRESS/DEFENSIVE/EMERGENCY), the smoothing slows by 50%
  1970→ *    to avoid chasing volatile prices. Fast target jumps (|Δ| > 5%) are
  1971→ *    damped by up to 50% (whip-saw guard). */
  1972→export function advanceMase(s: ReserveState, fx: FxRates): void {
  1973→  const prices = buildPriceData(fx);
  1974→  const vols = estimateVolsFromState(s);
  1975→  const regime = detectRegime(
  1976→    s.lastVix > 0 ? s.lastVix : fx.VIX,
  1977→    s.lastDxy > 0 ? s.lastDxy : fx.DXY,
  1978→    vols.Gold,
  1979→  );
  1980→  const mase = maseEnsemble(vols, prices, null, regime);
  1981→  const constrained = applyEnvelopes(mase.target);
  1982→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1983→  // §8.4 + §7.4 — velocity + stress-adaptive smoothing.
  1984→  const velocity = targetVelocity(prevSmoothed, constrained);
  1985→  const sl = stressLevel(s.riskState.state);
  1986→  s.maseSmoothed = smoothWeightsAdaptive(prevSmoothed, constrained, SMOOTHING_LAMBDA, velocity, sl);
  1987→  s.maseLastAt = Date.now();
  1988→  // P0-FIX-1 — commit the new smoothed weights to the chain index. This
