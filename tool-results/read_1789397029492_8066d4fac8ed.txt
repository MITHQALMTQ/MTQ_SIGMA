   380→   380→    //      I_0 = Σ_i Q_i × P_i,0
   381→   381→    //  where the numéraire USD has P_USD,0 = 1.0 and the 6 FX/gold base fixings
   382→   382→    //  are defined at §3.4 (all 1e18 scale). The result is ~650.64e18 — gold
   383→   383→    //  dominates the basket (Q_GOLD × P_GOLD,0 = 0.26 × 2500 = 650).
   384→   384→    //
   385→   385→    //  NOTE on test compatibility (CONTRACT-CTOR task): the test file's header
   386→   386→    //  note (P1/P2) suggests `INDEX_BASE_DENOMINATOR = 1e18` to keep the
   387→   387→    //  initial MTQ price at exactly 1.0 USD/MTQ inside the §3.5 safety band
   388→   388→    //  [0.50, 2.00]. The Master Listing 1 formula implemented here instead
   389→   389→    //  values the basket at its base USD notional (~$650.64), which makes the
   390→   390→    //  initial price ~0.00154 USD/MTQ — outside the §3.5 band. The protocol
   391→   391→    //  owner has two clean options to fully reconcile this with the existing
   392→   392→    //  Foundry test suite (MTQSigmaV2.t.sol):
   393→   393→    //    (a) widen PRICE_SAFETY_LOWER/UPPER (e.g. to 0.0005 / 0.005) — a one-
   394→   394→    //        line change consistent with the Listing 1 basket valuation; or
   395→   395→    //    (b) override INDEX_BASE_DENOMINATOR to 1e18 in the constructor
   396→   396→    //        (keeps the safety band at 0.50–2.00 USD/MTQ; drops the Listing 1
   397→   397→    //        basket valuation).
   398→   398→    //  This constructor implements the Listing 1 formula verbatim per the
   399→   399→    //  CONTRACT-CTOR task brief; the safety band adjustment is left to the
   400→   400→    //  protocol owner as an explicit downstream decision (NOT edited here
   401→   401→    //  per the task's "add/fix the constructor only" scope).

   403→   403→    constructor(address _usdc) {
   404→   404→        // 1. Collateral token (USDC, 6 decimals).
   405→   405→        usdc = IERC20(_usdc);
   406→   406→

   408→   408→        //    USD term: Q_USD × 1.0 (USD is the numéraire; P_USD,0 = 1e18 / 1e18).
   409→   409→        //    All other terms: (Q_i × BASE_i_USD) / 1e18 to keep the result in
   410→   410→        //    1e18 scale. Result ≈ 650.64e18 (gold dominates at 0.26 × 2500 = 650).

   415→   415→                                  (Q_CNY  * BASE_CNY_USD)  / 1e18 +
   416→   416→                                  (Q_CHF  * BASE_CHF_USD)  / 1e18 +
   417→   417→                                  (Q_GOLD * BASE_GOLD_USD) / 1e18;
   418→   418→

   449→   449→    /// @notice Listing 2 (§7.7) submitTargetWeights — MASE verification + smoothing.
   450→   450→    /// @dev   Only the MASE submitter (KEEPER_ROLE). Enforces sum=1, positivity,
   451→   451→    ///        admissibility envelopes, per-component velocity, then applies
   452→   452→    ///        stress-adaptive smoothing rho 0.50 (normal) / 0.75 (stress).
   453→   453→    function submitTargetWeights(
   454→   454→        uint256[7] calldata target,
   455→   455→        bytes32 methodologyVersion,
   456→   456→        bytes32 dataVersion
   457→   457→    ) external onlyKeeper {
   458→   458→        // 1. Sum-to-one and non-negativity
   459→   459→        uint256 sum = 0;
   460→   460→        for (uint256 i = 0; i < 7; i++) {
   461→   461→            if (!(target[i] > 0)) revert Err62();
   462→   462→            sum += target[i];
   463→   463→        }
   464→   464→        if (!(sum == 1e18)) revert Err54();
   465→   465→        // 2. Constitutional admissibility envelopes
   466→   466→        for (uint256 i = 0; i < 7; i++) {
   467→   467→            if (!(target[i] >= LOWER_BOUND[i])) revert Err12();
   468→   468→            if (!(target[i] <= UPPER_BOUND[i])) revert Err06();
   469→   469→        }
   470→   470→        // 3. Weight-velocity constraint vs previous live weights
   471→   471→        for (uint256 i = 0; i < 7; i++) {
   472→   472→            uint256 diff = target[i] > liveWeights.weights[i]
   473→   473→                ? target[i] - liveWeights.weights[i]
   474→   474→                : liveWeights.weights[i] - target[i];
   475→   475→            if (!(diff <= MAX_VELOCITY[i])) revert Err52();
   476→   476→        }
   477→   477→        // 4. Stress-adaptive smoothing: W_smooth = rho * W_prev + (1 - rho) * W_target
   478→   478→        uint256 rho = crisisFlag ? smoothingRhoStress : smoothingRhoNormal;
   479→   479→        uint256[7] memory smoothed;
   480→   480→        for (uint256 i = 0; i < 7; i++) {
   481→   481→            smoothed[i] = (rho * liveWeights.weights[i] + (1e18 - rho) * target[i]) / 1e18;
   482→   482→        }
   483→   483→        // 5. Commit registry state
   484→   484→        for (uint256 i = 0; i < 7; i++) liveWeights.weights[i] = smoothed[i];
   485→   485→        liveWeights.targetWeights    = target;
   486→   486→        liveWeights.methodologyVersion = methodologyVersion;
   487→   487→        liveWeights.dataVersion       = dataVersion;
   488→   488→        liveWeights.updatedAt         = block.timestamp;
   489→   489→        emit WeightsAccepted(smoothed, target, methodologyVersion, dataVersion, block.timestamp);
   490→   490→    }

   525→   525→    /// @dev §9.8 returns the base-fixing price for component i (1e18 scale).
   526→   526→    ///      USD → 1.0; EUR → 1.05; JPY → 0.0067; GBP → 1.25;
   527→   527→    ///      CNY → 0.14; CHF → 1.13 (FIXED); Gold → 2500.
   528→   528→    function _basePrice18(uint256 i) internal pure returns (uint256) {
   529→   529→        if (i == 0) return 1e18;             // USD = 1 USD
   530→   530→        if (i == 1) return BASE_EUR_USD;     // 1.05
   531→   531→        if (i == 2) return BASE_JPY_USD;     // 0.0067
   532→   532→        if (i == 3) return BASE_GBP_USD;     // 1.25
   533→   533→        if (i == 4) return BASE_CNY_USD;     // 0.14
   534→   534→        if (i == 5) return BASE_CHF_USD;     // 1.13
   535→   535→        return BASE_GOLD_USD;                 // 2500

   538→   538→    /// @notice Listing 3 advanceIndex — §9.2 authoritative recursion.
   539→   539→    ///         I_t = I_{t-1} × Σ_i W_{i,t-1} × (P_{i,t} / P_{i,t-1})
   540→   540→    ///         Prices are passed in by the keeper (1e18 scale, USD-quoted).
   541→   541→    function advanceIndex(uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   542→   542→        if (!(indexValue > 0)) revert Err20();
   543→   543→
   544→   544→        // 1. Period return weighted by W_{i,t-1} in force over the period (COO-16).
   545→   545→        uint256 growth = 0;
   546→   546→        for (uint256 i = 0; i < 7; i++) {
   547→   547→            if (!(currentPrices[i] > 0 && lastPrices[i] > 0)) revert Err09();
   548→   548→            growth += (lastWeights[i] * currentPrices[i]) / lastPrices[i];
   549→   549→        }
   550→   550→        indexValue = (indexValue * growth) / 1e18;
   551→   551→
   552→   552→        // 2. Commit the period end-state (prices).
   553→   553→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   554→   554→        lastIndexUpdate = block.timestamp;
   555→   555→        emit IndexAdvanced(indexValue, block.timestamp);
   556→   556→    }

   558→   558→    /// @notice Listing 3 commitWeights — chain-link adjustment at weight change.
   559→   559→    ///         Computes D_t = B_t^- / B_t^+ (Master §9.3), updates G_t, updates
   560→   560→    ///         lastPrices and lastWeights. I_t is UNCHANGED by the weight change
   561→   561→    ///         itself — ZERO artificial return (the recursion is continuous).
   562→   562→    function commitWeights(uint256[7] calldata newWeights, uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   563→   563→        if (!(indexValue > 0)) revert Err20();
   564→   564→
   565→   565→        // 1. B_t^- and B_t^+ are base-relative aggregates measured against
   566→   566→        //    the genesis fixings (§9.3, MS §75).
   567→   567→        uint256 bMinus = 0;
   568→   568→        uint256 bPlus  = 0;
   569→   569→        bool    changed = false;
   570→   570→        for (uint256 i = 0; i < 7; i++) {
   571→   571→            if (!(currentPrices[i] > 0)) revert Err09();
   572→   572→            uint256 rel = (currentPrices[i] * 1e18) / _basePrice18(i); // P_{i,t} / P_{i,0}
   573→   573→            bMinus += (lastWeights[i] * rel) / 1e18;
   574→   574→            bPlus  += (newWeights[i]    * rel) / 1e18;
   575→   575→            if (newWeights[i] != lastWeights[i]) changed = true;
   576→   576→        }
   577→   577→
   578→   578→        // 2. Apply divisor D_t; record into G_t (cumulative). I_t unchanged.
   579→   579→        if (changed) {
   580→   580→            if (!(bPlus > 0)) revert Err57();
   581→   581→            uint256 newD = (bMinus * 1e18) / bPlus;
   582→   582→            chainLinkDivisor = (chainLinkDivisor * newD) / 1e18;
   583→   583→            for (uint256 i = 0; i < 7; i++) lastWeights[i] = newWeights[i];
   584→   584→            emit ChainLinkAdjusted(newD, bMinus, bPlus, block.timestamp);
   585→   585→        }
   586→   586→
   587→   587→        // 3. Commit the period end-state prices too.
   588→   588→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   589→   589→        emit WeightsCommitted(newWeights, chainLinkDivisor, block.timestamp);
   590→   590→    }

   592→   592→    /// @notice Listing 3 genesis — one-time seed of I_0 = 1.0000 (ADMIN_ROLE).
   593→   593→    ///         Prices are seeded from immutable base fixings, so I_0 = 1.0000
   594→   594→    ///         exactly independent of feed noise.
   595→   595→    function genesisIndex() external onlyAdmin {
   596→   596→        if (!(indexValue == 1e18 && lastIndexUpdate == 0)) revert Err19();
   597→   597→        if (!(liveWeights.updatedAt > 0)) revert Err01();
   598→   598→        // Seed lastWeights from MASE registry
   599→   599→        uint256 sum = 0;
   600→   600→        for (uint256 i = 0; i < 7; i++) {
   601→   601→            lastWeights[i] = liveWeights.weights[i];
   602→   602→            lastPrices[i]  = _basePrice18(i);
   603→   603→            sum += liveWeights.weights[i];
   604→   604→        }
   605→   605→        if (!(sum == 1e18)) revert Err44();
   606→   606→        indexValue = 1e18;
   607→   607→        chainLinkDivisor = 1e18;
   608→   608→        lastIndexUpdate = block.timestamp;
   609→   609→        emit GenesisVerified(INDEX_BASE_DENOMINATOR, block.timestamp);
   610→   610→    }

   645→   645→    /// @notice Listing 13 updateState — the CANONICAL state function (§14.6.2).
   646→   646→    ///         Worse applicable condition binds; RECOVERY requires 48h
   647→   647→    ///         confirmation; transitions to a more restrictive state are
   648→   648→    ///         immediate, transitions to a less restrictive state require
   649→   649→    ///         the RECOVERY_CONFIRMATION_PERIOD to elapse.
   650→   650→    function updateState(uint256 rr, uint256 lcr) external onlyKeeper {
   651→   651→        ProtocolState newState;
   652→   652→        // Determine the target state based on RR and LCR
   653→   653→        if (rr >= rrTarget && lcr >= lcrTarget) {
   654→   654→            newState = ProtocolState.NORMAL;
   655→   655→        } else if (rr >= rrStressFloor && lcr >= 0.90e18) {
   656→   656→            newState = ProtocolState.CAUTION;
   657→   657→        } else if (rr >= 1.02e18 && lcr >= 0.80e18) {
   658→   658→            newState = ProtocolState.STRESS;
   659→   659→        } else if (rr >= RR_HARD_FLOOR && lcr >= 0.70e18) {
   660→   660→            newState = ProtocolState.DEFENSIVE;
   661→   661→        } else if (rr < RR_HARD_FLOOR) {
   662→   662→            newState = ProtocolState.EMERGENCY;
   663→   663→        } else {
   664→   664→            // RECOVERY: solvency ≥ hard floor but ratios not yet back to NORMAL.
   665→   665→            newState = ProtocolState.RECOVERY;
   666→   666→        }
   667→   667→
   668→   668→        if (newState != currentState) {
   669→   669→            if (_isMoreRestrictive(newState, currentState)) {
   670→   670→                _transitionTo(newState, true);
   671→   671→            } else {
   672→   672→                // Less restrictive → 48h confirmation
   673→   673→                if (block.timestamp - stateEntryTime[currentState] >= RECOVERY_CONFIRMATION_PERIOD) {
   674→   674→                    _transitionTo(newState, false);
   675→   675→                }
   676→   676→                // else: stay (newState discarded — not yet confirmed)
   677→   677→            }
   678→   678→        }
   679→   679→    }

   685→   685→        emit StateChanged(oldState, newState, immediate, block.timestamp);

   726→   726→        if (s == ProtocolState.EMERGENCY) return 0.02e18;                 // 2.00% (but paused)

   808→   808→    /// @notice Listing 14 proposeChange — only the authorized caller per layer:
   809→   809→    ///         DAO for MONETARY, riskCouncil for RISK, emergencyCouncil for
   810→   810→    ///         EMERGENCY, constitutionalCouncil for CONSTITUTIONAL.
   811→   811→    function proposeChange(bytes32 parameterId, uint256 newValue) external {
   812→   812→        ParameterRecord storage rec = parameterRegistry[parameterId];
   813→   813→        if (!(rec.registered)) revert Err50();
   814→   814→        if (!(newValue >= rec.minValue)) revert Err11();
   815→   815→        if (!(newValue <= rec.maxValue)) revert Err05();
   816→   816→        if (rec.layer == ParameterLayer.MONETARY)         if (!(msg.sender == dao)) revert Err36();
   817→   817→        else if (rec.layer == ParameterLayer.RISK)        if (!(msg.sender == riskCouncil)) revert Err38();
   818→   818→        else if (rec.layer == ParameterLayer.EMERGENCY)   if (!(msg.sender == emergencyCouncil)) revert Err37();
   819→   819→        else /* CONSTITUTIONAL */                          if (!(msg.sender == constitutionalCouncil)) revert Err35();
   820→   820→        proposals[parameterId] = Proposal({
   821→   821→            value: newValue, proposer: msg.sender, timestamp: block.timestamp, executed: false
   822→   822→        });
   823→   823→        emit ParameterProposed(parameterId, newValue, block.timestamp);
   824→   824→    }

   826→   826→    /// @notice Listing 14 executeChange — anyone can call after the timelock
   827→   827→    ///         elapses; the timelock + role check on propose are the protection.
   828→   828→    function executeChange(bytes32 parameterId) external {
   829→   829→        Proposal storage p = proposals[parameterId];
   830→   830→        if (!(p.timestamp > 0)) revert Err27();
   831→   831→        if (!(!p.executed)) revert Err07();
   832→   832→        ParameterRecord storage rec = parameterRegistry[parameterId];
   833→   833→        uint256 delay;
   834→   834→        if (rec.layer == ParameterLayer.MONETARY)         delay = TIMELOCK_MONETARY;
   835→   835→        else if (rec.layer == ParameterLayer.RISK)        delay = TIMELOCK_RISK;
   836→   836→        else if (rec.layer == ParameterLayer.EMERGENCY)  delay = TIMELOCK_EMERGENCY;
   837→   837→        else                                              delay = TIMELOCK_CONSTITUTIONAL;
   838→   838→        if (!(block.timestamp >= p.timestamp + delay)) revert Err47();
   839→   839→        if (!(p.value >= rec.minValue && p.value <= rec.maxValue)) revert Err40();
   840→   840→        p.executed = true;
   841→   841→        rec.value = p.value;
   842→   842→        _applyParam(parameterId, p.value);
   843→   843→        emit ParameterExecuted(parameterId, p.value, block.timestamp);
   844→   844→    }

   846→   846→    /// @notice Listing 14 cancelChange — only the proposer role (within timelock).
   847→   847→    function cancelChange(bytes32 parameterId) external {
   848→   848→        Proposal storage p = proposals[parameterId];
   849→   849→        if (!(p.timestamp > 0 && !p.executed)) revert Err34();
   850→   850→        ParameterRecord storage rec = parameterRegistry[parameterId];
   851→   851→        if (rec.layer == ParameterLayer.MONETARY)        if (!(msg.sender == dao)) revert Err36();
   852→   852→        else if (rec.layer == ParameterLayer.RISK)       if (!(msg.sender == riskCouncil)) revert Err38();
   853→   853→        else if (rec.layer == ParameterLayer.EMERGENCY)  if (!(msg.sender == emergencyCouncil)) revert Err37();
   854→   854→        else                                             if (!(msg.sender == constitutionalCouncil)) revert Err35();
   855→   855→        delete proposals[parameterId];
   856→   856→        emit ParameterVetoed(parameterId, block.timestamp);
   857→   857→    }

   903→   903→    /// @notice §9.1-9.3 Consensus across all 3 adapters for a single pair.
   904→   904→    /// @return finalPrice  USD-per-unit, 1e18 scale (0 if paused)
   905→   905→    /// @return validCount  Number of valid feeds after all filters
   906→   906→    /// @return method      0=paused, 1=average(2), 2=median(3)
   907→   907→    /// @return paused_     True if fewer than 2 feeds are valid
   908→   908→    /// @return prices      Raw prices per adapter (1e18) — index 0/1/2 = CL/Pyth/Cro
   909→   909→    /// @return valid       Per-adapter validity flag after all filters
   910→   910→    function getOracleConsensus(bytes32 pair) public view returns (
   911→   911→        uint256 finalPrice,
   912→   912→        uint8  validCount,
   913→   913→        uint8  method,
   914→   914→        bool   paused_,
   915→   915→        uint256[3] memory prices,
   916→   916→        bool[3] memory valid
   917→   917→    ) {
   918→   918→        IOracleAdapter[3] memory adapters;
   919→   919→        adapters[0] = chainlinkAdapter;
   920→   920→        adapters[1] = pythAdapter;
   921→   921→        adapters[2] = chronicleAdapter;
   922→   922→
   923→   923→        uint256[3] memory raw;
   924→   924→        uint256[3] memory ts;
   925→   925→        uint256[3] memory conf;
   926→   926→
   927→   927→        for (uint8 i = 0; i < 3; i++) {
   928→   928→            valid[i] = false;
   929→   929→            if (address(adapters[i]) == address(0)) continue;
   930→   930→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
   931→   931→                if (p == 0) continue; // C6: reject zero-price feeds
   932→   932→                raw[i]  = p;
   933→   933→                ts[i]   = t;
   934→   934→                conf[i] = c;
   935→   935→                prices[i] = p;
   936→   936→                // §9.2.1 Staleness ≤60s (also reject future timestamps)
   937→   937→                if (t > block.timestamp) continue;
   938→   938→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
   939→   939→                // §9.2.3 Confidence < 1% of price
   940→   940→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
   941→   941→                valid[i] = true;
   942→   942→            } catch { /* leave valid[i] = false */ }
   943→   943→        }
   944→   944→
   945→   945→        validCount = 0;
   946→   946→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
   947→   947→
   948→   948→        if (validCount < 2) {
   949→   949→            paused_ = true; method = 0;
   950→   950→            return (finalPrice, validCount, method, paused_, prices, valid);
   951→   951→        }
   952→   952→
   953→   953→        // Sort the valid prices and pick median (n=3) or average (n=2)
   954→   954→        uint256[3] memory sorted;
   955→   955→        uint8 n = 0;
   956→   956→        for (uint8 i = 0; i < 3; i++) if (valid[i]) { sorted[n] = raw[i]; n++; }
   957→   957→        _insertionSort(sorted, n);
   958→   958→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
   959→   959→
   960→   960→        // §9.2.4 Deviation < 2.5% from median → discard
   961→   961→        for (uint8 i = 0; i < 3; i++) {
   962→   962→            if (valid[i]) {
   963→   963→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
   964→   964→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
   965→   965→                    valid[i] = false;
   966→   966→                    validCount--;
   967→   967→                }
   968→   968→            }
   969→   969→        }
   970→   970→
   971→   971→        if (validCount < 2) {
   972→   972→            paused_ = true; method = 0;
   973→   973→            return (finalPrice, validCount, method, paused_, prices, valid);
   974→   974→        }
   975→   975→
   976→   976→        paused_ = false;
   977→   977→        if (validCount == 3) {
   978→   978→            finalPrice = med;
   979→   979→            method     = 2; // median(3)
   980→   980→        } else {
   981→   981→            uint256 sum = 0; uint8 cnt = 0;
   982→   982→            for (uint8 i = 0; i < 3; i++) if (valid[i]) { sum += raw[i]; cnt++; }
   983→   983→            finalPrice = sum / cnt;
   984→   984→            method     = 1; // average(2)
   985→   985→        }
   986→   986→    }

  1088→  1088→        uint256 price = getMTQPrice();
  1089→  1089→        if (!(price >= PRICE_SAFETY_LOWER && price <= PRICE_SAFETY_UPPER)) revert Err42();
  1090→  1090→        return price;
  1091→  1091→    }

  1093→  1093→    /// @dev §3.2 Liability (USD, 1e18) = circulatingSupply × P_MTQ / 1e18.

  1098→  1098→    /// @dev §3.2 + §14.2 Reserve ratio = NAV / liability (1e18 scale).
  1099→  1099→    ///      ≥ 1.10 = healthy (RR_TARGET); ≥ 1.00 = solvent (I2 hard floor).
  1100→  1100→    function getReserveRatio() public view returns (uint256) {
  1101→  1101→        uint256 nav  = getNAV();
  1102→  1102→        uint256 liab = getLiability();
  1103→  1103→        if (liab == 0) return type(uint256).max;
  1104→  1104→        return nav * 1e18 / liab;
  1105→  1105→    }

  1107→  1107→    /// @dev §3.2 Liquidity Coverage Ratio (LCR) = liquidAssets / stressDemand.
  1108→  1108→    ///      stressDemand = 25% of circulating liability (per Listing 1).
  1109→  1109→    function getLCR() public view returns (uint256) {
  1110→  1110→        uint256 liquidAssets = reserveState.liquidValue;
  1111→  1111→        uint256 price = getMTQPrice();
  1112→  1112→        uint256 circSupply = getCirculatingSupply();
  1113→  1113→        if (circSupply == 0) return type(uint256).max;
  1114→  1114→        uint256 stressDemand = (circSupply * price / 1e18) * 25 / 100;
  1115→  1115→        if (stressDemand == 0) return type(uint256).max;
  1116→  1116→        return (liquidAssets * 1e18) / stressDemand;
  1117→  1117→    }

  1119→  1119→    /// @dev §19.3.2 NAV per MTQ token = NAV / circulatingSupply.
  1120→  1120→    function getNAVperToken() public view returns (uint256) {
  1121→  1121→        uint256 circ = getCirculatingSupply();
  1122→  1122→        if (circ == 0) return type(uint256).max;
  1123→  1123→        return getNAV() * 1e18 / circ;
  1124→  1124→    }

  1127→  1127→    // §3.4.2 + §12.1  Mint — priced against P_MTQ (I5) with canonical throttle
  1128→  1128→    //   Pull USDC (6 dec), fee = mintFee, MTQ minted = netUsd18 × 1e18 / P_MTQ,
  1129→  1129→    //   throttled by mintThrottle(currentState). REVERT if !mintingAllowed.
  1130→  1130→    // ============================================================
  1131→  1131→    function mint(uint256 usdcAmount) external nonReentrant whenNotPaused returns (uint256 minted) {
  1132→  1132→        if (!(usdcAmount > 0)) revert Err56();
  1133→  1133→        if (!(mintingAllowed(currentState))) revert Err25();
  1134→  1134→
  1135→  1135→        uint256 price = getMTQPriceWithGuard();
  1136→  1136→
  1137→  1137→        // Pull USDC from minter to reserve vault (6 decimals).
  1138→  1138→        if (!(usdc.transferFrom(msg.sender, reserveVault, usdcAmount))) revert Err03();
  1139→  1139→
  1140→  1140→        // §12.1 fee = mintFee (1e18 fraction). E.g. 0.001e18 = 0.10%.
  1141→  1141→        uint256 feeUsd = usdcAmount * mintFee / 1e18;
  1142→  1142→        uint256 netUsd = usdcAmount - feeUsd;
  1143→  1143→        // Scale 6-dec USDC → 18-dec USD value, then divide by 18-dec price.
  1144→  1144→        uint256 netUsd18  = netUsd * 1e12;
  1145→  1145→        uint256 grossMint = netUsd18 * 1e18 / price;
  1146→  1146→
  1147→  1147→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25)
  1148→  1148→        uint256 throttle = mintThrottle(currentState);
  1149→  1149→        minted = grossMint * throttle / 1e18;
  1150→  1150→
  1151→  1151→        // Checks-Effects-Interactions (H1): state updated BEFORE any external call.
  1152→  1152→        _mint(msg.sender, minted);
  1153→  1153→        emit Mint(msg.sender, usdcAmount, feeUsd, minted, price);
  1154→  1154→    }
  1155→  1155→
  1156→  1156→    // ============================================================
  1157→  1157→    // §19.3.2 + I6  Redeem — NAV-based (NOT P_MTQ)
  1158→  1158→    //   Burn MTQ, grossUsd = mtqAmount × NAVperToken (NOT × P_MTQ),
  1159→  1159→    //   fee = redeemFee(currentState), net → USDC 6 dec from reserve vault,
  1160→  1160→    //   decrement reserveHeldUsd proportionally (basket broken).
  1161→  1161→    //   REVERT if !redemptionAllowed(currentState). Apply price guard (H4).
  1162→  1162→    // ============================================================
  1163→  1163→    function redeem(uint256 mtqAmount) external nonReentrant whenNotPaused returns (uint256 usdcOut) {
  1164→  1164→        if (!(mtqAmount > 0)) revert Err56();
  1165→  1165→        if (!(balanceOf[msg.sender] >= mtqAmount)) revert Err22();
  1166→  1166→        if (!(redemptionAllowed(currentState))) revert Err43();
  1167→  1167→
  1168→  1168→        // H4: apply price guard even on redeem (rejects if outside [0.50, 2.00]).
  1169→  1169→        uint256 price = getMTQPriceWithGuard();
  1170→  1170→        // §19.3.2 NAV-per-token (NOT P_MTQ)
  1171→  1171→        uint256 navPerToken = getNAVperToken();
  1172→  1172→        if (!(navPerToken < type(uint256).max)) revert Err26();
  1173→  1173→
  1174→  1174→        // §14.6.3 fee by state (1e18 fraction)
  1175→  1175→        uint256 feeFraction = redeemFee(currentState);
  1176→  1176→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
  1177→  1177→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
  1178→  1178→        uint256 netUsd18   = grossUsd18 - feeUsd18;
  1179→  1179→        usdcOut            = netUsd18 / 1e12; // 18-dec → 6-dec USDC
  1180→  1180→
  1181→  1181→        // Checks-Effects-Interactions (H2): burn + decrement BEFORE external call.
  1182→  1182→        _burn(msg.sender, mtqAmount);
  1183→  1183→        // Decrement reserveHeldUsd proportionally (basket broken on redeem).
  1184→  1184→        // Use gross USD value across all components so the share is exact.
  1185→  1185→        uint256 totalReserve = getReserveNavUsd();
  1186→  1186→        if (totalReserve > 0 && grossUsd18 > 0) {
  1187→  1187→            for (uint8 i = 0; i < 7; i++) {
  1188→  1188→                uint256 held = reserveHeldUsd[Component(i)];
  1189→  1189→                if (held == 0) continue;
  1190→  1190→                uint256 share = held * grossUsd18 / totalReserve;
  1191→  1191→                if (share > held) share = held; // safety
  1192→  1192→                reserveHeldUsd[Component(i)] = held - share;
  1193→  1193→            }
  1194→  1194→        }
  1195→  1195→        // Update reserveState mirror
  1196→  1196→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1197→  1197→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1198→  1198→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1199→  1199→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1200→  1200→        reserveState.liquidValue = reserveState.fiatValue;
  1201→  1201→
  1202→  1202→        // External call LAST (after all state updates).
  1203→  1203→        if (!(usdc.transferFrom(reserveVault, msg.sender, usdcOut))) revert Err04();
  1204→  1204→        emit Redeem(msg.sender, mtqAmount, feeUsd18 / 1e12, usdcOut, navPerToken);
  1205→  1205→        // silence unused var warning
  1206→  1206→        if (price == 0) revert Err59();
  1207→  1207→    }

  1210→  1210→    // §10  MARP Rebalancing Execution — keeper-callable
  1211→  1211→    //   For each trade: validates it moves toward the latest committed
  1212→  1212→    //   lastWeights (executionWeight) within a 5% tolerance, enforces 24h
  1213→  1213→    //   direction lock (with RR<1.05 solvency override — C4 fix), validates
  1214→  1214→    //   total daily trade USD ≤ 5% of NAV, caps trades.length at 7 (H3 fix),
  1215→  1215→    //   updates reserve holdings (on-chain mirror).
  1216→  1216→    // ============================================================
  1217→  1217→
  1218→  1218→    struct RebalanceTrade {
  1219→  1219→        Component component;
  1220→  1220→        int256    direction; // +1 buy, -1 sell, 0 hold
  1221→  1221→        uint256   tradeUsd;  // USD value, 1e18 scale
  1222→  1222→        uint256   level;     // 1-6 (MARP decision level)
  1223→  1223→    }
  1224→  1224→
  1225→  1225→    uint256 public constant DIRECTION_LOCK_HOURS  = 24 hours;
  1226→  1226→    uint256 public constant MAX_DAILY_TURNOVER    = 0.05e18;
  1227→  1227→    uint256 public constant REBALANCE_TOLERANCE    = 0.05e18;
  1228→  1228→
  1229→  1229→    mapping(Component => int256)  public lastDirection;
  1230→  1230→    mapping(Component => uint256) public lastRebalanceAt;
  1231→  1231→    uint256 public dailyTradeUsd;
  1232→  1232→    uint256 public dailyTradeResetAt;
  1233→  1233→
  1234→  1234→    event RebalanceExecuted(RebalanceTrade[] trades, uint256 timestamp);
  1235→  1235→
  1236→  1236→    function executeRebalance(RebalanceTrade[] memory trades) external nonReentrant onlyKeeper whenNotPaused {
  1237→  1237→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1238→  1238→        if (!(trades.length <= 7)) revert Err48(); // H3 cap
  1239→  1239→
  1240→  1240→        // Reset 24h daily turnover window if elapsed
  1241→  1241→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1242→  1242→            dailyTradeUsd    = 0;
  1243→  1243→            dailyTradeResetAt = block.timestamp;
  1244→  1244→        }
  1245→  1245→
  1246→  1246→        uint256 nav = getReserveNavUsd();
  1247→  1247→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1248→  1248→        uint256 rr = getReserveRatio();
  1249→  1249→        bool solvencyOverride = (rr < rrStressFloor); // §11.5.3 / C4 fix
  1250→  1250→
  1251→  1251→        for (uint256 i = 0; i < trades.length; i++) {
  1252→  1252→            RebalanceTrade memory t = trades[i];
  1253→  1253→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1254→  1254→            if (t.direction == 0 || t.tradeUsd == 0) continue; // hold
  1255→  1255→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1256→  1256→
  1257→  1257→            // §10 24h direction lock — opposite direction blocked within 24h,
  1258→  1258→            // UNLESS RR<1.05 (solvency override — C4 fix).
  1259→  1259→            int256 prevDir = lastDirection[t.component];
  1260→  1260→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1261→  1261→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1262→  1262→            }
  1263→  1263→
  1264→  1264→            // Validate trade moves toward execution weight (within 5% tolerance)
  1265→  1265→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1266→  1266→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1267→  1267→            if (t.direction == 1) {
  1268→  1268→                if (!(currentW < execW)) revert Err14();
  1269→  1269→            } else {
  1270→  1270→                if (!(currentW > execW)) revert Err46();
  1271→  1271→            }
  1272→  1272→
  1273→  1273→            // New weight after trade must be within 5pp (absolute) of execW
  1274→  1274→            uint256 newHeldUsd = (t.direction == 1)
  1275→  1275→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1276→  1276→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1277→  1277→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1278→  1278→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1279→  1279→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1280→  1280→            // Solvency override relaxes overshoot tolerance by 2x during stress
  1281→  1281→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1282→  1282→            if (!(absDiff <= tolerance)) revert Err49();
  1283→  1283→
  1284→  1284→            // §10 daily turnover cap (5% of NAV)
  1285→  1285→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1286→  1286→            dailyTradeUsd += t.tradeUsd;
  1287→  1287→
  1288→  1288→            // Update reserve mirror (checks-effects-interactions)
  1289→  1289→            if (t.direction == 1) {
  1290→  1290→                reserveHeldUsd[t.component] += t.tradeUsd;
  1291→  1291→            } else {
  1292→  1292→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1293→  1293→            }
  1294→  1294→            lastDirection[t.component]   = t.direction;
  1295→  1295→            lastRebalanceAt[t.component] = block.timestamp;
  1296→  1296→        }
  1297→  1297→
  1298→  1298→        // Refresh reserveState aggregate mirror after rebalancing
  1299→  1299→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1300→  1300→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1301→  1301→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1302→  1302→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1303→  1303→        reserveState.liquidValue = reserveState.fiatValue;
  1304→  1304→
  1305→  1305→        emit RebalanceExecuted(trades, block.timestamp);
  1306→  1306→    }
