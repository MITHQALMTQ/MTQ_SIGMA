   255→function seedHistory(mean: number, sd: number, n: number): number[] {
   256→  const out: number[] = [];
   257→  let last = mean;
   258→  for (let i = 0; i < n; i++) {
   259→    // simple mean-reverting walk
   260→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   261→    last = last + (mean - last) * 0.1 + shock;
   262→    out.push(Math.max(0.01, last));
   263→  }
   264→  return out;
   265→}

   267→// --- §2 GFB Index (v1.0 — 7-component chain-linked) -------------------------
   268→// GFB_t = Σ_i W^Prior_i × P_{i,t}  (then normalised by GFB_BASE_DENOMINATOR so
   269→// GFB = 1.00 exactly at the base date). The 7 components are the Strategic
   270→// Prior: USD, EUR, JPY, GBP, CNY, CHF, Gold. Gold is now a first-class index
   271→// component (P_Gold = XAU_USD).
   272→export function computeGfbIndex(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): number {
   273→  // Raw USD value of the basket at time t. USD is the unit of account (price 1.0);
   274→  // every other component contributes its USD-equivalent via its FX rate.
   275→  const numerator =
   276→    STRATEGIC_PRIOR.USD  * 1.0 +
   277→    STRATEGIC_PRIOR.EUR  * fx.EUR_USD +
   278→    STRATEGIC_PRIOR.JPY  * fx.JPY_USD +
   279→    STRATEGIC_PRIOR.GBP  * fx.GBP_USD +
   280→    STRATEGIC_PRIOR.CNY  * fx.CNY_USD +
   281→    STRATEGIC_PRIOR.CHF  * fx.CHF_USD +
   282→    STRATEGIC_PRIOR.Gold * fx.XAU_USD;
   283→  // Normalised: GFB_t = numerator / GFB_BASE_DENOMINATOR  (=1.0 at base date).
   284→  // GFB_BASE_DENOMINATOR is the v1.0 chain-linked denominator (includes Gold
   285→  // and CHF at base fixings — see blueprint.ts).
   286→  return numerator / GFB_BASE_DENOMINATOR;
   287→}

   301→export function reserveAssetValues(s: ReserveState, fx: FxRates) {

   306→  const cnyGross = s.cny * fx.CNY_USD;
   307→  const chfGross = s.chf * fx.CHF_USD; // NEW v1.0 — CHF is a first-class reserve component
   308→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross + chfGross;
   309→
   310→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   311→  const goldRef = fx.XAU_USD;
   312→  const goldExec = fx.XAU_USD * 0.995;
   313→  const goldPrice = Math.min(goldRef, goldExec);
   314→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   315→
   316→  // Net values (after haircuts)
   317→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   318→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   319→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   320→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   321→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   322→  const chfNet = chfGross * (1 - HAIRCUTS.CHF); // NEW v1.0
   323→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   324→
   325→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet + goldNet;
   326→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet;
   327→
   328→  // Per-issuer breakdown (for concentration visualization)
   329→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   330→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   331→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   332→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);

   335→  return {
   336→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, chfGross, fiatGross,
   337→    goldGross, goldPrice,
   338→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, chfNet, goldNet,
   339→    nav, fiatNet,
   340→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   341→  };

   374→// --- §6 Adaptive Macro Engine -----------------------------------------------
   375→function rollingStats(hist: number[]): { mean: number; sd: number } {
   376→  const filled = hist.filter((x) => x > 0);
   377→  if (filled.length === 0) return { mean: 0, sd: 0 };
   378→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   379→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   380→  return { mean, sd: Math.sqrt(variance) };
   381→}
   382→
   383→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   384→  const v = rollingStats(s.vixHistory);
   385→  const d = rollingStats(s.dxyHistory);
   386→  const eps = 1e-9;
   387→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   388→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   389→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   390→}

   392→export function computeRawTargetTheta(zVix: number, zDxy: number): number {
   393→  const theta = ALPHA * zVix + BETA * zDxy;
   394→  return clampSym(theta, -THETA_MAX, THETA_MAX);
   395→}

   397→export function computeRawTargetGoldWeight(zVix: number, zDxy: number): number {
   398→  const theta = computeRawTargetTheta(zVix, zDxy);
   399→  return clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   400→}

   402→export function smoothGoldWeight(prev: number, target: number): number {
   403→  if (prev <= 0) return target;
   404→  return SMOOTHING_LAMBDA * target + (1 - SMOOTHING_LAMBDA) * prev;
   405→}

   407→// --- §8 Dynamic Buffer -------------------------------------------------------
   408→export function bufferBaseGoldRatio(rr: number): number {
   409→  if (rr >= RR_TARGET) return BUFFER_GOLD_BASE; // 62.5% → total 26.25%
   410→  if (rr >= RR_STRESS) return BUFFER_GOLD_STRESS; // 85% → total 28.5%
   411→  return BUFFER_GOLD_EMERGENCY; // 100% → total 30%
   412→}
   413→
   414→export function currentBufferGoldRatio(s: ReserveState): number {
   415→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   416→  if (s.rampStartAt === 0) {
   417→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   418→  }
   419→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   420→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   421→  const progress = elapsedH / RAMP_DURATION_HOURS;
   422→  const from = s.rampFromRatio;
   423→  const to = s.rampToRatio;
   424→  return from + (to - from) * progress;
   425→}
   426→
   427→export function bufferBaseFromRatio(rr: number): "BASE" | "STRESS" | "EMERGENCY" {
   428→  if (rr >= RR_TARGET) return "BASE";
   429→  if (rr >= RR_STRESS) return "STRESS";
   430→  return "EMERGENCY";
   431→}

   433→// Unified target gold weight (tokenomics reconciliation):
   434→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   435→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   436→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   437→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   438→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   439→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   440→}

   442→// --- §7 Rebalancing decision (simplified cost-benefit) ----------------------
   443→export interface RebalanceDecision {
   444→  shouldRebalance: boolean;
   445→  direction: 0 | 1 | -1;
   446→  observedGoldWeight: number;
   447→  targetGoldWeight: number;
   448→  deviation: number; // observed - target (signed)

   453→export function maybeResetDailyTurnover(s: ReserveState): void {
   454→  if (Date.now() - s.lastTurnoverResetAt >= 86_400_000) {
   455→    s.dailyTurnoverUsd = 0;
   456→    s.lastTurnoverResetAt = Date.now();
   457→  }

   460→export function evaluateRebalance(
   461→  s: ReserveState,
   462→  vals: { nav: number; goldNet: number; fiatNet: number },
   463→  rr: number,
   464→): RebalanceDecision {
   465→  const nav = vals.nav;
   466→  if (nav <= 0) {
   467→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   468→  }
   469→  const observed = vals.goldNet / nav;
   470→  const target = computeTargetGoldWeight(s, rr);
   471→  const deviation = observed - target;
   472→
   473→  if (Math.abs(deviation) < 0.005) {
   474→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   475→  }
   476→
   477→  // Direction lock (overridden in stress)
   478→  const inStress = rr < RR_STRESS;
   479→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   480→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   481→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   482→    }
   483→  }
   484→
   485→  // Trade sizing
   486→  const surplusUsd = Math.abs(deviation) * nav;
   487→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   488→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   489→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   490→  maybeResetDailyTurnover(s);
   491→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   492→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   493→
   494→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   495→  const estCost = tradeUsd * 0.001 + 50;
   496→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   497→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   498→
   499→  if (benefit <= estCost) {
   500→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   501→  }
   502→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   503→}
   504→
   505→// Apply a rebalance trade in USD terms (mutates state).
   506→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   507→  if (!decision.shouldRebalance) return;
   508→  const usd = decision.tradeUsd;
   509→  if (decision.direction === -1) {
   510→    // sell gold → buy USDC (simplified: convert to USDC)
   511→    const goldUnits = usd / goldPrice;
   512→    s.paxg = Math.max(0, s.paxg - goldUnits);
   513→    s.usdc += usd;
   514→  } else {
   515→    // buy gold → spend USDC
   516→    s.usdc = Math.max(0, s.usdc - usd);
   517→    const goldUnits = usd / goldPrice;
   518→    s.paxg += goldUnits;
   519→  }
   520→  s.lastTradeDir = decision.direction;
   521→  s.lastTradeAt = Date.now();
   522→  s.dailyTurnoverUsd += usd;
   523→  s.updatedAt = Date.now();
   524→}

   526→// --- §11 Geopolitical Eject -------------------------------------------------
   527→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   528→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   529→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   530→  const drift = (cur: number, amp: number) => {
   531→    const shock = (Math.random() - 0.5) * 2 * amp;
   532→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   533→  };
   534→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   535→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   536→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   537→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   538→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   539→
   540→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   541→  for (const c of currencies) {
   542→    const p = s.pegHealth[c];
   543→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   544→    if (outside) {
   545→      s.depegHours[c] += dtHours;
   546→    } else {
   547→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   548→    }
   549→    // Staged ladder
   550→    const h = s.depegHours[c];
   551→    let stage = 0;
   552→    if (h > 48) stage = 3;
   553→    else if (h > 24) stage = 2;
   554→    else if (h > 12) stage = 1;
   555→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   556→    s.ejectStage[c] = stage;
   557→  }
   558→}

   573→
   574→export function applyMint(
   575→  s: ReserveState,
   576→  fx: FxRates,
   577→  status: ProtocolStatus,
   578→  inputUsd: number,
   579→): MintResult {
   580→  const gfb = computeGfbIndex(fx);
   581→  const price = computeMtqPrice(gfb);
   582→  if (!priceInSafetyBand(price)) {
   583→    return { ok: false, reason: "MTQ price outside safety band (0.50–2.00 USD); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   584→  }
   585→  if (status === "DEFENSIVE" || status === "EMERGENCY") {
   586→    return { ok: false, reason: `${status}: minting paused by Risk State Machine.`, inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   587→  }
   588→  const throttle = status === "CAUTION" ? 0.5 : status === "RECOVERY" ? 0.25 : 1.0;
   589→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
   590→  const netUsd = inputUsd - feeUsd;
   591→  const minted = (netUsd / price) * throttle;
   592→
   593→  // Mutate reserve: deposit USDC (split across 3 USD issuers per concentration policy),
   594→  // mint MTQ to user (circulating supply rises). The concentration rebalancer in the
   595→  // tick loop will re-optimize the split if any issuer exceeds 25%.
   596→  const usdThird = netUsd / 3;
   597→  s.usdc += usdThird;
   598→  s.usdp += usdThird;
   599→  s.usdt += usdThird;
   600→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet (§13.2 sweep target)
   601→  s.totalSupply += minted;
   602→  s.updatedAt = Date.now();
   603→
   604→  const vals = reserveAssetValues(s, fx);
   605→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
   606→  return {
   607→    ok: true,
   608→    inputUsd,
   609→    feeUsd,
   610→    netUsd,
   611→    mtqPrice: price,
   612→    mtqMinted: minted,
   613→    throttleFactor: throttle,
   614→    newCirculatingSupply: circulatingSupply(s),
   615→    newReserveRatio: newRr,
   616→  };
   617→}

At RR>100% the §12.2 literal pays redeemers MORE than §3.4.2, draining the buffer surplus via arbitrage. We settle on the arbitrage-safe §3.4.2 value; the §12.2 figure is shown for audit only.";

   667→  }
   668→  // Fee by status (§14.1): NORMAL/CAUTION 0.15%, DEFENSIVE 0.5%, EMERGENCY 2%
   669→  let feeBps = REDEEM_FEE_BPS;
   670→  if (status === "DEFENSIVE") feeBps = 50;
   671→  if (status === "EMERGENCY") feeBps = 200;

   673→  const feeUsd = grossUsd * (feeBps / 10_000);
   674→  const netUsd = grossUsd - feeUsd;
   675→
   676→  // Release actual reserve composition proportionally (§12.2):
   677→  // gold portion = netUsd * W_target (legacy v1.2 buffer-derived target —
   678→  // retained until the MASE ensemble replaces it). fiat portion split per the
   679→  // 6 non-gold Strategic Prior components (USD/EUR/JPY/GBP/CNY/CHF) renormalised
   680→  // to the non-gold total.
   681→  const vals = reserveAssetValues(s, fx);
   682→  const rr = computeReserveRatio(vals.nav, computeLiability(s, price));
   683→  const wGold = computeTargetGoldWeight(s, rr);
   684→  const goldUsd = netUsd * wGold;
   685→  const fiatUsd = netUsd * (1 - wGold);
   686→  const goldPrice = vals.goldPrice;
   687→  const goldPaxg = goldUsd / goldPrice;
   688→
   689→  // v1.0: Strategic Prior weights for the 6 non-gold components, renormalised
   690→  // to the non-gold total (0.27 + 0.20 + 0.09 + 0.08 + 0.05 + 0.05 = 0.74).
   691→  const basketFiatTotal =
   692→    STRATEGIC_PRIOR.USD + STRATEGIC_PRIOR.EUR + STRATEGIC_PRIOR.JPY +
   693→    STRATEGIC_PRIOR.GBP + STRATEGIC_PRIOR.CNY + STRATEGIC_PRIOR.CHF;
   694→  const w = (k: keyof typeof STRATEGIC_PRIOR) => STRATEGIC_PRIOR[k] / basketFiatTotal;
   695→  const basket: RedeemBasket[] = [
   696→    { currency: "USD", token: "USDC", usdValue: fiatUsd * w("USD"), nativeAmount: fiatUsd * w("USD"), weight: STRATEGIC_PRIOR.USD },
   697→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * w("EUR"), nativeAmount: (fiatUsd * w("EUR")) / fx.EUR_USD, weight: STRATEGIC_PRIOR.EUR },
   698→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * w("JPY"), nativeAmount: (fiatUsd * w("JPY")) / fx.JPY_USD, weight: STRATEGIC_PRIOR.JPY },
   699→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * w("GBP"), nativeAmount: (fiatUsd * w("GBP")) / fx.GBP_USD, weight: STRATEGIC_PRIOR.GBP },
   700→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * w("CNY"), nativeAmount: (fiatUsd * w("CNY")) / fx.CNY_USD, weight: STRATEGIC_PRIOR.CNY },
   701→    { currency: "CHF", token: "CHF₿", usdValue: fiatUsd * w("CHF"), nativeAmount: (fiatUsd * w("CHF")) / fx.CHF_USD, weight: STRATEGIC_PRIOR.CHF }, // NEW v1.0
   702→  ];
   703→
   704→  // Mutate reserve: burn MTQ from circulating (reduce totalSupply), release assets
   705→  // USD released proportionally across USDC + USDP + USDT; gold across PAXG + XAUT
   706→  s.totalSupply -= inputMtq;
   707→  const usdRelease = basket[0].nativeAmount;

   710→  s.usdp = Math.max(0, s.usdp - usdThirdRelease);
   711→  s.usdt = Math.max(0, s.usdt - usdThirdRelease);
   712→  s.eurc = Math.max(0, s.eurc - basket[1].nativeAmount);

   714→  s.gbp  = Math.max(0, s.gbp  - basket[3].nativeAmount);
   715→  s.cny  = Math.max(0, s.cny  - basket[4].nativeAmount);
   716→  s.chf  = Math.max(0, s.chf  - basket[5].nativeAmount); // NEW v1.0
   717→  // Gold released 50/50 PAXG + XAUT
   718→  const goldHalf = goldPaxg / 2;
   719→  s.paxg = Math.max(0, s.paxg - goldHalf);
   720→  s.xaut = Math.max(0, s.xaut - goldHalf);
   721→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet
   722→  s.updatedAt = Date.now();
   723→
   724→  const newVals = reserveAssetValues(s, fx);
   725→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
   726→  return {

   735→    goldPaxg,

   737→    newCirculatingSupply: circulatingSupply(s),
   738→    newReserveRatio: newRr,
   739→    auditNavPerToken: circ > 0 ? vals0.nav / circ : 0,
   740→    auditGrossUsdNav,
   741→    auditDeltaUsd: auditGrossUsdNav - grossUsd,
   742→    auditNote,

   903→// --- §6/§7/§8.4/§10 v1.0 MASE + 4-state weights + MARP ---------------------
   904→// The Multi-Asset Stochastic Ensemble (§6/§7) runs 6 candidate models, blends
   905→// them into a single target weight vector, clamps to per-component
   906→// admissibility envelopes (§8.1), and EMA-smooths toward the constrained target
   907→// (§8.4). MARP (§10) then decides per-component whether the deviation between
   908→// the smoothed target and the observed execution weight justifies a trade
   909→// (urgency test → no-trade zone → cost-benefit gate → partial correction →
   910→// daily-turnover cap → execute).
   911→//
   912→// `advanceMase(s, fx)` is called ONCE per tick from the pilot-state tick loop.
   913→// It is the only function that mutates `s.maseSmoothed`. `buildMaseSnapshot()`
   914→// is a pure helper used by `computeSnapshot()` — it recomputes target/smoothed/
   915→// MARP from the current fx + vals using the persisted prev-smoothed as the EMA
   916→// prior, so multiple computeSnapshot calls within a single tick return the
   917→// same value (idempotent).
   918→
   919→interface MaseSnapshotData {
   920→  mase: NonNullable<MetricsSnapshot["mase"]>;
   921→  weightStates: NonNullable<MetricsSnapshot["weightStates"]>;
   922→  marp: NonNullable<MetricsSnapshot["marp"]>;
   923→  envelopes: MetricsSnapshot["envelopes"];
   924→}

   926→/** Estimate per-component annualized volatility from the engine's macro state.
   927→ *  Pilot approximation: scale baseline vols by VIX / 18.5 (the "normal" VIX
   928→ *  baseline), so a 30 VIX roughly doubles estimated vols and a 12 VIX halves
   929→ *  them. Production should use a rolling covariance from real price history. */
   930→function estimateVolsFromState(s: ReserveState): VolatilityData {
   931→  const vix = s.lastVix > 0 ? s.lastVix : 18.5;
   932→  const f = vix / 18.5;
   933→  return {
   934→    USD: 0.05 * f,
   935→    EUR: 0.10 * f,
   936→    JPY: 0.12 * f,
   937→    GBP: 0.11 * f,
   938→    CNY: 0.09 * f,
   939→    CHF: 0.10 * f,
   940→    Gold: 0.15 * f,
   941→  };
   942→}
   943→
   944→/** Detect the market regime from VIX (primary) + DXY (secondary).
   945→ *  0 = calm, 1 = normal, 2 = stress, 3 = crisis.
   946→ *  VIX thresholds: <15 calm, <22 normal, <30 stress, ≥30 crisis.
   947→ *  DXY extremes (<88 or >115) bump the regime up to ≥ stress. */
   948→function detectRegime(vix: number, dxy: number, goldVol: number): MarketRegime {
   949→  let r: 0 | 1 | 2 | 3 = 1;
   950→  if (vix >= 30) r = 3;
   951→  else if (vix >= 22) r = 2;
   952→  else if (vix >= 15) r = 1;
   953→  else r = 0;
   954→  if ((dxy >= 115 || dxy <= 88) && r < 2) r = 2;
   955→  return { regime: r, vix, dxy, goldVol };
   956→}

   972→function buildObservedWeights(
   973→  vals: ReturnType<typeof reserveAssetValues>,
   974→  nav: number,

   990→/** Pure helper — computes the MASE + 4-state + MARP + envelope snapshot data.
   991→ *  Reads `s.maseSmoothed` (does NOT mutate). */
   992→function buildMaseSnapshot(
   993→  s: ReserveState,
   994→  fx: FxSnapshot,
   995→  vals: ReturnType<typeof reserveAssetValues>,
   996→  nav: number,
   997→  rr: number,
   998→): MaseSnapshotData {
   999→  const prices = buildPriceData(fx);
  1000→  const vols = estimateVolsFromState(s);
  1001→  const regime = detectRegime(s.lastVix > 0 ? s.lastVix : fx.VIX, s.lastDxy > 0 ? s.lastDxy : fx.DXY, vols.Gold);
  1002→
  1003→  // §6/§7 — MASE ensemble → 6 candidate models + target weight vector
  1004→  const mase = maseEnsemble(vols, prices, null, regime);
  1005→  // §8.1 — Constrain target to per-component admissibility envelopes
  1006→  const constrained = applyEnvelopes(mase.target);
  1007→  // §8.4 — EMA-smooth toward the constrained target (uses prev persisted as prior)
  1008→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1009→  const smoothed = smoothWeights(prevSmoothed, constrained);
  1010→  // §2.3 — Execution = actual observed reserve composition
  1011→  const observed = buildObservedWeights(vals, nav);
  1012→
  1013→  // §10 — MARP per-component rebalancing decision
  1014→  const volsRecord: Record<Component, number> = {
  1015→    USD: vols.USD, EUR: vols.EUR, JPY: vols.JPY, GBP: vols.GBP,
  1016→    CNY: vols.CNY, CHF: vols.CHF, Gold: vols.Gold,
  1017→  };
  1018→  const marpDecisions: MarpDecision[] = marpDecision(observed, smoothed, nav, rr, volsRecord);
  1019→  const totalTradeUsd = marpDecisions
  1020→    .filter((d) => d.shouldTrade)
  1021→    .reduce((a, d) => a + d.tradeUsd, 0);
  1022→
  1023→  // §8.1 — Admissibility envelope status per component
  1024→  // "warn" = within 10% of the band edge; "breach" = outside the envelope.
  1025→  const envelopes = COMPONENTS.map((c) => {
  1026→    const env = ADMISSIBILITY_ENVELOPES[c];
  1027→    const current = observed[c];
  1028→    const lower = env.lower;
  1029→    const upper = env.upper;
  1030→    const margin = (upper - lower) * 0.1;
  1031→    let status: "ok" | "warn" | "breach" = "ok";
  1032→    if (current < lower || current > upper) status = "breach";
  1033→    else if (current < lower + margin || current > upper - margin) status = "warn";
  1034→    return { component: c, lower, upper, current, status };
  1035→  });
  1036→
  1037→  return {
  1038→    mase: {
  1039→      models: mase.models.map((m) => ({
  1040→        id: m.id,
  1041→        name: m.name,
  1042→        weights: m.weights as Record<string, number>,
  1043→      })),
  1044→      ensembleTarget: mase.target as Record<string, number>,
  1045→    },
  1046→    weightStates: {
  1047→      prior: STRATEGIC_PRIOR as Record<string, number>,
  1048→      target: constrained as Record<string, number>,
  1049→      smoothed: smoothed as Record<string, number>,
  1050→      execution: observed as Record<string, number>,
  1051→    },
  1052→    marp: {
  1053→      decisions: marpDecisions.map((d) => ({
  1054→        shouldTrade: d.shouldTrade,
  1055→        component: d.component,
  1056→        direction: d.direction,
  1057→        tradeUsd: d.tradeUsd,
  1058→        urgency: d.urgency,
  1059→        reason: d.reason,
  1060→        level: d.level,
  1061→      })),
  1062→      totalTradeUsd,
  1063→    },
  1064→    envelopes,
  1065→  };
  1066→}
  1067→
  1068→/** Advance MASE state once per tick. Mutates `s.maseSmoothed` (EMA prior for
  1069→ *  the next tick). Called from the pilot-state tick loop, AFTER advanceMacro
  1070→ *  (so lastVix/lastDxy are fresh) and BEFORE the first computeSnapshot of the
  1071→ *  tick (so the snapshot reads the freshly-persisted smoothed weights). */
  1072→export function advanceMase(s: ReserveState, fx: FxRates): void {
  1073→  const prices = buildPriceData(fx);
  1074→  const vols = estimateVolsFromState(s);
  1075→  const regime = detectRegime(
  1076→    s.lastVix > 0 ? s.lastVix : fx.VIX,
  1077→    s.lastDxy > 0 ? s.lastDxy : fx.DXY,
  1078→    vols.Gold,
  1079→  );
  1080→  const mase = maseEnsemble(vols, prices, null, regime);
  1081→  const constrained = applyEnvelopes(mase.target);
  1082→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1083→  s.maseSmoothed = smoothWeights(prevSmoothed, constrained);
  1084→  s.maseLastAt = Date.now();
  1085→}

  1095→// Advance macro history (call ~ every tick; dtHours is simulated elapsed)
  1096→export function advanceMacro(s: ReserveState, vix: number, dxy: number, dtHours: number) {
  1097→  const v = clamp(vix, VIX_MIN, VIX_MAX);
  1098→  const d = clamp(dxy, DXY_MIN, DXY_MAX);
  1099→  s.lastVix = v;
  1100→  s.lastDxy = d;
  1101→  s.vixHistory.push(v);
  1102→  s.dxyHistory.push(d);
  1103→  if (s.vixHistory.length > ROLLING_WINDOW_DAYS) s.vixHistory.shift();
  1104→  if (s.dxyHistory.length > ROLLING_WINDOW_DAYS) s.dxyHistory.shift();
  1105→  const vStats = rollingStats(s.vixHistory);
  1106→  const dStats = rollingStats(s.dxyHistory);
  1107→  const zVix = vStats.sd > 1e-9 ? (v - vStats.mean) / (vStats.sd + 1e-9) : 0;
  1108→  const zDxy = dStats.sd > 1e-9 ? (d - dStats.mean) / (dStats.sd + 1e-9) : 0;
  1109→  const rawTarget = computeRawTargetGoldWeight(zVix, zDxy);
  1110→  s.smoothedGoldWeight = smoothGoldWeight(s.smoothedGoldWeight, rawTarget);
  1111→}

  1135→// --- §11.3 Reintegration Score (Anti-Gaming) --------------------------------
  1136→// R_score = w1·TimeInBand(48h) + w2·LiquidityDepth + w3·OracleAgreement + w4·(1−Volatility)
  1137→// All inputs normalised to [0,1]; threshold >0.80 → staged re-purchase 25%→50%→100%.
  1138→export function updateReintegration(
  1139→  s: ReserveState,
  1140→  board: OracleBoard | null,
  1141→  dtHours: number,
  1142→) {
  1143→  const currencies: ("USD" | "EUR" | "GBP" | "JPY" | "CNY")[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
  1144→  for (const c of currencies) {
  1145→    const r = s.reintegration[c];
  1146→    // Time-in-band: increases when peg is inside [0.98,1.02]; decays when outside.
  1147→    const inBand = s.pegHealth[c] >= EJECT_DEPEG_BAND_LOWER && s.pegHealth[c] <= EJECT_DEPEG_BAND_UPPER;
  1148→    // dtHours is simulated elapsed; we approximate 48h window fraction.
  1149→    if (inBand) {
  1150→      r.timeInBand48h = Math.min(1, r.timeInBand48h + dtHours / 48);
  1151→    } else {
  1152→      r.timeInBand48h = Math.max(0, r.timeInBand48h - dtHours / 48);
  1153→    }
  1154→    // Liquidity depth: randomised around per-currency baseline (honest: simulated).
  1155→    const base = c === "USD" ? 1.0 : c === "EUR" ? 0.95 : c === "CNY" ? 0.4 : 0.5;
  1156→    r.liquidityDepth = clamp(base + (Math.random() - 0.5) * 0.1, 0, 1);
  1157→    // Oracle agreement: valid feeds / 3 (from oracle board if present, else last known)
  1158→    if (board) {
  1159→      const pairMap: Record<string, string> = { USD: "—", EUR: "EUR/USD", GBP: "GBP/USD", JPY: "JPY/USD", CNY: "CNY/USD" };
  1160→      const cons = board.pairs.find((p) => p.pair === pairMap[c]);
  1161→      if (cons) {
  1162→        r.oracleAgreement = cons.validCount / 3;
  1163→        // Volatility from oracle spread (bps → fraction)
  1164→        r.volatility = clamp(cons.spreadBps / 100, 0, 1);
  1165→      }
  1166→    }
  1167→    // R_score
  1168→    r.score =
  1169→      REINTEGRATION_WEIGHTS.w1_TimeInBand * r.timeInBand48h +
  1170→      REINTEGRATION_WEIGHTS.w2_LiquidityDepth * r.liquidityDepth +
  1171→      REINTEGRATION_WEIGHTS.w3_OracleAgreement * r.oracleAgreement +
  1172→      REINTEGRATION_WEIGHTS.w4_OneMinusVolatility * (1 - r.volatility);
  1173→    // Repurchase staging: only for currencies that were previously ejected
  1174→    if (s.ejectStage[c] > 0 && r.score > REINTEGRATION_THRESHOLD) {
  1175→      r.repurchaseStage = Math.min(3, r.repurchaseStage + 1);
  1176→      // As repurchase progresses, reduce eject stage
  1177→      if (r.repurchaseStage >= 3) {
  1178→        s.ejectStage[c] = Math.max(0, s.ejectStage[c] - 1);
  1179→        if (s.ejectStage[c] === 0) r.repurchaseStage = 0;
  1180→      }
  1181→    } else if (r.score < REINTEGRATION_THRESHOLD * 0.6) {
  1182→      r.repurchaseStage = 0;
  1183→    }
  1184→  }
  1185→}

  1204→// --- §8.5 First-Loss Waterfall ----------------------------------------------
  1205→// 5 layers: (1) Operational Surplus → (2) Buffer Fiat → (3) Buffer Gold →
  1206→// (4) Core Fiat → (5) Core Gold. Apply a loss; track consumed per layer.
  1207→export function applyLoss(s: ReserveState, vals: { fiatNet: number; goldNet: number }, lossUsd: number) {
  1208→  let remaining = lossUsd;
  1209→  s.waterfall.consumed.surplus = 0; // reset for this loss event tracking is cumulative; we add
  1210→  const layers = [
  1211→    { name: "surplus", cap: () => s.treasury.hotWalletUsd + s.waterfall.operationalSurplus, consume: (amt: number) => {
  1212→      const fromSurplus = Math.min(s.treasury.hotWalletUsd, amt);
  1213→      s.treasury.hotWalletUsd -= fromSurplus;
  1214→      s.waterfall.operationalSurplus -= Math.min(s.waterfall.operationalSurplus, amt - fromSurplus);
  1215→      s.waterfall.consumed.surplus += amt;
  1216→    } },
  1217→    { name: "bufferFiat", cap: () => vals.fiatNet * 0.10 * (1 - currentBufferGoldRatio(s)), consume: (amt: number) => { s.waterfall.consumed.bufferFiat += amt; } },
  1218→    { name: "bufferGold", cap: () => vals.goldNet * 0.10 * currentBufferGoldRatio(s), consume: (amt: number) => { s.waterfall.consumed.bufferGold += amt; } },
  1219→    { name: "coreFiat", cap: () => vals.fiatNet * 0.80, consume: (amt: number) => { s.waterfall.consumed.coreFiat += amt; } },
  1220→    { name: "coreGold", cap: () => vals.goldNet * 0.20, consume: (amt: number) => { s.waterfall.consumed.coreGold += amt; } },
  1221→  ] as const;
  1222→  for (let i = 0; i < layers.length && remaining > 0; i++) {
  1223→    const layer = layers[i];
  1224→    const cap = layer.cap();
  1225→    const consumed = Math.min(cap, remaining);
  1226→    layer.consume(consumed);
  1227→    remaining -= consumed;
  1228→    if (remaining > 0) s.waterfall.currentLayer = (i + 2) as 1 | 2 | 3 | 4 | 5;
  1229→  }
  1230→  if (remaining <= 0) s.waterfall.currentLayer = 1;
  1231→}

  1233→// --- §3.6 PriceUpdated events (0.5% change threshold) -----------------------
  1234→export function maybePriceEvent(s: ReserveState, newPrice: number) {
  1235→  if (s.lastLoggedPrice <= 0) { s.lastLoggedPrice = newPrice; return; }
  1236→  const diff = Math.abs(newPrice - s.lastLoggedPrice);
  1237→  const changePct = diff / s.lastLoggedPrice;
  1238→  if (changePct >= PRICE_EVENT_THRESHOLD) {
  1239→    s.priceEvents.unshift({ ts: Date.now(), oldPrice: s.lastLoggedPrice, newPrice, changePct });
  1240→    if (s.priceEvents.length > 20) s.priceEvents.pop();
  1241→    s.lastLoggedPrice = newPrice;
  1242→  }
  1243→}

  1255→export function simulateQuote(amountUsd: number, isBuyGold: boolean, oracleGoldPrice: number): ExecQuote {
  1256→  const provider: "1inch" | "Paraswap" = Math.random() < 0.5 ? "1inch" : "Paraswap";

  1258→  const slip = 0.0005 + Math.random() * 0.0035;
  1259→  const oracleRate = oracleGoldPrice;
  1260→  const quoteRate = isBuyGold ? oracleRate * (1 + slip) : oracleRate * (1 - slip);
  1261→  const bound = isBuyGold ? oracleRate * (1 + SLIPPAGE_TOLERANCE) : oracleRate * (1 - SLIPPAGE_TOLERANCE);
  1262→  const withinBound = isBuyGold ? quoteRate <= bound : quoteRate >= bound;
  1263→  return {
  1264→    provider, oracleRate, quoteRate, bound, withinBound,
  1265→    effectiveSlippageBps: Math.round(slip * 10_000),
  1266→    amountUsd,
  1267→  };
  1268→}

  1270→// --- FIX: §5.6 Concentration Rebalancing -------------------------------------
  1271→// Adjusts the USDC/USDP split to bring the maximum issuer share under the
  1272→// --- FIX: §5.6 Multi-Issuer Concentration Optimizer -------------------------
  1273→// 3 issuers: CIRCLE (USDC + EURC), PAXOS (USDP + PAXG), TETHER (USDT + XAUT).
  1274→// EURC is fixed to Circle (no second EUR issuer yet). GBP/JPY/CNY are TBD.
  1275→// The optimizer adjusts the splittable pools (USDC/USDP/USDT, PAXG/XAUT) to
  1276→// minimize the maximum issuer share, targeting ≤ 25% (warn threshold,
  1277→// headroom under the 30% hard limit).
  1278→export interface ConcentrationFixResult {
  1279→  adjusted: boolean;
  1280→  sharesBefore: { CIRCLE: number; PAXOS: number; TETHER: number };
  1281→  sharesAfter: { CIRCLE: number; PAXOS: number; TETHER: number };
  1282→  splits: { usdc: number; usdp: number; usdt: number; paxg: number; xaut: number };
  1283→  target: number; // 0.25
  1284→  reason: string;
  1285→}
  1286→
  1287→export function rebalanceForConcentration(
  1288→  s: ReserveState,
  1289→  vals: { usdcUsd: number; usdpUsd: number; usdtUsd: number; paxgUsd: number; xautUsd: number; nav: number; eurNet: number },
  1290→): ConcentrationFixResult {
  1291→  const nav = vals.nav;
  1292→  const TARGET = 0.25;
  1293→  const usdNetFactor = 1 - HAIRCUTS.USD;
  1294→  const goldNetFactor = 1 - HAIRCUTS.XAU;
  1295→
  1296→  const sharesBefore = {
  1297→    CIRCLE: nav > 0 ? (vals.usdcUsd + vals.eurNet) / nav : 0,
  1298→    PAXOS: nav > 0 ? (vals.usdpUsd + vals.paxgUsd) / nav : 0,
  1299→    TETHER: nav > 0 ? (vals.usdtUsd + vals.xautUsd) / nav : 0,
  1300→  };
  1301→  const maxBefore = Math.max(sharesBefore.CIRCLE, sharesBefore.PAXOS, sharesBefore.TETHER);
  1302→
  1303→  if (nav <= 0) {
  1304→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "NAV ≤ 0" };
  1305→  }
  1306→  if (maxBefore <= TARGET) {
  1307→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "All issuers ≤ 25% warn threshold" };
  1308→  }
  1309→
  1310→  // Total splittable pools (in gross token units)
  1311→  const totalUsdGross = s.usdc + s.usdp + s.usdt;
  1312→  const totalGoldGross = s.paxg + s.xaut;
  1313→
  1314→  // Circle's floor = EURC (fixed). Circle's USDC budget = max(0, TARGET×NAV − EURC_usd)
  1315→  const circleUsdNetBudget = Math.max(0, TARGET * nav - vals.eurNet);
  1316→  const usdcGross = Math.min(totalUsdGross, circleUsdNetBudget / usdNetFactor);
  1317→  const remainingUsdGross = Math.max(0, totalUsdGross - usdcGross);
  1318→
  1319→  // Split remaining USD equally between USDP (Paxos) and USDT (Tether)
  1320→  const usdpGross = remainingUsdGross / 2;
  1321→  const usdtGross = remainingUsdGross / 2;
  1322→
  1323→  // Now balance gold between PAXG (Paxos) and XAUT (Tether) so both ≤ 25%.
  1324→  // Clean approach: split gold 50/50 by default; only deviate if one issuer
  1325→  // would breach the 25% cap. This guarantees a real split (the previous
  1326→  // min-budget logic could dump all gold into one issuer when budgets were
  1327→  // asymmetric or when one issuer's USD share was already small).
  1328→  const paxosUsdNet = usdpGross * usdNetFactor;
  1329→  const tetherUsdNet = usdtGross * usdNetFactor;
  1330→  const halfGoldGross = totalGoldGross / 2;
  1331→  let paxgGross = halfGoldGross;
  1332→  let xautGross = halfGoldGross;
  1333→  // Cap check 1: Paxos = USDP_net + PAXG_net ≤ TARGET×NAV
  1334→  const paxosWithHalf = paxosUsdNet + halfGoldGross * goldNetFactor;
  1335→  if (paxosWithHalf > TARGET * nav) {
  1336→    const maxPaxgNet = Math.max(0, TARGET * nav - paxosUsdNet);
  1337→    paxgGross = Math.min(halfGoldGross, maxPaxgNet / goldNetFactor);
  1338→    xautGross = totalGoldGross - paxgGross;
  1339→  }
  1340→  // Cap check 2: Tether = USDT_net + XAUT_net ≤ TARGET×NAV
  1341→  const tetherWithXaut = tetherUsdNet + xautGross * goldNetFactor;
  1342→  if (tetherWithXaut > TARGET * nav) {
  1343→    const maxXautNet = Math.max(0, TARGET * nav - tetherUsdNet);
  1344→    xautGross = Math.min(xautGross, maxXautNet / goldNetFactor);
  1345→    paxgGross = totalGoldGross - xautGross;
  1346→  }
  1347→
  1348→  s.usdc = usdcGross;
  1349→  s.usdp = usdpGross;
  1350→  s.usdt = usdtGross;
  1351→  s.paxg = paxgGross;
  1352→  s.xaut = xautGross;
  1353→
  1354→  const sharesAfter = {
  1355→    CIRCLE: (usdcGross * usdNetFactor + vals.eurNet) / nav,
  1356→    PAXOS: (usdpGross * usdNetFactor + paxgGross * goldNetFactor) / nav,
  1357→    TETHER: (usdtGross * usdNetFactor + xautGross * goldNetFactor) / nav,
  1358→  };
  1359→
  1360→  const maxAfter = Math.max(sharesAfter.CIRCLE, sharesAfter.PAXOS, sharesAfter.TETHER);
  1361→  return {
  1362→    adjusted: true,
  1363→    sharesBefore, sharesAfter,
  1364→    splits: { usdc: usdcGross, usdp: usdpGross, usdt: usdtGross, paxg: paxgGross, xaut: xautGross },
  1365→    target: TARGET,
  1366→    reason: maxAfter <= TARGET
  1367→      ? "Multi-issuer split: all 3 issuers ≤ 25% warn threshold"
  1368→      : `Optimized: max issuer ${(maxAfter * 100).toFixed(1)}% (Circle EURC floor limits further reduction)`,
  1369→  };
  1370→}
