   194→export function initReserveState(goldPrice: number): ReserveState {
   195→  const initialUsdDeposit = 1_100_000;
   196→

   198→  // Each component's USD-equivalent notional = deposit × W^Prior_i. The
   199→  // resulting portfolio is the genesis GFB basket; RR = deposit / supply = 1.10.

   201→  const eurcUsd   = initialUsdDeposit * STRATEGIC_PRIOR.EUR;  // $220K
   202→  const jpyUsd   = initialUsdDeposit * STRATEGIC_PRIOR.JPY;  //  $99K
   203→  const gbpUsd   = initialUsdDeposit * STRATEGIC_PRIOR.GBP;  //  $88K
   204→  const cnyUsd   = initialUsdDeposit * STRATEGIC_PRIOR.CNY;  //  $55K
   205→  const chfUsd   = initialUsdDeposit * STRATEGIC_PRIOR.CHF;  //  $55K (NEW v1.0)
   206→  const goldSpend = initialUsdDeposit * STRATEGIC_PRIOR.Gold; // $286K (Gold is now in the index)
   207→

   209→  const usdc = usdTotal / 3;
   210→  const usdp = usdTotal / 3;
   211→  const usdt = usdTotal / 3;
   212→
   213→  // Convert USD-equivalent fiat holdings into native token units (using the
   214→  // base-date fixings so genesis lands at exactly GFB = 1.00).
   215→  const eurc = eurcUsd / 1.05;            // € per $1.05 at base date

   225→  // index vault (locked, backs the 26% Strategic Prior Gold weight) and the
   226→  // reserve buffer (the gold MARP rebalances). 50/50 was chosen over the
   227→  // "26% to index / 74% to reserve" alternative because:

   229→  //     A 26/74 split would leave index gold at only ~6.8% of NAV — far below
   230→  //     the 26% index weight it is supposed to back, which is misleading.
   231→  //   - A 50/50 split gives both pools meaningful starting capital (13% of NAV
   232→  //     each), so MARP has a real buffer to rebalance AND the index gold is a

   236→  // move gold between the two pools.
   237→  const indexPaxg = paxg * 0.5;
   238→  const indexXaut = xaut * 0.5;
   239→  const reservePaxg = paxg - indexPaxg;
   240→  const reserveXaut = xaut - indexXaut;

   249→    // §14.1 — index/reserve gold split (50/50 at genesis; index locked)

   253→    reserveXaut,

   305→function seedHistory(mean: number, sd: number, n: number): number[] {
   306→  const out: number[] = [];
   307→  let last = mean;
   308→  for (let i = 0; i < n; i++) {
   309→    // simple mean-reverting walk
   310→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   311→    last = last + (mean - last) * 0.1 + shock;
   312→    out.push(Math.max(0.01, last));
   313→  }
   314→  return out;
   315→}

   317→// --- §14.1 Constitutional Separation helpers ---------------------------------
   318→// Maintain the §14.1 invariant: paxg = indexPaxg + reservePaxg (and same for
   319→// xaut). Index gold is LOCKED (only changes via commitIndexGold); reserve gold
   320→// is the buffer MARP rebalances. These two helpers are the only sanctioned
   321→// ways to keep the splits consistent after any mutation to total gold:
   322→
   323→/** Re-derive reserve gold from total − index (call after the legacy path or
   324→ *  concentration optimizer mutates `s.paxg` / `s.xaut` directly). Index gold
   325→ *  stays locked. */
   326→export function syncReserveFromTotal(s: ReserveState): void {
   327→  s.reservePaxg = Math.max(0, s.paxg - s.indexPaxg);
   328→  s.reserveXaut = Math.max(0, s.xaut - s.indexXaut);
   329→}
   330→
   331→/** Re-derive total gold from index + reserve (call after the MARP path mutates
   332→ *  `s.reservePaxg` / `s.reserveXaut`). Index gold stays locked. */
   333→export function syncTotalFromReserve(s: ReserveState): void {
   334→  s.paxg = s.indexPaxg + Math.max(0, s.reservePaxg);
   335→  s.xaut = s.indexXaut + Math.max(0, s.reserveXaut);
   336→}

   338→/** §14.1 commitIndexGold — the keeper-role equivalent in the TS engine.
   339→ *  Sets the index gold holdings (PAXG + XAUT) that back the Strategic Prior
   340→ *  Gold weight (26%). Only this function may move gold between the index
   341→ *  vault and the reserve buffer. Recomputes the totals via the §14.1 invariant
   342→ *  (reserve = total − index). Emits an IndexGoldCommitted event-equivalent
   343→ *  (console log — production should emit an on-chain event). */
   344→export function commitIndexGold(
   345→  s: ReserveState,
   346→  indexPaxg: number,
   347→  indexXaut: number,
   348→): { indexPaxg: number; indexXaut: number; reservePaxg: number; reserveXaut: number; totalPaxg: number; totalXaut: number } {
   349→  const safeIndexPaxg = Math.max(0, indexPaxg);
   350→  const safeIndexXaut = Math.max(0, indexXaut);
   351→  s.indexPaxg = safeIndexPaxg;
   352→  s.indexXaut = safeIndexXaut;
   353→  // Reserve = max(0, total − index) — never let reserve go negative even if
   354→  // the keeper over-commits (log a warning in that case).
   355→  if (s.paxg < safeIndexPaxg) {
   356→    console.warn(`[mtq-engine] commitIndexGold: indexPaxg ${safeIndexPaxg.toFixed(4)} > total paxg ${s.paxg.toFixed(4)} — reserve went to 0; total bumped to match index.`);
   357→    s.paxg = safeIndexPaxg;
   358→  }
   359→  if (s.xaut < safeIndexXaut) {
   360→    console.warn(`[mtq-engine] commitIndexGold: indexXaut ${safeIndexXaut.toFixed(4)} > total xaut ${s.xaut.toFixed(4)} — reserve went to 0; total bumped to match index.`);
   361→    s.xaut = safeIndexXaut;
   362→  }
   363→  syncReserveFromTotal(s);
   364→  s.updatedAt = Date.now();
   365→  console.log(`[mtq-engine] IndexGoldCommitted: indexPaxg=${s.indexPaxg.toFixed(4)}, indexXaut=${s.indexXaut.toFixed(4)}, reservePaxg=${s.reservePaxg.toFixed(4)}, reserveXaut=${s.reserveXaut.toFixed(4)}`);
   366→  return {
   367→    indexPaxg: s.indexPaxg,
   368→    indexXaut: s.indexXaut,
   369→    reservePaxg: s.reservePaxg,
   370→    reserveXaut: s.reserveXaut,
   371→    totalPaxg: s.paxg,
   372→    totalXaut: s.xaut,
   373→  };
   374→}

   376→// --- §2 GFB Index (v1.0 — 7-component chain-linked) -------------------------
   377→// GFB_t = Σ_i W^Prior_i × P_{i,t}  (then normalised by GFB_BASE_DENOMINATOR so
   378→// GFB = 1.00 exactly at the base date). The 7 components are the Strategic
   379→// Prior: USD, EUR, JPY, GBP, CNY, CHF, Gold. Gold is now a first-class index
   380→// component (P_Gold = XAU_USD).
   381→export function computeGfbIndex(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): number {
   382→  // Raw USD value of the basket at time t. USD is the unit of account (price 1.0);
   383→  // every other component contributes its USD-equivalent via its FX rate.
   384→  const numerator =
   385→    STRATEGIC_PRIOR.USD  * 1.0 +
   386→    STRATEGIC_PRIOR.EUR  * fx.EUR_USD +
   387→    STRATEGIC_PRIOR.JPY  * fx.JPY_USD +
   388→    STRATEGIC_PRIOR.GBP  * fx.GBP_USD +
   389→    STRATEGIC_PRIOR.CNY  * fx.CNY_USD +
   390→    STRATEGIC_PRIOR.CHF  * fx.CHF_USD +
   391→    STRATEGIC_PRIOR.Gold * fx.XAU_USD;
   392→  // Normalised: GFB_t = numerator / GFB_BASE_DENOMINATOR  (=1.0 at base date).
   393→  // GFB_BASE_DENOMINATOR is the v1.0 chain-linked denominator (includes Gold
   394→  // and CHF at base fixings — see blueprint.ts).
   395→  return numerator / GFB_BASE_DENOMINATOR;
   396→}

   408→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   409→// USD value of each holding (gross), then apply haircuts for NAV.
   410→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   411→  const usdGross = (s.usdc + s.usdp + s.usdt) * 1.0; // USDC + USDP + USDT combined
   412→  const eurGross = s.eurc * fx.EUR_USD;
   413→  const gbpGross = s.gbp * fx.GBP_USD;
   414→  const jpyGross = s.jpy * fx.JPY_USD;
   415→  const cnyGross = s.cny * fx.CNY_USD;
   416→  const chfGross = s.chf * fx.CHF_USD; // NEW v1.0 — CHF is a first-class reserve component
   417→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross + chfGross;
   418→
   419→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   420→  const goldRef = fx.XAU_USD;
   421→  const goldExec = fx.XAU_USD * 0.995;
   422→  const goldPrice = Math.min(goldRef, goldExec);
   423→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   424→
   425→  // Net values (after haircuts)
   426→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   427→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   428→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   429→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   430→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   431→  const chfNet = chfGross * (1 - HAIRCUTS.CHF); // NEW v1.0
   432→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   433→
   434→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet + goldNet;
   435→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet;
   436→
   437→  // Per-issuer breakdown (for concentration visualization)
   438→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   439→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   440→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   441→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);
   442→  const xautUsd = s.xaut * goldPrice * (1 - HAIRCUTS.XAU);
   443→
   444→  // §14.1 — index vs reserve gold split (USD net of haircut).
   445→  // Index gold is locked (backs the 26% Strategic Prior Gold weight); reserve
   446→  // gold is the buffer MARP rebalances. The total gold (= index + reserve) is
   447→  // what serves the index weight for MASE/MARP observed-weight purposes.
   448→  const indexPaxgUsd = s.indexPaxg * goldPrice * (1 - HAIRCUTS.XAU);
   449→  const indexXautUsd = s.indexXaut * goldPrice * (1 - HAIRCUTS.XAU);
   450→  const reservePaxgUsd = (s.reservePaxg >= 0 ? s.reservePaxg : Math.max(0, s.paxg - s.indexPaxg)) * goldPrice * (1 - HAIRCUTS.XAU);
   451→  const reserveXautUsd = (s.reserveXaut >= 0 ? s.reserveXaut : Math.max(0, s.xaut - s.indexXaut)) * goldPrice * (1 - HAIRCUTS.XAU);
   452→  const indexGoldNet = indexPaxgUsd + indexXautUsd;
   453→  const reserveGoldNet = reservePaxgUsd + reserveXautUsd;
   454→
   455→  return {
   456→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, chfGross, fiatGross,
   457→    goldGross, goldPrice,
   458→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, chfNet, goldNet,
   459→    nav, fiatNet,
   460→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   461→    // §14.1 gold split
   462→    indexPaxgUsd, indexXautUsd, reservePaxgUsd, reserveXautUsd,
   463→    indexGoldNet, reserveGoldNet,
   464→  };
   465→}

   497→// --- §6 Adaptive Macro Engine -----------------------------------------------
   498→function rollingStats(hist: number[]): { mean: number; sd: number } {
   499→  const filled = hist.filter((x) => x > 0);
   500→  if (filled.length === 0) return { mean: 0, sd: 0 };
   501→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   502→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   503→  return { mean, sd: Math.sqrt(variance) };
   504→}
   505→
   506→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   507→  const v = rollingStats(s.vixHistory);
   508→  const d = rollingStats(s.dxyHistory);
   509→  const eps = 1e-9;
   510→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   511→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   512→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   513→}

   515→export function computeRawTargetTheta(zVix: number, zDxy: number): number {
   516→  const theta = ALPHA * zVix + BETA * zDxy;
   517→  return clampSym(theta, -THETA_MAX, THETA_MAX);
   518→}

   520→export function computeRawTargetGoldWeight(zVix: number, zDxy: number): number {
   521→  const theta = computeRawTargetTheta(zVix, zDxy);
   522→  return clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);

   525→export function smoothGoldWeight(prev: number, target: number): number {
   526→  if (prev <= 0) return target;
   527→  return SMOOTHING_LAMBDA * target + (1 - SMOOTHING_LAMBDA) * prev;

   530→// --- §8 Dynamic Buffer -------------------------------------------------------
   531→export function bufferBaseGoldRatio(rr: number): number {
   532→  if (rr >= RR_TARGET) return BUFFER_GOLD_BASE; // 62.5% → total 26.25%
   533→  if (rr >= RR_STRESS) return BUFFER_GOLD_STRESS; // 85% → total 28.5%
   534→  return BUFFER_GOLD_EMERGENCY; // 100% → total 30%

   537→export function currentBufferGoldRatio(s: ReserveState): number {
   538→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   539→  if (s.rampStartAt === 0) {
   540→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   541→  }
   542→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   543→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   544→  const progress = elapsedH / RAMP_DURATION_HOURS;
   545→  const from = s.rampFromRatio;
   546→  const to = s.rampToRatio;
   547→  return from + (to - from) * progress;
   548→}

   550→export function bufferBaseFromRatio(rr: number): "BASE" | "STRESS" | "EMERGENCY" {
   551→  if (rr >= RR_TARGET) return "BASE";
   552→  if (rr >= RR_STRESS) return "STRESS";
   553→  return "EMERGENCY";
   554→}

   556→// Unified target gold weight (tokenomics reconciliation):
   557→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   558→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   559→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   560→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   561→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   562→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   563→}

   565→// --- §7 Rebalancing decision (simplified cost-benefit) ----------------------
   566→export interface RebalanceDecision {
   567→  shouldRebalance: boolean;
   568→  direction: 0 | 1 | -1;
   569→  observedGoldWeight: number;
   570→  targetGoldWeight: number;
   571→  deviation: number; // observed - target (signed)
   572→  tradeUsd: number;
   573→  reason: string;
   574→}

   576→export function maybeResetDailyTurnover(s: ReserveState): void {
   577→  if (Date.now() - s.lastTurnoverResetAt >= 86_400_000) {
   578→    s.dailyTurnoverUsd = 0;
   579→    s.lastTurnoverResetAt = Date.now();
   580→  }
   581→}

   583→export function evaluateRebalance(
   584→  s: ReserveState,
   585→  vals: { nav: number; goldNet: number; fiatNet: number },
   586→  rr: number,
   587→): RebalanceDecision {
   588→  const nav = vals.nav;
   589→  if (nav <= 0) {
   590→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   591→  }
   592→  const observed = vals.goldNet / nav;
   593→  const target = computeTargetGoldWeight(s, rr);
   594→  const deviation = observed - target;
   595→
   596→  if (Math.abs(deviation) < 0.005) {
   597→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   598→  }
   599→
   600→  // Direction lock (overridden in stress)
   601→  const inStress = rr < RR_STRESS;
   602→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   603→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   604→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   605→    }
   606→  }
   607→
   608→  // Trade sizing
   609→  const surplusUsd = Math.abs(deviation) * nav;
   610→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   611→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   612→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   613→  maybeResetDailyTurnover(s);
   614→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   615→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   616→
   617→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   618→  const estCost = tradeUsd * 0.001 + 50;
   619→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   620→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   621→
   622→  if (benefit <= estCost) {
   623→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   624→  }
   625→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   626→}

   628→// Apply a rebalance trade in USD terms (mutates state).
   629→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   630→  if (!decision.shouldRebalance) return;
   631→  const usd = decision.tradeUsd;
   632→  if (decision.direction === -1) {
   633→    // sell gold → buy USDC (simplified: convert to USDC)
   634→    const goldUnits = usd / goldPrice;
   635→    s.paxg = Math.max(0, s.paxg - goldUnits);
   636→    // §14.1 — the sold gold comes from the RESERVE buffer (index gold is locked).
   637→    // Maintain the invariant paxg = indexPaxg + reservePaxg.
   638→    s.reservePaxg = Math.max(0, s.reservePaxg - goldUnits);
   639→    s.usdc += usd;
   640→  } else {
   641→    // buy gold → spend USDC
   642→    s.usdc = Math.max(0, s.usdc - usd);
   643→    const goldUnits = usd / goldPrice;
   644→    s.paxg += goldUnits;
   645→    // §14.1 — the bought gold goes to the RESERVE buffer (index gold is locked).
   646→    s.reservePaxg += goldUnits;
   647→  }
   648→  s.lastTradeDir = decision.direction;
   649→  s.lastTradeAt = Date.now();
   650→  s.dailyTurnoverUsd += usd;
   651→  s.updatedAt = Date.now();
   652→}
   653→
   654→// --- §14.1 + §10 MARP per-component rebalance execution ---------------------
   655→// applyMarpRebalance() is the v1.0 production-target rebalance path. Unlike the
   656→// legacy §7 single-direction `applyRebalanceTrade` (which only trades gold ↔
   657→// USDC), this takes the per-component MarpDecision[] (output of marpDecision())
   658→// and applies each "would-execute" trade to the RESERVE holdings — index gold
   659→// is LOCKED per §14.1 and never touched here.
   660→//
   661→// Trade execution model (pilot):
   662→//   - For non-USD components (EUR/JPY/GBP/CNY/CHF/Gold): each trade pairs with
   663→//     USD as the unit of account. A "sell" of component X converts X → USD
   664→//     (split 1/3 across USDC/USDP/USDT, respecting the §5.6 concentration
   665→//     target). A "buy" of component X converts USD → X.
   666→//   - For USD: each trade pairs with Gold (the other deep pool). A "sell" of
   667→//     USD converts USD → Gold (split 50/50 PAXG/XAUT in the RESERVE buffer).
   668→//     A "buy" of USD converts Gold → USD.
   669→//   - For Gold: a "sell" reduces reservePaxg + reserveXaut (50/50); a "buy"
   670→//     increases them. Index gold is never touched.
   671→//
   672→// Constraints enforced (§10):
   673→//   - MAX_DAILY_TURNOVER (5% of NAV) — trades that would breach are skipped.
   674→//   - DIRECTION_LOCK_HOURS (24h) — trades that reverse the last direction
   675→//     within 24h are skipped (whipsaw guard).
   676→//   - Only decisions with shouldTrade=true AND level>=6 are executed
   677→//     (level 1-5 are advisory: no-trade zone / low urgency / cost-benefit
   678→//     fail / turnover cap).
   679→//
   680→// Mutates: s.usdc/usdp/usdt, s.eurc/gbp/jpy/cny/chf, s.reservePaxg/reserveXaut
   681→//          (and re-syncs s.paxg/xaut totals via the §14.1 invariant),
   682→//          s.lastTradeDir/lastTradeAt, s.dailyTurnoverUsd, s.updatedAt.
   683→// Index gold (s.indexPaxg, s.indexXaut) is NEVER touched.
   684→export interface MarpExecDecision {
   685→  shouldTrade: boolean;
   686→  component: string;
   687→  direction: string;
   688→  tradeUsd: number;
   689→  level: number;
   690→}
   691→
   692→export interface MarpExecResult {
   693→  appliedCount: number;
   694→  skippedCount: number;
   695→  totalTradeUsd: number;
   696→  path: 'legacy' | 'marp';
   697→  // Per-decision trace: which were applied vs skipped (and why).
   698→  traces: { component: string; direction: string; tradeUsd: number; level: number; applied: boolean; skipReason?: string }[];
   699→}
   700→
   701→export function applyMarpRebalance(
   702→  s: ReserveState,
   703→  fx: FxRates,
   704→  decisions: MarpExecDecision[],
   705→): MarpExecResult {
   706→  let appliedCount = 0;
   707→  let skippedCount = 0;
   708→  let totalTradeUsd = 0;
   709→  const traces: MarpExecResult['traces'] = [];
   710→
   711→  maybeResetDailyTurnover(s);
   712→  const vals = reserveAssetValues(s, fx);
   713→  const nav = vals.nav;
   714→  if (nav <= 0) {
   715→    // Nothing to do — surface a single skip trace for visibility.
   716→    for (const d of decisions) {
   717→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'NAV ≤ 0' });
   718→    }
   719→    return { appliedCount: 0, skippedCount: decisions.length, totalTradeUsd: 0, path: 'marp', traces };
   720→  }
   721→  const goldPrice = vals.goldPrice;
   722→
   723→  for (const d of decisions) {
   724→    // Level gate: only execute shouldTrade && level >= 6 decisions.
   725→    if (!d.shouldTrade || d.level < 6) {
   726→      skippedCount++;
   727→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: `level ${d.level} (advisory)` });
   728→      continue;
   729→    }
   730→    let tradeUsd = d.tradeUsd;
   731→    // Enforce MAX_DAILY_TURNOVER (5% of NAV)
   732→    const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   733→    if (tradeUsd > remainingDaily) {
   734→      tradeUsd = Math.max(0, remainingDaily);
   735→    }
   736→    if (tradeUsd <= 0) {
   737→      skippedCount++;
   738→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'daily turnover cap' });
   739→      continue;
   740→    }
   741→    // Direction (+1 buy / -1 sell / 0 hold)
   742→    const dir: 0 | 1 | -1 = d.direction === 'buy' ? 1 : d.direction === 'sell' ? -1 : 0;
   743→    if (dir === 0) {
   744→      skippedCount++;
   745→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'hold' });
   746→      continue;
   747→    }
   748→    // Enforce DIRECTION_LOCK_HOURS (24h whipsaw guard)
   749→    if (Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   750→      if ((dir === 1 && s.lastTradeDir === -1) || (dir === -1 && s.lastTradeDir === 1)) {
   751→        skippedCount++;
   752→        traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'direction lock (24h whipsaw guard)' });
   753→        continue;
   754→      }
   755→    }
   756→
   757→    // Apply the trade to the RESERVE holdings.
   758→    if (d.component === 'Gold') {
   759→      // Gold ↔ USD. Split 50/50 across reservePaxg + reserveXaut.
   760→      const goldUnits = tradeUsd / goldPrice;
   761→      const half = goldUnits / 2;
   762→      if (dir === -1) {
   763→        // sell gold → USD
   764→        s.reservePaxg = Math.max(0, s.reservePaxg - half);
   765→        s.reserveXaut = Math.max(0, s.reserveXaut - half);
   766→        const usdThird = tradeUsd / 3;
   767→        s.usdc += usdThird;
   768→        s.usdp += usdThird;
   769→        s.usdt += usdThird;
   770→      } else {
   771→        // buy gold ← USD
   772→        const usdThird = tradeUsd / 3;
   773→        s.usdc = Math.max(0, s.usdc - usdThird);
   774→        s.usdp = Math.max(0, s.usdp - usdThird);
   775→        s.usdt = Math.max(0, s.usdt - usdThird);
   776→        s.reservePaxg += half;
   777→        s.reserveXaut += half;
   778→      }
   779→      // §14.1 — re-sync total paxg/xaut = index + reserve
   780→      syncTotalFromReserve(s);
   781→    } else if (d.component === 'USD') {
   782→      // USD ↔ Gold. Split 50/50 across reservePaxg + reserveXaut.
   783→      const goldUnits = tradeUsd / goldPrice;
   784→      const half = goldUnits / 2;
   785→      if (dir === -1) {
   786→        // sell USD → gold
   787→        const usdThird = tradeUsd / 3;
   788→        s.usdc = Math.max(0, s.usdc - usdThird);
   789→        s.usdp = Math.max(0, s.usdp - usdThird);
   790→        s.usdt = Math.max(0, s.usdt - usdThird);
   791→        s.reservePaxg += half;
   792→        s.reserveXaut += half;
   793→      } else {
   794→        // buy USD ← gold
   795→        s.reservePaxg = Math.max(0, s.reservePaxg - half);
   796→        s.reserveXaut = Math.max(0, s.reserveXaut - half);
   797→        const usdThird = tradeUsd / 3;
   798→        s.usdc += usdThird;
   799→        s.usdp += usdThird;
   800→        s.usdt += usdThird;
   801→      }
   802→      // §14.1 — re-sync total paxg/xaut = index + reserve
   803→      syncTotalFromReserve(s);
   804→    } else {
   805→      // Non-USD fiat component (EUR/JPY/GBP/CNY/CHF) ↔ USD.
   806→      // Convert tradeUsd to native units via the FX rate.
   807→      const fxRate =
   808→        d.component === 'EUR' ? fx.EUR_USD :
   809→        d.component === 'JPY' ? fx.JPY_USD :
   810→        d.component === 'GBP' ? fx.GBP_USD :
   811→        d.component === 'CNY' ? fx.CNY_USD :
   812→        d.component === 'CHF' ? fx.CHF_USD : 1.0;
   813→      const nativeUnits = tradeUsd / fxRate;
   814→      const usdThird = tradeUsd / 3;
   815→      if (dir === -1) {
   816→        // sell component → USD
   817→        if (d.component === 'EUR') s.eurc = Math.max(0, s.eurc - nativeUnits);
   818→        else if (d.component === 'JPY') s.jpy = Math.max(0, s.jpy - nativeUnits);
   819→        else if (d.component === 'GBP') s.gbp = Math.max(0, s.gbp - nativeUnits);
   820→        else if (d.component === 'CNY') s.cny = Math.max(0, s.cny - nativeUnits);
   821→        else if (d.component === 'CHF') s.chf = Math.max(0, s.chf - nativeUnits);
   822→        s.usdc += usdThird;
   823→        s.usdp += usdThird;
   824→        s.usdt += usdThird;
   825→      } else {
   826→        // buy component ← USD
   827→        s.usdc = Math.max(0, s.usdc - usdThird);
   828→        s.usdp = Math.max(0, s.usdp - usdThird);
   829→        s.usdt = Math.max(0, s.usdt - usdThird);
   830→        if (d.component === 'EUR') s.eurc += nativeUnits;
   831→        else if (d.component === 'JPY') s.jpy += nativeUnits;
   832→        else if (d.component === 'GBP') s.gbp += nativeUnits;
   833→        else if (d.component === 'CNY') s.cny += nativeUnits;
   834→        else if (d.component === 'CHF') s.chf += nativeUnits;
   835→      }
   836→    }
   837→
   838→    s.lastTradeDir = dir;
   839→    s.lastTradeAt = Date.now();
   840→    s.dailyTurnoverUsd += tradeUsd;
   841→    totalTradeUsd += tradeUsd;
   842→    appliedCount++;
   843→    traces.push({ component: d.component, direction: d.direction, tradeUsd, level: d.level, applied: true });
   844→  }
   845→
   846→  s.updatedAt = Date.now();
   847→  return { appliedCount, skippedCount, totalTradeUsd, path: 'marp', traces };
   848→}

   850→// --- §11 Geopolitical Eject -------------------------------------------------
   851→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   852→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   853→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   854→  const drift = (cur: number, amp: number) => {
   855→    const shock = (Math.random() - 0.5) * 2 * amp;
   856→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   857→  };
   858→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   859→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   860→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   861→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   862→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   863→
   864→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   865→  for (const c of currencies) {
   866→    const p = s.pegHealth[c];
   867→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   868→    if (outside) {
   869→      s.depegHours[c] += dtHours;
   870→    } else {
   871→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   872→    }
   873→    // Staged ladder
   874→    const h = s.depegHours[c];
   875→    let stage = 0;
   876→    if (h > 48) stage = 3;
   877→    else if (h > 24) stage = 2;
   878→    else if (h > 12) stage = 1;
   879→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   880→    s.ejectStage[c] = stage;
   881→  }
   882→}

   898→export function applyMint(
   899→  s: ReserveState,

   902→  inputUsd: number,
   903→): MintResult {
   904→  const gfb = computeGfbIndex(fx);
   905→  const price = computeMtqPrice(gfb);
   906→  if (!priceInSafetyBand(price)) {
   907→    return { ok: false, reason: "MTQ price outside safety band (0.50–2.00 USD); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   908→  }
   909→  if (status === "DEFENSIVE" || status === "EMERGENCY") {
   910→    return { ok: false, reason: `${status}: minting paused by Risk State Machine.`, inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   911→  }
   912→  const throttle = status === "CAUTION" ? 0.5 : status === "RECOVERY" ? 0.25 : 1.0;
   913→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
   914→  const netUsd = inputUsd - feeUsd;
   915→  const minted = (netUsd / price) * throttle;
   916→
   917→  // Mutate reserve: deposit USDC (split across 3 USD issuers per concentration policy),
   918→  // mint MTQ to user (circulating supply rises). The concentration rebalancer in the
   919→  // tick loop will re-optimize the split if any issuer exceeds 25%.
   920→  const usdThird = netUsd / 3;
   921→  s.usdc += usdThird;
   922→  s.usdp += usdThird;
   923→  s.usdt += usdThird;
   924→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet (§13.2 sweep target)
   925→  s.totalSupply += minted;
   926→  s.updatedAt = Date.now();
   927→
   928→  const vals = reserveAssetValues(s, fx);
   929→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
   930→  return {
   931→    ok: true,
   932→    inputUsd,
   933→    feeUsd,
   934→    netUsd,
   935→    mtqPrice: price,
   936→    mtqMinted: minted,
   937→    throttleFactor: throttle,
   938→    newCirculatingSupply: circulatingSupply(s),

   976→  s: ReserveState,

   980→): RedeemResult {

   982→  const price = computeMtqPrice(gfb);

At RR>100% the §12.2 literal pays redeemers MORE than §3.4.2, draining the buffer surplus via arbitrage. We settle on the arbitrage-safe §3.4.2 value; the §12.2 figure is shown for audit only.";
   989→  if (inputMtq > circ) {
   990→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)), auditNavPerToken, auditGrossUsdNav, auditDeltaUsd: auditGrossUsdNav, auditNote };
   991→  }

   993→  let feeBps = REDEEM_FEE_BPS;
   994→  if (status === "DEFENSIVE") feeBps = 50;
   995→  if (status === "EMERGENCY") feeBps = 200;
   996→  const grossUsd = inputMtq * price; // redemption is priced against the GFB Index (§3.4.2)
   997→  const feeUsd = grossUsd * (feeBps / 10_000);
   998→  const netUsd = grossUsd - feeUsd;
   999→
  1000→  // Release actual reserve composition proportionally (§12.2):
  1001→  // gold portion = netUsd * W_target (legacy v1.2 buffer-derived target —
  1002→  // retained until the MASE ensemble replaces it). fiat portion split per the
  1003→  // 6 non-gold Strategic Prior components (USD/EUR/JPY/GBP/CNY/CHF) renormalised
  1004→  // to the non-gold total.

  1009→  const fiatUsd = netUsd * (1 - wGold);
  1010→  const goldPrice = vals.goldPrice;
  1011→  const goldPaxg = goldUsd / goldPrice;
  1012→
  1013→  // v1.0: Strategic Prior weights for the 6 non-gold components, renormalised
  1014→  // to the non-gold total (0.27 + 0.20 + 0.09 + 0.08 + 0.05 + 0.05 = 0.74).
  1015→  const basketFiatTotal =
  1016→    STRATEGIC_PRIOR.USD + STRATEGIC_PRIOR.EUR + STRATEGIC_PRIOR.JPY +
  1017→    STRATEGIC_PRIOR.GBP + STRATEGIC_PRIOR.CNY + STRATEGIC_PRIOR.CHF;
  1018→  const w = (k: keyof typeof STRATEGIC_PRIOR) => STRATEGIC_PRIOR[k] / basketFiatTotal;
  1019→  const basket: RedeemBasket[] = [
  1020→    { currency: "USD", token: "USDC", usdValue: fiatUsd * w("USD"), nativeAmount: fiatUsd * w("USD"), weight: STRATEGIC_PRIOR.USD },
  1021→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * w("EUR"), nativeAmount: (fiatUsd * w("EUR")) / fx.EUR_USD, weight: STRATEGIC_PRIOR.EUR },
  1022→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * w("JPY"), nativeAmount: (fiatUsd * w("JPY")) / fx.JPY_USD, weight: STRATEGIC_PRIOR.JPY },
  1023→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * w("GBP"), nativeAmount: (fiatUsd * w("GBP")) / fx.GBP_USD, weight: STRATEGIC_PRIOR.GBP },
  1024→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * w("CNY"), nativeAmount: (fiatUsd * w("CNY")) / fx.CNY_USD, weight: STRATEGIC_PRIOR.CNY },
  1025→    { currency: "CHF", token: "CHF₿", usdValue: fiatUsd * w("CHF"), nativeAmount: (fiatUsd * w("CHF")) / fx.CHF_USD, weight: STRATEGIC_PRIOR.CHF }, // NEW v1.0
  1026→  ];
  1027→
  1028→  // Mutate reserve: burn MTQ from circulating (reduce totalSupply), release assets
  1029→  // USD released proportionally across USDC + USDP + USDT; gold across PAXG + XAUT
  1030→  s.totalSupply -= inputMtq;
  1031→  const usdRelease = basket[0].nativeAmount;
  1032→  const usdThirdRelease = usdRelease / 3;
  1033→  s.usdc = Math.max(0, s.usdc - usdThirdRelease);
  1034→  s.usdp = Math.max(0, s.usdp - usdThirdRelease);
  1035→  s.usdt = Math.max(0, s.usdt - usdThirdRelease);
  1036→  s.eurc = Math.max(0, s.eurc - basket[1].nativeAmount);
  1037→  s.jpy  = Math.max(0, s.jpy  - basket[2].nativeAmount);
  1038→  s.gbp  = Math.max(0, s.gbp  - basket[3].nativeAmount);
  1039→  s.cny  = Math.max(0, s.cny  - basket[4].nativeAmount);
  1040→  s.chf  = Math.max(0, s.chf  - basket[5].nativeAmount); // NEW v1.0
  1041→  // Gold released 50/50 PAXG + XAUT
  1042→  const goldHalf = goldPaxg / 2;
  1043→  s.paxg = Math.max(0, s.paxg - goldHalf);
  1044→  s.xaut = Math.max(0, s.xaut - goldHalf);
  1045→  // §14.1 — gold released to the redeemer comes from the RESERVE buffer first
  1046→  // (index gold is locked). syncReserveFromTotal re-derives reserve = total − index
  1047→  // so the invariant paxg = indexPaxg + reservePaxg is preserved.
  1048→  syncReserveFromTotal(s);
  1049→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet
  1050→  s.updatedAt = Date.now();
  1051→
  1052→  const newVals = reserveAssetValues(s, fx);
  1053→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
  1054→  return {
  1055→    ok: true,
  1056→    inputMtq,
  1057→    mtqPrice: price,
  1058→    grossUsd,
  1059→    feeBps,
  1060→    feeUsd,
  1061→    netUsd,
  1062→    goldUsd,
  1063→    goldPaxg,
  1064→    basket,
  1065→    newCirculatingSupply: circulatingSupply(s),
  1066→    newReserveRatio: newRr,
  1067→    auditNavPerToken: circ > 0 ? vals0.nav / circ : 0,
  1068→    auditGrossUsdNav,
  1069→    auditDeltaUsd: auditGrossUsdNav - grossUsd,
  1070→    auditNote,
  1071→  };
  1072→}

  1259→// --- §6/§7/§8.4/§10 v1.0 MASE + 4-state weights + MARP ---------------------
  1260→// The Multi-Asset Stochastic Ensemble (§6/§7) runs 6 candidate models, blends
  1261→// them into a single target weight vector, clamps to per-component
  1262→// admissibility envelopes (§8.1), and EMA-smooths toward the constrained target
  1263→// (§8.4). MARP (§10) then decides per-component whether the deviation between
  1264→// the smoothed target and the observed execution weight justifies a trade
  1265→// (urgency test → no-trade zone → cost-benefit gate → partial correction →
  1266→// daily-turnover cap → execute).
  1267→//
  1268→// `advanceMase(s, fx)` is called ONCE per tick from the pilot-state tick loop.
  1269→// It is the only function that mutates `s.maseSmoothed`. `buildMaseSnapshot()`
  1270→// is a pure helper used by `computeSnapshot()` — it recomputes target/smoothed/
  1271→// MARP from the current fx + vals using the persisted prev-smoothed as the EMA
  1272→// prior, so multiple computeSnapshot calls within a single tick return the
  1273→// same value (idempotent).
  1274→
  1275→interface MaseSnapshotData {
  1276→  mase: NonNullable<MetricsSnapshot["mase"]>;
  1277→  weightStates: NonNullable<MetricsSnapshot["weightStates"]>;
  1278→  marp: NonNullable<MetricsSnapshot["marp"]>;
  1279→  // §14.1 — read-only projection of what MARP WOULD do this tick.
  1280→  marpExecution: NonNullable<MetricsSnapshot["marpExecution"]>;
  1281→  envelopes: MetricsSnapshot["envelopes"];
  1282→}

  1284→/** Estimate per-component annualized volatility from the engine's macro state.
  1285→ *  Pilot approximation: scale baseline vols by VIX / 18.5 (the "normal" VIX
  1286→ *  baseline), so a 30 VIX roughly doubles estimated vols and a 12 VIX halves
  1287→ *  them. Production should use a rolling covariance from real price history. */
  1288→function estimateVolsFromState(s: ReserveState): VolatilityData {
  1289→  const vix = s.lastVix > 0 ? s.lastVix : 18.5;
  1290→  const f = vix / 18.5;
  1291→  return {
  1292→    USD: 0.05 * f,
  1293→    EUR: 0.10 * f,
  1294→    JPY: 0.12 * f,
  1295→    GBP: 0.11 * f,
  1296→    CNY: 0.09 * f,
  1297→    CHF: 0.10 * f,
  1298→    Gold: 0.15 * f,
  1299→  };
  1300→}

  1302→/** Detect the market regime from VIX (primary) + DXY (secondary).
  1303→ *  0 = calm, 1 = normal, 2 = stress, 3 = crisis.
  1304→ *  VIX thresholds: <15 calm, <22 normal, <30 stress, ≥30 crisis.
  1305→ *  DXY extremes (<88 or >115) bump the regime up to ≥ stress. */
  1306→function detectRegime(vix: number, dxy: number, goldVol: number): MarketRegime {
  1307→  let r: 0 | 1 | 2 | 3 = 1;
  1308→  if (vix >= 30) r = 3;
  1309→  else if (vix >= 22) r = 2;
  1310→  else if (vix >= 15) r = 1;
  1311→  else r = 0;
  1312→  if ((dxy >= 115 || dxy <= 88) && r < 2) r = 2;
  1313→  return { regime: r, vix, dxy, goldVol };
  1314→}

  1329→/** Build the observed execution-weight vector from the actual reserve composition.
  1330→ *  §14.1 note: the Gold component's observed weight uses the TOTAL gold
  1331→ *  (= indexPaxg + indexXaut + reservePaxg + reserveXaut, USD-net). Both the
  1332→ *  index gold (locked) and the reserve gold (MARP-rebalanced) serve the
  1333→ *  Strategic Prior Gold weight (26%) — but only the RESERVE gold is available
  1334→ *  for MARP to rebalance. The index gold is locked and only changes when
  1335→ *  commitIndexGold() is called (keeper-role equivalent). */
  1336→function buildObservedWeights(
  1337→  vals: ReturnType<typeof reserveAssetValues>,
  1338→  nav: number,

  1343→  return {

  1356→/** Pure helper — computes the MASE + 4-state + MARP + envelope snapshot data.
  1357→ *  Reads `s.maseSmoothed` (does NOT mutate). */
  1358→function buildMaseSnapshot(
  1359→  s: ReserveState,
  1360→  fx: FxSnapshot,
  1361→  vals: ReturnType<typeof reserveAssetValues>,
  1362→  nav: number,
  1363→  rr: number,
  1364→): MaseSnapshotData {
  1365→  const prices = buildPriceData(fx);
  1366→  const vols = estimateVolsFromState(s);
  1367→  const regime = detectRegime(s.lastVix > 0 ? s.lastVix : fx.VIX, s.lastDxy > 0 ? s.lastDxy : fx.DXY, vols.Gold);
  1368→
  1369→  // §6/§7 — MASE ensemble → 6 candidate models + target weight vector
  1370→  const mase = maseEnsemble(vols, prices, null, regime);
  1371→  // §8.1 — Constrain target to per-component admissibility envelopes
  1372→  const constrained = applyEnvelopes(mase.target);
  1373→  // §8.4 — EMA-smooth toward the constrained target (uses prev persisted as prior)
  1374→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1375→  const smoothed = smoothWeights(prevSmoothed, constrained);
  1376→  // §2.3 — Execution = actual observed reserve composition
  1377→  const observed = buildObservedWeights(vals, nav);
  1378→
  1379→  // §10 — MARP per-component rebalancing decision
  1380→  const volsRecord: Record<Component, number> = {
  1381→    USD: vols.USD, EUR: vols.EUR, JPY: vols.JPY, GBP: vols.GBP,
  1382→    CNY: vols.CNY, CHF: vols.CHF, Gold: vols.Gold,
  1383→  };
  1384→  const marpDecisions: MarpDecision[] = marpDecision(observed, smoothed, nav, rr, volsRecord);
  1385→  const totalTradeUsd = marpDecisions
  1386→    .filter((d) => d.shouldTrade)
  1387→    .reduce((a, d) => a + d.tradeUsd, 0);
  1388→
  1389→  // §14.1 — per-component MARP execution summary (READ-ONLY projection).
  1390→  // appliedCount = # of decisions with shouldTrade && level >= 6 (would-execute).
  1391→  // skippedCount  = # of decisions held back (no-trade zone / low urgency /
  1392→  //                 cost-benefit fail / turnover cap / sub-level-6).
  1393→  // totalTradeUsd = Σ tradeUsd for the would-execute decisions.
  1394→  // path = which execution path is currently mutating state (mirrors s.rebalancePath).
  1395→  // NOTE: this is what MARP WOULD do based on the current decisions. The actual
  1396→  // mutation happens in applyMarpRebalance() only when USE_MARP_EXECUTION=true
  1397→  // in pilot-state.ts. applyMarpRebalance may further skip decisions that
  1398→  // breach MAX_DAILY_TURNOVER or DIRECTION_LOCK_HOURS at execution time.
  1399→  const wouldExecute = marpDecisions.filter((d) => d.shouldTrade && d.level >= 6);
  1400→  const marpExecution = {
  1401→    appliedCount: wouldExecute.length,
  1402→    skippedCount: marpDecisions.length - wouldExecute.length,
  1403→    totalTradeUsd: wouldExecute.reduce((a, d) => a + d.tradeUsd, 0),
  1404→    path: s.rebalancePath,
  1405→  };
  1406→
  1407→  // §8.1 — Admissibility envelope status per component
  1408→  // "warn" = within 10% of the band edge; "breach" = outside the envelope.
  1409→  const envelopes = COMPONENTS.map((c) => {
  1410→    const env = ADMISSIBILITY_ENVELOPES[c];
  1411→    const current = observed[c];
  1412→    const lower = env.lower;
  1413→    const upper = env.upper;
  1414→    const margin = (upper - lower) * 0.1;
  1415→    let status: "ok" | "warn" | "breach" = "ok";
  1416→    if (current < lower || current > upper) status = "breach";
  1417→    else if (current < lower + margin || current > upper - margin) status = "warn";
  1418→    return { component: c, lower, upper, current, status };
  1419→  });
  1420→
  1421→  return {
  1422→    mase: {
  1423→      models: mase.models.map((m) => ({
  1424→        id: m.id,
  1425→        name: m.name,
  1426→        weights: m.weights as Record<string, number>,
  1427→      })),
  1428→      ensembleTarget: mase.target as Record<string, number>,
  1429→    },
  1430→    weightStates: {
  1431→      prior: STRATEGIC_PRIOR as Record<string, number>,
  1432→      target: constrained as Record<string, number>,
  1433→      smoothed: smoothed as Record<string, number>,
  1434→      execution: observed as Record<string, number>,
  1435→    },
  1436→    marp: {
  1437→      decisions: marpDecisions.map((d) => ({
  1438→        shouldTrade: d.shouldTrade,
  1439→        component: d.component,
  1440→        direction: d.direction,
  1441→        tradeUsd: d.tradeUsd,
  1442→        urgency: d.urgency,
  1443→        reason: d.reason,
  1444→        level: d.level,
  1445→      })),
  1446→      totalTradeUsd,
  1447→    },
  1448→    // §14.1 — MARP execution summary (read-only projection; mirrors s.rebalancePath).
  1449→    marpExecution,
  1450→    envelopes,
  1451→  };
  1452→}

  1454→/** Advance MASE state once per tick. Mutates `s.maseSmoothed` (EMA prior for
  1455→ *  the next tick). Called from the pilot-state tick loop, AFTER advanceMacro
  1456→ *  (so lastVix/lastDxy are fresh) and BEFORE the first computeSnapshot of the
  1457→ *  tick (so the snapshot reads the freshly-persisted smoothed weights). */
  1458→export function advanceMase(s: ReserveState, fx: FxRates): void {

  1471→}

  1481→// Advance macro history (call ~ every tick; dtHours is simulated elapsed)
  1482→export function advanceMacro(s: ReserveState, vix: number, dxy: number, dtHours: number) {
  1483→  const v = clamp(vix, VIX_MIN, VIX_MAX);

  1485→  s.lastVix = v;
  1486→  s.lastDxy = d;
  1487→  s.vixHistory.push(v);
  1488→  s.dxyHistory.push(d);
  1489→  if (s.vixHistory.length > ROLLING_WINDOW_DAYS) s.vixHistory.shift();
  1490→  if (s.dxyHistory.length > ROLLING_WINDOW_DAYS) s.dxyHistory.shift();
  1491→  const vStats = rollingStats(s.vixHistory);
  1492→  const dStats = rollingStats(s.dxyHistory);
  1493→  const zVix = vStats.sd > 1e-9 ? (v - vStats.mean) / (vStats.sd + 1e-9) : 0;
  1494→  const zDxy = dStats.sd > 1e-9 ? (d - dStats.mean) / (dStats.sd + 1e-9) : 0;
  1495→  const rawTarget = computeRawTargetGoldWeight(zVix, zDxy);
  1496→  s.smoothedGoldWeight = smoothGoldWeight(s.smoothedGoldWeight, rawTarget);
  1497→}

  1521→// --- §11.3 Reintegration Score (Anti-Gaming) --------------------------------
  1522→// R_score = w1·TimeInBand(48h) + w2·LiquidityDepth + w3·OracleAgreement + w4·(1−Volatility)
  1523→// All inputs normalised to [0,1]; threshold >0.80 → staged re-purchase 25%→50%→100%.
  1524→export function updateReintegration(
  1525→  s: ReserveState,
  1526→  board: OracleBoard | null,
  1527→  dtHours: number,
  1528→) {
  1529→  const currencies: ("USD" | "EUR" | "GBP" | "JPY" | "CNY")[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
  1530→  for (const c of currencies) {
  1531→    const r = s.reintegration[c];
  1532→    // Time-in-band: increases when peg is inside [0.98,1.02]; decays when outside.
  1533→    const inBand = s.pegHealth[c] >= EJECT_DEPEG_BAND_LOWER && s.pegHealth[c] <= EJECT_DEPEG_BAND_UPPER;
  1534→    // dtHours is simulated elapsed; we approximate 48h window fraction.
  1535→    if (inBand) {
  1536→      r.timeInBand48h = Math.min(1, r.timeInBand48h + dtHours / 48);
  1537→    } else {
  1538→      r.timeInBand48h = Math.max(0, r.timeInBand48h - dtHours / 48);
  1539→    }
  1540→    // Liquidity depth: randomised around per-currency baseline (honest: simulated).
  1541→    const base = c === "USD" ? 1.0 : c === "EUR" ? 0.95 : c === "CNY" ? 0.4 : 0.5;
  1542→    r.liquidityDepth = clamp(base + (Math.random() - 0.5) * 0.1, 0, 1);
  1543→    // Oracle agreement: valid feeds / 3 (from oracle board if present, else last known)
  1544→    if (board) {
  1545→      const pairMap: Record<string, string> = { USD: "—", EUR: "EUR/USD", GBP: "GBP/USD", JPY: "JPY/USD", CNY: "CNY/USD" };
  1546→      const cons = board.pairs.find((p) => p.pair === pairMap[c]);
  1547→      if (cons) {
  1548→        r.oracleAgreement = cons.validCount / 3;
  1549→        // Volatility from oracle spread (bps → fraction)
  1550→        r.volatility = clamp(cons.spreadBps / 100, 0, 1);
  1551→      }
  1552→    }
  1553→    // R_score
  1554→    r.score =
  1555→      REINTEGRATION_WEIGHTS.w1_TimeInBand * r.timeInBand48h +
  1556→      REINTEGRATION_WEIGHTS.w2_LiquidityDepth * r.liquidityDepth +
  1557→      REINTEGRATION_WEIGHTS.w3_OracleAgreement * r.oracleAgreement +
  1558→      REINTEGRATION_WEIGHTS.w4_OneMinusVolatility * (1 - r.volatility);
  1559→    // Repurchase staging: only for currencies that were previously ejected
  1560→    if (s.ejectStage[c] > 0 && r.score > REINTEGRATION_THRESHOLD) {
  1561→      r.repurchaseStage = Math.min(3, r.repurchaseStage + 1);
  1562→      // As repurchase progresses, reduce eject stage
  1563→      if (r.repurchaseStage >= 3) {
  1564→        s.ejectStage[c] = Math.max(0, s.ejectStage[c] - 1);
  1565→        if (s.ejectStage[c] === 0) r.repurchaseStage = 0;
  1566→      }
  1567→    } else if (r.score < REINTEGRATION_THRESHOLD * 0.6) {
  1568→      r.repurchaseStage = 0;
  1569→    }
  1570→  }
  1571→}

  1590→// --- §8.5 First-Loss Waterfall ----------------------------------------------
  1591→// 5 layers: (1) Operational Surplus → (2) Buffer Fiat → (3) Buffer Gold →
  1592→// (4) Core Fiat → (5) Core Gold. Apply a loss; track consumed per layer.
  1593→export function applyLoss(s: ReserveState, vals: { fiatNet: number; goldNet: number }, lossUsd: number) {
  1594→  let remaining = lossUsd;
  1595→  s.waterfall.consumed.surplus = 0; // reset for this loss event tracking is cumulative; we add
  1596→  const layers = [
  1597→    { name: "surplus", cap: () => s.treasury.hotWalletUsd + s.waterfall.operationalSurplus, consume: (amt: number) => {
  1598→      const fromSurplus = Math.min(s.treasury.hotWalletUsd, amt);
  1599→      s.treasury.hotWalletUsd -= fromSurplus;
  1600→      s.waterfall.operationalSurplus -= Math.min(s.waterfall.operationalSurplus, amt - fromSurplus);
  1601→      s.waterfall.consumed.surplus += amt;
  1602→    } },
  1603→    { name: "bufferFiat", cap: () => vals.fiatNet * 0.10 * (1 - currentBufferGoldRatio(s)), consume: (amt: number) => { s.waterfall.consumed.bufferFiat += amt; } },
  1604→    { name: "bufferGold", cap: () => vals.goldNet * 0.10 * currentBufferGoldRatio(s), consume: (amt: number) => { s.waterfall.consumed.bufferGold += amt; } },
  1605→    { name: "coreFiat", cap: () => vals.fiatNet * 0.80, consume: (amt: number) => { s.waterfall.consumed.coreFiat += amt; } },
  1606→    { name: "coreGold", cap: () => vals.goldNet * 0.20, consume: (amt: number) => { s.waterfall.consumed.coreGold += amt; } },
  1607→  ] as const;
  1608→  for (let i = 0; i < layers.length && remaining > 0; i++) {
  1609→    const layer = layers[i];
  1610→    const cap = layer.cap();
  1611→    const consumed = Math.min(cap, remaining);
  1612→    layer.consume(consumed);
  1613→    remaining -= consumed;
  1614→    if (remaining > 0) s.waterfall.currentLayer = (i + 2) as 1 | 2 | 3 | 4 | 5;
  1615→  }
  1616→  if (remaining <= 0) s.waterfall.currentLayer = 1;
  1617→}

  1620→export function maybePriceEvent(s: ReserveState, newPrice: number) {
  1621→  if (s.lastLoggedPrice <= 0) { s.lastLoggedPrice = newPrice; return; }
  1622→  const diff = Math.abs(newPrice - s.lastLoggedPrice);
  1623→  const changePct = diff / s.lastLoggedPrice;
  1624→  if (changePct >= PRICE_EVENT_THRESHOLD) {
  1625→    s.priceEvents.unshift({ ts: Date.now(), oldPrice: s.lastLoggedPrice, newPrice, changePct });
  1626→    if (s.priceEvents.length > 20) s.priceEvents.pop();
  1627→    s.lastLoggedPrice = newPrice;
  1628→  }
  1629→}

  1647→  const bound = isBuyGold ? oracleRate * (1 + SLIPPAGE_TOLERANCE) : oracleRate * (1 - SLIPPAGE_TOLERANCE);
  1648→  const withinBound = isBuyGold ? quoteRate <= bound : quoteRate >= bound;
  1649→  return {
  1650→    provider, oracleRate, quoteRate, bound, withinBound,
  1651→    effectiveSlippageBps: Math.round(slip * 10_000),
  1652→    amountUsd,

  1657→// Adjusts the USDC/USDP split to bring the maximum issuer share under the
  1658→// --- FIX: §5.6 Multi-Issuer Concentration Optimizer -------------------------
  1659→// 3 issuers: CIRCLE (USDC + EURC), PAXOS (USDP + PAXG), TETHER (USDT + XAUT).
  1660→// EURC is fixed to Circle (no second EUR issuer yet). GBP/JPY/CNY are TBD.
  1661→// The optimizer adjusts the splittable pools (USDC/USDP/USDT, PAXG/XAUT) to
  1662→// minimize the maximum issuer share, targeting ≤ 25% (warn threshold,
  1663→// headroom under the 30% hard limit).
  1664→export interface ConcentrationFixResult {
  1665→  adjusted: boolean;
  1666→  sharesBefore: { CIRCLE: number; PAXOS: number; TETHER: number };
  1667→  sharesAfter: { CIRCLE: number; PAXOS: number; TETHER: number };
  1668→  splits: { usdc: number; usdp: number; usdt: number; paxg: number; xaut: number };
  1669→  target: number; // 0.25
  1670→  reason: string;
  1671→}

  1673→export function rebalanceForConcentration(
  1674→  s: ReserveState,
  1675→  vals: { usdcUsd: number; usdpUsd: number; usdtUsd: number; paxgUsd: number; xautUsd: number; nav: number; eurNet: number },
  1676→): ConcentrationFixResult {
  1677→  const nav = vals.nav;
  1678→  const TARGET = 0.25;
  1679→  const usdNetFactor = 1 - HAIRCUTS.USD;
  1680→  const goldNetFactor = 1 - HAIRCUTS.XAU;
  1681→
  1682→  const sharesBefore = {
  1683→    CIRCLE: nav > 0 ? (vals.usdcUsd + vals.eurNet) / nav : 0,
  1684→    PAXOS: nav > 0 ? (vals.usdpUsd + vals.paxgUsd) / nav : 0,
  1685→    TETHER: nav > 0 ? (vals.usdtUsd + vals.xautUsd) / nav : 0,
  1686→  };
  1687→  const maxBefore = Math.max(sharesBefore.CIRCLE, sharesBefore.PAXOS, sharesBefore.TETHER);
  1688→
  1689→  if (nav <= 0) {
  1690→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "NAV ≤ 0" };
  1691→  }
  1692→  if (maxBefore <= TARGET) {
  1693→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "All issuers ≤ 25% warn threshold" };
  1694→  }
  1695→
  1696→  // Total splittable pools (in gross token units)
  1697→  const totalUsdGross = s.usdc + s.usdp + s.usdt;
  1698→  const totalGoldGross = s.paxg + s.xaut;
  1699→
  1700→  // Circle's floor = EURC (fixed). Circle's USDC budget = max(0, TARGET×NAV − EURC_usd)
  1701→  const circleUsdNetBudget = Math.max(0, TARGET * nav - vals.eurNet);
  1702→  const usdcGross = Math.min(totalUsdGross, circleUsdNetBudget / usdNetFactor);
  1703→  const remainingUsdGross = Math.max(0, totalUsdGross - usdcGross);
  1704→
  1705→  // Split remaining USD equally between USDP (Paxos) and USDT (Tether)
  1706→  const usdpGross = remainingUsdGross / 2;
  1707→  const usdtGross = remainingUsdGross / 2;
  1708→
  1709→  // Now balance gold between PAXG (Paxos) and XAUT (Tether) so both ≤ 25%.
  1710→  // Clean approach: split gold 50/50 by default; only deviate if one issuer
  1711→  // would breach the 25% cap. This guarantees a real split (the previous
  1712→  // min-budget logic could dump all gold into one issuer when budgets were
  1713→  // asymmetric or when one issuer's USD share was already small).
  1714→  const paxosUsdNet = usdpGross * usdNetFactor;
  1715→  const tetherUsdNet = usdtGross * usdNetFactor;
  1716→  const halfGoldGross = totalGoldGross / 2;
  1717→  let paxgGross = halfGoldGross;
  1718→  let xautGross = halfGoldGross;
  1719→  // Cap check 1: Paxos = USDP_net + PAXG_net ≤ TARGET×NAV
  1720→  const paxosWithHalf = paxosUsdNet + halfGoldGross * goldNetFactor;
  1721→  if (paxosWithHalf > TARGET * nav) {
  1722→    const maxPaxgNet = Math.max(0, TARGET * nav - paxosUsdNet);
  1723→    paxgGross = Math.min(halfGoldGross, maxPaxgNet / goldNetFactor);
  1724→    xautGross = totalGoldGross - paxgGross;
  1725→  }
  1726→  // Cap check 2: Tether = USDT_net + XAUT_net ≤ TARGET×NAV
  1727→  const tetherWithXaut = tetherUsdNet + xautGross * goldNetFactor;
  1728→  if (tetherWithXaut > TARGET * nav) {
  1729→    const maxXautNet = Math.max(0, TARGET * nav - tetherUsdNet);
  1730→    xautGross = Math.min(xautGross, maxXautNet / goldNetFactor);
  1731→    paxgGross = totalGoldGross - xautGross;
  1732→  }
  1733→
  1734→  s.usdc = usdcGross;
  1735→  s.usdp = usdpGross;
  1736→  s.usdt = usdtGross;
  1737→  s.paxg = paxgGross;
  1738→  s.xaut = xautGross;
  1739→  // §14.1 — concentration optimizer re-splits the TOTAL PAXG/XAUT pool across
  1740→  // issuers; the index gold (locked) keeps its PAXG/XAUT split, so we re-derive
  1741→  // the reserve split as (total − index). This preserves the invariant
  1742→  // paxg = indexPaxg + reservePaxg (and same for xaut).
  1743→  syncReserveFromTotal(s);
  1744→
  1745→  const sharesAfter = {
  1746→    CIRCLE: (usdcGross * usdNetFactor + vals.eurNet) / nav,
  1747→    PAXOS: (usdpGross * usdNetFactor + paxgGross * goldNetFactor) / nav,
  1748→    TETHER: (usdtGross * usdNetFactor + xautGross * goldNetFactor) / nav,
  1749→  };
  1750→
  1751→  const maxAfter = Math.max(sharesAfter.CIRCLE, sharesAfter.PAXOS, sharesAfter.TETHER);
  1752→  return {
  1753→    adjusted: true,
  1754→    sharesBefore, sharesAfter,
  1755→    splits: { usdc: usdcGross, usdp: usdpGross, usdt: usdtGross, paxg: paxgGross, xaut: xautGross },
  1756→    target: TARGET,
  1757→    reason: maxAfter <= TARGET
  1758→      ? "Multi-issuer split: all 3 issuers ≤ 25% warn threshold"
  1759→      : `Optimized: max issuer ${(maxAfter * 100).toFixed(1)}% (Circle EURC floor limits further reduction)`,
  1760→  };
  1761→}
