   388→        uint256 allowed = allowance[from][msg.sender];
   389→        if (allowed != type(uint256).max) {
   390→            require(allowed >= amount, "MTQV2: insufficient allowance");
   391→            allowance[from][msg.sender] = allowed - amount;

   396→    function _transfer(address from, address to, uint256 amount) internal {
   397→        require(balanceOf[from] >= amount, "MTQV2: insufficient balance");
   398→        balanceOf[from] -= amount;
   399→        balanceOf[to]   += amount;
   400→        emit Transfer(from, to, amount);
   401→    }
   402→    function _mint(address to, uint256 amount) internal {
   403→        totalSupply   += amount;

   407→    function _burn(address from, uint256 amount) internal {
   408→        require(balanceOf[from] >= amount, "MTQV2: burn exceeds balance");
   409→        balanceOf[from] -= amount;
   410→        totalSupply     -= amount;
   411→        emit Transfer(from, address(0), amount);
   412→    }

   458→    // §3.4.2 + §12.1  Mint — priced against the GFB Index
   459→    //   Pull USDC (6 dec), fee 0.10%, MTQ minted = netUsd18 × 1e18 / price.
   460→    //   Throttled by status: NORMAL 100%, CAUTION 50%, RECOVERY 25%,
   461→    //   DEFENSIVE/EMERGENCY paused.
   462→    // ==================================================================
   463→    function mint(uint256 usdcAmount) external whenNotPaused returns (uint256 minted) {
   464→        require(usdcAmount > 0, "MTQV2: zero amount");
   465→        require(
   466→            protocolStatus != Status.DEFENSIVE && protocolStatus != Status.EMERGENCY,
   467→            "MTQV2: mint paused by risk state"
   468→        );
   469→
   470→        uint256 price = getMTQPriceWithGuard();
   471→
   472→        // Pull USDC from minter to reserve vault (6 decimals).
   473→        require(usdc.transferFrom(msg.sender, reserveVault, usdcAmount), "MTQV2: USDC pull failed");
   474→
   475→        // §12.1  0.10% fee (configurable via timelock)
   476→        uint256 feeUsd = usdcAmount * mintFeeBps / 10000;
   477→        uint256 netUsd = usdcAmount - feeUsd;
   478→        // Scale 6-dec USDC → 18-dec USD value, then divide by 18-dec price.
   479→        uint256 netUsd18  = netUsd * 1e12;
   480→        uint256 grossMint = netUsd18 * 1e18 / price;
   481→
   482→        // §14.1  Throttle by status
   483→        uint256 throttle = 1e18;
   484→        if (protocolStatus == Status.CAUTION)  throttle = 0.50e18;
   485→        if (protocolStatus == Status.RECOVERY) throttle = 0.25e18;
   486→        minted = grossMint * throttle / 1e18;
   487→
   488→        _mint(msg.sender, minted);
   489→        emit Mint(msg.sender, usdcAmount, feeUsd, minted, price);
   490→    }

   492→    // ==================================================================
   493→    // §3.4.2 + §12.2  Redeem — priced against the GFB Index (arbitrage-safe)
   494→    //   Burn MTQ, grossUsd18 = mtq × price / 1e18, fee by status:
   495→    //   NORMAL/CAUTION 0.15%, DEFENSIVE 0.5%, EMERGENCY 2%, net → USDC 6 dec,
   496→    //   released from reserve vault.
   497→    // ==================================================================
   498→    function redeem(uint256 mtqAmount) external whenNotPaused returns (uint256 usdcOut) {
   499→        require(mtqAmount > 0, "MTQV2: zero amount");
   500→        require(balanceOf[msg.sender] >= mtqAmount, "MTQV2: insufficient balance");
   501→
   502→        uint256 price = getMTQPrice();
   503→
   504→        // §14.1  Fee by status
   505→        uint256 feeBps = redeemFeeBps;
   506→        if (protocolStatus == Status.DEFENSIVE) feeBps = 50;   // 0.5%
   507→        if (protocolStatus == Status.EMERGENCY) feeBps = 200; // 2.0%
   508→
   509→        uint256 grossUsd18 = mtqAmount * price / 1e18;
   510→        uint256 feeUsd18   = grossUsd18 * feeBps / 10000;
   511→        uint256 netUsd18   = grossUsd18 - feeUsd18;
   512→        usdcOut            = netUsd18 / 1e12; // 18-dec → 6-dec USDC
   513→
   514→        _burn(msg.sender, mtqAmount);
   515→        // Release USDC from reserve vault to minter (vault must approve this contract).
   516→        require(usdc.transferFrom(reserveVault, msg.sender, usdcOut), "MTQV2: USDC release failed");
   517→        emit Redeem(msg.sender, mtqAmount, feeUsd18 / 1e12, usdcOut, price);
   518→    }

   531→    function envelopeLower(Component c) public pure returns (uint256) {
   532→        if (c == Component.USD)  return ENV_USD_LOWER;
   533→        if (c == Component.EUR)  return ENV_EUR_LOWER;
   534→        if (c == Component.JPY)  return ENV_JPY_LOWER;
   535→        if (c == Component.GBP)  return ENV_GBP_LOWER;
   536→        if (c == Component.CNY)  return ENV_CNY_LOWER;
   537→        if (c == Component.CHF)  return ENV_CHF_LOWER;
   538→        return ENV_GOLD_LOWER;
   539→    }
   540→    function envelopeUpper(Component c) public pure returns (uint256) {
   541→        if (c == Component.USD)  return ENV_USD_UPPER;
   542→        if (c == Component.EUR)  return ENV_EUR_UPPER;
   543→        if (c == Component.JPY)  return ENV_JPY_UPPER;
   544→        if (c == Component.GBP)  return ENV_GBP_UPPER;
   545→        if (c == Component.CNY)  return ENV_CNY_UPPER;
   546→        if (c == Component.CHF)  return ENV_CHF_UPPER;
   547→        return ENV_GOLD_UPPER;
   548→    }
   549→    function _assertEnvelope(Component c, uint256 w) internal pure {
   550→        uint256 lo = envelopeLower(c);
   551→        uint256 hi = envelopeUpper(c);
   552→        require(w >= lo && w <= hi, "MTQV2: weight outside envelope");
   553→    }

   556→    // §8.1  On-Chain MASE Weight Registry — commit (oracle/keeper only)
   557→    //   Validates each weight against its per-component admissibility envelope
   558→    //   (hard bounds) — reverts on envelope breach.
   559→    // ==================================================================
   560→    function commitWeights(
   561→        uint256[7] calldata target,
   562→        uint256[7] calldata smoothed,
   563→        uint256[7] calldata execution
   564→    ) external onlyOracleOrKeeper {
   565→        for (uint8 i = 0; i < 7; i++) {
   566→            Component c = Component(i);
   567→            _assertEnvelope(c, target[i]);
   568→            _assertEnvelope(c, smoothed[i]);
   569→            _assertEnvelope(c, execution[i]);
   570→            weights.targetWeights[i]    = target[i];
   571→            weights.smoothedWeights[i] = smoothed[i];
   572→            weights.executionWeights[i] = execution[i];
   573→        }
   574→        weights.lastUpdatedAt = block.timestamp;
   575→        emit WeightsCommitted(target, smoothed, execution, block.timestamp);
   576→    }

   596→    // ==================================================================
   597→    // §10  MARP Rebalancing Execution — keeper-callable
   598→    //   For each trade: validates it moves toward the latest committed
   599→    //   executionWeight within a 5% tolerance, enforces 24h direction lock,
   600→    //   validates total daily trade USD ≤ 5% of NAV, and updates reserve
   601→    //   holdings (on-chain mirror).
   602→    // ==================================================================
   603→    function executeRebalance(RebalanceTrade[] calldata trades) external onlyKeeper whenNotPaused {
   604→        require(weights.lastUpdatedAt > 0, "MTQV2: no weights committed");
   605→
   606→        // Reset 24h daily turnover window if elapsed
   607→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
   608→            dailyTradeUsd    = 0;
   609→            dailyTradeResetAt = block.timestamp;
   610→        }
   611→
   612→        uint256 nav = getReserveNavUsd();
   613→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
   614→
   615→        for (uint256 i = 0; i < trades.length; i++) {
   616→            RebalanceTrade calldata t = trades[i];
   617→            require(t.level >= 1 && t.level <= 6, "MTQV2: invalid MARP level");
   618→
   619→            // Hold trade — no action
   620→            if (t.direction == 0 || t.tradeUsd == 0) continue;
   621→
   622→            require(t.direction == 1 || t.direction == -1, "MTQV2: bad direction");
   623→
   624→            // §10  24h direction lock — opposite direction blocked within 24h
   625→            int256 prevDir = lastDirection[t.component];
   626→            if (prevDir != 0 && prevDir != t.direction) {
   627→                require(
   628→                    block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS,
   629→                    "MTQV2: direction lock"
   630→                );
   631→            }
   632→
   633→            // Validate trade moves toward execution weight (within 5% tolerance)
   634→            uint256 execW    = weights.executionWeights[uint256(t.component)];
   635→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
   636→            if (t.direction == 1) {
   637→                require(currentW < execW, "MTQV2: buy but already above target");
   638→            } else {
   639→                require(currentW > execW, "MTQV2: sell but already below target");
   640→            }
   641→
   642→            // New weight after trade must be within 5pp (absolute) of execW
   643→            uint256 newHeldUsd = (t.direction == 1)
   644→                ? reserveHeldUsd[t.component] + t.tradeUsd
   645→                : reserveHeldUsd[t.component] - t.tradeUsd;
   646→            require(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1, "MTQV2: insufficient reserve");
   647→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
   648→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
   649→            require(absDiff <= REBALANCE_TOLERANCE, "MTQV2: trade overshoots execution weight by >5%");
   650→
   651→            // §10  Daily turnover cap (5% of NAV)
   652→            require(dailyTradeUsd + t.tradeUsd <= maxDaily, "MTQV2: daily turnover cap");
   653→            dailyTradeUsd += t.tradeUsd;
   654→
   655→            // Update reserve mirror
   656→            if (t.direction == 1) {
   657→                reserveHeldUsd[t.component] += t.tradeUsd;
   658→            } else {
   659→                reserveHeldUsd[t.component] -= t.tradeUsd;
   660→            }
   661→            lastDirection[t.component]   = t.direction;
   662→            lastRebalanceAt[t.component] = block.timestamp;
   663→        }
   664→
   665→        emit RebalanceExecuted(trades, block.timestamp);
   666→    }

   681→    /// @notice §9.1-9.3  Consensus across all 3 adapters for a single pair.
   682→    /// @return finalPrice  USD-per-unit, 1e18 scale (0 if paused)
   683→    /// @return validCount  Number of valid feeds after all filters
   684→    /// @return method      0=paused, 1=average(2), 2=median(3)
   685→    /// @return paused_     True if fewer than 2 feeds are valid
   686→    /// @return prices      Raw prices per adapter (1e18) — index 0/1/2 = CL/Pyth/Cro
   687→    /// @return valid       Per-adapter validity flag after all filters
   688→    function getOracleConsensus(bytes32 pair) public view returns (
   689→        uint256 finalPrice,
   690→        uint8  validCount,
   691→        uint8  method,
   692→        bool   paused_,
   693→        uint256[3] memory prices,
   694→        bool[3] memory valid
   695→    ) {
   696→        // Fetch from each adapter (try/catch — missing adapters are skipped)
   697→        IOracleAdapter[3] memory adapters;
   698→        adapters[0] = chainlinkAdapter;
   699→        adapters[1] = pythAdapter;
   700→        adapters[2] = chronicleAdapter;
   701→
   702→        uint256[3] memory raw;
   703→        uint256[3] memory ts;
   704→        uint256[3] memory conf;
   705→
   706→        for (uint8 i = 0; i < 3; i++) {
   707→            valid[i] = false;
   708→            if (address(adapters[i]) == address(0)) continue;
   709→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
   710→                raw[i]  = p;
   711→                ts[i]   = t;
   712→                conf[i] = c;
   713→                prices[i] = p;
   714→                // §9.2.1  staleness ≤60s (also reject future timestamps)
   715→                if (t > block.timestamp) continue;
   716→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
   717→                // §9.2.3  confidence < 1% of price
   718→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
   719→                valid[i] = true;
   720→            } catch {
   721→                // leave valid[i] = false
   722→            }
   723→        }
   724→
   725→        // Count valid
   726→        validCount = 0;
   727→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
   728→
   729→        if (validCount < 2) {
   730→            paused_ = true;
   731→            method  = 0;
   732→            return (finalPrice, validCount, method, paused_, prices, valid);
   733→        }
   734→
   735→        // Sort the valid prices and pick median (n=3) or average (n=2)
   736→        uint256[3] memory sorted;
   737→        uint8 n = 0;
   738→        for (uint8 i = 0; i < 3; i++) {
   739→            if (valid[i]) { sorted[n] = raw[i]; n++; }
   740→        }
   741→        _insertionSort(sorted, n);
   742→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
   743→
   744→        // §9.2.4  Deviation < 2.5% from median → discard
   745→        for (uint8 i = 0; i < 3; i++) {
   746→            if (valid[i]) {
   747→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
   748→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
   749→                    valid[i] = false;
   750→                    validCount--;
   751→                }
   752→            }
   753→        }
   754→
   755→        if (validCount < 2) {
   756→            paused_ = true;
   757→            method  = 0;
   758→            return (finalPrice, validCount, method, paused_, prices, valid);
   759→        }
   760→
   761→        paused_ = false;
   762→        if (validCount == 3) {
   763→            finalPrice = med;
   764→            method     = 2; // median(3)
   765→        } else {
   766→            // Average of the 2 remaining valid feeds
   767→            uint256 sum = 0;
   768→            uint8  cnt = 0;
   769→            for (uint8 i = 0; i < 3; i++) {
   770→                if (valid[i]) { sum += raw[i]; cnt++; }
   771→            }
   772→            finalPrice = sum / cnt;
   773→            method     = 1; // average(2)
   774→        }
   775→    }

   851→    /// @notice §5 + §14.1  Reserve Net Asset Value (USD, 1e18) — applies
   852→    ///         per-asset haircuts pulled from the registry. Frozen/delisted
   853→    ///         assets contribute 0 to NAV. If the registry is unset, falls
   854→    ///         back to the raw gross sum.
   855→    function getReserveNetAssetValue() public view returns (uint256) {
   856→        if (address(assetRegistry) == address(0)) {
   857→            return getReserveNavUsd();
   858→        }
   859→        bytes32[7] memory codes;
   860→        codes[0] = bytes32("USD");
   861→        codes[1] = bytes32("EUR");
   862→        codes[2] = bytes32("JPY");
   863→        codes[3] = bytes32("GBP");
   864→        codes[4] = bytes32("CNY");
   865→        codes[5] = bytes32("CHF");
   866→        codes[6] = bytes32("XAU");
   867→
   868→        uint256 nav = 0;
   869→        for (uint8 i = 0; i < 7; i++) {
   870→            uint256 held = reserveHeldUsd[Component(i)];
   871→            if (held == 0) continue;
   872→            try IAssetRegistry(assetRegistry).getAsset(codes[i]) returns (
   873→                address, uint256 haircut, uint8 state, bytes32
   874→            ) {
   875→                if (state == 2 || state == 3) continue; // frozen / delisted → 0 NAV
   876→                // Apply haircut: nav += held × (1 − haircut)
   877→                nav += held * (1e18 - haircut) / 1e18;
   878→            } catch {
   879→                // Registry call failed — fall back to no haircut for this asset
   880→                nav += held;
   881→            }
   882→        }
   883→        return nav;
   884→    }

   887→    ///         ≥ 1.10 = healthy (RR_TARGET); ≥ 1.00 = solvent (I2 hard floor).
   888→    function getReserveRatio() public view returns (uint256) {
   889→        uint256 nav  = getReserveNetAssetValue();
   890→        uint256 liab = getLiability();
   891→        if (liab == 0) return type(uint256).max;
   892→        return nav * 1e18 / liab;

   900→    /// @notice Queue a parameter change (ADMIN_ROLE). Executes after 48h.
   901→    function queueChange(bytes32 key, uint256 newValue) external onlyAdmin {
   902→        ParameterChange storage pc = paramChanges[key];
   903→        require(!pc.executed, "MTQV2: already executed");
   904→        pc.key        = key;
   905→        pc.newValue   = newValue;
   906→        pc.queuedAt   = block.timestamp;
   907→        pc.executesAt = block.timestamp + TIMELOCK_DELAY;
   908→        pc.executed   = false;

   912→    /// @notice Execute a queued parameter change after the timelock elapses.
   913→    ///         Anyone can call (the timelock is the protection, not the caller).
   914→    function executeChange(bytes32 key) external {
   915→        ParameterChange storage pc = paramChanges[key];
   916→        require(pc.queuedAt > 0, "MTQV2: not queued");
   917→        require(!pc.executed, "MTQV2: already executed");
   918→        require(block.timestamp >= pc.executesAt, "MTQV2: timelock not elapsed");
   919→        pc.executed = true;
   920→        _applyParam(key, pc.newValue);
   921→        emit ParameterExecuted(key, pc.newValue, block.timestamp);
   922→    }

  1045→    /// @dev In-place insertion sort of sorted[0..n), ascending.
  1046→    function _insertionSort(uint256[3] memory arr, uint8 n) internal pure {
  1047→        for (uint8 i = 1; i < n; i++) {
  1048→            uint256 key = arr[i];
  1049→            uint8 j = i;
  1050→            while (j > 0 && arr[j - 1] > key) {
  1051→                arr[j] = arr[j - 1];
  1052→                j--;
  1053→            }
  1054→            arr[j] = key;
  1055→        }
  1056→    }
