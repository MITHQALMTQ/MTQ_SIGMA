   366→function seedHistory(mean: number, sd: number, n: number): number[] {
   367→  const out: number[] = [];
   368→  let last = mean;
   369→  for (let i = 0; i < n; i++) {
   370→    // simple mean-reverting walk
   371→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   372→    last = last + (mean - last) * 0.1 + shock;
   373→    out.push(Math.max(0.01, last));
   374→  }
   375→  return out;
   376→}
   377→
   378→// --- §14.1 Constitutional Separation helpers ---------------------------------
   379→// Maintain the §14.1 invariant: paxg = indexPaxg + reservePaxg (and same for
   380→// xaut). Index gold is LOCKED (only changes via commitIndexGold); reserve gold
   381→// is the buffer MARP rebalances. These two helpers are the only sanctioned
   382→// ways to keep the splits consistent after any mutation to total gold:
   383→
   384→/** Re-derive reserve gold from total − index (call after the legacy path or
   385→ *  concentration optimizer mutates `s.paxg` / `s.xaut` directly). Index gold
   386→ *  stays locked. */
   387→export function syncReserveFromTotal(s: ReserveState): void {
   388→  s.reservePaxg = Math.max(0, s.paxg - s.indexPaxg);
   389→  s.reserveXaut = Math.max(0, s.xaut - s.indexXaut);
   390→}
   391→
   392→/** Re-derive total gold from index + reserve (call after the MARP path mutates
   393→ *  `s.reservePaxg` / `s.reserveXaut`). Index gold stays locked. */
   394→export function syncTotalFromReserve(s: ReserveState): void {
   395→  s.paxg = s.indexPaxg + Math.max(0, s.reservePaxg);
   396→  s.xaut = s.indexXaut + Math.max(0, s.reserveXaut);
   397→}

   399→/** §14.1 commitIndexGold — the keeper-role equivalent in the TS engine.
   400→ *  Sets the index gold holdings (PAXG + XAUT) that back the Strategic Prior
   401→ *  Gold weight (26%). Only this function may move gold between the index
   402→ *  vault and the reserve buffer. Recomputes the totals via the §14.1 invariant
   403→ *  (reserve = total − index). Emits an IndexGoldCommitted event-equivalent
   404→ *  (console log — production should emit an on-chain event). */
   405→export function commitIndexGold(
   406→  s: ReserveState,
   407→  indexPaxg: number,
   408→  indexXaut: number,
   409→): { indexPaxg: number; indexXaut: number; reservePaxg: number; reserveXaut: number; totalPaxg: number; totalXaut: number } {
   410→  const safeIndexPaxg = Math.max(0, indexPaxg);
   411→  const safeIndexXaut = Math.max(0, indexXaut);
   412→  s.indexPaxg = safeIndexPaxg;
   413→  s.indexXaut = safeIndexXaut;
   414→  // Reserve = max(0, total − index) — never let reserve go negative even if
   415→  // the keeper over-commits (log a warning in that case).
   416→  if (s.paxg < safeIndexPaxg) {

   437→// --- §2 GFB Index (v1.0 — 7-component chain-linked) -------------------------
   438→// GFB_t = Σ_i W^Prior_i × P_{i,t}  (then normalised by GFB_BASE_DENOMINATOR so
   439→// GFB = 1.00 exactly at the base date). The 7 components are the Strategic
   440→// Prior: USD, EUR, JPY, GBP, CNY, CHF, Gold. Gold is now a first-class index
   441→// component (P_Gold = XAU_USD).
   442→//
   443→// DEPRECATED (P0-FIX-1): this is the LEGACY fixed-base Laspeyres index. It
   444→// weights each component by its raw USD notional share (gold contributes
   445→// 99.9% of the index by notional despite a 26% Strategic Prior weight), which
   446→// creates a structural short-gold exposure that crashed the §23 validation
   447→// program (S5: 0% survival, RR crashed to 0.83 in 100% of gold +50% runs).
   448→//
   449→// The CANONICAL chain-linked index lives in `src/lib/mtq/chain-index.ts`. The
   450→// engine's `computeSnapshot` now uses `getMTQPrice(s.chainIndex)` (the
   451→// chain-linked form) for both `gfbIndex` and `mtqPrice`. This legacy
   452→// `computeGfbIndex` is RETAINED for backward compatibility with the
   453→// standalone audit-stress.ts simulations (which need a pure function of fx,
   454→// not the persisted chain index state).
   455→export function computeGfbIndex(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): number {
   456→  // Raw USD value of the basket at time t. USD is the unit of account (price 1.0);
   457→  // every other component contributes its USD-equivalent via its FX rate.
   458→  const numerator =
   459→    STRATEGIC_PRIOR.USD  * 1.0 +
   460→    STRATEGIC_PRIOR.EUR  * fx.EUR_USD +
   461→    STRATEGIC_PRIOR.JPY  * fx.JPY_USD +
   462→    STRATEGIC_PRIOR.GBP  * fx.GBP_USD +
   463→    STRATEGIC_PRIOR.CNY  * fx.CNY_USD +
   464→    STRATEGIC_PRIOR.CHF  * fx.CHF_USD +
   465→    STRATEGIC_PRIOR.Gold * fx.XAU_USD;
   466→  // Normalised: GFB_t = numerator / GFB_BASE_DENOMINATOR  (=1.0 at base date).
   467→  // GFB_BASE_DENOMINATOR is the v1.0 chain-linked denominator (includes Gold
   468→  // and CHF at base fixings — see blueprint.ts).
   469→  return numerator / GFB_BASE_DENOMINATOR;
   470→}

   519→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   520→// USD value of each holding (gross), then apply haircuts for NAV.
   521→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   522→  const usdGross = (s.usdc + s.usdp + s.usdt) * 1.0; // USDC + USDP + USDT combined
   523→  const eurGross = s.eurc * fx.EUR_USD;
   524→  const gbpGross = s.gbp * fx.GBP_USD;
   525→  const jpyGross = s.jpy * fx.JPY_USD;
   526→  const cnyGross = s.cny * fx.CNY_USD;
   527→  const chfGross = s.chf * fx.CHF_USD; // NEW v1.0 — CHF is a first-class reserve component
   528→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross + chfGross;
   529→
   530→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   531→  const goldRef = fx.XAU_USD;
   532→  const goldExec = fx.XAU_USD * 0.995;
   533→  const goldPrice = Math.min(goldRef, goldExec);
   534→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   535→
   536→  // Net values (after haircuts)
   537→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   538→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   539→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   540→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   541→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   542→  const chfNet = chfGross * (1 - HAIRCUTS.CHF); // NEW v1.0
   543→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   544→
   545→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet + goldNet;
   546→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet;
   547→
   548→  // Per-issuer breakdown (for concentration visualization)
   549→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   550→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   551→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   552→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);
   553→  const xautUsd = s.xaut * goldPrice * (1 - HAIRCUTS.XAU);
   554→
   555→  // §14.1 — index vs reserve gold split (USD net of haircut).
   556→  // Index gold is locked (backs the 26% Strategic Prior Gold weight); reserve
   557→  // gold is the buffer MARP rebalances. The total gold (= index + reserve) is
   558→  // what serves the index weight for MASE/MARP observed-weight purposes.
   559→  const indexPaxgUsd = s.indexPaxg * goldPrice * (1 - HAIRCUTS.XAU);
   560→  const indexXautUsd = s.indexXaut * goldPrice * (1 - HAIRCUTS.XAU);
   561→  const reservePaxgUsd = (s.reservePaxg >= 0 ? s.reservePaxg : Math.max(0, s.paxg - s.indexPaxg)) * goldPrice * (1 - HAIRCUTS.XAU);
   562→  const reserveXautUsd = (s.reserveXaut >= 0 ? s.reserveXaut : Math.max(0, s.xaut - s.indexXaut)) * goldPrice * (1 - HAIRCUTS.XAU);
   563→  const indexGoldNet = indexPaxgUsd + indexXautUsd;
   564→  const reserveGoldNet = reservePaxgUsd + reserveXautUsd;
   565→
   566→  return {
   567→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, chfGross, fiatGross,
   568→    goldGross, goldPrice,
   569→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, chfNet, goldNet,
   570→    nav, fiatNet,
   571→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   572→    // §14.1 gold split
   573→    indexPaxgUsd, indexXautUsd, reservePaxgUsd, reserveXautUsd,
   574→    indexGoldNet, reserveGoldNet,
   575→  };
   576→}

   600→// P0-FIX-3: the canonical classification lives in state-machine.ts. This

   616→  // condition-binds rule exactly.

   618→  const cs = determineState(rr, lcr, prev, Date.now(), Date.now());
   619→  return cs.state;

   661→// --- §6 Adaptive Macro Engine -----------------------------------------------
   662→function rollingStats(hist: number[]): { mean: number; sd: number } {
   663→  const filled = hist.filter((x) => x > 0);
   664→  if (filled.length === 0) return { mean: 0, sd: 0 };
   665→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   666→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   667→  return { mean, sd: Math.sqrt(variance) };
   668→}
   669→
   670→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   671→  const v = rollingStats(s.vixHistory);
   672→  const d = rollingStats(s.dxyHistory);
   673→  const eps = 1e-9;
   674→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   675→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   676→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   677→}

   679→export function computeRawTargetTheta(zVix: number, zDxy: number): number {

   696→  if (rr >= RR_TARGET) return BUFFER_GOLD_BASE; // 62.5% → total 26.25%
   697→  if (rr >= RR_STRESS) return BUFFER_GOLD_STRESS; // 85% → total 28.5%

   701→export function currentBufferGoldRatio(s: ReserveState): number {
   702→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   703→  if (s.rampStartAt === 0) {
   704→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   705→  }
   706→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   707→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   708→  const progress = elapsedH / RAMP_DURATION_HOURS;
   709→  const from = s.rampFromRatio;
   710→  const to = s.rampToRatio;
   711→  return from + (to - from) * progress;
   712→}

   723→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp

   741→  if (Date.now() - s.lastTurnoverResetAt >= 86_400_000) {

   747→export function evaluateRebalance(
   748→  s: ReserveState,
   749→  vals: { nav: number; goldNet: number; fiatNet: number },
   750→  rr: number,
   751→): RebalanceDecision {
   752→  const nav = vals.nav;
   753→  if (nav <= 0) {
   754→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   755→  }
   756→  const observed = vals.goldNet / nav;
   757→  const target = computeTargetGoldWeight(s, rr);
   758→  const deviation = observed - target;
   759→
   760→  if (Math.abs(deviation) < 0.005) {
   761→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   762→  }
   763→
   764→  // Direction lock (overridden in stress)
   765→  const inStress = rr < RR_STRESS;
   766→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   767→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   768→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   769→    }
   770→  }
   771→
   772→  // Trade sizing
   773→  const surplusUsd = Math.abs(deviation) * nav;
   774→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   775→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   776→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   777→  maybeResetDailyTurnover(s);
   778→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   779→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   780→
   781→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   782→  const estCost = tradeUsd * 0.001 + 50;
   783→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   784→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   785→
   786→  if (benefit <= estCost) {
   787→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   788→  }
   789→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   790→}

   792→// Apply a rebalance trade in USD terms (mutates state).
   793→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   794→  if (!decision.shouldRebalance) return;
   795→  const usd = decision.tradeUsd;
   796→  if (decision.direction === -1) {
   797→    // sell gold → buy USDC (simplified: convert to USDC)
   798→    const goldUnits = usd / goldPrice;
   799→    s.paxg = Math.max(0, s.paxg - goldUnits);
   800→    // §14.1 — the sold gold comes from the RESERVE buffer (index gold is locked).
   801→    // Maintain the invariant paxg = indexPaxg + reservePaxg.
   802→    s.reservePaxg = Math.max(0, s.reservePaxg - goldUnits);
   803→    s.usdc += usd;
   804→  } else {
   805→    // buy gold → spend USDC
   806→    s.usdc = Math.max(0, s.usdc - usd);
   807→    const goldUnits = usd / goldPrice;
   808→    s.paxg += goldUnits;
   809→    // §14.1 — the bought gold goes to the RESERVE buffer (index gold is locked).
   810→    s.reservePaxg += goldUnits;
   811→  }
   812→  s.lastTradeDir = decision.direction;
   813→  s.lastTradeAt = Date.now();
   814→  s.dailyTurnoverUsd += usd;
   815→  s.updatedAt = Date.now();
   816→}

   818→// --- §14.1 + §10 MARP per-component rebalance execution ---------------------
   819→// applyMarpRebalance() is the v1.0 production-target rebalance path. Unlike the
   820→// legacy §7 single-direction `applyRebalanceTrade` (which only trades gold ↔
   821→// USDC), this takes the per-component MarpDecision[] (output of marpDecision())
   822→// and applies each "would-execute" trade to the RESERVE holdings — index gold
   823→// is LOCKED per §14.1 and never touched here.
   824→//
   825→// Trade execution model (pilot):
   826→//   - For non-USD components (EUR/JPY/GBP/CNY/CHF/Gold): each trade pairs with
   827→//     USD as the unit of account. A "sell" of component X converts X → USD
   828→//     (split 1/3 across USDC/USDP/USDT, respecting the §5.6 concentration
   829→//     target). A "buy" of component X converts USD → X.
   830→//   - For USD: each trade pairs with Gold (the other deep pool). A "sell" of
   831→//     USD converts USD → Gold (split 50/50 PAXG/XAUT in the RESERVE buffer).
   832→//     A "buy" of USD converts Gold → USD.
   833→//   - For Gold: a "sell" reduces reservePaxg + reserveXaut (50/50); a "buy"
   834→//     increases them. Index gold is never touched.
   835→//
   836→// Constraints enforced (§10):
   837→//   - MAX_DAILY_TURNOVER (5% of NAV) — trades that would breach are skipped.
   838→//   - DIRECTION_LOCK_HOURS (24h) — trades that reverse the last direction
   839→//     within 24h are skipped (whipsaw guard).
   840→//   - Only decisions with shouldTrade=true AND level>=6 are executed
   841→//     (level 1-5 are advisory: no-trade zone / low urgency / cost-benefit
   842→//     fail / turnover cap).
   843→//
   844→// Mutates: s.usdc/usdp/usdt, s.eurc/gbp/jpy/cny/chf, s.reservePaxg/reserveXaut
   845→//          (and re-syncs s.paxg/xaut totals via the §14.1 invariant),
   846→//          s.lastTradeDir/lastTradeAt, s.dailyTurnoverUsd, s.updatedAt.
   847→// Index gold (s.indexPaxg, s.indexXaut) is NEVER touched.
   848→export interface MarpExecDecision {
   849→  shouldTrade: boolean;
   850→  component: string;
   851→  direction: string;
   852→  tradeUsd: number;
   853→  level: number;
   854→}
   855→
   856→export interface MarpExecResult {
   857→  appliedCount: number;
   858→  skippedCount: number;
   859→  totalTradeUsd: number;
   860→  path: 'legacy' | 'marp';
   861→  // Per-decision trace: which were applied vs skipped (and why).
   862→  traces: { component: string; direction: string; tradeUsd: number; level: number; applied: boolean; skipReason?: string }[];
   863→}
   864→
   865→export function applyMarpRebalance(
   866→  s: ReserveState,
   867→  fx: FxRates,
   868→  decisions: MarpExecDecision[],
   869→): MarpExecResult {
   870→  let appliedCount = 0;
   871→  let skippedCount = 0;
   872→  let totalTradeUsd = 0;
   873→  const traces: MarpExecResult['traces'] = [];
   874→
   875→  maybeResetDailyTurnover(s);
   876→  const vals = reserveAssetValues(s, fx);
   877→  const nav = vals.nav;
   878→  if (nav <= 0) {
   879→    // Nothing to do — surface a single skip trace for visibility.
   880→    for (const d of decisions) {
   881→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'NAV ≤ 0' });
   882→    }
   883→    return { appliedCount: 0, skippedCount: decisions.length, totalTradeUsd: 0, path: 'marp', traces };
   884→  }
   885→  const goldPrice = vals.goldPrice;
   886→
   887→  for (const d of decisions) {
   888→    // Level gate: only execute shouldTrade && level >= 6 decisions.
   889→    if (!d.shouldTrade || d.level < 6) {
   890→      skippedCount++;
   891→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: `level ${d.level} (advisory)` });
   892→      continue;
   893→    }
   894→    let tradeUsd = d.tradeUsd;
   895→    // Enforce MAX_DAILY_TURNOVER (5% of NAV)
   896→    const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   897→    if (tradeUsd > remainingDaily) {
   898→      tradeUsd = Math.max(0, remainingDaily);
   899→    }
   900→    if (tradeUsd <= 0) {
   901→      skippedCount++;
   902→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'daily turnover cap' });
   903→      continue;
   904→    }
   905→    // Direction (+1 buy / -1 sell / 0 hold)
   906→    const dir: 0 | 1 | -1 = d.direction === 'buy' ? 1 : d.direction === 'sell' ? -1 : 0;
   907→    if (dir === 0) {
   908→      skippedCount++;
   909→      traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'hold' });
   910→      continue;
   911→    }
   912→    // Enforce DIRECTION_LOCK_HOURS (24h whipsaw guard)
   913→    if (Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   914→      if ((dir === 1 && s.lastTradeDir === -1) || (dir === -1 && s.lastTradeDir === 1)) {
   915→        skippedCount++;
   916→        traces.push({ component: d.component, direction: d.direction, tradeUsd: d.tradeUsd, level: d.level, applied: false, skipReason: 'direction lock (24h whipsaw guard)' });
   917→        continue;
   918→      }
   919→    }
   920→
   921→    // Apply the trade to the RESERVE holdings.
   922→    if (d.component === 'Gold') {
   923→      // Gold ↔ USD. Split 50/50 across reservePaxg + reserveXaut.
   924→      const goldUnits = tradeUsd / goldPrice;
   925→      const half = goldUnits / 2;
   926→      if (dir === -1) {
   927→        // sell gold → USD
   928→        s.reservePaxg = Math.max(0, s.reservePaxg - half);
   929→        s.reserveXaut = Math.max(0, s.reserveXaut - half);
   930→        const usdThird = tradeUsd / 3;
   931→        s.usdc += usdThird;
   932→        s.usdp += usdThird;
   933→        s.usdt += usdThird;
   934→      } else {
   935→        // buy gold ← USD
   936→        const usdThird = tradeUsd / 3;
   937→        s.usdc = Math.max(0, s.usdc - usdThird);
   938→        s.usdp = Math.max(0, s.usdp - usdThird);
   939→        s.usdt = Math.max(0, s.usdt - usdThird);
   940→        s.reservePaxg += half;
   941→        s.reserveXaut += half;
   942→      }
   943→      // §14.1 — re-sync total paxg/xaut = index + reserve
   944→      syncTotalFromReserve(s);
   945→    } else if (d.component === 'USD') {
   946→      // USD ↔ Gold. Split 50/50 across reservePaxg + reserveXaut.
   947→      const goldUnits = tradeUsd / goldPrice;
   948→      const half = goldUnits / 2;
   949→      if (dir === -1) {
   950→        // sell USD → gold
   951→        const usdThird = tradeUsd / 3;
   952→        s.usdc = Math.max(0, s.usdc - usdThird);
   953→        s.usdp = Math.max(0, s.usdp - usdThird);
   954→        s.usdt = Math.max(0, s.usdt - usdThird);
   955→        s.reservePaxg += half;
   956→        s.reserveXaut += half;
   957→      } else {
   958→        // buy USD ← gold
   959→        s.reservePaxg = Math.max(0, s.reservePaxg - half);
   960→        s.reserveXaut = Math.max(0, s.reserveXaut - half);
   961→        const usdThird = tradeUsd / 3;
   962→        s.usdc += usdThird;
   963→        s.usdp += usdThird;
   964→        s.usdt += usdThird;
   965→      }
   966→      // §14.1 — re-sync total paxg/xaut = index + reserve
   967→      syncTotalFromReserve(s);
   968→    } else {
   969→      // Non-USD fiat component (EUR/JPY/GBP/CNY/CHF) ↔ USD.
   970→      // Convert tradeUsd to native units via the FX rate.
   971→      const fxRate =
   972→        d.component === 'EUR' ? fx.EUR_USD :
   973→        d.component === 'JPY' ? fx.JPY_USD :
   974→        d.component === 'GBP' ? fx.GBP_USD :
   975→        d.component === 'CNY' ? fx.CNY_USD :
   976→        d.component === 'CHF' ? fx.CHF_USD : 1.0;
   977→      const nativeUnits = tradeUsd / fxRate;
   978→      const usdThird = tradeUsd / 3;
   979→      if (dir === -1) {
   980→        // sell component → USD
   981→        if (d.component === 'EUR') s.eurc = Math.max(0, s.eurc - nativeUnits);
   982→        else if (d.component === 'JPY') s.jpy = Math.max(0, s.jpy - nativeUnits);
   983→        else if (d.component === 'GBP') s.gbp = Math.max(0, s.gbp - nativeUnits);
   984→        else if (d.component === 'CNY') s.cny = Math.max(0, s.cny - nativeUnits);
   985→        else if (d.component === 'CHF') s.chf = Math.max(0, s.chf - nativeUnits);
   986→        s.usdc += usdThird;
   987→        s.usdp += usdThird;
   988→        s.usdt += usdThird;
   989→      } else {
   990→        // buy component ← USD
   991→        s.usdc = Math.max(0, s.usdc - usdThird);
   992→        s.usdp = Math.max(0, s.usdp - usdThird);
   993→        s.usdt = Math.max(0, s.usdt - usdThird);
   994→        if (d.component === 'EUR') s.eurc += nativeUnits;
   995→        else if (d.component === 'JPY') s.jpy += nativeUnits;
   996→        else if (d.component === 'GBP') s.gbp += nativeUnits;
   997→        else if (d.component === 'CNY') s.cny += nativeUnits;
   998→        else if (d.component === 'CHF') s.chf += nativeUnits;
   999→      }
  1000→    }
  1001→
  1002→    s.lastTradeDir = dir;
  1003→    s.lastTradeAt = Date.now();
  1004→    s.dailyTurnoverUsd += tradeUsd;
  1005→    totalTradeUsd += tradeUsd;
  1006→    appliedCount++;
  1007→    traces.push({ component: d.component, direction: d.direction, tradeUsd, level: d.level, applied: true });
  1008→  }
  1009→
  1010→  s.updatedAt = Date.now();
  1011→  return { appliedCount, skippedCount, totalTradeUsd, path: 'marp', traces };
  1012→}

  1014→// --- §11 Geopolitical Eject -------------------------------------------------
  1015→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
  1016→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
  1017→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
  1018→  const drift = (cur: number, amp: number) => {
  1019→    const shock = (Math.random() - 0.5) * 2 * amp;
  1020→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
  1021→  };
  1022→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
  1023→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
  1024→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
  1025→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
  1026→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
  1027→
  1028→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
  1029→  for (const c of currencies) {
  1030→    const p = s.pegHealth[c];
  1031→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
  1032→    if (outside) {
  1033→      s.depegHours[c] += dtHours;
  1034→    } else {
  1035→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
  1036→    }
  1037→    // Staged ladder
  1038→    const h = s.depegHours[c];
  1039→    let stage = 0;
  1040→    if (h > 48) stage = 3;
  1041→    else if (h > 24) stage = 2;
  1042→    else if (h > 12) stage = 1;
  1043→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
  1044→    s.ejectStage[c] = stage;
  1045→  }
  1046→}

  1062→export function applyMint(
  1063→  s: ReserveState,
  1064→  fx: FxRates,
  1065→  status: ProtocolStatus,
  1066→  inputUsd: number,
  1067→): MintResult {
  1068→  // P0-FIX-1: use the canonical chain-linked index for the MTQ price (the
  1069→  // legacy Laspeyres form had a structural short-gold bug — see chain-index.ts).
  1070→  const price = getMtqPriceFromState(s);
  1071→  if (!priceInSafetyBand(price)) {
  1072→    return { ok: false, reason: "Reference Value outside safety band (0.50–2.00 USD reporting); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
  1073→  }
  1074→  // P0-FIX-3: use the canonical mint throttle / minting-allowed from
  1075→  // state-machine.ts. STRESS now also pauses minting (previously only
  1076→  // DEFENSIVE/EMERGENCY did). NORMAL=1.0, CAUTION=0.5, STRESS/DEFENSIVE/EMERGENCY=0 (paused), RECOVERY=0.25.
  1077→  if (!mintingAllowed(status)) {
  1078→    return { ok: false, reason: `${status}: minting paused by canonical Risk State Machine (§21.3).`, inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
  1079→  }
  1080→  const throttle = mintThrottle(status);
  1081→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
  1082→  const netUsd = inputUsd - feeUsd;
  1083→  const minted = (netUsd / price) * throttle;
  1084→
  1085→  // Mutate reserve: deposit USDC (split across 3 USD issuers per concentration policy),
  1086→  // mint MTQ to user (circulating supply rises). The concentration rebalancer in the
  1087→  // tick loop will re-optimize the split if any issuer exceeds 25%.
  1088→  const usdThird = netUsd / 3;
  1089→  s.usdc += usdThird;
  1090→  s.usdp += usdThird;
  1091→  s.usdt += usdThird;
  1092→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet (§13.2 sweep target)
  1093→  s.totalSupply += minted;
  1094→  s.updatedAt = Date.now();
  1095→
  1096→  const vals = reserveAssetValues(s, fx);
  1097→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
  1098→  return {
  1099→    ok: true,
  1100→    inputUsd,
  1101→    feeUsd,
  1102→    netUsd,
  1103→    mtqPrice: price,
  1104→    mtqMinted: minted,
  1105→    throttleFactor: throttle,
  1106→    newCirculatingSupply: circulatingSupply(s),
  1107→    newReserveRatio: newRr,
  1108→  };
  1109→}

  1147→export function applyRedeem(
  1148→  s: ReserveState,
  1149→  fx: FxRates,
  1150→  status: ProtocolStatus,
  1151→  inputMtq: number,
  1152→): RedeemResult {
  1153→  // P0-FIX-1: use the canonical chain-linked index for P_MTQ.
  1154→  const price = getMtqPriceFromState(s);
  1155→  const circ = circulatingSupply(s);
  1156→  // Pre-settlement reserve values (used to compute NAV_t and the audit fields).
  1157→  const vals0 = reserveAssetValues(s, fx);
  1158→  // P0-FIX-2 (Master §19.3.2, Invariant I6): NAV-based redemption.
  1159→  //   NAV_t = V_net / circulatingSupply; grossUsd = inputMtq × NAV_t
  1160→  //
  1161→  // Genesis fallback: at genesis (or any time circulatingSupply = 0), NAV_t
  1162→  // is undefined (division by zero). We fall back to P_MTQ in this case:
  1163→  //   - At genesis, there is no circulating MTQ to redeem (the inputMtq > circ
  1164→  //     guard below catches this case explicitly).
  1165→  //   - In practice the fallback is exercised only when the protocol has not
  1166→  //     yet minted any circulating supply — the redemption would fail with
  1167→  //     "insufficient supply" before reaching the NAV computation.
  1168→  //   - After the first mint, circulating supply > 0 and NAV_t = V_net / S_circ
  1169→  //     will be very high (e.g. $42/MTQ when RR = 2400%) — this is CORRECT:
  1170→  //     redeeming 1 MTQ returns $42 worth of reserve, which IS the actual
  1171→  //     value of the token. The Index/NAV divergence analysis (§12) surfaces
  1172→  //     when this matters (e.g. when NAV_per_MTQ diverges from P_MTQ by >5%
  1173→  //     → "monitor" status; >10% → "stress").
  1174→  const navPerMtq = circ > 0 ? vals0.nav / circ : price;
  1175→  // Honest audit fields (computed pre-settlement so they exist on the error path too).
  1176→  // P0-FIX-2: the primary settlement now uses NAV (§12.2 — formerly audit-only).
  1177→  // The legacy §3.4.2 (index-priced) is now the AUDIT-ONLY alternative.
  1178→  const auditNavPerToken = navPerMtq;
  1179→  const auditGrossUsdNav = inputMtq * navPerMtq;
  1180→  const auditGrossUsdIndex = inputMtq * price;
  1181→  const auditNote =
  1182→    "P0-FIX-2 (Master §19.3.2, Invariant I6): redemption is now priced against " +
  1183→    "NAV_t = V_net / circulatingSupply (the §12.2 form, formerly audit-only). " +
  1184→    "The legacy §3.4.2 (redeem at P_MTQ) is now the AUDIT-ONLY alternative " +
  1185→    "(auditGrossUsdIndex). At RR>100% the NAV pays redeemers MORE than the index " +
  1186→    "price (auditDeltaUsd > 0), which IS the intent: the redeemer receives the " +
  1187→    "actual book value of their token, not just the index-tracked price. The " +
  1188→    "§12 Index/NAV divergence analysis surfaces when this gap becomes material " +
  1189→    "(>5% monitor, >10% stress).";
  1190→  if (inputMtq > circ) {
  1191→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)), navPerMtq, auditNavPerToken, auditGrossUsdNav, auditGrossUsdIndex, auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex, auditNote };
  1192→  }
  1193→  // P0-FIX-3: §21.4 — redemption is PAUSED in EMERGENCY (the reconciliation
  1194→  // resolves the §16.2 vs §21.4 contradiction in favour of §21.4: pause).
  1195→  if (!redemptionAllowed(status)) {
  1196→    return { ok: false, reason: `${status}: redemption paused per §21.4 (canonical Risk State Machine).`, inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)), navPerMtq, auditNavPerToken, auditGrossUsdNav, auditGrossUsdIndex, auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex, auditNote };
  1197→  }
  1198→  // P0-FIX-3: state-dependent fee from the canonical 6-state risk machine.
  1199→  // NORMAL/CAUTION 0.15% (15 bps), STRESS 0.50% (50 bps), DEFENSIVE 1.00%
  1200→  // (100 bps), EMERGENCY 2.00% (200 bps), RECOVERY 0.50% (50 bps).
  1201→  const feeFraction = redeemFee(status);
  1202→  const feeBps = Math.round(feeFraction * 10_000);
  1203→  // P0-FIX-2: grossUsd is now NAV-based (was index-based in the legacy engine).
  1204→  const grossUsd = inputMtq * navPerMtq;
  1205→  const feeUsd = grossUsd * feeFraction;
  1206→  const netUsd = grossUsd - feeUsd;
  1207→
  1208→  // Release actual reserve composition proportionally (§12.2):
  1209→  // gold portion = netUsd * W_target (legacy v1.2 buffer-derived target —
  1210→  // retained until the MASE ensemble replaces it). fiat portion split per the
  1211→  // 6 non-gold Strategic Prior components (USD/EUR/JPY/GBP/CNY/CHF) renormalised
  1212→  // to the non-gold total.
  1213→  const vals = reserveAssetValues(s, fx);
  1214→  const rr = computeReserveRatio(vals.nav, computeLiability(s, price));
  1215→  const wGold = computeTargetGoldWeight(s, rr);
  1216→  const goldUsd = netUsd * wGold;
  1217→  const fiatUsd = netUsd * (1 - wGold);
  1218→  const goldPrice = vals.goldPrice;
  1219→  const goldPaxg = goldUsd / goldPrice;
  1220→
  1221→  // v1.0: Strategic Prior weights for the 6 non-gold components, renormalised
  1222→  // to the non-gold total (0.27 + 0.20 + 0.09 + 0.08 + 0.05 + 0.05 = 0.74).
  1223→  const basketFiatTotal =
  1224→    STRATEGIC_PRIOR.USD + STRATEGIC_PRIOR.EUR + STRATEGIC_PRIOR.JPY +
  1225→    STRATEGIC_PRIOR.GBP + STRATEGIC_PRIOR.CNY + STRATEGIC_PRIOR.CHF;
  1226→  const w = (k: keyof typeof STRATEGIC_PRIOR) => STRATEGIC_PRIOR[k] / basketFiatTotal;
  1227→  const basket: RedeemBasket[] = [
  1228→    { currency: "USD", token: "USDC", usdValue: fiatUsd * w("USD"), nativeAmount: fiatUsd * w("USD"), weight: STRATEGIC_PRIOR.USD },
  1229→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * w("EUR"), nativeAmount: (fiatUsd * w("EUR")) / fx.EUR_USD, weight: STRATEGIC_PRIOR.EUR },
  1230→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * w("JPY"), nativeAmount: (fiatUsd * w("JPY")) / fx.JPY_USD, weight: STRATEGIC_PRIOR.JPY },
  1231→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * w("GBP"), nativeAmount: (fiatUsd * w("GBP")) / fx.GBP_USD, weight: STRATEGIC_PRIOR.GBP },
  1232→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * w("CNY"), nativeAmount: (fiatUsd * w("CNY")) / fx.CNY_USD, weight: STRATEGIC_PRIOR.CNY },
  1233→    { currency: "CHF", token: "CHF₿", usdValue: fiatUsd * w("CHF"), nativeAmount: (fiatUsd * w("CHF")) / fx.CHF_USD, weight: STRATEGIC_PRIOR.CHF }, // NEW v1.0
  1234→  ];
  1235→
  1236→  // Mutate reserve: burn MTQ from circulating (reduce totalSupply), release assets
  1237→  // USD released proportionally across USDC + USDP + USDT; gold across PAXG + XAUT
  1238→  // B9 FIX: Pre-check that every component has sufficient holdings before
  1239→  // mutating. Previously, Math.max(0, holding - amount) silently floored
  1240→  // negative holdings to 0 — the deficit disappeared, breaking conservation
  1241→  // of value and creating phantom surplus over time. Now we revert the entire
  1242→  // redeem if any component is short. This is the safe choice for mainnet.
  1243→  const usdRelease = basket[0].nativeAmount;
  1244→  const usdThirdRelease = usdRelease / 3;
  1245→  const goldHalf = goldPaxg / 2;
  1246→
  1247→  // B9: Check all holdings are sufficient. If any is short, return ok:false
  1248→  // with a clear reason — do NOT silently floor to 0.
  1249→  if (s.usdc < usdThirdRelease || s.usdp < usdThirdRelease || s.usdt < usdThirdRelease) {
  1250→    return {
  1251→      ok: false, reason: "Insufficient USD stable holdings for redemption basket (B9 conservation guard).",
  1252→      inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0,
  1253→      goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ,
  1254→      newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)),
  1255→      navPerMtq, auditNavPerToken, auditGrossUsdNav, auditGrossUsdIndex,
  1256→      auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex, auditNote,
  1257→    };
  1258→  }
  1259→  if (s.eurc < basket[1].nativeAmount || s.jpy < basket[2].nativeAmount ||
  1260→      s.gbp < basket[3].nativeAmount || s.cny < basket[4].nativeAmount ||
  1261→      s.chf < basket[5].nativeAmount) {
  1262→    return {
  1263→      ok: false, reason: "Insufficient fiat token holdings for redemption basket (B9 conservation guard).",
  1264→      inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0,
  1265→      goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ,
  1266→      newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)),
  1267→      navPerMtq, auditNavPerToken, auditGrossUsdNav, auditGrossUsdIndex,
  1268→      auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex, auditNote,
  1269→    };
  1270→  }
  1271→  if (s.paxg < goldHalf || s.xaut < goldHalf) {
  1272→    return {
  1273→      ok: false, reason: "Insufficient gold (PAXG/XAUT) holdings for redemption basket (B9 conservation guard).",
  1274→      inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0,
  1275→      goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ,
  1276→      newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)),
  1277→      navPerMtq, auditNavPerToken, auditGrossUsdNav, auditGrossUsdIndex,
  1278→      auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex, auditNote,
  1279→    };
  1280→  }
  1281→
  1282→  s.totalSupply -= inputMtq;
  1283→  // B9: All holdings verified sufficient above — subtract directly (no Math.max needed).
  1284→  s.usdc -= usdThirdRelease;
  1285→  s.usdp -= usdThirdRelease;
  1286→  s.usdt -= usdThirdRelease;
  1287→  s.eurc -= basket[1].nativeAmount;
  1288→  s.jpy  -= basket[2].nativeAmount;
  1289→  s.gbp  -= basket[3].nativeAmount;
  1290→  s.cny  -= basket[4].nativeAmount;
  1291→  s.chf  -= basket[5].nativeAmount; // NEW v1.0
  1292→  // Gold released 50/50 PAXG + XAUT
  1293→  s.paxg -= goldHalf;
  1294→  s.xaut -= goldHalf;
  1295→  // §14.1 — gold released to the redeemer comes from the RESERVE buffer first
  1296→  // (index gold is locked). syncReserveFromTotal re-derives reserve = total − index
  1297→  // so the invariant paxg = indexPaxg + reservePaxg is preserved.
  1298→  syncReserveFromTotal(s);
  1299→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet
  1300→  s.updatedAt = Date.now();
  1301→
  1302→  const newVals = reserveAssetValues(s, fx);
  1303→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
  1304→  return {
  1305→    ok: true,
  1306→    inputMtq,
  1307→    mtqPrice: price,
  1308→    grossUsd,
  1309→    feeBps,
  1310→    feeUsd,
  1311→    netUsd,
  1312→    goldUsd,
  1313→    goldPaxg,
  1314→    basket,
  1315→    newCirculatingSupply: circulatingSupply(s),
  1316→    newReserveRatio: newRr,
  1317→    navPerMtq,
  1318→    auditNavPerToken,
  1319→    auditGrossUsdNav,
  1320→    auditGrossUsdIndex,
  1321→    auditDeltaUsd: auditGrossUsdNav - auditGrossUsdIndex,
  1322→    auditNote,
  1323→  };
  1324→}

  1461→export function computeSnapshot(s: ReserveState, fx: FxSnapshot, ctx?: { oracle?: OracleBoard | null; registry?: AssetRecord[] | null }): MetricsSnapshot {

  1467→  const gfb = price; // gfbIndex now mirrors the chain-linked I_t (P_MTQ = I_t × PAR = I_t)

  1469→  const nav = vals.nav;
  1470→  const liability = computeLiability(s, price);
  1471→  const rr = computeReserveRatio(nav, liability);

  1474→  // state (s.riskState.state) — this preserves the RECOVERY hysteresis (the

  1587→    marpExecution: maseData.marpExecution,

  1628→// --- §6/§7/§8.4/§10 v1.0 MASE + 4-state weights + MARP ---------------------
  1629→// The Multi-Asset Stochastic Ensemble (§6/§7) runs 6 candidate models, blends
  1630→// them into a single target weight vector, clamps to per-component
  1631→// admissibility envelopes (§8.1), and EMA-smooths toward the constrained target
  1632→// (§8.4). MARP (§10) then decides per-component whether the deviation between
  1633→// the smoothed target and the observed execution weight justifies a trade
  1634→// (urgency test → no-trade zone → cost-benefit gate → partial correction →
  1635→// daily-turnover cap → execute).
  1636→//
  1637→// `advanceMase(s, fx)` is called ONCE per tick from the pilot-state tick loop.
  1638→// It is the only function that mutates `s.maseSmoothed`. `buildMaseSnapshot()`
  1639→// is a pure helper used by `computeSnapshot()` — it recomputes target/smoothed/
  1640→// MARP from the current fx + vals using the persisted prev-smoothed as the EMA
  1641→// prior, so multiple computeSnapshot calls within a single tick return the
  1642→// same value (idempotent).
  1643→
  1644→interface MaseSnapshotData {
  1645→  mase: NonNullable<MetricsSnapshot["mase"]>;
  1646→  weightStates: NonNullable<MetricsSnapshot["weightStates"]>;
  1647→  marp: NonNullable<MetricsSnapshot["marp"]>;
  1648→  // §14.1 — read-only projection of what MARP WOULD do this tick.
  1649→  marpExecution: NonNullable<MetricsSnapshot["marpExecution"]>;
  1650→  envelopes: MetricsSnapshot["envelopes"];
  1651→}

  1653→/** Estimate per-component annualized volatility from the engine's macro state.
  1654→ *  Pilot approximation: scale baseline vols by VIX / 18.5 (the "normal" VIX
  1655→ *  baseline), so a 30 VIX roughly doubles estimated vols and a 12 VIX halves
  1656→ *  them. Production should use a rolling covariance from real price history. */
  1657→function estimateVolsFromState(s: ReserveState): VolatilityData {
  1658→  const vix = s.lastVix > 0 ? s.lastVix : 18.5;
  1659→  const f = vix / 18.5;
  1660→  return {
  1661→    USD: 0.05 * f,
  1662→    EUR: 0.10 * f,
  1663→    JPY: 0.12 * f,
  1664→    GBP: 0.11 * f,
  1665→    CNY: 0.09 * f,
  1666→    CHF: 0.10 * f,
  1667→    Gold: 0.15 * f,
  1668→  };
  1669→}

  1671→/** Detect the market regime from VIX (primary) + DXY (secondary).
  1672→ *  0 = calm, 1 = normal, 2 = stress, 3 = crisis.
  1673→ *  VIX thresholds: <15 calm, <22 normal, <30 stress, ≥30 crisis.
  1674→ *  DXY extremes (<88 or >115) bump the regime up to ≥ stress. */
  1675→function detectRegime(vix: number, dxy: number, goldVol: number): MarketRegime {
  1676→  let r: 0 | 1 | 2 | 3 = 1;
  1677→  if (vix >= 30) r = 3;
  1678→  else if (vix >= 22) r = 2;
  1679→  else if (vix >= 15) r = 1;
  1680→  else r = 0;
  1681→  if ((dxy >= 115 || dxy <= 88) && r < 2) r = 2;
  1682→  return { regime: r, vix, dxy, goldVol };
  1683→}

  1698→/** Build the observed execution-weight vector from the actual reserve composition.
  1699→ *  §14.1 note: the Gold component's observed weight uses the TOTAL gold
  1700→ *  (= indexPaxg + indexXaut + reservePaxg + reserveXaut, USD-net). Both the
  1701→ *  index gold (locked) and the reserve gold (MARP-rebalanced) serve the
  1702→ *  Strategic Prior Gold weight (26%) — but only the RESERVE gold is available
  1703→ *  for MARP to rebalance. The index gold is locked and only changes when
  1704→ *  commitIndexGold() is called (keeper-role equivalent). */
  1705→function buildObservedWeights(
  1706→  vals: ReturnType<typeof reserveAssetValues>,
  1707→  nav: number,
  1708→): WeightVector {
  1709→  if (nav <= 0) {
  1710→    return { USD: 0, EUR: 0, JPY: 0, GBP: 0, CNY: 0, CHF: 0, Gold: 0 };
  1711→  }
  1712→  return {
  1713→    USD: vals.usdNet / nav,
  1714→    EUR: vals.eurNet / nav,
  1715→    JPY: vals.jpyNet / nav,
  1716→    GBP: vals.gbpNet / nav,
  1717→    CNY: vals.cnyNet / nav,
  1718→    CHF: vals.chfNet / nav,
  1719→    // §14.1 — gold weight = TOTAL gold (index + reserve) / NAV. Both pools
  1720→    // back the 26% Strategic Prior Gold weight; only reserve is MARP-rebalanced.
  1721→    Gold: vals.goldNet / nav,
  1722→  };
  1723→}

  1725→/** Pure helper — computes the MASE + 4-state + MARP + envelope snapshot data.
  1726→ *  Reads `s.maseSmoothed` (does NOT mutate). */
  1727→function buildMaseSnapshot(
  1728→  s: ReserveState,
  1729→  fx: FxSnapshot,
  1730→  vals: ReturnType<typeof reserveAssetValues>,
  1731→  nav: number,
  1732→  rr: number,
  1733→): MaseSnapshotData {
  1734→  const prices = buildPriceData(fx);
  1735→  const vols = estimateVolsFromState(s);
  1736→  const regime = detectRegime(s.lastVix > 0 ? s.lastVix : fx.VIX, s.lastDxy > 0 ? s.lastDxy : fx.DXY, vols.Gold);
  1737→
  1738→  // §6/§7 — MASE ensemble → 6 candidate models + target weight vector
  1739→  const mase = maseEnsemble(vols, prices, null, regime);
  1740→  // §8.1 — Constrain target to per-component admissibility envelopes
  1741→  const constrained = applyEnvelopes(mase.target);
  1742→  // §8.4 — EMA-smooth toward the constrained target (uses prev persisted as prior)
  1743→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1744→  const smoothed = smoothWeights(prevSmoothed, constrained);
  1745→  // §2.3 — Execution = actual observed reserve composition
  1746→  const observed = buildObservedWeights(vals, nav);
  1747→
  1748→  // §10 — MARP per-component rebalancing decision
  1749→  const volsRecord: Record<Component, number> = {
  1750→    USD: vols.USD, EUR: vols.EUR, JPY: vols.JPY, GBP: vols.GBP,
  1751→    CNY: vols.CNY, CHF: vols.CHF, Gold: vols.Gold,
  1752→  };
  1753→  const marpDecisions: MarpDecision[] = marpDecision(observed, smoothed, nav, rr, volsRecord);
  1754→  const totalTradeUsd = marpDecisions
  1755→    .filter((d) => d.shouldTrade)
  1756→    .reduce((a, d) => a + d.tradeUsd, 0);
  1757→
  1758→  // §14.1 — per-component MARP execution summary (READ-ONLY projection).
  1759→  // appliedCount = # of decisions with shouldTrade && level >= 6 (would-execute).
  1760→  // skippedCount  = # of decisions held back (no-trade zone / low urgency /
  1761→  //                 cost-benefit fail / turnover cap / sub-level-6).
  1762→  // totalTradeUsd = Σ tradeUsd for the would-execute decisions.
  1763→  // path = which execution path is currently mutating state (mirrors s.rebalancePath).
  1764→  // NOTE: this is what MARP WOULD do based on the current decisions. The actual
  1765→  // mutation happens in applyMarpRebalance() only when USE_MARP_EXECUTION=true
  1766→  // in pilot-state.ts. applyMarpRebalance may further skip decisions that
  1767→  // breach MAX_DAILY_TURNOVER or DIRECTION_LOCK_HOURS at execution time.
  1768→  const wouldExecute = marpDecisions.filter((d) => d.shouldTrade && d.level >= 6);
  1769→  const marpExecution = {
  1770→    appliedCount: wouldExecute.length,
  1771→    skippedCount: marpDecisions.length - wouldExecute.length,
  1772→    totalTradeUsd: wouldExecute.reduce((a, d) => a + d.tradeUsd, 0),
  1773→    path: s.rebalancePath,
  1774→  };
  1775→
  1776→  // §8.1 — Admissibility envelope status per component
  1777→  // "warn" = within 10% of the band edge; "breach" = outside the envelope.
  1778→  const envelopes = COMPONENTS.map((c) => {
  1779→    const env = ADMISSIBILITY_ENVELOPES[c];
  1780→    const current = observed[c];
  1781→    const lower = env.lower;
  1782→    const upper = env.upper;
  1783→    const margin = (upper - lower) * 0.1;
  1784→    let status: "ok" | "warn" | "breach" = "ok";
  1785→    if (current < lower || current > upper) status = "breach";
  1786→    else if (current < lower + margin || current > upper - margin) status = "warn";
  1787→    return { component: c, lower, upper, current, status };
  1788→  });
  1789→
  1790→  return {
  1791→    mase: {
  1792→      models: mase.models.map((m) => ({
  1793→        id: m.id,
  1794→        name: m.name,
  1795→        weights: m.weights as Record<string, number>,
  1796→      })),
  1797→      ensembleTarget: mase.target as Record<string, number>,
  1798→    },
  1799→    weightStates: {
  1800→      prior: STRATEGIC_PRIOR as Record<string, number>,
  1801→      target: constrained as Record<string, number>,
  1802→      smoothed: smoothed as Record<string, number>,
  1803→      execution: observed as Record<string, number>,
  1804→    },
  1805→    marp: {
  1806→      decisions: marpDecisions.map((d) => ({
  1807→        shouldTrade: d.shouldTrade,
  1808→        component: d.component,
  1809→        direction: d.direction,
  1810→        tradeUsd: d.tradeUsd,
  1811→        urgency: d.urgency,
  1812→        reason: d.reason,
  1813→        level: d.level,
  1814→      })),
  1815→      totalTradeUsd,
  1816→    },
  1817→    // §14.1 — MARP execution summary (read-only projection; mirrors s.rebalancePath).
  1818→    marpExecution,
  1819→    envelopes,
  1820→  };
  1821→}
  1822→
  1823→/** Advance MASE state once per tick. Mutates `s.maseSmoothed` (EMA prior for
  1824→ *  the next tick) and `s.chainIndex` (commits the new weights to preserve
  1825→ *  zero-artificial-return continuity per Master §9.2 COO-16). Called from
  1826→ *  the pilot-state tick loop, AFTER advanceMacro (so lastVix/lastDxy are
  1827→ *  fresh) and BEFORE the first computeSnapshot of the tick (so the snapshot
  1828→ *  reads the freshly-persisted smoothed weights).
  1829→ *
  1830→ *  P0-FIX-1 (chain-linking): the new smoothed weights are committed to the
  1831→ *    chain index via commitChainIndexWeights(). This updates G_t (the
  1832→ *    cumulative chain-link factor) and the prevWeights / prevPrices used by
  1833→ *    the next advanceIndex() call. The divisor D_t = B_t^- / B_t^+ ensures
  1834→ *    the rebalance creates ZERO artificial index return.
  1835→ *
  1836→ *  P0-FIX-3 + §8.4/§7.4 (velocity + stress-adaptive smoothing): the EMA
  1837→ *    smoothing now uses smoothWeightsAdaptive() with the canonical risk
  1838→ *    state's stress level and the target velocity (max component-wise |Δ|
  1839→ *    between the previous smoothed and the new constrained target). In
  1840→ *    stress states (STRESS/DEFENSIVE/EMERGENCY), the smoothing slows by 50%
  1841→ *    to avoid chasing volatile prices. Fast target jumps (|Δ| > 5%) are
  1842→ *    damped by up to 50% (whip-saw guard). */
  1843→export function advanceMase(s: ReserveState, fx: FxRates): void {
  1844→  const prices = buildPriceData(fx);
  1845→  const vols = estimateVolsFromState(s);
  1846→  const regime = detectRegime(
  1847→    s.lastVix > 0 ? s.lastVix : fx.VIX,
  1848→    s.lastDxy > 0 ? s.lastDxy : fx.DXY,
  1849→    vols.Gold,
  1850→  );
  1851→  const mase = maseEnsemble(vols, prices, null, regime);
  1852→  const constrained = applyEnvelopes(mase.target);
  1853→  const prevSmoothed = s.maseSmoothed ?? (STRATEGIC_PRIOR as WeightVector);
  1854→  // §8.4 + §7.4 — velocity + stress-adaptive smoothing.
  1855→  const velocity = targetVelocity(prevSmoothed, constrained);
  1856→  const sl = stressLevel(s.riskState.state);
  1857→  s.maseSmoothed = smoothWeightsAdaptive(prevSmoothed, constrained, SMOOTHING_LAMBDA, velocity, sl);
  1858→  s.maseLastAt = Date.now();
  1859→  // P0-FIX-1 — commit the new smoothed weights to the chain index. This
  1860→  // computes the divisor D_t = B_t^- / B_t^+ that preserves index continuity
  1861→  // (zero artificial return), updates G_t, and updates prevWeights +
  1862→  // prevPrices for the next advanceIndex() call. Wrapped in try/catch so a
  1863→  // chain-index commit failure doesn't break the tick loop.
  1864→  try {
  1865→    commitChainIndexWeights(s, fx, s.maseSmoothed);
  1866→  } catch (e) {
  1867→    if (typeof console !== "undefined") console.error("[mtq-engine] chain index commit error:", e);
  1868→  }
  1869→}

  1879→// Advance macro history (call ~ every tick; dtHours is simulated elapsed)
  1880→export function advanceMacro(s: ReserveState, vix: number, dxy: number, dtHours: number) {
  1881→  const v = clamp(vix, VIX_MIN, VIX_MAX);
  1882→  const d = clamp(dxy, DXY_MIN, DXY_MAX);
  1883→  s.lastVix = v;
  1884→  s.lastDxy = d;
  1885→  s.vixHistory.push(v);
  1886→  s.dxyHistory.push(d);
  1887→  if (s.vixHistory.length > ROLLING_WINDOW_DAYS) s.vixHistory.shift();
  1888→  if (s.dxyHistory.length > ROLLING_WINDOW_DAYS) s.dxyHistory.shift();

  1890→  const dStats = rollingStats(s.dxyHistory);
  1891→  const zVix = vStats.sd > 1e-9 ? (v - vStats.mean) / (vStats.sd + 1e-9) : 0;
  1892→  const zDxy = dStats.sd > 1e-9 ? (d - dStats.mean) / (dStats.sd + 1e-9) : 0;
  1893→  const rawTarget = computeRawTargetGoldWeight(zVix, zDxy);
  1894→  s.smoothedGoldWeight = smoothGoldWeight(s.smoothedGoldWeight, rawTarget);

  1919→// --- §11.3 Reintegration Score (Anti-Gaming) --------------------------------
  1920→// R_score = w1·TimeInBand(48h) + w2·LiquidityDepth + w3·OracleAgreement + w4·(1−Volatility)
  1921→// All inputs normalised to [0,1]; threshold >0.80 → staged re-purchase 25%→50%→100%.
  1922→export function updateReintegration(
  1923→  s: ReserveState,
  1924→  board: OracleBoard | null,
  1925→  dtHours: number,
  1926→) {
  1927→  const currencies: ("USD" | "EUR" | "GBP" | "JPY" | "CNY")[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
  1928→  for (const c of currencies) {
  1929→    const r = s.reintegration[c];
  1930→    // Time-in-band: increases when peg is inside [0.98,1.02]; decays when outside.
  1931→    const inBand = s.pegHealth[c] >= EJECT_DEPEG_BAND_LOWER && s.pegHealth[c] <= EJECT_DEPEG_BAND_UPPER;
  1932→    // dtHours is simulated elapsed; we approximate 48h window fraction.
  1933→    if (inBand) {
  1934→      r.timeInBand48h = Math.min(1, r.timeInBand48h + dtHours / 48);
  1935→    } else {
  1936→      r.timeInBand48h = Math.max(0, r.timeInBand48h - dtHours / 48);
  1937→    }
  1938→    // Liquidity depth: randomised around per-currency baseline (honest: simulated).
  1939→    const base = c === "USD" ? 1.0 : c === "EUR" ? 0.95 : c === "CNY" ? 0.4 : 0.5;
  1940→    r.liquidityDepth = clamp(base + (Math.random() - 0.5) * 0.1, 0, 1);
  1941→    // Oracle agreement: valid feeds / 3 (from oracle board if present, else last known)
  1942→    if (board) {
  1943→      const pairMap: Record<string, string> = { USD: "—", EUR: "EUR/USD", GBP: "GBP/USD", JPY: "JPY/USD", CNY: "CNY/USD" };
  1944→      const cons = board.pairs.find((p) => p.pair === pairMap[c]);
  1945→      if (cons) {
  1946→        r.oracleAgreement = cons.validCount / 3;
  1947→        // Volatility from oracle spread (bps → fraction)
  1948→        r.volatility = clamp(cons.spreadBps / 100, 0, 1);
  1949→      }
  1950→    }
  1951→    // R_score
  1952→    r.score =
  1953→      REINTEGRATION_WEIGHTS.w1_TimeInBand * r.timeInBand48h +
  1954→      REINTEGRATION_WEIGHTS.w2_LiquidityDepth * r.liquidityDepth +
  1955→      REINTEGRATION_WEIGHTS.w3_OracleAgreement * r.oracleAgreement +
  1956→      REINTEGRATION_WEIGHTS.w4_OneMinusVolatility * (1 - r.volatility);
  1957→    // Repurchase staging: only for currencies that were previously ejected
  1958→    if (s.ejectStage[c] > 0 && r.score > REINTEGRATION_THRESHOLD) {
  1959→      r.repurchaseStage = Math.min(3, r.repurchaseStage + 1);
  1960→      // As repurchase progresses, reduce eject stage
  1961→      if (r.repurchaseStage >= 3) {
  1962→        s.ejectStage[c] = Math.max(0, s.ejectStage[c] - 1);
  1963→        if (s.ejectStage[c] === 0) r.repurchaseStage = 0;
  1964→      }
  1965→    } else if (r.score < REINTEGRATION_THRESHOLD * 0.6) {
  1966→      r.repurchaseStage = 0;
  1967→    }
  1968→  }
  1969→}

  1988→// --- §8.5 First-Loss Waterfall ----------------------------------------------
  1989→// 5 layers: (1) Operational Surplus → (2) Buffer Fiat → (3) Buffer Gold →
  1990→// (4) Core Fiat → (5) Core Gold. Apply a loss; track consumed per layer.
  1991→export function applyLoss(s: ReserveState, vals: { fiatNet: number; goldNet: number }, lossUsd: number) {
  1992→  let remaining = lossUsd;
