   114→// ============================================================================
   115→//  MaliciousUSDC — re-enters mtq.mint() / mtq.redeem() during transferFrom
   116→//  (H1 / H2 reentrancy-guard audit vector).
   117→// ----------------------------------------------------------------------------
   118→//  The V3 contract pulls USDC via `usdc.transferFrom(msg.sender, reserveVault,
   119→//  usdcAmount)` during mint() and pushes USDC via `usdc.transferFrom(
   120→//  reserveVault, msg.sender, usdcOut)` during redeem(). An ERC-777-style token
   121→//  with a callback would re-enter the outer function. This mock simulates the
   122→//  vector: the malicious token's transferFrom() calls back into mtq.mint() /
   123→//  mtq.redeem() while the outer call is still inside the nonReentrant scope.
   124→//  The nonReentrant guard MUST block the inner call (Err02). The MaliciousUSDC
   125→//  swallows the inner revert via try/catch and exposes `innerReverted` so the
   126→//  test can assert the guard held.
   127→// ============================================================================
   128→contract MaliciousUSDC is MockUSDC {
   129→    MTQSigmaV2 public target;
   130→    enum Vector { NONE, MINT, REDEEM }
   131→    Vector  public vector = Vector.NONE;
   132→    uint256 public reenterAmount;
   133→    bool    public innerReverted;   // true if the inner call was blocked by nonReentrant
   134→    bool    public triggered;      // true if the malicious callback fired
   135→
   136→    function setTarget(MTQSigmaV2 _t) external { target = _t; }
   137→
   138→    function armMint(uint256 amount) external {
   139→        vector = Vector.MINT;
   140→        reenterAmount = amount;
   141→        innerReverted = false;
   142→        triggered = false;
   143→    }
   144→
   145→    function armRedeem(uint256 amount) external {
   146→        vector = Vector.REDEEM;
   147→        reenterAmount = amount;
   148→        innerReverted = false;
   149→        triggered = false;
   150→    }
   151→
   152→    function disarm() external { vector = Vector.NONE; }
   153→
   154→    function transferFrom(address from, address to, uint256 amount) public override returns (bool) {
   155→        // 1. Execute the legitimate transfer first so the contract's
   156→        //    accounting updates have already happened once.
   157→        bool ok = super.transferFrom(from, to, amount);
   158→
   159→        // 2. Attempt re-entry only on the armed vector.
   160→        if (address(target) != address(0)) {
   161→            if (vector == Vector.MINT) {
   162→                vector = Vector.NONE;          // one-shot
   163→                triggered = true;
   164→                this.approve(address(target), type(uint256).max);
   165→                try target.mint(reenterAmount) {
   166→                    innerReverted = false;    // guard FAILED — re-entry succeeded
   167→                } catch {
   168→                    innerReverted = true;     // guard held — nonReentrant reverted
   169→                }
   170→            } else if (vector == Vector.REDEEM) {
   171→                vector = Vector.NONE;
   172→                triggered = true;
   173→                try target.redeem(reenterAmount) {
   174→                    innerReverted = false;    // guard FAILED
   175→                } catch {
   176→                    innerReverted = true;     // guard held
   177→                }
   178→            }
   179→        }
   180→        return ok;
   181→    }
   182→}

   184→// ============================================================================
   185→//  MaliciousAssetRegistry — re-enters mtq.executeRebalance() during getAsset
   186→//  (H1 reentrancy-guard audit vector for executeRebalance).
   187→// ----------------------------------------------------------------------------
   188→//  executeRebalance calls getReserveRatio() → getNAV(), and getNAV() does
   189→//  `try IAssetRegistry(assetRegistry).getAsset(codes[i])`. We replace the
   190→//  asset registry with this malicious implementation that attempts to call
   191→//  mtq.executeRebalance(innerTrades) from inside its getAsset(). The inner
   192→//  call MUST be blocked by nonReentrant. The try/catch swallows the inner
   193→//  revert and exposes `innerReverted` for the test to assert. The outer
   194→//  executeRebalance call continues (the catch block in getNAV swallows the
   195→//  revert and `nav += held`).
   196→// ============================================================================
   197→contract MaliciousAssetRegistry {
   198→    MTQSigmaV2                    public target;
   199→    MTQSigmaV2.RebalanceTrade[]   public innerTrades;
   200→    bool                          public triggered;
   201→    bool                          public innerReverted;
   202→
   203→    function setTarget(MTQSigmaV2 _t) external { target = _t; }
   204→    function setInnerTrades(MTQSigmaV2.RebalanceTrade[] memory t) external {
   205→        // Solidity 0.8.20 can't copy struct arrays to storage in one assignment;
   206→        // copy element-by-element via push().
   207→        for (uint256 i = innerTrades.length; i > 0; i--) innerTrades.pop();
   208→        for (uint256 i = 0; i < t.length; i++) innerTrades.push(t[i]);
   209→    }
   210→
   211→    function getAsset(bytes32) external returns (address, uint256, uint8, bytes32) {
   212→        if (!triggered && innerTrades.length > 0) {
   213→            triggered = true;
   214→            try target.executeRebalance(innerTrades) {
   215→                innerReverted = false;    // guard FAILED
   216→            } catch {
   217→                innerReverted = true;     // guard held
   218→            }
   219→        }
   220→        return (address(0), 0, 0, bytes32(0));
   221→    }
   222→}

   952→    // ---- 7.1 S5: Gold +50% shock — survival ≥ 95% (chain-linked index) ----
   953→    // The S5 stochastic stress test. Fuzz the seed, apply gold +50% at tick 1
   954→    // with light gaussian-ish noise afterward, advance the chain index,
   955→    // revalue the gold reserve mirror to the new price (the production
   956→    // keeper's role), and assert RR >= 1.00 at every tick.
   957→    //
   958→    // Pseudocode:
   959→    //   for each fuzz seed:
   960→    //     rng = mulberry32(seed)
   961→    //     prices = base_fixings
   962→    //     gold_qty = reserveHeldUsd[Gold] / GOLD_BASE   // physical gold units
   963→    //     survived = true
   964→    //     for t in 0..30:
   965→    //       if t == 1: prices[6] *= 1.50       // +50% gold shock
   966→    //       else if t > 1: prices[i] += gaussian(rng, 0, 0.001) for each i
   967→    //       if t > 0:
   968→    //         mtq.advanceIndex(prices)
   969→    //         mtq.setReserveHolding(Gold, gold_qty * prices[6] / 1e18) // revalue
   970→    //       if mtq.getReserveRatio() < 1.00e18: survived = false
   971→    //     assert survived
   972→    //
   973→    // With the chain-linked index, gold +50% → +13% index growth (NOT +50%),
   974→    // so the reserve mirror (revalued to track the gold price) keeps RR ≥ 1.00.
   975→    // REQUIRES P1/P2 — getReserveRatio calls getLiability → getMTQPrice.
   976→    function testFuzz_S5_GoldPlus50_Survival100Pct(uint256 seed) public {
   977→        vm.assume(seed < 1000);
   978→
   979→        // Fresh contract per fuzz case (forge calls setUp before each case).
   980→        MTQSigmaV2 mtqFuzz = _deployFuzzMTQ();
   981→
   982→        uint256 rngState = seed ^ 0x6d2b79f5;
   983→        uint256[7] memory prices = BASE_PRICES;
   984→
   985→        // Capture gold "physical" units (USD value / USD per ounce).
   986→        uint256 goldUsd0 = mtqFuzz.reserveHeldUsd(MTQSigmaV2.Component.Gold);
   987→        uint256 goldQty  = goldUsd0 * 1e18 / GOLD_BASE; // 1e18-scale gold units
   988→
   989→        bool survived = true;
   990→        for (uint256 t = 0; t < 30; t++) {
   991→            if (t == 1) {
   992→                // +50% gold shock (the headline test condition).
   993→                prices[6] = prices[6] * 150 / 100;
   994→            } else if (t > 1) {
   995→                // Light gaussian-ish noise on every component (±0.1%).
   996→                for (uint256 i = 0; i < 7; i++) {
   997→                    rngState = uint256(keccak256(abi.encodePacked(rngState, i, t)));
   998→                    int256 signed = int256(uint256(rngState)) - int256(type(int128).max);
   999→                    int256 scaled = signed / 1e15; // ±0.1% noise
  1000→                    if (scaled != 0) {
  1001→                        uint256 oldP = prices[i];
  1002→                        if (scaled > 0) {
  1003→                            prices[i] = oldP + (oldP * uint256(scaled)) / 1e18;
  1004→                        } else {
  1005→                            uint256 dec = oldP * uint256(-scaled) / 1e18;
  1006→                            prices[i] = (dec < oldP) ? (oldP - dec) : 1;
  1007→                        }
  1008→                    }
  1009→                }
  1010→            }
  1011→
  1012→            if (t > 0) {
  1013→                vm.prank(address(this));
  1014→                mtqFuzz.advanceIndex(prices);
  1015→                // Revalue the gold reserve to the new price (production
  1016→                // keeper's role — mirrors the TS engine's mark-to-market).
  1017→                uint256 newGoldUsd = goldQty * prices[6] / 1e18;
  1018→                vm.prank(address(this));
  1019→                mtqFuzz.setReserveHolding(MTQSigmaV2.Component.Gold, newGoldUsd);
  1020→            }
  1021→
  1022→            uint256 rr = mtqFuzz.getReserveRatio();
  1023→            if (rr < 1.00e18) {
  1024→                survived = false;
  1025→            }
  1026→        }
  1027→        assertTrue(survived, "S5 (gold +50%) - RR stays >= 1.00 (chain-linked dampens the shock)");
  1028→    }

  1148→    // ---- 7.5 State transitions are monotonic (no backward jumps w/o confirmation) ----
  1149→    // Fuzz a sequence of (rr, lcr) inputs; assert that any transition to a
  1150→    // LESS-restrictive state only happens after the 48h confirmation period.
  1151→    // More-restrictive transitions are immediate. Pure check — updateState
  1152→    // is independent of P1/P2 (no getMTQPrice call).
  1153→    function testFuzz_StateTransitions_Monotonic(uint256 seed) public {
  1154→        vm.assume(seed < 1000);
  1155→
  1156→        // Fresh contract so currentState = NORMAL.
  1157→        MTQSigmaV2 mtqFuzz = _deployFreshMTQ();
  1158→
  1159→        uint256 rngState = seed ^ 0x517cc1b7;
  1160→        uint256 lastEntryTime = mtqFuzz.stateEntryTime(mtqFuzz.currentState());
  1161→
  1162→        for (uint256 t = 0; t < 20; t++) {
  1163→            // Derive (rr, lcr) from the rng. Bounded to plausible ranges.
  1164→            rngState = uint256(keccak256(abi.encodePacked(rngState, t)));
  1165→            uint256 rr  = 0.95e18 + (rngState % 0.30e18);   // RR ∈ [0.95, 1.25]

  1167→            uint256 lcr = 0.70e18 + (rngState % 0.40e18);   // LCR ∈ [0.70, 1.10]
  1168→

  1170→            uint256 warpDelta = rngState % 49 hours;
  1171→            vm.warp(block.timestamp + warpDelta);
  1172→

  1174→            vm.prank(address(this));
  1175→            mtqFuzz.updateState(rr, lcr);
  1176→            uint256 stateAfter = uint256(mtqFuzz.currentState());
  1177→
  1178→            if (stateAfter != stateBefore) {
  1179→                // Determine restrictiveness (lower number = less restrictive).
  1180→                // NORMAL=0, CAUTION=1, RECOVERY=2, STRESS=3, DEFENSIVE=4, EMERGENCY=5
  1181→                // (per the contract's _restrictiveness function — note that
  1182→                // RECOVERY is ranked BETWEEN CAUTION and STRESS, even though
  1183→                // its enum index is 5).
  1184→                uint256 rankBefore = _rankOf(stateBefore);
  1185→                uint256 rankAfter  = _rankOf(stateAfter);
  1186→                bool moreRestrictive = rankAfter > rankBefore;
  1187→

  1189→                    // More-restrictive transitions are immediate — no warp requirement.
  1190→                    assertTrue(true, "more-restrictive transition (immediate)");

  1192→                    // Less-restrictive transition: 48h must have elapsed since
  1193→                    // the current state was entered.

  1196→                        mtqFuzz.RECOVERY_CONFIRMATION_PERIOD(),
  1197→                        "less-restrictive transition only after 48h confirmation"
  1198→                    );
  1199→                }
  1200→                lastEntryTime = mtqFuzz.stateEntryTime(mtqFuzz.currentState());
  1201→            }
  1202→        }
  1203→    }

  1209→    /// @dev Mirror of the contract's _restrictiveness rank function.
  1210→    ///      NORMAL=0, CAUTION=1, RECOVERY=2, STRESS=3, DEFENSIVE=4, EMERGENCY=5.
  1211→    function _rankOf(uint256 s) internal pure returns (uint256) {
  1212→        if (s == 0) return 0; // NORMAL
  1213→        if (s == 1) return 1; // CAUTION
  1214→        if (s == 5) return 2; // RECOVERY
  1215→        if (s == 2) return 3; // STRESS
  1216→        if (s == 3) return 4; // DEFENSIVE
  1217→        return 5;             // EMERGENCY (4)
  1218→    }
