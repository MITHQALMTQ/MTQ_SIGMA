    97→    97→function mulberry32(seed: number): () => number {
    98→    98→  let a = seed | 0;
    99→    99→  return () => {
   100→   100→    a |= 0;
   101→   101→    a = (a + 0x6D2B79F5) | 0;
   102→   102→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   103→   103→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   104→   104→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   105→   105→  };
   106→   106→}
   107→   107→
   108→   108→function gaussian(rng: () => number, mean = 0, sd = 1): number {
   109→   109→  const u1 = Math.max(1e-9, rng());
   110→   110→  const u2 = rng();
   111→   111→  return mean + sd * Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
   112→   112→}
   113→   113→
   114→   114→// Cauchy distribution — heavy-tailed (no finite variance).
   115→   115→function cauchy(rng: () => number, location = 0, scale = 1): number {
   116→   116→  return location + scale * Math.tan(Math.PI * (rng() - 0.5));
   117→   117→}

   119→   119→function clamp(v: number, lo: number, hi: number): number {
   120→   120→  return Math.max(lo, Math.min(hi, v));

   123→   123→function clampSym(v: number, lo: number, hi: number): number {

   159→   159→interface TickMetrics {
   160→   160→  rr: number;
   161→   161→  lcr: number;

   170→   170→
   171→   171→function stepEngine(
   172→   172→  s: ReserveState,
   173→   173→  fx: FxSnapshot,
   174→   174→  opts: { doRebalance?: boolean; doMacro?: boolean } = {},
   175→   175→): TickMetrics {
   176→   176→  const doRebalance = opts.doRebalance ?? true;
   177→   177→  const doMacro = opts.doMacro ?? true;
   178→   178→
   179→   179→  // Simulate "1 day has passed" so the rebalancer's daily-turnover cap and
   180→   180→  // 24h direction lock do not permanently block trades inside a tight MC
   181→   181→  // loop (the engine uses wall-clock time, which doesn't advance here).
   182→   182→  // This is the standard "1 tick = 1 day" interpretation for MC stress sims.
   183→   183→  s.dailyTurnoverUsd = 0;
   184→   184→  s.lastTurnoverResetAt = Date.now();
   185→   185→  s.lastTradeAt = 0;
   186→   186→
   187→   187→  const gfb = computeGfbIndex(fx);
   188→   188→  const price = computeMtqPrice(gfb);
   189→   189→  const vals = reserveAssetValues(s, fx);
   190→   190→  const nav = vals.nav;
   191→   191→  const liability = computeLiability(s, price);
   192→   192→  const rr = computeReserveRatio(nav, liability);
   193→   193→  const lcr = computeLcr(s, vals, price);
   194→   194→  const status = determineStatus(rr, lcr);
   195→   195→
   196→   196→  if (doMacro) advanceMacro(s, fx.VIX, fx.DXY, 24);
   197→   197→  updateBufferState(s, rr);
   198→   198→
   199→   199→  if (doRebalance && Number.isFinite(rr)) {
   200→   200→    const decision = evaluateRebalance(s, vals, rr);
   201→   201→    if (decision.shouldRebalance) {
   202→   202→      applyRebalanceTrade(s, decision, vals.goldPrice);
   203→   203→    }
   204→   204→  }
   205→   205→
   206→   206→  return {
   207→   207→    rr, lcr, status, price, nav, liability,
   208→   208→    inBand: priceInSafetyBand(price),
   209→   209→    goldPrice: vals.goldPrice,
   210→   210→    goldNet: vals.goldNet,
   211→   211→  };
   212→   212→}

   214→   214→// Initialise reserve + mint sufficient circulating supply to bring RR close to 1.10.
   215→   215→// Returns the post-mint circulating supply and starting snapshot.
   216→   216→//
   217→   217→// CRITICAL AUDIT NOTE: After `applyMint(9M USD)`, the gold weight of NAV drops to
   218→   218→// ~2.8% (genesis had 26% gold but the $9M user mint adds only USD). The engine's
   219→   219→// rebalancer would restore gold to ~26% over ~7 days, but in a tight MC loop the
   220→   220→// daily-turnover reset uses wall-clock time (so the cap is never lifted within a
   221→   221→// sub-second run). We therefore:
   222→   222→//   (a) Call `redistributeToStrategicPrior(s, fx)` after the initial mint — this
   223→   223→//       preserves total NAV + total supply but re-splits the holdings across all
   224→   224→//       7 Strategic Prior components (so gold = 26% of NAV from the start).
   225→   225→//   (b) Reset `s.dailyTurnoverUsd`, `s.lastTurnoverResetAt`, `s.lastTradeAt` at the
   226→   226→//       start of each tick in `stepEngine` so each tick simulates "1 day passed"
   227→   227→//       (the rebalancer gets fresh daily turnover + lifted direction lock).
   228→   228→function setupInitialState(initialMintUsd = 9_000_000): { state: ReserveState; fx: FxSnapshot; circ: number; startRr: number } {
   229→   229→  const fx = makeFx();
   230→   230→  const s = initReserveState(BASE_FIXINGS.XAU_USD);
   231→   231→  const snap0 = computeSnapshot(s, fx);
   232→   232→  applyMint(s, fx, snap0.status, initialMintUsd);
   233→   233→  // Redistribute holdings to Strategic Prior composition (preserves total NAV,
   234→   234→  // restores gold weight to 26%). This faithfully matches the engine's design
   235→   235→  // intent (Strategic Prior is the long-term anchor), and removes the
   236→   236→  // "post-mint USD overweight" transient that would otherwise dominate the
   237→   237→  // early MC ticks.
   238→   238→  redistributeToStrategicPrior(s, fx);
   239→   239→  const vals = reserveAssetValues(s, fx);
   240→   240→  const price = computeMtqPrice(computeGfbIndex(fx));
   241→   241→  const liability = computeLiability(s, price);
   242→   242→  const rr = computeReserveRatio(vals.nav, liability);
   243→   243→  return { state: s, fx, circ: circulatingSupply(s), startRr: rr };
   244→   244→}

   246→   246→// Re-split the reserve holdings across the 7 Strategic Prior components.
   247→   247→// Preserves total NAV (modulo tiny haircut rounding) and totalSupply.
   248→   248→function redistributeToStrategicPrior(s: ReserveState, fx: FxSnapshot): void {
   249→   249→  const vals = reserveAssetValues(s, fx);
   250→   250→  const navNet = vals.nav;
   251→   251→  if (navNet <= 0) return;
   252→   252→  // Target USD-notional (net of haircut) per component
   253→   253→  const usdNetTarget = navNet * STRATEGIC_PRIOR.USD;
   254→   254→  const eurNetTarget = navNet * STRATEGIC_PRIOR.EUR;
   255→   255→  const jpyNetTarget = navNet * STRATEGIC_PRIOR.JPY;
   256→   256→  const gbpNetTarget = navNet * STRATEGIC_PRIOR.GBP;
   257→   257→  const cnyNetTarget = navNet * STRATEGIC_PRIOR.CNY;
   258→   258→  const chfNetTarget = navNet * STRATEGIC_PRIOR.CHF;
   259→   259→  const goldNetTarget = navNet * STRATEGIC_PRIOR.Gold;
   260→   260→  // Convert net → gross via haircut factors, then to native units via FX
   261→   261→  const usdGrossTotal = usdNetTarget / (1 - HAIRCUTS.USD);
   262→   262→  const eurGross = eurNetTarget / (1 - HAIRCUTS.EUR) / fx.EUR_USD;
   263→   263→  const jpyGross = jpyNetTarget / (1 - HAIRCUTS.JPY) / fx.JPY_USD;
   264→   264→  const gbpGross = gbpNetTarget / (1 - HAIRCUTS.GBP) / fx.GBP_USD;
   265→   265→  const cnyGross = cnyNetTarget / (1 - HAIRCUTS.CNY) / fx.CNY_USD;
   266→   266→  const chfGross = chfNetTarget / (1 - HAIRCUTS.CHF) / fx.CHF_USD;
   267→   267→  const goldGrossTotal = goldNetTarget / (1 - HAIRCUTS.XAU) / fx.XAU_USD;
   268→   268→  // USD split across 3 issuers (1/3 each) per §5.6 concentration policy
   269→   269→  s.usdc = usdGrossTotal / 3;
   270→   270→  s.usdp = usdGrossTotal / 3;
   271→   271→  s.usdt = usdGrossTotal / 3;
   272→   272→  s.eurc = eurGross;
   273→   273→  s.jpy = jpyGross;
   274→   274→  s.gbp = gbpGross;
   275→   275→  s.cny = cnyGross;
   276→   276→  s.chf = chfGross;
   277→   277→  // Gold split 50/50 PAXG/XAUT (per §5.6), then split 50/50 index/reserve (per §14.1)
   278→   278→  s.paxg = goldGrossTotal / 2;
   279→   279→  s.xaut = goldGrossTotal / 2;
   280→   280→  s.indexPaxg = s.paxg / 2;
   281→   281→  s.indexXaut = s.xaut / 2;
   282→   282→  s.reservePaxg = s.paxg / 2;
   283→   283→  s.reserveXaut = s.xaut / 2;
   284→   284→  s.updatedAt = Date.now();
   285→   285→}

   301→   301→function summarizeTrajectory(
   302→   302→  minRR: number,
   303→   303→  worstStatus: string,
   304→   304→  finalStatus: string,
   305→   305→  recoveryTick: number,
   306→   306→  pegStableTicks: number,

   308→   308→  statusTransitions: number,

   310→   310→  return {
   311→   311→    survived: Number.isFinite(minRR) ? minRR >= RR_HARD : true,
   312→   312→    minRR: Number.isFinite(minRR) ? minRR : 999,
   313→   313→    worstStatus,
   314→   314→    finalStatus,
   315→   315→    recoveryTick,

   321→   321→// Aggregate TrajectoryResult[] → StressResult
   322→   322→function aggregateRuns(
   323→   323→  scenario: string,
   324→   324→  description: string,
   325→   325→  runs: TrajectoryResult[],
   326→   326→  notes: string[] = [],
   327→   327→): StressResult {
   328→   328→  const n = runs.length;
   329→   329→  const survivors = runs.filter((r) => r.survived);
   330→   330→  const breaches = runs.filter((r) => !r.survived);
   331→   331→  const minRRs = runs.map((r) => r.minRR);
   332→   332→  const finiteMinRRs = minRRs.filter((x) => Number.isFinite(x));
   333→   333→  const meanMinRR = finiteMinRRs.length > 0
   334→   334→    ? finiteMinRRs.reduce((a, b) => a + b, 0) / finiteMinRRs.length
   335→   335→    : 999;
   336→   336→  const worstMinRR = finiteMinRRs.length > 0 ? Math.min(...finiteMinRRs) : 0;
   337→   337→  const recoveryTicks = runs
   338→   338→    .filter((r) => r.recoveryTick > 0)
   339→   339→    .map((r) => r.recoveryTick);
   340→   340→  const meanTimeToRecover = recoveryTicks.length > 0
   341→   341→    ? recoveryTicks.reduce((a, b) => a + b, 0) / recoveryTicks.length
   342→   342→    : -1;
   343→   343→  const pegStabilityPct = runs.reduce((a, r) => a + r.pegStablePct, 0) / n;
   344→   344→  // Most common final status
   345→   345→  const statusCounts: Record<string, number> = {};
   346→   346→  for (const r of runs) statusCounts[r.finalStatus] = (statusCounts[r.finalStatus] ?? 0) + 1;
   347→   347→  const finalStatus = Object.entries(statusCounts).sort((a, b) => b[1] - a[1])[0][0];
   348→   348→  // Worst status entered across all runs (max rank)
   349→   349→  const worstStatus = runs.reduce(
   350→   350→    (acc, r) => (STATUS_RANK[r.worstStatus] > STATUS_RANK[acc] ? r.worstStatus : acc),
   351→   351→    "NORMAL",
   352→   352→  );
   353→   353→  return {
   354→   354→    scenario,
   355→   355→    description,
   356→   356→    runs: n,
   357→   357→    survivalRate: survivors.length / n,
   358→   358→    meanMinRR,
   359→   359→    worstMinRR,
   360→   360→    meanTimeToRecover,
   361→   361→    pegStabilityPct,
   362→   362→    finalStatus,
   363→   363→    worstStatus,
   364→   364→    breachCount: breaches.length,
   365→   365→    notes,
   366→   366→  };
   367→   367→}

   373→   373→interface HistoryRow {
   374→   374→  EUR: number; GBP: number; JPY: number; CNY: number; CHF: number; XAU: number;
   375→   375→  VIX: number; DXY: number;
   376→   376→}
   377→   377→
   378→   378→function buildSyntheticHistory(seed: number, days: number): HistoryRow[] {
   379→   379→  const rng = mulberry32(seed);
   380→   380→  const rows: HistoryRow[] = [];
   381→   381→  let EUR = 1.05, GBP = 1.25, JPY = 0.0067, CNY = 0.14, CHF = 0.88, XAU = 2500, VIX = 18.5, DXY = 104.2;
   382→   382→  for (let i = 0; i < days; i++) {
   383→   383→    EUR *= (1 + gaussian(rng, 0, 0.005));
   384→   384→    GBP *= (1 + gaussian(rng, 0, 0.005));
   385→   385→    JPY *= (1 + gaussian(rng, 0, 0.006));
   386→   386→    CNY *= (1 + gaussian(rng, 0, 0.004));
   387→   387→    CHF *= (1 + gaussian(rng, 0, 0.005));
   388→   388→    XAU *= (1 + gaussian(rng, 0, 0.012));
   389→   389→    VIX = clamp(VIX + gaussian(rng, 0, 1.5), 10, 80);
   390→   390→    DXY = clamp(DXY + gaussian(rng, 0, 0.5), 80, 120);
   391→   391→    rows.push({ EUR, GBP, JPY, CNY, CHF, XAU, VIX, DXY });
   392→   392→  }
   393→   393→  return rows;
   394→   394→}

   396→   396→// Convert a sequence of HistoryRow absolute prices into a sequence of returns.
   397→   397→function historyToReturns(rows: HistoryRow[]): HistoryRow[] {
   398→   398→  const rets: HistoryRow[] = [];
   399→   399→  for (let i = 1; i < rows.length; i++) {
   400→   400→    const a = rows[i - 1], b = rows[i];
   401→   401→    rets.push({
   402→   402→      EUR: b.EUR / a.EUR - 1,
   403→   403→      GBP: b.GBP / a.GBP - 1,
   404→   404→      JPY: b.JPY / a.JPY - 1,
   405→   405→      CNY: b.CNY / a.CNY - 1,
   406→   406→      CHF: b.CHF / a.CHF - 1,
   407→   407→      XAU: b.XAU / a.XAU - 1,
   408→   408→      VIX: b.VIX - a.VIX,
   409→   409→      DXY: b.DXY - a.DXY,
   410→   410→    });
   411→   411→  }
   412→   412→  return rets;
   413→   413→}

   422→   422→function runTrajectory(
   423→   423→  seed: number,
   424→   424→  ticks: number,
   425→   425→  gen: FxGen,
   426→   426→  opts: { initialMintUsd?: number; doRebalance?: boolean; applyShockAtTick?: (s: ReserveState, fx: FxSnapshot, tick: number) => void } = {},
   427→   427→): TrajectoryResult {
   428→   428→  const { state: s, fx: _fx } = setupInitialState(opts.initialMintUsd ?? 9_000_000);
   429→   429→  void _fx;
   430→   430→
   431→   431→  const rng = mulberry32(seed);
   432→   432→  // Start FX at base fixings
   433→   433→  let cur: HistoryRow = {
   434→   434→    EUR: BASE_FIXINGS.EUR_USD, GBP: BASE_FIXINGS.GBP_USD, JPY: BASE_FIXINGS.JPY_USD,
   435→   435→    CNY: BASE_FIXINGS.CNY_USD, CHF: BASE_FIXINGS.CHF_USD, XAU: BASE_FIXINGS.XAU_USD,
   436→   436→    VIX: 18.5, DXY: 104.2,
   437→   437→  };
   438→   438→
   439→   439→  let minRR = Infinity;
   440→   440→  let worstStatus = "NORMAL";
   441→   441→  let finalStatus = "NORMAL";
   442→   442→  let recoveryTick = -1;
   443→   443→  let hasBeenStressed = false;
   444→   444→  let pegStableTicks = 0;
   445→   445→  let prevStatus = "NORMAL";
   446→   446→  let statusTransitions = 0;
   447→   447→
   448→   448→  for (let t = 0; t < ticks; t++) {
   449→   449→    const nxt = gen(rng, t, cur);
   450→   450→    cur = nxt;
   451→   451→    const fx = makeFx({ EUR: nxt.EUR, GBP: nxt.GBP, JPY: nxt.JPY, CNY: nxt.CNY, CHF: nxt.CHF, XAU: nxt.XAU, VIX: nxt.VIX, DXY: nxt.DXY });
   452→   452→
   453→   453→    if (opts.applyShockAtTick) opts.applyShockAtTick(s, fx, t);
   454→   454→
   455→   455→    const m = stepEngine(s, fx, { doRebalance: opts.doRebalance ?? true });
   456→   456→    finalStatus = m.status;
   457→   457→    if (STATUS_RANK[m.status] > STATUS_RANK[worstStatus]) worstStatus = m.status;
   458→   458→    if (m.status !== prevStatus) {
   459→   459→      statusTransitions++;
   460→   460→      prevStatus = m.status;
   461→   461→    }
   462→   462→
   463→   463→    if (Number.isFinite(m.rr)) {
   464→   464→      if (m.rr < minRR) minRR = m.rr;
   465→   465→      if (m.rr < RR_TARGET) hasBeenStressed = true;
   466→   466→      if (recoveryTick === -1 && hasBeenStressed && m.rr >= RR_TARGET) recoveryTick = t;
   467→   467→    }
   468→   468→    if (m.inBand) pegStableTicks++;
   469→   469→  }
   470→   470→
   471→   471→  return summarizeTrajectory(minRR, worstStatus, finalStatus, recoveryTick, pegStableTicks, ticks, statusTransitions);
   472→   472→}

  1161→  1161→// Self-contained engine sim with perturbed parameters (mirrors engine.ts formulas).
  1162→  1162→// We use this because the blueprint/engine constants are immutable `const` exports
  1163→  1163→// — we cannot monkey-patch them in a TypeScript-safe way. The sim below faithfully
  1164→  1164→// reproduces the engine's:
  1165→  1165→//   - GFB Index (7-component chain-linked)
  1166→  1166→//   - Reserve NAV (with haircuts)
  1167→  1167→//   - Reserve Ratio + status determination
  1168→  1168→//   - §6 raw target gold weight: θ = α·zVix + β·zDxy, clamped to ±θ_max
  1169→  1169→//   - §8.4 EMA smoothing: λ · target + (1-λ) · prev
  1170→  1170→//   - §7 rebalance cost-benefit: benefit = |dev|·NAV·λ1 + (RR<TARGET)?(1/RR)·NAV·λ3 ; cost = tradeUsd·slippage + gas
  1171→  1171→function runPerturbedSim(seed: number, ticks: number, params: PerturbParams): TrajectoryResult {
  1172→  1172→  const rng = mulberry32(seed);
  1173→  1173→  // Initial state mirrors setupInitialState(): $1.1M genesis deposit + $9M user mint,
  1174→  1174→  // with the holdings redistributed to Strategic Prior composition (gold = 26% of NAV).
  1175→  1175→  // This matches the engine's design intent and removes the post-mint USD overweight
  1176→  1176→  // transient. The total NAV is ($1.1M + $9M) = $10.1M (modulo haircuts).
  1177→  1177→  const TOTAL_DEPOSIT = 1_100_000 + 9_000_000;
  1178→  1178→  let usdUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.USD;
  1179→  1179→  let eurUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.EUR;
  1180→  1180→  let jpyUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.JPY;
  1181→  1181→  let gbpUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.GBP;
  1182→  1182→  let cnyUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.CNY;
  1183→  1183→  let chfUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.CHF;
  1184→  1184→  let goldUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.Gold;
  1185→  1185→  let circulating = 9_000_000; // post-mint circulating (excludes 1M genesis)
  1186→  1186→  let smoothedTheta = 0;
  1187→  1187→  let vixHist: number[] = Array(90).fill(18.5);
  1188→  1188→  let dxyHist: number[] = Array(90).fill(104.2);
  1189→  1189→  let lastTradeDir: 0 | 1 | -1 = 0;
  1190→  1190→  let lastTradeAt = 0;
  1191→  1191→
  1192→  1192→  // FX state — start at base fixings
  1193→  1193→  let EUR = BASE_FIXINGS.EUR_USD, GBP = BASE_FIXINGS.GBP_USD, JPY = BASE_FIXINGS.JPY_USD;
  1194→  1194→  let CNY = BASE_FIXINGS.CNY_USD, CHF = BASE_FIXINGS.CHF_USD, XAU = BASE_FIXINGS.XAU_USD;
  1195→  1195→  let VIX = 18.5, DXY = 104.2;
  1196→  1196→
  1197→  1197→  let minRR = Infinity;
  1198→  1198→  let worstStatus = "NORMAL";
  1199→  1199→  let finalStatus = "NORMAL";
  1200→  1200→  let recoveryTick = -1;
  1201→  1201→  let hasBeenStressed = false;
  1202→  1202→  let pegStableTicks = 0;
  1203→  1203→  let prevStatus = "NORMAL";
  1204→  1204→  let statusTransitions = 0;
  1205→  1205→
  1206→  1206→  for (let t = 0; t < ticks; t++) {
  1207→  1207→    // Gaussian returns (modest stress: 1.5x normal vol to test perturbations)
  1208→  1208→    EUR = Math.max(0.4, EUR * (1 + gaussian(rng, 0, 0.0075)));
  1209→  1209→    GBP = Math.max(0.5, GBP * (1 + gaussian(rng, 0, 0.0075)));
  1210→  1210→    JPY = Math.max(0.003, JPY * (1 + gaussian(rng, 0, 0.009)));
  1211→  1211→    CNY = Math.max(0.06, CNY * (1 + gaussian(rng, 0, 0.006)));
  1212→  1212→    CHF = Math.max(0.4, CHF * (1 + gaussian(rng, 0, 0.0075)));
  1213→  1213→    XAU = Math.max(500, XAU * (1 + gaussian(rng, 0, 0.018)));
  1214→  1214→    VIX = clamp(VIX + gaussian(rng, 0, 2.0), 10, 80);
  1215→  1215→    DXY = clamp(DXY + gaussian(rng, 0, 0.7), 80, 120);
  1216→  1216→
  1217→  1217→    vixHist.push(VIX); if (vixHist.length > 90) vixHist.shift();
  1218→  1218→    dxyHist.push(DXY); if (dxyHist.length > 90) dxyHist.shift();
  1219→  1219→
  1220→  1220→    // Z-scores
  1221→  1221→    const vMean = avg(vixHist), vSd = stdev(vixHist);
  1222→  1222→    const dMean = avg(dxyHist), dSd = stdev(dxyHist);
  1223→  1223→    const zVix = vSd > 1e-9 ? (VIX - vMean) / vSd : 0;
  1224→  1224→    const zDxy = dSd > 1e-9 ? (DXY - dMean) / dSd : 0;
  1225→  1225→
  1226→  1226→    // §6 raw target gold weight with PERTURBED α, β, θ_max
  1227→  1227→    const theta = clampSym(params.alpha * zVix + params.beta * zDxy, -params.thetaMax, params.thetaMax);
  1228→  1228→    const target = clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
  1229→  1229→    // §8.4 EMA-smoothed with PERTURBED λ
  1230→  1230→    const smoothed = params.smoothing * target + (1 - params.smoothing) * (BASE_GOLD_WEIGHT + smoothedTheta);
  1231→  1231→    smoothedTheta = smoothed - BASE_GOLD_WEIGHT;
  1232→  1232→
  1233→  1233→    // GFB price (7-component chain-linked)
  1234→  1234→    const gfbNum =
  1235→  1235→      STRATEGIC_PRIOR.USD * 1.0 +
  1236→  1236→      STRATEGIC_PRIOR.EUR * EUR +
  1237→  1237→      STRATEGIC_PRIOR.JPY * JPY +
  1238→  1238→      STRATEGIC_PRIOR.GBP * GBP +
  1239→  1239→      STRATEGIC_PRIOR.CNY * CNY +
  1240→  1240→      STRATEGIC_PRIOR.CHF * CHF +
  1241→  1241→      STRATEGIC_PRIOR.Gold * XAU;
  1242→  1242→    const price = gfbNum / GFB_BASE_DENOMINATOR;
  1243→  1243→    const inBand = price >= PRICE_SAFETY_LOWER && price <= PRICE_SAFETY_UPPER;
  1244→  1244→    if (inBand) pegStableTicks++;
  1245→  1245→
  1246→  1246→    // NAV (with haircuts)
  1247→  1247→    const usdNet = usdUsd * (1 - HAIRCUTS.USD);
  1248→  1248→    const eurNet = eurUsd * (1 - HAIRCUTS.EUR);
  1249→  1249→    const jpyNet = jpyUsd * (1 - HAIRCUTS.JPY);
  1250→  1250→    const gbpNet = gbpUsd * (1 - HAIRCUTS.GBP);
  1251→  1251→    const cnyNet = cnyUsd * (1 - HAIRCUTS.CNY);
  1252→  1252→    const chfNet = chfUsd * (1 - HAIRCUTS.CHF);
  1253→  1253→    const goldNet = goldUsd * (1 - HAIRCUTS.XAU);
  1254→  1254→    const nav = usdNet + eurNet + jpyNet + gbpNet + cnyNet + chfNet + goldNet;
  1255→  1255→    const fiatNet = nav - goldNet;
  1256→  1256→
  1257→  1257→    // RR / LCR / status
  1258→  1258→    const liability = circulating * price;
  1259→  1259→    const rr = liability > 0 ? nav / liability : Infinity;
  1260→  1260→    const stressDemand = circulating * price * 0.25;
  1261→  1261→    const lcr = stressDemand > 0 ? fiatNet / stressDemand : Infinity;
  1262→  1262→    let status = "NORMAL";
  1263→  1263→    if (rr >= RR_TARGET && lcr >= 1) status = "NORMAL";
  1264→  1264→    else if (rr >= RR_STRESS && lcr >= 0.9) status = "CAUTION";
  1265→  1265→    else if (rr >= RR_HARD) status = "DEFENSIVE";
  1266→  1266→    else if (rr < RR_HARD) status = "EMERGENCY";
  1267→  1267→    finalStatus = status;
  1268→  1268→    if (STATUS_RANK[status] > STATUS_RANK[worstStatus]) worstStatus = status;
  1269→  1269→    if (status !== prevStatus) { statusTransitions++; prevStatus = status; }
  1270→  1270→
  1271→  1271→    if (Number.isFinite(rr)) {
  1272→  1272→      if (rr < minRR) minRR = rr;
  1273→  1273→      if (rr < RR_TARGET) hasBeenStressed = true;
  1274→  1274→      if (recoveryTick === -1 && hasBeenStressed && rr >= RR_TARGET) recoveryTick = t;
  1275→  1275→    }
  1276→  1276→
  1277→  1277→    // Rebalance with PERTURBED λ1, λ3
  1278→  1278→    if (Number.isFinite(rr) && nav > 0) {
  1279→  1279→      const observed = goldNet / nav;
  1280→  1280→      const deviation = observed - target;
  1281→  1281→      if (Math.abs(deviation) >= 0.005) {
  1282→  1282→        // Direction lock (24h whipsaw guard) — perturbation only changes λ's, not the lock
  1283→  1283→        const inStress = rr < RR_STRESS;
  1284→  1284→        const dir: 0 | 1 | -1 = deviation > 0 ? -1 : 1;
  1285→  1285→        const canTrade = inStress || lastTradeDir === 0 || dir === lastTradeDir || (t - lastTradeAt) >= 1;
  1286→  1286→        if (canTrade) {
  1287→  1287→          const surplusUsd = Math.abs(deviation) * nav;
  1288→  1288→          const tradeUsd = Math.min(surplusUsd, 400_000, MAX_DAILY_TURNOVER_PERTURB * nav);
  1289→  1289→          const cost = tradeUsd * 0.001 + 50;
  1290→  1290→          // §7 cost-benefit with PERTURBED λ1 (deviation) and λ3 (RR stress)
  1291→  1291→          const benefit =
  1292→  1292→            Math.abs(deviation) * nav * params.lambda1 +
  1293→  1293→            (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * params.lambda3 : 0);
  1294→  1294→          if (benefit > cost) {
  1295→  1295→            // Apply trade: gold ↔ USD
  1296→  1296→            if (dir === -1) {
  1297→  1297→              goldUsd -= tradeUsd;
  1298→  1298→              usdUsd += tradeUsd;
  1299→  1299→            } else {
  1300→  1300→              usdUsd -= tradeUsd;
  1301→  1301→              goldUsd += tradeUsd;
  1302→  1302→            }
  1303→  1303→            lastTradeDir = dir;
  1304→  1304→            lastTradeAt = t;
  1305→  1305→          }
  1306→  1306→        }
  1307→  1307→      }
  1308→  1308→    }
  1309→  1309→  }
  1310→  1310→
  1311→  1311→  return summarizeTrajectory(minRR, worstStatus, finalStatus, recoveryTick, pegStableTicks, ticks, statusTransitions);
  1312→  1312→}
  1313→  1313→
  1314→  1314→const MAX_DAILY_TURNOVER_PERTURB = 0.05; // 5% of NAV (matches blueprint)
  1315→  1315→
  1316→  1316→function avg(a: number[]): number {
  1317→  1317→  return a.reduce((x, y) => x + y, 0) / a.length;
  1318→  1318→}
  1319→  1319→function stdev(a: number[]): number {
  1320→  1320→  if (a.length === 0) return 0;
  1321→  1321→  const m = avg(a);
  1322→  1322→  return Math.sqrt(a.reduce((s, x) => s + (x - m) * (x - m), 0) / a.length);
  1323→  1323→}
