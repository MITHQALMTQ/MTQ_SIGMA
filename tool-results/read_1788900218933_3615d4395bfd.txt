    72→              <tr>
    73→                {headers.map((h, i) => (

    75→                    key={i}

    81→              </tr>

    88→                >

    90→                    <td
    91→                      key={ci}

   110→    return (

   113→        fixed
   114→      </Pill>
   115→    );
   116→  }
   117→  if (severity === "outstanding") {
   118→    return (
   119→      <Pill tone="amber">
   120→        <GlowDot color="amber" size="h-1.5 w-1.5" />
   121→        outstanding
   122→      </Pill>
   123→    );

   125→  return (
   126→    <Pill tone="muted">
   127→      <GlowDot color="gold" size="h-1.5 w-1.5" />
   128→      informational
   129→    </Pill>
   130→  );
   131→}
