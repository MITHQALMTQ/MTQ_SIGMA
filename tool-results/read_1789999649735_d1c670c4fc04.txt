   497→    // The Constitutional Council (7/7 multi-sig) authorizes banks. Each

   502→    struct BankRecord {
   503→        address bankAddress;
   504→        string  name;
   505→        string  jurisdiction;   // ISO 3166-1 alpha-2 ("US", "AE", "SA", ...)
   506→        bool    isAuthorized;
   507→        uint256 dailyMintCap;   // USD, 1e18 scale
   508→        uint256 dailyMintUsed;  // USD, 1e18 scale (resets daily)

   510→        uint256 authorizedAt;

   513→    mapping(address => BankRecord) public bankRegistry;
   514→    address[] public authorizedBanks;

   516→    function authorizeBank(
   517→        address bank,
   518→        string calldata _name,
   519→        string calldata _jurisdiction,
   520→        uint256 _dailyCap
   521→    ) external onlyConstitutionalCouncil {
   522→        if (bank == address(0)) revert Err55();

   531→            dailyMintUsed: 0,

   536→        emit BankAuthorized(bank, _name, _jurisdiction, _dailyCap, block.timestamp);
   537→    }

   542→        emit BankRevoked(bank, block.timestamp);
   543→    }

   553→    /// @dev Rolls the per-bank daily mint counter at UTC midnight.
   554→    ///      Idempotent within the same UTC day. Called internally by mint
   555→    ///      paths; also callable by anyone (no-op if already current).
   556→    function _rollDailyCap(address bank) internal {
   557→        BankRecord storage rec = bankRegistry[bank];
   558→        if (rec.dailyMintResetAt == 0) return;
   559→        uint256 today = (block.timestamp / 1 days) * 1 days;
   560→        if (rec.dailyMintResetAt < today) {
   561→            uint256 oldUsed = rec.dailyMintUsed;
   562→            rec.dailyMintUsed = 0;
   563→            rec.dailyMintResetAt = today;
   564→            emit DailyCapReset(bank, oldUsed, 0, block.timestamp);
   565→        }
   566→    }

   569→    // === V3 NEW §7 — AvailableBackingCertificate (ABC) ===
   570→    // ------------------------------------------------------------
   571→    // Banks submit an ABC proving they have backing assets held by a
   572→    // qualified custodian. The keeper verifies the custodian attestation
   573→    // (production: via oracle; testnet: auto-verify via verifyBackingCertificate).
   574→    //
   575→    // MITHQAL NEVER takes custody — it only verifies evidence.
   576→    // ============================================================
   577→
   578→    struct AvailableBackingCertificate {
   579→        bytes32 certificateId;
   580→        address bank;
   581→        uint256 amountUsd;                   // 1e18 scale
   582→        bytes32 custodianAttestationHash;    // hash of custodian's attestation
   583→        uint256 custodianAttestationExpiry;  // unix seconds
   584→        bytes32 reserveAssetBreakdownHash;   // hash of the reserve composition
   585→        uint256 issuedAt;
   586→        bool    verified;
   587→    }
   588→
   589→    mapping(bytes32 => AvailableBackingCertificate) public backingCertificates;
   590→
   591→    function submitBackingCertificate(AvailableBackingCertificate calldata cert) external onlyAuthorizedBank {
   592→        // The submitting bank must match the certificate's bank field.
   593→        if (cert.bank != msg.sender) revert Err92_BackdingCertMismatch();
   594→        if (cert.amountUsd == 0) revert Err86_InvalidRequest();
   595→        if (cert.custodianAttestationExpiry <= block.timestamp) revert Err91_InvalidExpiry();
   596→        if (cert.certificateId == bytes32(0)) revert Err86_InvalidRequest();
   597→        // Reject duplicate certificate IDs.
   598→        if (backingCertificates[cert.certificateId].issuedAt != 0) revert Err69_NonceReused();
   599→
   600→        backingCertificates[cert.certificateId] = AvailableBackingCertificate({
   601→            certificateId: cert.certificateId,
   602→            bank: cert.bank,
   603→            amountUsd: cert.amountUsd,
   604→            custodianAttestationHash: cert.custodianAttestationHash,
   605→            custodianAttestationExpiry: cert.custodianAttestationExpiry,
   606→            reserveAssetBreakdownHash: cert.reserveAssetBreakdownHash,
   607→            issuedAt: block.timestamp,
   608→            verified: false
   609→        });
   610→        emit BackingCertificateSubmitted(cert.certificateId, cert.bank, cert.amountUsd, block.timestamp);
   611→    }
   612→
   613→    /// @notice Verify an ABC's custodian attestation. Production: oracle call.
   614→    ///         Testnet/pilot: keeper auto-verifies (the keeper is trusted).
   615→    function verifyBackingCertificate(bytes32 certId) external onlyKeeper returns (bool) {
   616→        AvailableBackingCertificate storage c = backingCertificates[certId];
   617→        if (c.issuedAt == 0) revert Err85_CertNotFound();
   618→        if (c.custodianAttestationExpiry <= block.timestamp) revert Err72_BackdingCertExpired();
   619→        c.verified = true;
   620→        // Credit the bank's verified-backing pool (evidence only — not custody).
   621→        backingVerified[c.bank] += c.amountUsd;
   622→        emit BackingCertificateVerified(certId, msg.sender, true, block.timestamp);
   623→        return true;
   624→    }

   626→    function getCertificate(bytes32 certId) external view returns (AvailableBackingCertificate memory) {
   627→        return backingCertificates[certId];
   628→    }

   630→    // ============================================================
   631→    // === V3 NEW §8 — 16-Step Bank Minting Workflow ===
   632→    // ------------------------------------------------------------
   633→    // BM-01 PENDING        — bank submits requestMint()
   634→    // BM-02 RECEIVED       — MBG gateway ack (keeper: API auth + schema OK)
   635→    // BM-03 KYC            — KYC passed (keeper: off-chain evidence)
   636→    // BM-04 AML            — AML/sanctions passed (keeper)
   637→    // BM-05 BACKING        — ABC submitted & in registry
   638→    // BM-06 EVIDENCE       — ABC verified (custodian attestation valid)
   639→    // BM-07 REQUESTED      — MBG has translated request to on-chain
   640→    // BM-08 TRANSLATED     — request payload validated against policy
   641→    // BM-09 ELIGIBLE       — bank + jurisdiction eligible
   642→    // BM-10 JURISDICTION   — jurisdiction checks (sanctions/exchange control)
   643→    // BM-11 BACKING_VERIFIED — backing ≥ mint amount
   644→    // BM-12 BANK_RISK      — bank-level risk (cap, concentration, credit)
   645→    // BM-13 SYSTEM_RISK    — system-level risk (RR/LCR/state machine)
   646→    // BM-14 DMCE           — Daily Mint Cap / Concentration / Exposure check
   647→    // BM-15 AUTHORIZED     — MITHQAL Monetary Control authorization (L4)
   648→    // BM-16 MINTED         — executeMint() final atomic write (L6+L7)
   649→    // REJECTED             — terminal failure state
   650→    // ============================================================
   651→
   652→    enum WorkflowState {
   653→        NONE,
   654→        BM01_PENDING,
   655→        BM02_RECEIVED,
   656→        BM03_KYC,
   657→        BM04_AML,
   658→        BM05_BACKING,
   659→        BM06_EVIDENCE,
   660→        BM07_REQUESTED,
   661→        BM08_TRANSLATED,
   662→        BM09_ELIGIBLE,
   663→        BM10_JURISDICTION,
   664→        BM11_BACKING_VERIFIED,
   665→        BM12_BANK_RISK,
   666→        BM13_SYSTEM_RISK,
   667→        BM14_DMCE,
   668→        BM15_AUTHORIZED,
   669→        BM16_MINTED,
   670→        REJECTED
   671→    }

   673→    struct BankMintRequest {
   674→        address bank;
   675→        bytes32 corporateCustomerHash;     // hash of the corporate customer identity
   676→        uint256 amountUsd;                  // 1e18 scale
   677→        bytes32 backingCertificateId;
   678→        string  jurisdiction;               // ISO 3166-1 alpha-2
   679→        uint256 nonce;                      // bank-unique anti-replay
   680→        uint256 expiry;                     // unix seconds; request expires after
   681→    }
   682→
   683→    struct MintRequest {
   684→        bytes32 requestId;
   685→        BankMintRequest request;
   686→        WorkflowState state;
   687→        uint256 createdAt;
   688→        uint256 authorizedAt;
   689→        uint256 mintedAt;
   690→        uint256 mintedAmount;
   691→        string  rejectReason;
   692→    }
   693→
   694→    mapping(bytes32 => MintRequest) public mintRequests;
   695→    /// @notice Anti-replay: bank → nonce → consumed.
   696→    mapping(address => mapping(uint256 => bool)) public usedNonces;
   697→    /// @notice Per-bank concentration: total outstanding MTQ minted via this bank.
   698→    mapping(address => uint256) public bankMintedTotal;
   699→    /// @notice DMCE: Daily Mint Cap & Concentration Exposure — single-bank cap
   700→    ///         as a fraction of total supply (1e18 scale). Default 5%.
   701→    uint256 public dmceBankConcentrationCap = 0.05e18;
   702→
   703→    /// @notice Request a mint — only an authorized bank. Creates state=BM01_PENDING.
   704→    function requestMint(BankMintRequest calldata req) external onlyAuthorizedBank whenNotPaused returns (bytes32 requestId) {
   705→        if (req.bank != msg.sender) revert Err70_JurisdictionMismatch();
   706→        if (req.amountUsd == 0) revert Err86_InvalidRequest();
   707→        if (req.expiry <= block.timestamp) revert Err91_InvalidExpiry();
   708→        if (usedNonces[msg.sender][req.nonce]) revert Err69_NonceReused();
   709→        usedNonces[msg.sender][req.nonce] = true;
   710→
   711→        // L3 POLICY (early): per-bank daily cap precheck (full enforcement
   712→        // happens at BM-12/BM-14, but we reject early to fail fast).
   713→        _rollDailyCap(msg.sender);
   714→        BankRecord storage bank = bankRegistry[msg.sender];
   715→        if (bank.dailyMintUsed + req.amountUsd > bank.dailyMintCap) revert Err67_DailyCapExceeded();
   716→
   717→        // The certificate must already exist (BM-05 will verify it is in the
   718→        // registry; BM-06 will verify it is verified).
   719→        if (backingCertificates[req.backingCertificateId].issuedAt == 0) revert Err85_CertNotFound();
   720→
   721→        requestId = keccak256(abi.encodePacked(
   722→            msg.sender,
   723→            req.corporateCustomerHash,
   724→            req.amountUsd,
   725→            req.backingCertificateId,
   726→            req.nonce,
   727→            block.chainid,
   728→            block.timestamp
   729→        ));
   730→        if (mintRequests[requestId].createdAt != 0) revert Err69_NonceReused();
   731→
   732→        mintRequests[requestId] = MintRequest({
   733→            requestId: requestId,
   734→            request: req,
   735→            state: WorkflowState.BM01_PENDING,
   736→            createdAt: block.timestamp,
   737→            authorizedAt: 0,
   738→            mintedAt: 0,
   739→            mintedAmount: 0,
   740→            rejectReason: ""
   741→        });
   742→        emit MintRequested(requestId, msg.sender, req.amountUsd, req.backingCertificateId, req.jurisdiction, block.timestamp);
   743→        emit WorkflowAdvanced(requestId, WorkflowState.NONE, WorkflowState.BM01_PENDING, msg.sender, block.timestamp);
   744→    }
   745→
   746→    /// @notice Advance a request through the workflow. Only KEEPER (BM-02..BM-14)
   747→    ///         or MONETARY_CONTROL (BM-15) may advance. State transitions must
   748→    ///         be strictly sequential (no skipping). The keeper performs the
   749→    ///         L3 policy checks (KYC/AML/eligibility/jurisdiction/bank-risk/
   750→    ///         system-risk/DMCE) off-chain and only calls advanceWorkflow when
   751→    ///         the corresponding gate has passed. The L4 monetary authorization
   752→    ///         gate (BM-15) requires the MONETARY_CONTROL_ROLE signer.
   753→    function advanceWorkflow(bytes32 requestId, WorkflowState toState) external whenNotPaused returns (bool) {
   754→        MintRequest storage r = mintRequests[requestId];
   755→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   756→        if (r.state == WorkflowState.BM16_MINTED) revert Err76_WorkflowAlreadyMinted();
   757→        if (r.state == WorkflowState.REJECTED) revert Err77_WorkflowAlreadyRejected();
   758→        if (r.request.expiry <= block.timestamp) revert Err68_RequestExpired();
   759→
   760→        // L2 WORKFLOW GATE — strict sequential progression.
   761→        // toState must equal (current + 1).
   762→        if (uint256(toState) != uint256(r.state) + 1) revert Err74_WorkflowNotSequential();
   763→
   764→        // Role gate per state:
   765→        //   BM-02..BM-14 → KEEPER_ROLE
   766→        //   BM-15        → MONETARY_CONTROL_ROLE (L4)
   767→        if (toState == WorkflowState.BM15_AUTHORIZED) {
   768→            if (!(_roles[MONETARY_CONTROL_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err80_L4MonetaryGate();
   769→        } else {
   770→            if (!(_roles[KEEPER_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err31();
   771→        }
   772→
   773→        // L3 POLICY GATE — keeper has run the off-chain policy engine and
   774→        // attests that all checks for `toState` have passed. We re-verify
   775→        // a small number of on-chain invariants per state for defense in depth.
   776→        if (toState == WorkflowState.BM05_BACKING) {
   777→            // The ABC must exist.
   778→            if (backingCertificates[r.request.backingCertificateId].issuedAt == 0) revert Err85_CertNotFound();
   779→        } else if (toState == WorkflowState.BM06_EVIDENCE) {
   780→            // The ABC must be verified.
   781→            AvailableBackingCertificate storage c = backingCertificates[r.request.backingCertificateId];
   782→            if (!c.verified) revert Err71_BackdingCertNotVerified();
   783→            if (c.custodianAttestationExpiry <= block.timestamp) revert Err72_BackdingCertExpired();
   784→        } else if (toState == WorkflowState.BM09_ELIGIBLE) {
   785→            // Bank must still be authorized.
   786→            if (!bankRegistry[r.request.bank].isAuthorized) revert Err66_BankNotAuthorized();
   787→        } else if (toState == WorkflowState.BM10_JURISDICTION) {
   788→            // Jurisdiction on request must match bank's registered jurisdiction.
   789→            BankRecord storage b = bankRegistry[r.request.bank];
   790→            if (keccak256(bytes(b.jurisdiction)) != keccak256(bytes(r.request.jurisdiction))) revert Err70_JurisdictionMismatch();
   791→        } else if (toState == WorkflowState.BM11_BACKING_VERIFIED) {
   792→            // Verified backing pool for the bank must be ≥ requested amount.
   793→            if (backingVerified[r.request.bank] < r.request.amountUsd) revert Err73_BackdingCertInsufficient();
   794→        } else if (toState == WorkflowState.BM12_BANK_RISK) {
   795→            // Per-bank daily cap precheck (rolls the day first).
   796→            _rollDailyCap(r.request.bank);
   797→            BankRecord storage br = bankRegistry[r.request.bank];
   798→            if (br.dailyMintUsed + r.request.amountUsd > br.dailyMintCap) revert Err67_DailyCapExceeded();
   799→        } else if (toState == WorkflowState.BM13_SYSTEM_RISK) {
   800→            // Minting must be allowed by the 6-state risk machine.
   801→            if (!mintingAllowed(currentState)) revert Err25();
   802→            // Oracle must be healthy (strict 3-source quorum on all 6 pairs).
   803→            _requireOracleHealthy();
   804→        } else if (toState == WorkflowState.BM14_DMCE) {
   805→            // DMCE: single-bank concentration cap.
   806→            uint256 concentration = (totalSupply == 0)
   807→                ? type(uint256).max
   808→                : bankMintedTotal[r.request.bank] * 1e18 / totalSupply;
   809→            // Reject if adding this mint would breach the concentration cap.
   810→            uint256 newMintedMTQ = r.request.amountUsd * 1e18 / getMTQPriceWithGuard();
   811→            uint256 newConcentration = (totalSupply + newMintedMTQ == 0)
   812→                ? 0
   813→                : (bankMintedTotal[r.request.bank] + newMintedMTQ) * 1e18 / (totalSupply + newMintedMTQ);
   814→            if (newConcentration > dmceBankConcentrationCap && concentration > dmceBankConcentrationCap) revert Err88_ConcentrationLimit();
   815→        } else if (toState == WorkflowState.BM15_AUTHORIZED) {
   816→            // L4 MONETARY CONTROL — the modifier already gated this.
   817→            // Mark the authorization timestamp.
   818→            r.authorizedAt = block.timestamp;
   819→        }
   820→
   821→        WorkflowState fromState = r.state;
   822→        r.state = toState;
   823→        emit WorkflowAdvanced(requestId, fromState, toState, msg.sender, block.timestamp);
   824→        return true;
   825→    }
   826→
   827→    /// @notice Reject a request (terminal). Keeper or Monetary Control only.
   828→    function rejectMint(bytes32 requestId, string calldata reason) external whenNotPaused returns (bool) {
   829→        MintRequest storage r = mintRequests[requestId];
   830→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   831→        if (r.state == WorkflowState.BM16_MINTED) revert Err76_WorkflowAlreadyMinted();
   832→        if (r.state == WorkflowState.REJECTED) revert Err77_WorkflowAlreadyRejected();
   833→        if (!(_roles[KEEPER_ROLE][msg.sender] || _roles[MONETARY_CONTROL_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err31();
   834→        WorkflowState fromState = r.state;
   835→        r.state = WorkflowState.REJECTED;
   836→        r.rejectReason = reason;
   837→        emit WorkflowAdvanced(requestId, fromState, WorkflowState.REJECTED, msg.sender, block.timestamp);
   838→        emit MintRejected(requestId, msg.sender, reason, block.timestamp);
   839→        return true;
   840→    }

   842→    /// @notice L5+L6+L7 FINALITY GATE — execute the mint atomically.
   843→    ///         Requires state == BM15_AUTHORIZED (L2/L5), oracle healthy (L3
   844→    ///         recheck), keeper signature (L7). Single transaction = atomic
   845→    ///         write (L6). Mints MTQ to the bank; the bank distributes to the
   846→    ///         corporate customer off-chain (sovereign currency settlement).
   847→    /// @dev LIQ-MGMT (audit H5 + M2): the caller passes a `minMinted` floor
   848→    ///      (1e18 scale MTQ). The mint reverts with Err93_SlippageExceeded if
   849→    ///      the actual minted amount (after price + throttle) falls below it.
   850→    ///      The mint also reverts with Err94_PerTxCapExceeded if the requested
   851→    ///      USD amount exceeds MAX_MINT_PER_TX ($500K). Pass minMinted = 0 to
   852→    ///      disable slippage protection (e.g. for keeper-relayed bank requests
   853→    ///      where the bank has signed off on the exact expected output).
   854→    function executeMint(bytes32 requestId, uint256 minMinted) external onlyKeeper whenNotPaused nonReentrant returns (uint256 minted) {
   855→        MintRequest storage r = mintRequests[requestId];
   856→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   857→        // L2 WORKFLOW GATE — must be at BM15_AUTHORIZED.
   858→        if (r.state != WorkflowState.BM15_AUTHORIZED) revert Err78_L2WorkflowGate();
   859→        // L5 LEDGER GATE — must not already be minted.
   860→        if (r.mintedAt != 0) revert Err76_WorkflowAlreadyMinted();
   861→        // L3 POLICY GATE (recheck) — minting allowed, oracle healthy, not expired.
   862→        if (!mintingAllowed(currentState)) revert Err25();
   863→        _requireOracleHealthy();
   864→        if (r.request.expiry <= block.timestamp) revert Err68_RequestExpired();
   865→        // Re-check backing is still verified and sufficient (defence in depth).
   866→        if (backingVerified[r.request.bank] < r.request.amountUsd) revert Err73_BackdingCertInsufficient();
   867→
   868→        // LIQ-MGMT M2: per-transaction cap (audit M2). Reject if the requested
   869→        // USD amount exceeds MAX_MINT_PER_TX. This caps the largest single
   870→        // mint the keeper can land per tx, limiting blast radius if the keeper
   871→        // key or the L4 monetary-control signer is compromised.
   872→        if (r.request.amountUsd > MAX_MINT_PER_TX) revert Err94_PerTxCapExceeded();
   873→
   874→        // L4 MONETARY AUTHORIZATION — already enforced at advanceWorkflow(BM15)
   875→        // by the onlyMonetaryControl modifier (we re-verify authorizedAt > 0).
   876→        if (r.authorizedAt == 0) revert Err80_L4MonetaryGate();
   877→
   878→        // L7 FINALITY ORACLE — the keeper signature (msg.sender = keeper) is
   879→        // the on-chain finality proof. Without it, no mint can land.
   880→
   881→        // L6 ATOMIC WRITE — all state changes below are in a single tx; if any
   882→        // reverts, the whole mint reverts (atomic, all-or-nothing).
   883→
   884→        // Compute mint amount: amountUsd (1e18) × 1e18 / P_MTQ (1e18).
   885→        uint256 price = getMTQPriceWithGuard();
   886→        uint256 grossMint = r.request.amountUsd * 1e18 / price;
   887→
   888→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25).
   889→        uint256 throttle = mintThrottle(currentState);
   890→        minted = grossMint * throttle / 1e18;
   891→
   892→        // LIQ-MGMT H5: slippage protection. The caller (keeper, relaying the
   893→        // bank's signed quote) commits to a minimum acceptable minted amount.
   894→        // If the on-chain computed minted (price + throttle) is below it, the
   895→        // whole tx reverts — protecting the bank's corporate customer from
   896→        // receiving fewer MTQ than their quote promised.
   897→        if (minted < minMinted) revert Err93_SlippageExceeded();
   898→
   899→        // State transitions + accounting.
   900→        r.state = WorkflowState.BM16_MINTED;
   901→        r.mintedAt = block.timestamp;
   902→        r.mintedAmount = minted;
   903→
   904→        // Decrement the bank's verified-backing pool by the minted USD amount
   905→        // (the backing is now "consumed" by the mint — the bank's custodian
   906→        // will release the corresponding reserves to the redeemer at redeem
   907→        // time, off-chain).
   908→        if (backingVerified[r.request.bank] >= r.request.amountUsd) {
   909→            backingVerified[r.request.bank] -= r.request.amountUsd;
   910→        } else {
   911→            // Should never hit thanks to the BM-11 check; defense in depth.
   912→            backingVerified[r.request.bank] = 0;
   913→        }
   914→
   915→        // Roll + charge the per-bank daily mint cap.
   916→        _rollDailyCap(r.request.bank);
   917→        bankRegistry[r.request.bank].dailyMintUsed += r.request.amountUsd;
   918→        bankMintedTotal[r.request.bank] += minted;
   919→
   920→        // L7 finality: mint to the bank (NOT the corporate customer — the bank
   921→        // settles with the customer off-chain per Blueprint v25.3 §6).
   922→        _mint(r.request.bank, minted);
   923→
   924→        emit WorkflowAdvanced(requestId, WorkflowState.BM15_AUTHORIZED, WorkflowState.BM16_MINTED, msg.sender, block.timestamp);
   925→        emit MintExecuted(requestId, r.request.bank, r.request.amountUsd, minted, price, block.timestamp);
   926→    }
   927→
   928→    /// @notice V3 non-custodial redeem — burns MTQ from the redeemer and emits
   929→    ///         a `RedeemSettlement` event. The bank settles with the redeemer
   930→    ///         off-chain (sovereign currency transfer). The contract itself
   931→    ///         does NOT push USDC: there is no reserveVault in V3.
   932→    ///         Only an authorized bank can call this on behalf of a redeemer
   933→    ///         (the redeemer must have approved the bank to burn their MTQ via
   934→    ///         the standard ERC-20 allowance).
   935→    function redeem(uint256 mtqAmount, address redeemer) external nonReentrant whenNotPaused returns (uint256 netUsd) {
   936→        if (!(mtqAmount > 0)) revert Err56();
   937→        if (!(redemptionAllowed(currentState))) revert Err43();
   938→        // Only an authorized bank can submit a redeem for settlement.
   939→        if (!bankRegistry[msg.sender].isAuthorized) revert Err90_NotAuthorizedBank();
   940→        // Redeemer must have approved the bank to burn their MTQ.
   941→        uint256 allowed = allowance[redeemer][msg.sender];
   942→        if (allowed != type(uint256).max) {
   943→            if (!(allowed >= mtqAmount)) revert Err21();
   944→            allowance[redeemer][msg.sender] = allowed - mtqAmount;
   945→        }
   946→        if (!(balanceOf[redeemer] >= mtqAmount)) revert Err22();
   947→
   948→        _requireOracleHealthy();
   949→        uint256 navPerToken = getNAVperToken();
   950→        if (!(navPerToken < type(uint256).max)) revert Err26();
   951→
   952→        uint256 feeFraction = redeemFee(currentState);
   953→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
   954→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
   955→        netUsd             = grossUsd18 - feeUsd18;
   956→
   957→        // LIQ-MGMT M2: per-transaction cap (audit M2). The gross USD value of
   958→        // a single redeem is capped at MAX_REDEEM_PER_TX ($500K). This is
   959→        // measured against grossUsd18 (NAV-per-token × mtqAmount), which is
   960→        // the actual settlement obligation the bank incurs — NOT the net the
   961→        // redeemer receives. Caps the largest single burn the protocol will
   962→        // honour per tx.
   963→        if (grossUsd18 > MAX_REDEEM_PER_TX) revert Err94_PerTxCapExceeded();
   964→
   965→        // LIQ-MGMT T7: daily redemption cap (audit T7). Rolls the 24h window
   966→        // first (idempotent within the same day), then enforces that adding
   967→        // this redeem's gross USD does not exceed dailyRedeemCap. The cap is
   968→        // measured in gross-USD (pre-fee) since that is the reserve outflow
   969→        // the protocol must honour. First redeem initializes the window.
   970→        if (dailyRedeemResetAt == 0 || block.timestamp >= dailyRedeemResetAt + 1 days) {
   971→            uint256 oldUsed = dailyRedeemedUsd;
   972→            dailyRedeemedUsd = 0;
   973→            dailyRedeemResetAt = block.timestamp;
   974→            emit DailyRedeemCapReset(oldUsed, 0, block.timestamp);
   975→        }
   976→        if (dailyRedeemedUsd + grossUsd18 > dailyRedeemCap) revert Err95_DailyRedeemCapExceeded();
   977→        dailyRedeemedUsd += grossUsd18;
   978→        emit DailyRedeemCapConsumed(msg.sender, grossUsd18, dailyRedeemedUsd, dailyRedeemCap, block.timestamp);
   979→
   980→        _burn(redeemer, mtqAmount);
   981→
   982→        // Decrement the on-chain reserve mirror proportionally (basket broken).
   983→        uint256 totalReserve = getReserveNavUsd();
   984→        if (totalReserve > 0 && grossUsd18 > 0) {
   985→            for (uint8 i = 0; i < 7; i++) {
   986→                uint256 held = reserveHeldUsd[Component(i)];
   987→                if (held == 0) continue;
   988→                uint256 share = held * grossUsd18 / totalReserve;
   989→                if (share > held) share = held;
   990→                reserveHeldUsd[Component(i)] = held - share;
   991→            }
   992→        }
   993→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
   994→                                  reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
   995→                                  reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
   996→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
   997→        reserveState.liquidValue = reserveState.fiatValue;
   998→
   999→        // Decrement the bank's outstanding MTQ counter (concentration reverse).
  1000→        if (bankMintedTotal[msg.sender] >= mtqAmount) {
  1001→            bankMintedTotal[msg.sender] -= mtqAmount;
  1002→        } else {
  1003→            bankMintedTotal[msg.sender] = 0;
  1004→        }
  1005→
  1006→        // The bank owes `netUsd` to the redeemer, settled off-chain via
  1007→        // sovereign currency transfer. Emit the settlement obligation.
  1008→        emit RedeemSettlement(redeemer, msg.sender, mtqAmount, grossUsd18, feeUsd18, netUsd, navPerToken, block.timestamp);
  1009→    }

  1081→    /// @notice LIQ-MGMT H4/O1: circuit-breaker helper. Compares the new MTQ
  1082→    ///         price (1e18 scale) against the price observed at the last
  1083→    ///         price-commit call. If the move exceeds MAX_PRICE_JUMP_PCT (10%),
  1084→    ///         the contract is auto-paused and `CircuitBreakerTriggered` is
  1085→    ///         emitted. `lastCommitPrice` is then updated to the new price.
  1086→    ///         The first commit (lastCommitPrice == 0) never triggers.
  1087→    /// @dev Uses getMTQPrice() (no guard) so the breaker can fire even when
  1088→    ///      the new price has exited the [0.50, 2.00] safety band — that is
  1089→    ///      exactly the scenario the breaker exists to catch. The pause takes
  1090→    ///      effect for all subsequent calls; the current commit call completes
  1091→    ///      (its state changes are NOT rolled back) so the offending price is
  1092→    ///      recorded on-chain for forensic review.
  1093→    function _circuitBreakerCheck(uint256 newPrice) internal {
  1094→        if (lastCommitPrice > 0 && newPrice > 0) {
  1095→            uint256 jumpPct = newPrice > lastCommitPrice
  1096→                ? ((newPrice - lastCommitPrice) * 100) / lastCommitPrice
  1097→                : ((lastCommitPrice - newPrice) * 100) / lastCommitPrice;
  1098→            if (jumpPct > MAX_PRICE_JUMP_PCT) {
  1099→                // Auto-pause on >10% price jump. The Pauser role can unpause
  1100→                // after the keeper has root-caused the move (oracle fault,
  1101→                // market shock, etc.) and re-verified the new price.
  1102→                paused = true;
  1103→                emit CircuitBreakerTriggered(newPrice, lastCommitPrice, jumpPct);
  1104→                emit Paused(msg.sender);
  1105→            }
  1106→        }
  1107→        lastCommitPrice = newPrice;
  1108→    }

  1240→    function updateState(uint256 rr, uint256 lcr) external onlyKeeper {
  1241→        ProtocolState newState;
  1242→        if (rr >= RR_TARGET && lcr >= lcrTarget) {
  1243→            newState = ProtocolState.NORMAL;
  1244→        } else if (rr >= RR_FLOOR && lcr >= 0.90e18) {

  1246→        } else if (rr >= 1.02e18 && lcr >= 0.80e18) {

  1248→        } else if (rr >= RR_HARD_FLOOR && lcr >= 0.70e18) {
  1249→            newState = ProtocolState.DEFENSIVE;
  1250→        } else if (rr < RR_HARD_FLOOR) {
  1251→            newState = ProtocolState.EMERGENCY;
  1252→        } else {
  1253→            newState = ProtocolState.RECOVERY;
  1254→        }
  1255→        if (newState != currentState) {
  1256→            if (_isMoreRestrictive(newState, currentState)) {
  1257→                _transitionTo(newState, true);
  1258→            } else {

  1260→                    _transitionTo(newState, false);
  1261→                }
  1262→            }
  1263→        }
  1264→    }

  1273→    function _restrictiveness(ProtocolState s) internal pure returns (uint256) {
  1274→        if (s == ProtocolState.NORMAL)    return 0;
  1275→        if (s == ProtocolState.CAUTION)   return 1;

  1405→    function getOracleConsensus(bytes32 pair) public view returns (
  1406→        uint256 finalPrice,
  1407→        uint8  validCount,
  1408→        uint8  method,
  1409→        bool   paused_,
  1410→        uint256[3] memory prices,
  1411→        bool[3] memory valid
  1412→    ) {
  1413→        IOracleAdapter[3] memory adapters;
  1414→        adapters[0] = chainlinkAdapter;
  1415→        adapters[1] = pythAdapter;
  1416→        adapters[2] = chronicleAdapter;
  1417→
  1418→        uint256[3] memory raw;
  1419→        uint256[3] memory ts;
  1420→        uint256[3] memory conf;
  1421→
  1422→        for (uint8 i = 0; i < 3; i++) {
  1423→            valid[i] = false;
  1424→            if (address(adapters[i]) == address(0)) continue;
  1425→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
  1426→                if (p == 0) continue;
  1427→                raw[i]  = p;
  1428→                ts[i]   = t;
  1429→                conf[i] = c;
  1430→                prices[i] = p;
  1431→                if (t > block.timestamp) continue;
  1432→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
  1433→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
  1434→                valid[i] = true;
  1435→            } catch { /* leave valid[i] = false */ }
  1436→        }
  1437→
  1438→        validCount = 0;
  1439→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
  1440→
  1441→        if (validCount < 2) {
  1442→            paused_ = true; method = 0;
  1443→            return (finalPrice, validCount, method, paused_, prices, valid);
  1444→        }
  1445→
  1446→        uint256[3] memory sorted;
  1447→        uint8 n = 0;
  1448→        for (uint8 i = 0; i < 3; i++) if (valid[i]) { sorted[n] = raw[i]; n++; }
  1449→        _insertionSort(sorted, n);
  1450→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
  1451→
  1452→        for (uint8 i = 0; i < 3; i++) {
  1453→            if (valid[i]) {
  1454→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
  1455→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
  1456→                    valid[i] = false;
  1457→                    validCount--;
  1458→                }
  1459→            }
  1460→        }
  1461→
  1462→        paused_ = validCount < 3;
  1463→        if (paused_) {
  1464→            method     = 0;
  1465→            finalPrice = 0;
  1466→            return (finalPrice, validCount, method, paused_, prices, valid);
  1467→        }
  1468→
  1469→        finalPrice = med;
  1470→        method     = 2;
  1471→    }

  1624→    uint256 public constant DIRECTION_LOCK_HOURS  = 24 hours;
  1625→    uint256 public constant MAX_DAILY_TURNOVER    = 0.05e18;
  1626→    uint256 public constant REBALANCE_TOLERANCE    = 0.05e18;
  1627→
  1628→    mapping(Component => int256)  public lastDirection;
  1629→    mapping(Component => uint256) public lastRebalanceAt;

  1635→    function executeRebalance(RebalanceTrade[] memory trades) external nonReentrant onlyKeeper whenNotPaused {
  1636→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1637→        if (!(trades.length <= 7)) revert Err48();
  1638→
  1639→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1640→            dailyTradeUsd    = 0;
  1641→            dailyTradeResetAt = block.timestamp;
  1642→        }
  1643→
  1644→        uint256 nav = getReserveNavUsd();
  1645→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1646→        uint256 rr = getReserveRatio();
  1647→        bool solvencyOverride = (rr < rrStressFloor);
  1648→
  1649→        for (uint256 i = 0; i < trades.length; i++) {
  1650→            RebalanceTrade memory t = trades[i];
  1651→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1652→            if (t.direction == 0 || t.tradeUsd == 0) continue;
  1653→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1654→
  1655→            int256 prevDir = lastDirection[t.component];
  1656→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1657→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1658→            }
  1659→
  1660→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1661→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1662→            if (t.direction == 1) {
  1663→                if (!(currentW < execW)) revert Err14();
  1664→            } else {
  1665→                if (!(currentW > execW)) revert Err46();
  1666→            }
  1667→
  1668→            uint256 newHeldUsd = (t.direction == 1)
  1669→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1670→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1671→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1672→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1673→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1674→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1675→            if (!(absDiff <= tolerance)) revert Err49();
  1676→
  1677→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1678→            dailyTradeUsd += t.tradeUsd;
  1679→
  1680→            if (t.direction == 1) {
  1681→                reserveHeldUsd[t.component] += t.tradeUsd;
  1682→            } else {
  1683→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1684→            }
  1685→            lastDirection[t.component]   = t.direction;
  1686→            lastRebalanceAt[t.component] = block.timestamp;
  1687→        }
  1688→
  1689→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1690→                                  reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1691→                                  reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1692→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1693→        reserveState.liquidValue = reserveState.fiatValue;
  1694→
  1695→        emit RebalanceExecuted(trades, block.timestamp);
  1696→    }
