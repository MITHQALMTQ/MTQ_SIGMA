   134→   134→function seedHistory(mean: number, sd: number, n: number): number[] {
   135→   135→  const out: number[] = [];
   136→   136→  let last = mean;
   137→   137→  for (let i = 0; i < n; i++) {
   138→   138→    // simple mean-reverting walk
   139→   139→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   140→   140→    last = last + (mean - last) * 0.1 + shock;
   141→   141→    out.push(Math.max(0.01, last));
   142→   142→  }
   143→   143→  return out;
   144→   144→}

   174→   174→  const gbpGross = s.gbp * fx.GBP_USD;

   176→   176→  const cnyGross = s.cny * fx.CNY_USD;

   182→   182→  const goldRef = fx.XAU_USD;
   183→   183→  const goldExec = fx.XAU_USD * 0.995;
   184→   184→  const goldPrice = Math.min(goldRef, goldExec);

   191→   191→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   192→   192→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   193→   193→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   194→   194→
   195→   195→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + goldNet;
   196→   196→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet;

   237→   237→function rollingStats(hist: number[]): { mean: number; sd: number } {
   238→   238→  const filled = hist.filter((x) => x > 0);
   239→   239→  if (filled.length === 0) return { mean: 0, sd: 0 };
   240→   240→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   241→   241→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   242→   242→  return { mean, sd: Math.sqrt(variance) };
   243→   243→}

   245→   245→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   246→   246→  const v = rollingStats(s.vixHistory);
   247→   247→  const d = rollingStats(s.dxyHistory);
   248→   248→  const eps = 1e-9;
   249→   249→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   250→   250→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   251→   251→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   252→   252→}

   276→   276→export function currentBufferGoldRatio(s: ReserveState): number {
   277→   277→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   278→   278→  if (s.rampStartAt === 0) {
   279→   279→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   280→   280→  }
   281→   281→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   282→   282→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   283→   283→  const progress = elapsedH / RAMP_DURATION_HOURS;
   284→   284→  const from = s.rampFromRatio;
   285→   285→  const to = s.rampToRatio;
   286→   286→  return from + (to - from) * progress;
   287→   287→}

   289→   289→export function bufferBaseFromRatio(rr: number): "BASE" | "STRESS" | "EMERGENCY" {
   290→   290→  if (rr >= RR_TARGET) return "BASE";
   291→   291→  if (rr >= RR_STRESS) return "STRESS";
   292→   292→  return "EMERGENCY";
   293→   293→}

   295→   295→// Unified target gold weight (tokenomics reconciliation):
   296→   296→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   297→   297→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   298→   298→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   299→   299→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   300→   300→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   301→   301→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   302→   302→}

   322→   322→export function evaluateRebalance(
   323→   323→  s: ReserveState,
   324→   324→  vals: { nav: number; goldNet: number; fiatNet: number },
   325→   325→  rr: number,
   326→   326→): RebalanceDecision {
   327→   327→  const nav = vals.nav;
   328→   328→  if (nav <= 0) {
   329→   329→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   330→   330→  }
   331→   331→  const observed = vals.goldNet / nav;
   332→   332→  const target = computeTargetGoldWeight(s, rr);
   333→   333→  const deviation = observed - target;
   334→   334→
   335→   335→  if (Math.abs(deviation) < 0.005) {
   336→   336→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   337→   337→  }
   338→   338→
   339→   339→  // Direction lock (overridden in stress)
   340→   340→  const inStress = rr < RR_STRESS;
   341→   341→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   342→   342→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   343→   343→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   344→   344→    }
   345→   345→  }
   346→   346→
   347→   347→  // Trade sizing
   348→   348→  const surplusUsd = Math.abs(deviation) * nav;
   349→   349→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   350→   350→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   351→   351→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   352→   352→  maybeResetDailyTurnover(s);
   353→   353→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   354→   354→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   355→   355→
   356→   356→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   357→   357→  const estCost = tradeUsd * 0.001 + 50;
   358→   358→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   359→   359→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   360→   360→
   361→   361→  if (benefit <= estCost) {
   362→   362→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   363→   363→  }
   364→   364→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   365→   365→}

   367→   367→// Apply a rebalance trade in USD terms (mutates state).
   368→   368→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   369→   369→  if (!decision.shouldRebalance) return;
   370→   370→  const usd = decision.tradeUsd;
   371→   371→  if (decision.direction === -1) {
   372→   372→    // sell gold → buy USDC (simplified: convert to USDC)
   373→   373→    const goldUnits = usd / goldPrice;
   374→   374→    s.paxg = Math.max(0, s.paxg - goldUnits);
   375→   375→    s.usdc += usd;
   376→   376→  } else {
   377→   377→    // buy gold → spend USDC
   378→   378→    s.usdc = Math.max(0, s.usdc - usd);
   379→   379→    const goldUnits = usd / goldPrice;
   380→   380→    s.paxg += goldUnits;
   381→   381→  }
   382→   382→  s.lastTradeDir = decision.direction;
   383→   383→  s.lastTradeAt = Date.now();
   384→   384→  s.dailyTurnoverUsd += usd;
   385→   385→  s.updatedAt = Date.now();
   386→   386→}

   388→   388→// --- §11 Geopolitical Eject -------------------------------------------------
   389→   389→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   390→   390→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   391→   391→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   392→   392→  const drift = (cur: number, amp: number) => {
   393→   393→    const shock = (Math.random() - 0.5) * 2 * amp;
   394→   394→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   395→   395→  };
   396→   396→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   397→   397→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   398→   398→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   399→   399→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   400→   400→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   401→   401→
   402→   402→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   403→   403→  for (const c of currencies) {
   404→   404→    const p = s.pegHealth[c];
   405→   405→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   406→   406→    if (outside) {
   407→   407→      s.depegHours[c] += dtHours;
   408→   408→    } else {
   409→   409→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   410→   410→    }
   411→   411→    // Staged ladder
   412→   412→    const h = s.depegHours[c];
   413→   413→    let stage = 0;
   414→   414→    if (h > 48) stage = 3;
   415→   415→    else if (h > 24) stage = 2;
   416→   416→    else if (h > 12) stage = 1;
   417→   417→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   418→   418→    s.ejectStage[c] = stage;
   419→   419→  }
   420→   420→}

   436→   436→export function applyMint(
   437→   437→  s: ReserveState,
   438→   438→  fx: FxRates,
   439→   439→  status: ProtocolStatus,
   440→   440→  inputUsd: number,
   441→   441→): MintResult {
   442→   442→  const gfb = computeGfbIndex(fx);
   443→   443→  const price = computeMtqPrice(gfb);
   444→   444→  if (!priceInSafetyBand(price)) {
   445→   445→    return { ok: false, reason: "MTQ price outside safety band (0.50–2.00 USD); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   446→   446→  }
   447→   447→  if (status === "DEFENSIVE" || status === "EMERGENCY") {
   448→   448→    return { ok: false, reason: `${status}: minting paused by Risk State Machine.`, inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   449→   449→  }
   450→   450→  const throttle = status === "CAUTION" ? 0.5 : status === "RECOVERY" ? 0.25 : 1.0;
   451→   451→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
   452→   452→  const netUsd = inputUsd - feeUsd;
   453→   453→  const minted = (netUsd / price) * throttle;
   454→   454→
   455→   455→  // Mutate reserve: deposit USDC, mint MTQ to user (circulating supply rises)
   456→   456→  s.usdc += netUsd + feeUsd; // fee stays in reserve (operational surplus)
   457→   457→  s.totalSupply += minted;
   458→   458→  s.updatedAt = Date.now();
   459→   459→
   460→   460→  const vals = reserveAssetValues(s, fx);
   461→   461→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
   462→   462→  return {
   463→   463→    ok: true,
   464→   464→    inputUsd,
   465→   465→    feeUsd,
   466→   466→    netUsd,
   467→   467→    mtqPrice: price,
   468→   468→    mtqMinted: minted,
   469→   469→    throttleFactor: throttle,
   470→   470→    newCirculatingSupply: circulatingSupply(s),
   471→   471→    newReserveRatio: newRr,
   472→   472→  };
   473→   473→}

   499→   499→export function applyRedeem(
   500→   500→  s: ReserveState,
   501→   501→  fx: FxRates,
   502→   502→  status: ProtocolStatus,
   503→   503→  inputMtq: number,
   504→   504→): RedeemResult {
   505→   505→  const gfb = computeGfbIndex(fx);
   506→   506→  const price = computeMtqPrice(gfb);
   507→   507→  const circ = circulatingSupply(s);
   508→   508→  if (inputMtq > circ) {
   509→   509→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   510→   510→  }
   511→   511→  // Fee by status (§14.1): NORMAL/CAUTION 0.15%, DEFENSIVE 0.5%, EMERGENCY 2%
   512→   512→  let feeBps = REDEEM_FEE_BPS;
   513→   513→  if (status === "DEFENSIVE") feeBps = 50;
   514→   514→  if (status === "EMERGENCY") feeBps = 200;
   515→   515→  const grossUsd = inputMtq * price; // redemption is priced against the GFB Index (§3.4.2)
   516→   516→  const feeUsd = grossUsd * (feeBps / 10_000);
   517→   517→  const netUsd = grossUsd - feeUsd;
   518→   518→
   519→   519→  // Release actual reserve composition proportionally (§12.2):
   520→   520→  // gold portion = netUsd * W_target ; fiat portion split per GFB weights.
   521→   521→  const vals = reserveAssetValues(s, fx);
   522→   522→  const rr = computeReserveRatio(vals.nav, computeLiability(s, price));
   523→   523→  const wGold = computeTargetGoldWeight(s, rr);
   524→   524→  const goldUsd = netUsd * wGold;
   525→   525→  const fiatUsd = netUsd * (1 - wGold);
   526→   526→  const goldPrice = vals.goldPrice;
   527→   527→  const goldPaxg = goldUsd / goldPrice;
   528→   528→
   529→   529→  const basketFiatTotal = Q_USD + Q_EUR + Q_GBP + Q_JPY + Q_CNY; // = 1.0
   530→   530→  const basket: RedeemBasket[] = [
   531→   531→    { currency: "USD", token: "USDC", usdValue: fiatUsd * (Q_USD / basketFiatTotal), nativeAmount: fiatUsd * (Q_USD / basketFiatTotal), weight: Q_USD },
   532→   532→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * (Q_EUR / basketFiatTotal), nativeAmount: (fiatUsd * (Q_EUR / basketFiatTotal)) / fx.EUR_USD, weight: Q_EUR },
   533→   533→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * (Q_GBP / basketFiatTotal), nativeAmount: (fiatUsd * (Q_GBP / basketFiatTotal)) / fx.GBP_USD, weight: Q_GBP },
   534→   534→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * (Q_JPY / basketFiatTotal), nativeAmount: (fiatUsd * (Q_JPY / basketFiatTotal)) / fx.JPY_USD, weight: Q_JPY },
   535→   535→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * (Q_CNY / basketFiatTotal), nativeAmount: (fiatUsd * (Q_CNY / basketFiatTotal)) / fx.CNY_USD, weight: Q_CNY },
   536→   536→  ];
   537→   537→
   538→   538→  // Mutate reserve: burn MTQ from circulating (reduce totalSupply), release assets
   539→   539→  s.totalSupply -= inputMtq;
   540→   540→  s.usdc = Math.max(0, s.usdc - basket[0].nativeAmount - (goldUsd > 0 && wGold < 1 ? 0 : 0));
   541→   541→  s.usdc = Math.max(0, s.usdc - basket[0].nativeAmount);
   542→   542→  s.eurc = Math.max(0, s.eurc - basket[1].nativeAmount);
   543→   543→  s.gbp = Math.max(0, s.gbp - basket[2].nativeAmount);
   544→   544→  s.jpy = Math.max(0, s.jpy - basket[3].nativeAmount);
   545→   545→  s.cny = Math.max(0, s.cny - basket[4].nativeAmount);
   546→   546→  s.paxg = Math.max(0, s.paxg - goldPaxg);
   547→   547→  s.updatedAt = Date.now();
   548→   548→
   549→   549→  const newVals = reserveAssetValues(s, fx);
   550→   550→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
   551→   551→  return {
   552→   552→    ok: true,
   553→   553→    inputMtq,
   554→   554→    mtqPrice: price,
   555→   555→    grossUsd,
   556→   556→    feeBps,
   557→   557→    feeUsd,
   558→   558→    netUsd,
   559→   559→    goldUsd,
   560→   560→    goldPaxg,
   561→   561→    basket,
   562→   562→    newCirculatingSupply: circulatingSupply(s),
   563→   563→    newReserveRatio: newRr,
   564→   564→  };
   565→   565→}

   660→   660→// Advance macro history (call ~ every tick; dtHours is simulated elapsed)

   662→   662→  const v = clamp(vix, VIX_MIN, VIX_MAX);
   663→   663→  const d = clamp(dxy, DXY_MIN, DXY_MAX);
   664→   664→  s.lastVix = v;
   665→   665→  s.lastDxy = d;
   666→   666→  s.vixHistory.push(v);
   667→   667→  s.dxyHistory.push(d);
   668→   668→  if (s.vixHistory.length > ROLLING_WINDOW_DAYS) s.vixHistory.shift();
   669→   669→  if (s.dxyHistory.length > ROLLING_WINDOW_DAYS) s.dxyHistory.shift();
   670→   670→  const vStats = rollingStats(s.vixHistory);
   671→   671→  const dStats = rollingStats(s.dxyHistory);
   672→   672→  const zVix = vStats.sd > 1e-9 ? (v - vStats.mean) / (vStats.sd + 1e-9) : 0;
   673→   673→  const zDxy = dStats.sd > 1e-9 ? (d - dStats.mean) / (dStats.sd + 1e-9) : 0;
   674→   674→  const rawTarget = computeRawTargetGoldWeight(zVix, zDxy);
