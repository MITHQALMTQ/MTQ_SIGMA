   203→   203→function seedHistory(mean: number, sd: number, n: number): number[] {
   204→   204→  const out: number[] = [];
   205→   205→  let last = mean;
   206→   206→  for (let i = 0; i < n; i++) {
   207→   207→    // simple mean-reverting walk
   208→   208→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   209→   209→    last = last + (mean - last) * 0.1 + shock;
   210→   210→    out.push(Math.max(0.01, last));
   211→   211→  }
   212→   212→  return out;
   213→   213→}

   239→   239→// USD value of each holding (gross), then apply haircuts for NAV.
   240→   240→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   241→   241→  const usdGross = (s.usdc + s.usdp + s.usdt) * 1.0; // USDC + USDP + USDT combined
   242→   242→  const eurGross = s.eurc * fx.EUR_USD;
   243→   243→  const gbpGross = s.gbp * fx.GBP_USD;
   244→   244→  const jpyGross = s.jpy * fx.JPY_USD;
   245→   245→  const cnyGross = s.cny * fx.CNY_USD;
   246→   246→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross;
   247→   247→
   248→   248→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   249→   249→  const goldRef = fx.XAU_USD;
   250→   250→  const goldExec = fx.XAU_USD * 0.995;
   251→   251→  const goldPrice = Math.min(goldRef, goldExec);
   252→   252→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   253→   253→
   254→   254→  // Net values (after haircuts)
   255→   255→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   256→   256→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   257→   257→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   258→   258→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   259→   259→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   260→   260→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   261→   261→
   262→   262→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + goldNet;
   263→   263→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet;
   264→   264→
   265→   265→  // Per-issuer breakdown (for concentration visualization)
   266→   266→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   267→   267→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   268→   268→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   269→   269→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);
   270→   270→  const xautUsd = s.xaut * goldPrice * (1 - HAIRCUTS.XAU);
   271→   271→
   272→   272→  return {
   273→   273→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, fiatGross,
   274→   274→    goldGross, goldPrice,
   275→   275→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, goldNet,
   276→   276→    nav, fiatNet,
   277→   277→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   278→   278→  };
   279→   279→}

   311→   311→// --- §6 Adaptive Macro Engine -----------------------------------------------
   312→   312→function rollingStats(hist: number[]): { mean: number; sd: number } {
   313→   313→  const filled = hist.filter((x) => x > 0);
   314→   314→  if (filled.length === 0) return { mean: 0, sd: 0 };
   315→   315→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   316→   316→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   317→   317→  return { mean, sd: Math.sqrt(variance) };
   318→   318→}
   319→   319→
   320→   320→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   321→   321→  const v = rollingStats(s.vixHistory);
   322→   322→  const d = rollingStats(s.dxyHistory);
   323→   323→  const eps = 1e-9;
   324→   324→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   325→   325→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   326→   326→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   327→   327→}

   329→   329→export function computeRawTargetTheta(zVix: number, zDxy: number): number {
   330→   330→  const theta = ALPHA * zVix + BETA * zDxy;
   331→   331→  return clampSym(theta, -THETA_MAX, THETA_MAX);
   332→   332→}

   334→   334→export function computeRawTargetGoldWeight(zVix: number, zDxy: number): number {
   335→   335→  const theta = computeRawTargetTheta(zVix, zDxy);
   336→   336→  return clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);

   339→   339→export function smoothGoldWeight(prev: number, target: number): number {
   340→   340→  if (prev <= 0) return target;
   341→   341→  return SMOOTHING_LAMBDA * target + (1 - SMOOTHING_LAMBDA) * prev;

   351→   351→export function currentBufferGoldRatio(s: ReserveState): number {
   352→   352→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   353→   353→  if (s.rampStartAt === 0) {
   354→   354→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   355→   355→  }
   356→   356→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   357→   357→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   358→   358→  const progress = elapsedH / RAMP_DURATION_HOURS;
   359→   359→  const from = s.rampFromRatio;
   360→   360→  const to = s.rampToRatio;
   361→   361→  return from + (to - from) * progress;
   362→   362→}

   365→   365→  if (rr >= RR_TARGET) return "BASE";
   366→   366→  if (rr >= RR_STRESS) return "STRESS";
   367→   367→  return "EMERGENCY";

   370→   370→// Unified target gold weight (tokenomics reconciliation):
   371→   371→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   372→   372→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   373→   373→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   374→   374→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   375→   375→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   376→   376→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);

   380→   380→export interface RebalanceDecision {
   381→   381→  shouldRebalance: boolean;
   382→   382→  direction: 0 | 1 | -1;
   383→   383→  observedGoldWeight: number;
   384→   384→  targetGoldWeight: number;

   386→   386→  tradeUsd: number;
   387→   387→  reason: string;

   390→   390→export function maybeResetDailyTurnover(s: ReserveState): void {
   391→   391→  if (Date.now() - s.lastTurnoverResetAt >= 86_400_000) {
   392→   392→    s.dailyTurnoverUsd = 0;
   393→   393→    s.lastTurnoverResetAt = Date.now();
   394→   394→  }
   395→   395→}
   396→   396→
   397→   397→export function evaluateRebalance(
   398→   398→  s: ReserveState,
   399→   399→  vals: { nav: number; goldNet: number; fiatNet: number },
   400→   400→  rr: number,
   401→   401→): RebalanceDecision {
   402→   402→  const nav = vals.nav;
   403→   403→  if (nav <= 0) {
   404→   404→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   405→   405→  }
   406→   406→  const observed = vals.goldNet / nav;
   407→   407→  const target = computeTargetGoldWeight(s, rr);
   408→   408→  const deviation = observed - target;
   409→   409→
   410→   410→  if (Math.abs(deviation) < 0.005) {
   411→   411→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   412→   412→  }
   413→   413→
   414→   414→  // Direction lock (overridden in stress)
   415→   415→  const inStress = rr < RR_STRESS;
   416→   416→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   417→   417→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   418→   418→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   419→   419→    }
   420→   420→  }
   421→   421→
   422→   422→  // Trade sizing
   423→   423→  const surplusUsd = Math.abs(deviation) * nav;
   424→   424→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   425→   425→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   426→   426→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   427→   427→  maybeResetDailyTurnover(s);
   428→   428→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   429→   429→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   430→   430→
   431→   431→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   432→   432→  const estCost = tradeUsd * 0.001 + 50;
   433→   433→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   434→   434→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   435→   435→
   436→   436→  if (benefit <= estCost) {
   437→   437→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   438→   438→  }
   439→   439→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   440→   440→}

   442→   442→// Apply a rebalance trade in USD terms (mutates state).
   443→   443→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   444→   444→  if (!decision.shouldRebalance) return;
   445→   445→  const usd = decision.tradeUsd;
   446→   446→  if (decision.direction === -1) {
   447→   447→    // sell gold → buy USDC (simplified: convert to USDC)
   448→   448→    const goldUnits = usd / goldPrice;
   449→   449→    s.paxg = Math.max(0, s.paxg - goldUnits);
   450→   450→    s.usdc += usd;
   451→   451→  } else {
   452→   452→    // buy gold → spend USDC
   453→   453→    s.usdc = Math.max(0, s.usdc - usd);
   454→   454→    const goldUnits = usd / goldPrice;
   455→   455→    s.paxg += goldUnits;
   456→   456→  }
   457→   457→  s.lastTradeDir = decision.direction;
   458→   458→  s.lastTradeAt = Date.now();
   459→   459→  s.dailyTurnoverUsd += usd;
   460→   460→  s.updatedAt = Date.now();
   461→   461→}

   463→   463→// --- §11 Geopolitical Eject -------------------------------------------------
   464→   464→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   465→   465→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   466→   466→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   467→   467→  const drift = (cur: number, amp: number) => {
   468→   468→    const shock = (Math.random() - 0.5) * 2 * amp;
   469→   469→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   470→   470→  };
   471→   471→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   472→   472→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   473→   473→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   474→   474→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   475→   475→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   476→   476→
   477→   477→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   478→   478→  for (const c of currencies) {
   479→   479→    const p = s.pegHealth[c];
   480→   480→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   481→   481→    if (outside) {
   482→   482→      s.depegHours[c] += dtHours;
   483→   483→    } else {
   484→   484→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   485→   485→    }
   486→   486→    // Staged ladder
   487→   487→    const h = s.depegHours[c];
   488→   488→    let stage = 0;
   489→   489→    if (h > 48) stage = 3;
   490→   490→    else if (h > 24) stage = 2;
   491→   491→    else if (h > 12) stage = 1;
   492→   492→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   493→   493→    s.ejectStage[c] = stage;
   494→   494→  }
   495→   495→}

   511→   511→export function applyMint(

   524→   524→  }

   529→   529→
   530→   530→  // Mutate reserve: deposit USDC (split across 3 USD issuers per concentration policy),
   531→   531→  // mint MTQ to user (circulating supply rises). The concentration rebalancer in the
   532→   532→  // tick loop will re-optimize the split if any issuer exceeds 25%.
   533→   533→  const usdThird = netUsd / 3;
   534→   534→  s.usdc += usdThird;
   535→   535→  s.usdp += usdThird;
   536→   536→  s.usdt += usdThird;

   538→   538→  s.totalSupply += minted;
   539→   539→  s.updatedAt = Date.now();
   540→   540→
   541→   541→  const vals = reserveAssetValues(s, fx);
   542→   542→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));

   546→   546→    feeUsd,
   547→   547→    netUsd,
   548→   548→    mtqPrice: price,
   549→   549→    mtqMinted: minted,
   550→   550→    throttleFactor: throttle,
   551→   551→    newCirculatingSupply: circulatingSupply(s),
   552→   552→    newReserveRatio: newRr,
   553→   553→  };

   809→   809→// Advance macro history (call ~ every tick; dtHours is simulated elapsed)
   810→   810→export function advanceMacro(s: ReserveState, vix: number, dxy: number, dtHours: number) {
   811→   811→  const v = clamp(vix, VIX_MIN, VIX_MAX);
   812→   812→  const d = clamp(dxy, DXY_MIN, DXY_MAX);

   849→   849→// --- §11.3 Reintegration Score (Anti-Gaming) --------------------------------
   850→   850→// R_score = w1·TimeInBand(48h) + w2·LiquidityDepth + w3·OracleAgreement + w4·(1−Volatility)
   851→   851→// All inputs normalised to [0,1]; threshold >0.80 → staged re-purchase 25%→50%→100%.
   852→   852→export function updateReintegration(
   853→   853→  s: ReserveState,
   854→   854→  board: OracleBoard | null,
   855→   855→  dtHours: number,
   856→   856→) {
   857→   857→  const currencies: ("USD" | "EUR" | "GBP" | "JPY" | "CNY")[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   858→   858→  for (const c of currencies) {
   859→   859→    const r = s.reintegration[c];
   860→   860→    // Time-in-band: increases when peg is inside [0.98,1.02]; decays when outside.
   861→   861→    const inBand = s.pegHealth[c] >= EJECT_DEPEG_BAND_LOWER && s.pegHealth[c] <= EJECT_DEPEG_BAND_UPPER;
   862→   862→    // dtHours is simulated elapsed; we approximate 48h window fraction.
   863→   863→    if (inBand) {
   864→   864→      r.timeInBand48h = Math.min(1, r.timeInBand48h + dtHours / 48);
   865→   865→    } else {
   866→   866→      r.timeInBand48h = Math.max(0, r.timeInBand48h - dtHours / 48);
   867→   867→    }
   868→   868→    // Liquidity depth: randomised around per-currency baseline (honest: simulated).
   869→   869→    const base = c === "USD" ? 1.0 : c === "EUR" ? 0.95 : c === "CNY" ? 0.4 : 0.5;
   870→   870→    r.liquidityDepth = clamp(base + (Math.random() - 0.5) * 0.1, 0, 1);
   871→   871→    // Oracle agreement: valid feeds / 3 (from oracle board if present, else last known)
   872→   872→    if (board) {
   873→   873→      const pairMap: Record<string, string> = { USD: "—", EUR: "EUR/USD", GBP: "GBP/USD", JPY: "JPY/USD", CNY: "CNY/USD" };
   874→   874→      const cons = board.pairs.find((p) => p.pair === pairMap[c]);
   875→   875→      if (cons) {
   876→   876→        r.oracleAgreement = cons.validCount / 3;
   877→   877→        // Volatility from oracle spread (bps → fraction)
   878→   878→        r.volatility = clamp(cons.spreadBps / 100, 0, 1);
   879→   879→      }
   880→   880→    }
   881→   881→    // R_score
   882→   882→    r.score =
   883→   883→      REINTEGRATION_WEIGHTS.w1_TimeInBand * r.timeInBand48h +
   884→   884→      REINTEGRATION_WEIGHTS.w2_LiquidityDepth * r.liquidityDepth +
   885→   885→      REINTEGRATION_WEIGHTS.w3_OracleAgreement * r.oracleAgreement +
   886→   886→      REINTEGRATION_WEIGHTS.w4_OneMinusVolatility * (1 - r.volatility);
   887→   887→    // Repurchase staging: only for currencies that were previously ejected
   888→   888→    if (s.ejectStage[c] > 0 && r.score > REINTEGRATION_THRESHOLD) {
   889→   889→      r.repurchaseStage = Math.min(3, r.repurchaseStage + 1);
   890→   890→      // As repurchase progresses, reduce eject stage
   891→   891→      if (r.repurchaseStage >= 3) {
   892→   892→        s.ejectStage[c] = Math.max(0, s.ejectStage[c] - 1);
   893→   893→        if (s.ejectStage[c] === 0) r.repurchaseStage = 0;
   894→   894→      }
   895→   895→    } else if (r.score < REINTEGRATION_THRESHOLD * 0.6) {
   896→   896→      r.repurchaseStage = 0;
   897→   897→    }
   898→   898→  }
   899→   899→}

   918→   918→// --- §8.5 First-Loss Waterfall ----------------------------------------------
   919→   919→// 5 layers: (1) Operational Surplus → (2) Buffer Fiat → (3) Buffer Gold →
   920→   920→// (4) Core Fiat → (5) Core Gold. Apply a loss; track consumed per layer.
   921→   921→export function applyLoss(s: ReserveState, vals: { fiatNet: number; goldNet: number }, lossUsd: number) {
   922→   922→  let remaining = lossUsd;
   923→   923→  s.waterfall.consumed.surplus = 0; // reset for this loss event tracking is cumulative; we add
   924→   924→  const layers = [
   925→   925→    { name: "surplus", cap: () => s.treasury.hotWalletUsd + s.waterfall.operationalSurplus, consume: (amt: number) => {
   926→   926→      const fromSurplus = Math.min(s.treasury.hotWalletUsd, amt);
   927→   927→      s.treasury.hotWalletUsd -= fromSurplus;
   928→   928→      s.waterfall.operationalSurplus -= Math.min(s.waterfall.operationalSurplus, amt - fromSurplus);
   929→   929→      s.waterfall.consumed.surplus += amt;
   930→   930→    } },
   931→   931→    { name: "bufferFiat", cap: () => vals.fiatNet * 0.10 * (1 - currentBufferGoldRatio(s)), consume: (amt: number) => { s.waterfall.consumed.bufferFiat += amt; } },
   932→   932→    { name: "bufferGold", cap: () => vals.goldNet * 0.10 * currentBufferGoldRatio(s), consume: (amt: number) => { s.waterfall.consumed.bufferGold += amt; } },
   933→   933→    { name: "coreFiat", cap: () => vals.fiatNet * 0.80, consume: (amt: number) => { s.waterfall.consumed.coreFiat += amt; } },
   934→   934→    { name: "coreGold", cap: () => vals.goldNet * 0.20, consume: (amt: number) => { s.waterfall.consumed.coreGold += amt; } },
   935→   935→  ] as const;
   936→   936→  for (let i = 0; i < layers.length && remaining > 0; i++) {
   937→   937→    const layer = layers[i];
   938→   938→    const cap = layer.cap();
   939→   939→    const consumed = Math.min(cap, remaining);
   940→   940→    layer.consume(consumed);
   941→   941→    remaining -= consumed;
   942→   942→    if (remaining > 0) s.waterfall.currentLayer = (i + 2) as 1 | 2 | 3 | 4 | 5;
   943→   943→  }
   944→   944→  if (remaining <= 0) s.waterfall.currentLayer = 1;
   945→   945→}

   955→   955→    s.lastLoggedPrice = newPrice;
   956→   956→  }

   984→   984→// --- FIX: §5.6 Concentration Rebalancing -------------------------------------
   985→   985→// Adjusts the USDC/USDP split to bring the maximum issuer share under the
   986→   986→// --- FIX: §5.6 Multi-Issuer Concentration Optimizer -------------------------
   987→   987→// 3 issuers: CIRCLE (USDC + EURC), PAXOS (USDP + PAXG), TETHER (USDT + XAUT).
   988→   988→// EURC is fixed to Circle (no second EUR issuer yet). GBP/JPY/CNY are TBD.
   989→   989→// The optimizer adjusts the splittable pools (USDC/USDP/USDT, PAXG/XAUT) to
   990→   990→// minimize the maximum issuer share, targeting ≤ 25% (warn threshold,
   991→   991→// headroom under the 30% hard limit).
   992→   992→export interface ConcentrationFixResult {
   993→   993→  adjusted: boolean;
   994→   994→  sharesBefore: { CIRCLE: number; PAXOS: number; TETHER: number };
   995→   995→  sharesAfter: { CIRCLE: number; PAXOS: number; TETHER: number };
   996→   996→  splits: { usdc: number; usdp: number; usdt: number; paxg: number; xaut: number };
   997→   997→  target: number; // 0.25
   998→   998→  reason: string;
   999→   999→}
  1000→  1000→
  1001→  1001→export function rebalanceForConcentration(
  1002→  1002→  s: ReserveState,
  1003→  1003→  vals: { usdcUsd: number; usdpUsd: number; usdtUsd: number; paxgUsd: number; xautUsd: number; nav: number; eurNet: number },
  1004→  1004→): ConcentrationFixResult {
  1005→  1005→  const nav = vals.nav;
  1006→  1006→  const TARGET = 0.25;
  1007→  1007→  const usdNetFactor = 1 - HAIRCUTS.USD;
  1008→  1008→  const goldNetFactor = 1 - HAIRCUTS.XAU;
  1009→  1009→
  1010→  1010→  const sharesBefore = {
  1011→  1011→    CIRCLE: nav > 0 ? (vals.usdcUsd + vals.eurNet) / nav : 0,
  1012→  1012→    PAXOS: nav > 0 ? (vals.usdpUsd + vals.paxgUsd) / nav : 0,
  1013→  1013→    TETHER: nav > 0 ? (vals.usdtUsd + vals.xautUsd) / nav : 0,
  1014→  1014→  };
  1015→  1015→  const maxBefore = Math.max(sharesBefore.CIRCLE, sharesBefore.PAXOS, sharesBefore.TETHER);
  1016→  1016→
  1017→  1017→  if (nav <= 0) {
  1018→  1018→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "NAV ≤ 0" };
  1019→  1019→  }
  1020→  1020→  if (maxBefore <= TARGET) {
  1021→  1021→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "All issuers ≤ 25% warn threshold" };
  1022→  1022→  }
  1023→  1023→
  1024→  1024→  // Total splittable pools (in gross token units)
  1025→  1025→  const totalUsdGross = s.usdc + s.usdp + s.usdt;
  1026→  1026→  const totalGoldGross = s.paxg + s.xaut;
  1027→  1027→
  1028→  1028→  // Circle's floor = EURC (fixed). Circle's USDC budget = max(0, TARGET×NAV − EURC_usd)
  1029→  1029→  const circleUsdNetBudget = Math.max(0, TARGET * nav - vals.eurNet);
  1030→  1030→  const usdcGross = Math.min(totalUsdGross, circleUsdNetBudget / usdNetFactor);
  1031→  1031→  const remainingUsdGross = Math.max(0, totalUsdGross - usdcGross);
  1032→  1032→
  1033→  1033→  // Split remaining USD equally between USDP (Paxos) and USDT (Tether)
  1034→  1034→  const usdpGross = remainingUsdGross / 2;
  1035→  1035→  const usdtGross = remainingUsdGross / 2;
  1036→  1036→
  1037→  1037→  // Now balance gold between PAXG (Paxos) and XAUT (Tether) so both ≤ 25%.
  1038→  1038→  // Clean approach: split gold 50/50 by default; only deviate if one issuer
  1039→  1039→  // would breach the 25% cap. This guarantees a real split (the previous
  1040→  1040→  // min-budget logic could dump all gold into one issuer when budgets were
  1041→  1041→  // asymmetric or when one issuer's USD share was already small).
  1042→  1042→  const paxosUsdNet = usdpGross * usdNetFactor;
  1043→  1043→  const tetherUsdNet = usdtGross * usdNetFactor;
  1044→  1044→  const halfGoldGross = totalGoldGross / 2;
  1045→  1045→  let paxgGross = halfGoldGross;
  1046→  1046→  let xautGross = halfGoldGross;
  1047→  1047→  // Cap check 1: Paxos = USDP_net + PAXG_net ≤ TARGET×NAV
  1048→  1048→  const paxosWithHalf = paxosUsdNet + halfGoldGross * goldNetFactor;
  1049→  1049→  if (paxosWithHalf > TARGET * nav) {
  1050→  1050→    const maxPaxgNet = Math.max(0, TARGET * nav - paxosUsdNet);
  1051→  1051→    paxgGross = Math.min(halfGoldGross, maxPaxgNet / goldNetFactor);
  1052→  1052→    xautGross = totalGoldGross - paxgGross;
