   514→    /// @notice Listing 2 (§7.7) submitTargetWeights — MASE verification + smoothing.
   515→    /// @dev   Only the MASE submitter (KEEPER_ROLE). Enforces sum=1, positivity,
   516→    ///        admissibility envelopes, per-component velocity, then applies
   517→    ///        stress-adaptive smoothing rho 0.50 (normal) / 0.75 (stress).
   518→    function submitTargetWeights(
   519→        uint256[7] calldata target,
   520→        bytes32 methodologyVersion,
   521→        bytes32 dataVersion
   522→    ) external onlyKeeper {
   523→        // 1. Sum-to-one and non-negativity
   524→        uint256 sum = 0;
   525→        for (uint256 i = 0; i < 7; i++) {
   526→            if (!(target[i] > 0)) revert Err62();
   527→            sum += target[i];
   528→        }
   529→        if (!(sum == 1e18)) revert Err54();
   530→        // 2. Constitutional admissibility envelopes
   531→        for (uint256 i = 0; i < 7; i++) {
   532→            if (!(target[i] >= LOWER_BOUND[i])) revert Err12();
   533→            if (!(target[i] <= UPPER_BOUND[i])) revert Err06();
   534→        }
   535→        // 3. Weight-velocity constraint vs previous live weights
   536→        for (uint256 i = 0; i < 7; i++) {
   537→            uint256 diff = target[i] > liveWeights.weights[i]
   538→                ? target[i] - liveWeights.weights[i]
   539→                : liveWeights.weights[i] - target[i];
   540→            if (!(diff <= MAX_VELOCITY[i])) revert Err52();
   541→        }
   542→        // 4. Stress-adaptive smoothing: W_smooth = rho * W_prev + (1 - rho) * W_target
   543→        uint256 rho = crisisFlag ? smoothingRhoStress : smoothingRhoNormal;
   544→        uint256[7] memory smoothed;
   545→        for (uint256 i = 0; i < 7; i++) {
   546→            smoothed[i] = (rho * liveWeights.weights[i] + (1e18 - rho) * target[i]) / 1e18;
   547→        }
   548→        // 5. Commit registry state
   549→        for (uint256 i = 0; i < 7; i++) liveWeights.weights[i] = smoothed[i];
   550→        liveWeights.targetWeights    = target;
   551→        liveWeights.methodologyVersion = methodologyVersion;
   552→        liveWeights.dataVersion       = dataVersion;
   553→        liveWeights.updatedAt         = block.timestamp;
   554→        emit WeightsAccepted(smoothed, target, methodologyVersion, dataVersion, block.timestamp);
   555→    }

   590→    /// @dev §9.8 returns the base-fixing price for component i (1e18 scale).
   591→    ///      USD → 1.0; EUR → 1.05; JPY → 0.0067; GBP → 1.25;
   592→    ///      CNY → 0.14; CHF → 1.13 (FIXED); Gold → 2500.
   593→    function _basePrice18(uint256 i) internal pure returns (uint256) {
   594→        if (i == 0) return 1e18;             // USD = 1 USD
   595→        if (i == 1) return BASE_EUR_USD;     // 1.05
   596→        if (i == 2) return BASE_JPY_USD;     // 0.0067
   597→        if (i == 3) return BASE_GBP_USD;     // 1.25
   598→        if (i == 4) return BASE_CNY_USD;     // 0.14
   599→        if (i == 5) return BASE_CHF_USD;     // 1.13
   600→        return BASE_GOLD_USD;                 // 2500

   603→    /// @notice Listing 3 advanceIndex — §9.2 authoritative recursion.
   604→    ///         I_t = I_{t-1} × Σ_i W_{i,t-1} × (P_{i,t} / P_{i,t-1})
   605→    ///         Prices are passed in by the keeper (1e18 scale, USD-quoted).
   606→    function advanceIndex(uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   607→        if (!(indexValue > 0)) revert Err20();
   608→
   609→        // 1. Period return weighted by W_{i,t-1} in force over the period (COO-16).
   610→        uint256 growth = 0;
   611→        for (uint256 i = 0; i < 7; i++) {
   612→            if (!(currentPrices[i] > 0 && lastPrices[i] > 0)) revert Err09();
   613→            growth += (lastWeights[i] * currentPrices[i]) / lastPrices[i];
   614→        }
   615→        indexValue = (indexValue * growth) / 1e18;
   616→
   617→        // 2. Commit the period end-state (prices).
   618→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   619→        lastIndexUpdate = block.timestamp;
   620→        emit IndexAdvanced(indexValue, block.timestamp);
   621→    }

   623→    /// @notice Listing 3 commitWeights — chain-link adjustment at weight change.
   624→    ///         Computes D_t = B_t^- / B_t^+ (Master §9.3), updates G_t, updates
   625→    ///         lastPrices and lastWeights. I_t is UNCHANGED by the weight change
   626→    ///         itself — ZERO artificial return (the recursion is continuous).
   627→    function commitWeights(uint256[7] calldata newWeights, uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   628→        if (!(indexValue > 0)) revert Err20();
   629→
   630→        // B11 FIX: Validate weights sum to 1e18 (within 1e14 tolerance = 0.01%).
   631→        // Without this, a buggy keeper can commit weights summing to 0.79 or 1.21,
   632→        // causing the chain-linked index to drift systematically. Err54 mirrors
   633→        // the sum-to-one check already enforced in submitTargetWeights (L480).
   634→        uint256 weightSum = 0;
   635→        for (uint256 i = 0; i < 7; i++) {
   636→            weightSum += newWeights[i];
   637→        }
   638→        if (weightSum > 1e18 + 1e14 || weightSum < 1e18 - 1e14) revert Err54();
   639→
   640→        // 1. B_t^- and B_t^+ are base-relative aggregates measured against
   641→        //    the genesis fixings (§9.3, MS §75).
   642→        uint256 bMinus = 0;
   643→        uint256 bPlus  = 0;
   644→        bool    changed = false;
   645→        for (uint256 i = 0; i < 7; i++) {
   646→            if (!(currentPrices[i] > 0)) revert Err09();
   647→            uint256 rel = (currentPrices[i] * 1e18) / _basePrice18(i); // P_{i,t} / P_{i,0}
   648→            bMinus += (lastWeights[i] * rel) / 1e18;
   649→            bPlus  += (newWeights[i]    * rel) / 1e18;
   650→            if (newWeights[i] != lastWeights[i]) changed = true;
   651→        }
   652→
   653→        // 2. Apply divisor D_t; record into G_t (cumulative). I_t unchanged.
   654→        if (changed) {
   655→            if (!(bPlus > 0)) revert Err57();
   656→            uint256 newD = (bMinus * 1e18) / bPlus;
   657→            chainLinkDivisor = (chainLinkDivisor * newD) / 1e18;
   658→            for (uint256 i = 0; i < 7; i++) lastWeights[i] = newWeights[i];
   659→            emit ChainLinkAdjusted(newD, bMinus, bPlus, block.timestamp);
   660→        }
   661→
   662→        // 3. Commit the period end-state prices too.
   663→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   664→        emit WeightsCommitted(newWeights, chainLinkDivisor, block.timestamp);
   665→    }

   667→    /// @notice Listing 3 genesis — one-time seed of I_0 = 1.0000 (ADMIN_ROLE).
   668→    ///         Prices are seeded from immutable base fixings, so I_0 = 1.0000
   669→    ///         exactly independent of feed noise.
   670→    function genesisIndex() external onlyAdmin {
   671→        if (!(indexValue == 1e18 && lastIndexUpdate == 0)) revert Err19();
   672→        if (!(liveWeights.updatedAt > 0)) revert Err01();
   673→        // Seed lastWeights from MASE registry
   674→        uint256 sum = 0;
   675→        for (uint256 i = 0; i < 7; i++) {
   676→            lastWeights[i] = liveWeights.weights[i];
   677→            lastPrices[i]  = _basePrice18(i);
   678→            sum += liveWeights.weights[i];
   679→        }
   680→        if (!(sum == 1e18)) revert Err44();
   681→        indexValue = 1e18;
   682→        chainLinkDivisor = 1e18;
   683→        lastIndexUpdate = block.timestamp;
   684→        emit GenesisVerified(INDEX_BASE_DENOMINATOR, block.timestamp);
   685→    }

   720→    /// @notice Listing 13 updateState — the CANONICAL state function (§14.6.2).
   721→    ///         Worse applicable condition binds; RECOVERY requires 48h
   722→    ///         confirmation; transitions to a more restrictive state are
   723→    ///         immediate, transitions to a less restrictive state require
   724→    ///         the RECOVERY_CONFIRMATION_PERIOD to elapse.
   725→    function updateState(uint256 rr, uint256 lcr) external onlyKeeper {
   726→        ProtocolState newState;
   727→        // Determine the target state based on RR and LCR
   728→        if (rr >= rrTarget && lcr >= lcrTarget) {
   729→            newState = ProtocolState.NORMAL;
   730→        } else if (rr >= rrStressFloor && lcr >= 0.90e18) {
   731→            newState = ProtocolState.CAUTION;
   732→        } else if (rr >= 1.02e18 && lcr >= 0.80e18) {
   733→            newState = ProtocolState.STRESS;
   734→        } else if (rr >= RR_HARD_FLOOR && lcr >= 0.70e18) {
   735→            newState = ProtocolState.DEFENSIVE;
   736→        } else if (rr < RR_HARD_FLOOR) {
   737→            newState = ProtocolState.EMERGENCY;
   738→        } else {
   739→            // RECOVERY: solvency ≥ hard floor but ratios not yet back to NORMAL.
   740→            newState = ProtocolState.RECOVERY;
   741→        }
   742→
   743→        if (newState != currentState) {
   744→            if (_isMoreRestrictive(newState, currentState)) {
   745→                _transitionTo(newState, true);
   746→            } else {
   747→                // Less restrictive → 48h confirmation
   748→                if (block.timestamp - stateEntryTime[currentState] >= RECOVERY_CONFIRMATION_PERIOD) {
   749→                    _transitionTo(newState, false);
   750→                }
   751→                // else: stay (newState discarded — not yet confirmed)
   752→            }
   753→        }
   754→    }

   883→    /// @notice Listing 14 proposeChange — only the authorized caller per layer:
   884→    ///         DAO for MONETARY, riskCouncil for RISK, emergencyCouncil for

   886→    function proposeChange(bytes32 parameterId, uint256 newValue) external {
   887→        ParameterRecord storage rec = parameterRegistry[parameterId];
   888→        if (!(rec.registered)) revert Err50();
   889→        if (!(newValue >= rec.minValue)) revert Err11();
   890→        if (!(newValue <= rec.maxValue)) revert Err05();
   891→        if (rec.layer == ParameterLayer.MONETARY)         if (!(msg.sender == dao)) revert Err36();
   892→        else if (rec.layer == ParameterLayer.RISK)        if (!(msg.sender == riskCouncil)) revert Err38();
   893→        else if (rec.layer == ParameterLayer.EMERGENCY)   if (!(msg.sender == emergencyCouncil)) revert Err37();
   894→        else /* CONSTITUTIONAL */                          if (!(msg.sender == constitutionalCouncil)) revert Err35();
   895→        proposals[parameterId] = Proposal({
   896→            value: newValue, proposer: msg.sender, timestamp: block.timestamp, executed: false
   897→        });
   898→        emit ParameterProposed(parameterId, newValue, block.timestamp);
   899→    }

   901→    /// @notice Listing 14 executeChange — anyone can call after the timelock
   902→    ///         elapses; the timelock + role check on propose are the protection.
   903→    function executeChange(bytes32 parameterId) external {
   904→        Proposal storage p = proposals[parameterId];
   905→        if (!(p.timestamp > 0)) revert Err27();
   906→        if (!(!p.executed)) revert Err07();
   907→        ParameterRecord storage rec = parameterRegistry[parameterId];
   908→        uint256 delay;
   909→        if (rec.layer == ParameterLayer.MONETARY)         delay = TIMELOCK_MONETARY;
   910→        else if (rec.layer == ParameterLayer.RISK)        delay = TIMELOCK_RISK;
   911→        else if (rec.layer == ParameterLayer.EMERGENCY)  delay = TIMELOCK_EMERGENCY;
   912→        else                                              delay = TIMELOCK_CONSTITUTIONAL;
   913→        if (!(block.timestamp >= p.timestamp + delay)) revert Err47();
   914→        if (!(p.value >= rec.minValue && p.value <= rec.maxValue)) revert Err40();
   915→        p.executed = true;
   916→        rec.value = p.value;
   917→        _applyParam(parameterId, p.value);
   918→        emit ParameterExecuted(parameterId, p.value, block.timestamp);
   919→    }

   924→        if (!(p.timestamp > 0 && !p.executed)) revert Err34();

   968→    /// @notice Set oracle adapter for a source (0=Chainlink, 1=Pyth, 2=Chronicle).
   969→    function setOracleAdapter(uint8 source, address adapter) external onlyRole(ORACLE_ROLE) {
   970→        if (!(source <= 2)) revert Err10();
   971→        if (!(adapter != address(0))) revert Err55(); // H8 source-independence
   972→        // B12 FIX: Enforce 3 distinct adapters — source independence (I9).
   973→        // Reject any (source, adapter) pair that would collide with an already-
   974→        // configured adapter. Without this, an admin could (maliciously or by
   975→        // mistake) point two of the three source slots at the same adapter,
   976→        // collapsing the 3-source median into a 2-of-1 self-consensus.
   977→        if (source == 0 && (adapter == address(pythAdapter)       || adapter == address(chronicleAdapter))) revert Err63();
   978→        if (source == 1 && (adapter == address(chainlinkAdapter)  || adapter == address(chronicleAdapter))) revert Err63();
   979→        if (source == 2 && (adapter == address(chainlinkAdapter)  || adapter == address(pythAdapter)))      revert Err63();
   980→        if (source == 0)      chainlinkAdapter  = IOracleAdapter(adapter);
   981→        else if (source == 1) pythAdapter       = IOracleAdapter(adapter);
   982→        else                  chronicleAdapter = IOracleAdapter(adapter);
   983→        emit OracleAdapterSet(source, adapter, msg.sender);
   984→    }

   986→    /// @notice §9.1-9.3 Consensus across all 3 adapters for a single pair.
   987→    /// @return finalPrice  USD-per-unit, 1e18 scale (0 if paused)
   988→    /// @return validCount  Number of valid feeds after all filters
   989→    /// @return method      0=paused, 1=average(2) [DEPRECATED by B6], 2=median(3)
   990→    /// @return paused_     True if fewer than 3 feeds are valid (B6: strict I9)
   991→    /// @return prices      Raw prices per adapter (1e18) — index 0/1/2 = CL/Pyth/Cro
   992→    /// @return valid       Per-adapter validity flag after all filters
   993→    function getOracleConsensus(bytes32 pair) public view returns (
   994→        uint256 finalPrice,
   995→        uint8  validCount,
   996→        uint8  method,
   997→        bool   paused_,
   998→        uint256[3] memory prices,
   999→        bool[3] memory valid
  1000→    ) {
  1001→        IOracleAdapter[3] memory adapters;
  1002→        adapters[0] = chainlinkAdapter;
  1003→        adapters[1] = pythAdapter;
  1004→        adapters[2] = chronicleAdapter;
  1005→
  1006→        uint256[3] memory raw;
  1007→        uint256[3] memory ts;
  1008→        uint256[3] memory conf;
  1009→
  1010→        for (uint8 i = 0; i < 3; i++) {
  1011→            valid[i] = false;
  1012→            if (address(adapters[i]) == address(0)) continue;
  1013→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
  1014→                if (p == 0) continue; // C6: reject zero-price feeds
  1015→                raw[i]  = p;
  1016→                ts[i]   = t;
  1017→                conf[i] = c;
  1018→                prices[i] = p;
  1019→                // §9.2.1 Staleness ≤60s (also reject future timestamps)
  1020→                if (t > block.timestamp) continue;
  1021→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
  1022→                // §9.2.3 Confidence < 1% of price
  1023→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
  1024→                valid[i] = true;
  1025→            } catch { /* leave valid[i] = false */ }
  1026→        }
  1027→
  1028→        validCount = 0;
  1029→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
  1030→
  1031→        if (validCount < 2) {
  1032→            paused_ = true; method = 0;
  1033→            return (finalPrice, validCount, method, paused_, prices, valid);
  1034→        }
  1035→
  1036→        // Sort the valid prices and pick median (n=3) or average (n=2)
  1037→        uint256[3] memory sorted;
  1038→        uint8 n = 0;
  1039→        for (uint8 i = 0; i < 3; i++) if (valid[i]) { sorted[n] = raw[i]; n++; }
  1040→        _insertionSort(sorted, n);
  1041→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
  1042→
  1043→        // §9.2.4 Deviation < 2.5% from median → discard
  1044→        for (uint8 i = 0; i < 3; i++) {
  1045→            if (valid[i]) {
  1046→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
  1047→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
  1048→                    valid[i] = false;
  1049→                    validCount--;
  1050→                }
  1051→            }
  1052→        }
  1053→
  1054→        // B6 FIX: Strict I9 — pause when <3 valid sources.
  1055→        // No 2-source averaging (governance override is off-chain only;
  1056→        // on-chain must always enforce strict 3-source quorum).
  1057→        paused_ = validCount < 3;
  1058→        if (paused_) {
  1059→            method     = 0; // paused (existing convention: 0 = paused)
  1060→            finalPrice = 0;
  1061→            return (finalPrice, validCount, method, paused_, prices, valid);
  1062→        }
  1063→
  1064→        // validCount == 3 — final price is the median.
  1065→        finalPrice = med;
  1066→        method     = 2; // median(3)
  1067→    }

  1127→    ///         contribute 0. If the registry is unset, falls back to gross.

  1130→        bytes32[7] memory codes;
  1131→        codes[0] = bytes32("USD"); codes[1] = bytes32("EUR");
  1132→        codes[2] = bytes32("JPY"); codes[3] = bytes32("GBP");
  1133→        codes[4] = bytes32("CNY"); codes[5] = bytes32("CHF");
  1134→        codes[6] = bytes32("XAU");
  1135→        uint256 nav = 0;
  1136→        for (uint8 i = 0; i < 7; i++) {
  1137→            uint256 held = reserveHeldUsd[Component(i)];

  1140→                address, uint256 haircut, uint8 state, bytes32

  1143→                nav += held * (1e18 - haircut) / 1e18;
  1144→            } catch { nav += held; }
  1145→        }
  1146→        return nav;
  1147→    }

  1234→    // §3.4.2 + §12.1  Mint — priced against P_MTQ (I5) with canonical throttle
  1235→    //   Pull USDC (6 dec), fee = mintFee, MTQ minted = netUsd18 × 1e18 / P_MTQ,
  1236→    //   throttled by mintThrottle(currentState). REVERT if !mintingAllowed.
  1237→    // ============================================================
  1238→    function mint(uint256 usdcAmount) external nonReentrant whenNotPaused returns (uint256 minted) {
  1239→        if (!(usdcAmount > 0)) revert Err56();
  1240→        if (!(mintingAllowed(currentState))) revert Err25();
  1241→
  1242→        // B3 FIX: Gate on oracle health — reject if any critical pair is paused.
  1243→        // Without this, users could mint against stale prices when the keeper
  1244→        // has missed a tick. This is the #1 historical stablecoin exploit vector.
  1245→        _requireOracleHealthy();
  1246→
  1247→        uint256 price = getMTQPriceWithGuard();
  1248→
  1249→        // Pull USDC from minter to reserve vault (6 decimals).
  1250→        if (!(usdc.transferFrom(msg.sender, reserveVault, usdcAmount))) revert Err03();
  1251→
  1252→        // §12.1 fee = mintFee (1e18 fraction). E.g. 0.001e18 = 0.10%.
  1253→        uint256 feeUsd = usdcAmount * mintFee / 1e18;
  1254→        uint256 netUsd = usdcAmount - feeUsd;
  1255→        // Scale 6-dec USDC → 18-dec USD value, then divide by 18-dec price.
  1256→        uint256 netUsd18  = netUsd * 1e12;
  1257→        uint256 grossMint = netUsd18 * 1e18 / price;
  1258→
  1259→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25)
  1260→        uint256 throttle = mintThrottle(currentState);
  1261→        minted = grossMint * throttle / 1e18;
  1262→
  1263→        // Checks-Effects-Interactions (H1): state updated BEFORE any external call.
  1264→        _mint(msg.sender, minted);
  1265→        emit Mint(msg.sender, usdcAmount, feeUsd, minted, price);
  1266→    }

  1270→    //   Burn MTQ, grossUsd = mtqAmount × NAVperToken (NOT × P_MTQ),
  1271→    //   fee = redeemFee(currentState), net → USDC 6 dec from reserve vault,
  1272→    //   decrement reserveHeldUsd proportionally (basket broken).

  1275→    function redeem(uint256 mtqAmount) external nonReentrant whenNotPaused returns (uint256 usdcOut) {
  1276→        if (!(mtqAmount > 0)) revert Err56();

  1278→        if (!(redemptionAllowed(currentState))) revert Err43();

  1280→        // B3 FIX: Gate on oracle health — same rationale as mint().
  1281→        _requireOracleHealthy();
  1282→
  1283→        // H4: apply price guard even on redeem (rejects if outside [0.50, 2.00]).
  1284→        uint256 price = getMTQPriceWithGuard();
  1285→        // §19.3.2 NAV-per-token (NOT P_MTQ)
  1286→        uint256 navPerToken = getNAVperToken();
  1287→        if (!(navPerToken < type(uint256).max)) revert Err26();
  1288→
  1289→        // §14.6.3 fee by state (1e18 fraction)
  1290→        uint256 feeFraction = redeemFee(currentState);
  1291→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
  1292→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
  1293→        uint256 netUsd18   = grossUsd18 - feeUsd18;
  1294→        usdcOut            = netUsd18 / 1e12; // 18-dec → 6-dec USDC
  1295→
  1296→        // Checks-Effects-Interactions (H2): burn + decrement BEFORE external call.
  1297→        _burn(msg.sender, mtqAmount);
  1298→        // Decrement reserveHeldUsd proportionally (basket broken on redeem).
  1299→        // Use gross USD value across all components so the share is exact.
  1300→        uint256 totalReserve = getReserveNavUsd();
  1301→        if (totalReserve > 0 && grossUsd18 > 0) {
  1302→            for (uint8 i = 0; i < 7; i++) {
  1303→                uint256 held = reserveHeldUsd[Component(i)];
  1304→                if (held == 0) continue;
  1305→                uint256 share = held * grossUsd18 / totalReserve;
  1306→                if (share > held) share = held; // safety
  1307→                reserveHeldUsd[Component(i)] = held - share;
  1308→            }
  1309→        }
  1310→        // Update reserveState mirror
  1311→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1312→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1313→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1314→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1315→        reserveState.liquidValue = reserveState.fiatValue;

  1317→        // External call LAST (after all state updates).
  1318→        if (!(usdc.transferFrom(reserveVault, msg.sender, usdcOut))) revert Err04();

  1320→        // silence unused var warning

  1325→    // §10  MARP Rebalancing Execution — keeper-callable
  1326→    //   For each trade: validates it moves toward the latest committed
  1327→    //   lastWeights (executionWeight) within a 5% tolerance, enforces 24h
  1328→    //   direction lock (with RR<1.05 solvency override — C4 fix), validates
  1329→    //   total daily trade USD ≤ 5% of NAV, caps trades.length at 7 (H3 fix),

  1333→    struct RebalanceTrade {
  1334→        Component component;
  1335→        int256    direction; // +1 buy, -1 sell, 0 hold
  1336→        uint256   tradeUsd;  // USD value, 1e18 scale
  1337→        uint256   level;     // 1-6 (MARP decision level)
  1338→    }
  1339→
  1340→    uint256 public constant DIRECTION_LOCK_HOURS  = 24 hours;
  1341→    uint256 public constant MAX_DAILY_TURNOVER    = 0.05e18;
  1342→    uint256 public constant REBALANCE_TOLERANCE    = 0.05e18;
  1343→
  1344→    mapping(Component => int256)  public lastDirection;
  1345→    mapping(Component => uint256) public lastRebalanceAt;
  1346→    uint256 public dailyTradeUsd;
  1347→    uint256 public dailyTradeResetAt;
  1348→
  1349→    event RebalanceExecuted(RebalanceTrade[] trades, uint256 timestamp);
  1350→
  1351→    function executeRebalance(RebalanceTrade[] memory trades) external nonReentrant onlyKeeper whenNotPaused {
  1352→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1353→        if (!(trades.length <= 7)) revert Err48(); // H3 cap
  1354→
  1355→        // Reset 24h daily turnover window if elapsed
  1356→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1357→            dailyTradeUsd    = 0;
  1358→            dailyTradeResetAt = block.timestamp;
  1359→        }
  1360→
  1361→        uint256 nav = getReserveNavUsd();
  1362→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1363→        uint256 rr = getReserveRatio();
  1364→        bool solvencyOverride = (rr < rrStressFloor); // §11.5.3 / C4 fix
  1365→
  1366→        for (uint256 i = 0; i < trades.length; i++) {
  1367→            RebalanceTrade memory t = trades[i];
  1368→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1369→            if (t.direction == 0 || t.tradeUsd == 0) continue; // hold
  1370→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1371→
  1372→            // §10 24h direction lock — opposite direction blocked within 24h,
  1373→            // UNLESS RR<1.05 (solvency override — C4 fix).
  1374→            int256 prevDir = lastDirection[t.component];
  1375→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1376→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1377→            }
  1378→
  1379→            // Validate trade moves toward execution weight (within 5% tolerance)
  1380→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1381→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1382→            if (t.direction == 1) {
  1383→                if (!(currentW < execW)) revert Err14();
  1384→            } else {
  1385→                if (!(currentW > execW)) revert Err46();
  1386→            }
  1387→
  1388→            // New weight after trade must be within 5pp (absolute) of execW
  1389→            uint256 newHeldUsd = (t.direction == 1)
  1390→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1391→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1392→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1393→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1394→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1395→            // Solvency override relaxes overshoot tolerance by 2x during stress
  1396→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1397→            if (!(absDiff <= tolerance)) revert Err49();
  1398→
  1399→            // §10 daily turnover cap (5% of NAV)
  1400→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1401→            dailyTradeUsd += t.tradeUsd;
  1402→
  1403→            // Update reserve mirror (checks-effects-interactions)
  1404→            if (t.direction == 1) {
  1405→                reserveHeldUsd[t.component] += t.tradeUsd;
  1406→            } else {
  1407→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1408→            }
  1409→            lastDirection[t.component]   = t.direction;
  1410→            lastRebalanceAt[t.component] = block.timestamp;
  1411→        }
  1412→
  1413→        // Refresh reserveState aggregate mirror after rebalancing
  1414→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1415→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1416→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1417→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1418→        reserveState.liquidValue = reserveState.fiatValue;
  1419→
  1420→        emit RebalanceExecuted(trades, block.timestamp);
  1421→    }
