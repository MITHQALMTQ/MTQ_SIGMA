   372→    // ============================================================
   373→    // §11.5 Stability Pool — Emergency Recovery Mechanism (T3 fix)
   374→    // ------------------------------------------------------------
   375→    //  When the 6-state risk machine enters EMERGENCY (RR < 1.0), there is
   376→    //  no automated recovery path in V2/early-V3. The Stability Pool gives
   377→    //  the protocol a USDC-denominated buffer that the keeper can deploy
   378→    //  to cover the deficit, restoring RR ≥ 1.0 and lifting the protocol
   379→    //  out of EMERGENCY once reserves are replenished off-chain.
   380→    //
   381→    //  Design (mirrors Listing 13 §11.5 + the audit T3 remediation brief):
   382→    //    • Users deposit USDC (6-dec) via `depositToStabilityPool`. The
   383→    //      contract takes custody of the USDC (the only ERC-20 the V3
   384→    //      contract ever custodies — every other reserve is held by banks'
   385→    //      qualified custodians per the ABC architecture).
   386→    //    • Depositors earn a share of rewards via the `stabilityRewardRate`
   387→    //      per-second fraction (1e18 scale). Rewards accrue as
   388→    //      `accumulatedRewardPerShare` (1e18 scale, USDC-denominated).
   389→    //    • In EMERGENCY (RR < RR_HARD_FLOOR = 1.0e18), the keeper calls
   390→    //      `useStabilityPoolForDeficit(deficitUsd)`. The pool covers up to

   594→    // === V3 NEW §6 — Bank Registry (Blueprint v25.3 §6) ===
   595→    // ------------------------------------------------------------
   596→    // The Constitutional Council (7/7 multi-sig) authorizes banks. Each

   601→    struct BankRecord {
   602→        address bankAddress;
   603→        string  name;
   604→        string  jurisdiction;   // ISO 3166-1 alpha-2 ("US", "AE", "SA", ...)
   605→        bool    isAuthorized;
   606→        uint256 dailyMintCap;   // USD, 1e18 scale
   607→        uint256 dailyMintUsed;  // USD, 1e18 scale (resets daily)
   608→        uint256 dailyMintResetAt; // unix seconds of last reset
   609→        uint256 authorizedAt;
   610→    }

   612→    mapping(address => BankRecord) public bankRegistry;
   613→    address[] public authorizedBanks;
   614→
   615→    function authorizeBank(
   616→        address bank,
   617→        string calldata _name,
   618→        string calldata _jurisdiction,
   619→        uint256 _dailyCap
   620→    ) external onlyConstitutionalCouncil {
   621→        if (bank == address(0)) revert Err55();
   622→        if (_dailyCap == 0) revert Err87_DailyCapZero();
   623→        bool already = bankRegistry[bank].isAuthorized;
   624→        bankRegistry[bank] = BankRecord({
   625→            bankAddress: bank,
   626→            name: _name,
   627→            jurisdiction: _jurisdiction,
   628→            isAuthorized: true,
   629→            dailyMintCap: _dailyCap,
   630→            dailyMintUsed: 0,
   631→            dailyMintResetAt: block.timestamp,
   632→            authorizedAt: block.timestamp
   633→        });
   634→        if (!already) authorizedBanks.push(bank);
   635→        emit BankAuthorized(bank, _name, _jurisdiction, _dailyCap, block.timestamp);
   636→    }

   639→        if (!bankRegistry[bank].isAuthorized) revert Err84_BankNotFound();
   640→        bankRegistry[bank].isAuthorized = false;
   641→        emit BankRevoked(bank, block.timestamp);
   642→    }

   652→    /// @dev Rolls the per-bank daily mint counter at UTC midnight.
   653→    ///      Idempotent within the same UTC day. Called internally by mint
   654→    ///      paths; also callable by anyone (no-op if already current).
   655→    function _rollDailyCap(address bank) internal {
   656→        BankRecord storage rec = bankRegistry[bank];
   657→        if (rec.dailyMintResetAt == 0) return;
   658→        uint256 today = (block.timestamp / 1 days) * 1 days;
   659→        if (rec.dailyMintResetAt < today) {
   660→            uint256 oldUsed = rec.dailyMintUsed;
   661→            rec.dailyMintUsed = 0;
   662→            rec.dailyMintResetAt = today;
   663→            emit DailyCapReset(bank, oldUsed, 0, block.timestamp);
   664→        }
   665→    }

   668→    // === V3 NEW §7 — AvailableBackingCertificate (ABC) ===
   669→    // ------------------------------------------------------------
   670→    // Banks submit an ABC proving they have backing assets held by a
   671→    // qualified custodian. The keeper verifies the custodian attestation
   672→    // (production: via oracle; testnet: auto-verify via verifyBackingCertificate).
   673→    //
   674→    // MITHQAL NEVER takes custody — it only verifies evidence.
   675→    // ============================================================
   676→
   677→    struct AvailableBackingCertificate {
   678→        bytes32 certificateId;
   679→        address bank;
   680→        uint256 amountUsd;                   // 1e18 scale
   681→        bytes32 custodianAttestationHash;    // hash of custodian's attestation
   682→        uint256 custodianAttestationExpiry;  // unix seconds
   683→        bytes32 reserveAssetBreakdownHash;   // hash of the reserve composition
   684→        uint256 issuedAt;
   685→        bool    verified;
   686→    }
   687→
   688→    mapping(bytes32 => AvailableBackingCertificate) public backingCertificates;
   689→
   690→    function submitBackingCertificate(AvailableBackingCertificate calldata cert) external onlyAuthorizedBank {
   691→        // The submitting bank must match the certificate's bank field.
   692→        if (cert.bank != msg.sender) revert Err92_BackdingCertMismatch();
   693→        if (cert.amountUsd == 0) revert Err86_InvalidRequest();
   694→        if (cert.custodianAttestationExpiry <= block.timestamp) revert Err91_InvalidExpiry();
   695→        if (cert.certificateId == bytes32(0)) revert Err86_InvalidRequest();
   696→        // Reject duplicate certificate IDs.
   697→        if (backingCertificates[cert.certificateId].issuedAt != 0) revert Err69_NonceReused();
   698→
   699→        backingCertificates[cert.certificateId] = AvailableBackingCertificate({
   700→            certificateId: cert.certificateId,
   701→            bank: cert.bank,
   702→            amountUsd: cert.amountUsd,
   703→            custodianAttestationHash: cert.custodianAttestationHash,
   704→            custodianAttestationExpiry: cert.custodianAttestationExpiry,
   705→            reserveAssetBreakdownHash: cert.reserveAssetBreakdownHash,
   706→            issuedAt: block.timestamp,
   707→            verified: false
   708→        });
   709→        emit BackingCertificateSubmitted(cert.certificateId, cert.bank, cert.amountUsd, block.timestamp);
   710→    }
   711→
   712→    /// @notice Verify an ABC's custodian attestation. Production: oracle call.
   713→    ///         Testnet/pilot: keeper auto-verifies (the keeper is trusted).
   714→    function verifyBackingCertificate(bytes32 certId) external onlyKeeper returns (bool) {
   715→        AvailableBackingCertificate storage c = backingCertificates[certId];
   716→        if (c.issuedAt == 0) revert Err85_CertNotFound();
   717→        if (c.custodianAttestationExpiry <= block.timestamp) revert Err72_BackdingCertExpired();
   718→        c.verified = true;
   719→        // Credit the bank's verified-backing pool (evidence only — not custody).
   720→        backingVerified[c.bank] += c.amountUsd;
   721→        emit BackingCertificateVerified(certId, msg.sender, true, block.timestamp);
   722→        return true;
   723→    }

   729→    // ============================================================
   730→    // === V3 NEW §8 — 16-Step Bank Minting Workflow ===
   731→    // ------------------------------------------------------------
   732→    // BM-01 PENDING        — bank submits requestMint()
   733→    // BM-02 RECEIVED       — MBG gateway ack (keeper: API auth + schema OK)
   734→    // BM-03 KYC            — KYC passed (keeper: off-chain evidence)
   735→    // BM-04 AML            — AML/sanctions passed (keeper)
   736→    // BM-05 BACKING        — ABC submitted & in registry
   737→    // BM-06 EVIDENCE       — ABC verified (custodian attestation valid)
   738→    // BM-07 REQUESTED      — MBG has translated request to on-chain
   739→    // BM-08 TRANSLATED     — request payload validated against policy
   740→    // BM-09 ELIGIBLE       — bank + jurisdiction eligible
   741→    // BM-10 JURISDICTION   — jurisdiction checks (sanctions/exchange control)
   742→    // BM-11 BACKING_VERIFIED — backing ≥ mint amount
   743→    // BM-12 BANK_RISK      — bank-level risk (cap, concentration, credit)
   744→    // BM-13 SYSTEM_RISK    — system-level risk (RR/LCR/state machine)
   745→    // BM-14 DMCE           — Daily Mint Cap / Concentration / Exposure check
   746→    // BM-15 AUTHORIZED     — MITHQAL Monetary Control authorization (L4)
   747→    // BM-16 MINTED         — executeMint() final atomic write (L6+L7)
   748→    // REJECTED             — terminal failure state
   749→    // ============================================================
   750→
   751→    enum WorkflowState {
   752→        NONE,
   753→        BM01_PENDING,
   754→        BM02_RECEIVED,
   755→        BM03_KYC,
   756→        BM04_AML,
   757→        BM05_BACKING,
   758→        BM06_EVIDENCE,
   759→        BM07_REQUESTED,
   760→        BM08_TRANSLATED,
   761→        BM09_ELIGIBLE,
   762→        BM10_JURISDICTION,
   763→        BM11_BACKING_VERIFIED,
   764→        BM12_BANK_RISK,
   765→        BM13_SYSTEM_RISK,
   766→        BM14_DMCE,
   767→        BM15_AUTHORIZED,
   768→        BM16_MINTED,
   769→        REJECTED
   770→    }

   772→    struct BankMintRequest {
   773→        address bank;
   774→        bytes32 corporateCustomerHash;     // hash of the corporate customer identity
   775→        uint256 amountUsd;                  // 1e18 scale
   776→        bytes32 backingCertificateId;
   777→        string  jurisdiction;               // ISO 3166-1 alpha-2
   778→        uint256 nonce;                      // bank-unique anti-replay
   779→        uint256 expiry;                     // unix seconds; request expires after
   780→    }

   782→    struct MintRequest {
   783→        bytes32 requestId;
   784→        BankMintRequest request;
   785→        WorkflowState state;
   786→        uint256 createdAt;
   787→        uint256 authorizedAt;
   788→        uint256 mintedAt;
   789→        uint256 mintedAmount;
   790→        string  rejectReason;
   791→    }

   793→    mapping(bytes32 => MintRequest) public mintRequests;
   794→    /// @notice Anti-replay: bank → nonce → consumed.
   795→    mapping(address => mapping(uint256 => bool)) public usedNonces;
   796→    /// @notice Per-bank concentration: total outstanding MTQ minted via this bank.
   797→    mapping(address => uint256) public bankMintedTotal;
   798→    /// @notice DMCE: Daily Mint Cap & Concentration Exposure — single-bank cap
   799→    ///         as a fraction of total supply (1e18 scale). Default 5%.
   800→    uint256 public dmceBankConcentrationCap = 0.05e18;

   802→    /// @notice Request a mint — only an authorized bank. Creates state=BM01_PENDING.
   803→    function requestMint(BankMintRequest calldata req) external onlyAuthorizedBank whenNotPaused returns (bytes32 requestId) {
   804→        if (req.bank != msg.sender) revert Err70_JurisdictionMismatch();
   805→        if (req.amountUsd == 0) revert Err86_InvalidRequest();
   806→        if (req.expiry <= block.timestamp) revert Err91_InvalidExpiry();
   807→        if (usedNonces[msg.sender][req.nonce]) revert Err69_NonceReused();
   808→        usedNonces[msg.sender][req.nonce] = true;
   809→
   810→        // L3 POLICY (early): per-bank daily cap precheck (full enforcement
   811→        // happens at BM-12/BM-14, but we reject early to fail fast).
   812→        _rollDailyCap(msg.sender);
   813→        BankRecord storage bank = bankRegistry[msg.sender];
   814→        if (bank.dailyMintUsed + req.amountUsd > bank.dailyMintCap) revert Err67_DailyCapExceeded();
   815→
   816→        // The certificate must already exist (BM-05 will verify it is in the
   817→        // registry; BM-06 will verify it is verified).
   818→        if (backingCertificates[req.backingCertificateId].issuedAt == 0) revert Err85_CertNotFound();
   819→
   820→        requestId = keccak256(abi.encodePacked(
   821→            msg.sender,
   822→            req.corporateCustomerHash,
   823→            req.amountUsd,
   824→            req.backingCertificateId,
   825→            req.nonce,
   826→            block.chainid,
   827→            block.timestamp
   828→        ));
   829→        if (mintRequests[requestId].createdAt != 0) revert Err69_NonceReused();
   830→
   831→        mintRequests[requestId] = MintRequest({
   832→            requestId: requestId,
   833→            request: req,
   834→            state: WorkflowState.BM01_PENDING,
   835→            createdAt: block.timestamp,
   836→            authorizedAt: 0,
   837→            mintedAt: 0,
   838→            mintedAmount: 0,
   839→            rejectReason: ""
   840→        });
   841→        emit MintRequested(requestId, msg.sender, req.amountUsd, req.backingCertificateId, req.jurisdiction, block.timestamp);
   842→        emit WorkflowAdvanced(requestId, WorkflowState.NONE, WorkflowState.BM01_PENDING, msg.sender, block.timestamp);
   843→    }
   844→
   845→    /// @notice Advance a request through the workflow. Only KEEPER (BM-02..BM-14)
   846→    ///         or MONETARY_CONTROL (BM-15) may advance. State transitions must
   847→    ///         be strictly sequential (no skipping). The keeper performs the
   848→    ///         L3 policy checks (KYC/AML/eligibility/jurisdiction/bank-risk/
   849→    ///         system-risk/DMCE) off-chain and only calls advanceWorkflow when
   850→    ///         the corresponding gate has passed. The L4 monetary authorization
   851→    ///         gate (BM-15) requires the MONETARY_CONTROL_ROLE signer.
   852→    function advanceWorkflow(bytes32 requestId, WorkflowState toState) external whenNotPaused returns (bool) {
   853→        MintRequest storage r = mintRequests[requestId];
   854→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   855→        if (r.state == WorkflowState.BM16_MINTED) revert Err76_WorkflowAlreadyMinted();
   856→        if (r.state == WorkflowState.REJECTED) revert Err77_WorkflowAlreadyRejected();
   857→        if (r.request.expiry <= block.timestamp) revert Err68_RequestExpired();
   858→
   859→        // L2 WORKFLOW GATE — strict sequential progression.
   860→        // toState must equal (current + 1).
   861→        if (uint256(toState) != uint256(r.state) + 1) revert Err74_WorkflowNotSequential();
   862→
   863→        // Role gate per state:
   864→        //   BM-02..BM-14 → KEEPER_ROLE
   865→        //   BM-15        → MONETARY_CONTROL_ROLE (L4)
   866→        if (toState == WorkflowState.BM15_AUTHORIZED) {
   867→            if (!(_roles[MONETARY_CONTROL_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err80_L4MonetaryGate();
   868→        } else {
   869→            if (!(_roles[KEEPER_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err31();
   870→        }
   871→
   872→        // L3 POLICY GATE — keeper has run the off-chain policy engine and
   873→        // attests that all checks for `toState` have passed. We re-verify
   874→        // a small number of on-chain invariants per state for defense in depth.
   875→        if (toState == WorkflowState.BM05_BACKING) {
   876→            // The ABC must exist.
   877→            if (backingCertificates[r.request.backingCertificateId].issuedAt == 0) revert Err85_CertNotFound();
   878→        } else if (toState == WorkflowState.BM06_EVIDENCE) {
   879→            // The ABC must be verified.
   880→            AvailableBackingCertificate storage c = backingCertificates[r.request.backingCertificateId];
   881→            if (!c.verified) revert Err71_BackdingCertNotVerified();
   882→            if (c.custodianAttestationExpiry <= block.timestamp) revert Err72_BackdingCertExpired();
   883→        } else if (toState == WorkflowState.BM09_ELIGIBLE) {
   884→            // Bank must still be authorized.
   885→            if (!bankRegistry[r.request.bank].isAuthorized) revert Err66_BankNotAuthorized();
   886→        } else if (toState == WorkflowState.BM10_JURISDICTION) {
   887→            // Jurisdiction on request must match bank's registered jurisdiction.
   888→            BankRecord storage b = bankRegistry[r.request.bank];
   889→            if (keccak256(bytes(b.jurisdiction)) != keccak256(bytes(r.request.jurisdiction))) revert Err70_JurisdictionMismatch();
   890→        } else if (toState == WorkflowState.BM11_BACKING_VERIFIED) {
   891→            // Verified backing pool for the bank must be ≥ requested amount.
   892→            if (backingVerified[r.request.bank] < r.request.amountUsd) revert Err73_BackdingCertInsufficient();
   893→        } else if (toState == WorkflowState.BM12_BANK_RISK) {
   894→            // Per-bank daily cap precheck (rolls the day first).
   895→            _rollDailyCap(r.request.bank);
   896→            BankRecord storage br = bankRegistry[r.request.bank];
   897→            if (br.dailyMintUsed + r.request.amountUsd > br.dailyMintCap) revert Err67_DailyCapExceeded();
   898→        } else if (toState == WorkflowState.BM13_SYSTEM_RISK) {
   899→            // Minting must be allowed by the 6-state risk machine.
   900→            if (!mintingAllowed(currentState)) revert Err25();
   901→            // Oracle must be healthy (strict 3-source quorum on all 6 pairs).
   902→            _requireOracleHealthy();
   903→        } else if (toState == WorkflowState.BM14_DMCE) {
   904→            // DMCE: single-bank concentration cap.
   905→            uint256 concentration = (totalSupply == 0)
   906→                ? type(uint256).max
   907→                : bankMintedTotal[r.request.bank] * 1e18 / totalSupply;
   908→            // Reject if adding this mint would breach the concentration cap.
   909→            uint256 newMintedMTQ = r.request.amountUsd * 1e18 / getMTQPriceWithGuard();
   910→            uint256 newConcentration = (totalSupply + newMintedMTQ == 0)
   911→                ? 0
   912→                : (bankMintedTotal[r.request.bank] + newMintedMTQ) * 1e18 / (totalSupply + newMintedMTQ);
   913→            if (newConcentration > dmceBankConcentrationCap && concentration > dmceBankConcentrationCap) revert Err88_ConcentrationLimit();
   914→        } else if (toState == WorkflowState.BM15_AUTHORIZED) {
   915→            // L4 MONETARY CONTROL — the modifier already gated this.
   916→            // Mark the authorization timestamp.
   917→            r.authorizedAt = block.timestamp;
   918→        }
   919→
   920→        WorkflowState fromState = r.state;
   921→        r.state = toState;
   922→        emit WorkflowAdvanced(requestId, fromState, toState, msg.sender, block.timestamp);
   923→        return true;
   924→    }

   926→    /// @notice Reject a request (terminal). Keeper or Monetary Control only.
   927→    function rejectMint(bytes32 requestId, string calldata reason) external whenNotPaused returns (bool) {
   928→        MintRequest storage r = mintRequests[requestId];
   929→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   930→        if (r.state == WorkflowState.BM16_MINTED) revert Err76_WorkflowAlreadyMinted();
   931→        if (r.state == WorkflowState.REJECTED) revert Err77_WorkflowAlreadyRejected();
   932→        if (!(_roles[KEEPER_ROLE][msg.sender] || _roles[MONETARY_CONTROL_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err31();
   933→        WorkflowState fromState = r.state;
   934→        r.state = WorkflowState.REJECTED;
   935→        r.rejectReason = reason;

   937→        emit MintRejected(requestId, msg.sender, reason, block.timestamp);
   938→        return true;

   941→    /// @notice L5+L6+L7 FINALITY GATE — execute the mint atomically.
   942→    ///         Requires state == BM15_AUTHORIZED (L2/L5), oracle healthy (L3
   943→    ///         recheck), keeper signature (L7). Single transaction = atomic
   944→    ///         write (L6). Mints MTQ to the bank; the bank distributes to the
   945→    ///         corporate customer off-chain (sovereign currency settlement).
   946→    /// @dev LIQ-MGMT (audit H5 + M2): the caller passes a `minMinted` floor
   947→    ///      (1e18 scale MTQ). The mint reverts with Err93_SlippageExceeded if
   948→    ///      the actual minted amount (after price + throttle) falls below it.
   949→    ///      The mint also reverts with Err94_PerTxCapExceeded if the requested
   950→    ///      USD amount exceeds MAX_MINT_PER_TX ($500K). Pass minMinted = 0 to
   951→    ///      disable slippage protection (e.g. for keeper-relayed bank requests
   952→    ///      where the bank has signed off on the exact expected output).
   953→    function executeMint(bytes32 requestId, uint256 minMinted) external onlyKeeper whenNotPaused nonReentrant returns (uint256 minted) {
   954→        MintRequest storage r = mintRequests[requestId];
   955→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   956→        // L2 WORKFLOW GATE — must be at BM15_AUTHORIZED.
   957→        if (r.state != WorkflowState.BM15_AUTHORIZED) revert Err78_L2WorkflowGate();
   958→        // L5 LEDGER GATE — must not already be minted.
   959→        if (r.mintedAt != 0) revert Err76_WorkflowAlreadyMinted();
   960→        // L3 POLICY GATE (recheck) — minting allowed, oracle healthy, not expired.
   961→        if (!mintingAllowed(currentState)) revert Err25();
   962→        _requireOracleHealthy();
   963→        if (r.request.expiry <= block.timestamp) revert Err68_RequestExpired();
   964→        // Re-check backing is still verified and sufficient (defence in depth).
   965→        if (backingVerified[r.request.bank] < r.request.amountUsd) revert Err73_BackdingCertInsufficient();
   966→
   967→        // LIQ-MGMT M2: per-transaction cap (audit M2). Reject if the requested
   968→        // USD amount exceeds MAX_MINT_PER_TX. This caps the largest single
   969→        // mint the keeper can land per tx, limiting blast radius if the keeper
   970→        // key or the L4 monetary-control signer is compromised.
   971→        if (r.request.amountUsd > MAX_MINT_PER_TX) revert Err94_PerTxCapExceeded();
   972→
   973→        // L4 MONETARY AUTHORIZATION — already enforced at advanceWorkflow(BM15)
   974→        // by the onlyMonetaryControl modifier (we re-verify authorizedAt > 0).
   975→        if (r.authorizedAt == 0) revert Err80_L4MonetaryGate();
   976→
   977→        // L7 FINALITY ORACLE — the keeper signature (msg.sender = keeper) is
   978→        // the on-chain finality proof. Without it, no mint can land.
   979→
   980→        // L6 ATOMIC WRITE — all state changes below are in a single tx; if any
   981→        // reverts, the whole mint reverts (atomic, all-or-nothing).
   982→
   983→        // Compute mint amount: amountUsd (1e18) × 1e18 / P_MTQ (1e18).
   984→        uint256 price = getMTQPriceWithGuard();
   985→        uint256 grossMint = r.request.amountUsd * 1e18 / price;
   986→
   987→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25).
   988→        uint256 throttle = mintThrottle(currentState);
   989→        minted = grossMint * throttle / 1e18;
   990→
   991→        // LIQ-MGMT H5: slippage protection. The caller (keeper, relaying the
   992→        // bank's signed quote) commits to a minimum acceptable minted amount.
   993→        // If the on-chain computed minted (price + throttle) is below it, the
   994→        // whole tx reverts — protecting the bank's corporate customer from
   995→        // receiving fewer MTQ than their quote promised.
   996→        if (minted < minMinted) revert Err93_SlippageExceeded();
   997→
   998→        // State transitions + accounting.
   999→        r.state = WorkflowState.BM16_MINTED;
  1000→        r.mintedAt = block.timestamp;
  1001→        r.mintedAmount = minted;
  1002→
  1003→        // Decrement the bank's verified-backing pool by the minted USD amount
  1004→        // (the backing is now "consumed" by the mint — the bank's custodian
  1005→        // will release the corresponding reserves to the redeemer at redeem
  1006→        // time, off-chain).
  1007→        if (backingVerified[r.request.bank] >= r.request.amountUsd) {
  1008→            backingVerified[r.request.bank] -= r.request.amountUsd;
  1009→        } else {
  1010→            // Should never hit thanks to the BM-11 check; defense in depth.
  1011→            backingVerified[r.request.bank] = 0;
  1012→        }
  1013→
  1014→        // Roll + charge the per-bank daily mint cap.
  1015→        _rollDailyCap(r.request.bank);
  1016→        bankRegistry[r.request.bank].dailyMintUsed += r.request.amountUsd;
  1017→        bankMintedTotal[r.request.bank] += minted;
  1018→
  1019→        // L7 finality: mint to the bank (NOT the corporate customer — the bank
  1020→        // settles with the customer off-chain per Blueprint v25.3 §6).
  1021→        _mint(r.request.bank, minted);
  1022→
  1023→        // §11.5.b T4 — Fee separation. Compute the mint-fee OBLIGATION on the
  1024→        // minted USD value and route it to the dedicated feeWallet (NOT
  1025→        // commingled with the reserve). This does NOT reduce the minted MTQ
  1026→        // amount (preserves backward compat with the LIQ-MGMT H5 tests that
  1027→        // assert minted == amountUsd at P_MTQ=1.0, throttle=1.0). The fee is
  1028→        // an on-chain obligation the bank owes to the feeWallet; settlement
  1029→        // happens off-chain via the same sovereign-currency rail as redeem.
  1030→        // If feeWallet has not been set yet, fees are still tracked in
  1031→        // accumulatedFees for later accounting (no revert — backward compat).
  1032→        {
  1033→            uint256 mintFeeUsd18 = r.request.amountUsd * mintFee / 1e18;
  1034→            if (mintFeeUsd18 > 0) {
  1035→                accumulatedFees += mintFeeUsd18;
  1036→                emit FeesCollected(mintFeeUsd18, 0, accumulatedFees, feeWallet);
  1037→            }
  1038→        }
  1039→
  1040→        emit WorkflowAdvanced(requestId, WorkflowState.BM15_AUTHORIZED, WorkflowState.BM16_MINTED, msg.sender, block.timestamp);
  1041→        emit MintExecuted(requestId, r.request.bank, r.request.amountUsd, minted, price, block.timestamp);
  1042→    }

  1044→    /// @notice V3 non-custodial redeem — burns MTQ from the redeemer and emits
  1045→    ///         a `RedeemSettlement` event. The bank settles with the redeemer
  1046→    ///         off-chain (sovereign currency transfer). The contract itself
  1047→    ///         does NOT push USDC: there is no reserveVault in V3.
  1048→    ///         Only an authorized bank can call this on behalf of a redeemer
  1049→    ///         (the redeemer must have approved the bank to burn their MTQ via
  1050→    ///         the standard ERC-20 allowance).
  1051→    function redeem(uint256 mtqAmount, address redeemer) external nonReentrant whenNotPaused returns (uint256 netUsd) {
  1052→        if (!(mtqAmount > 0)) revert Err56();
  1053→        if (!(redemptionAllowed(currentState))) revert Err43();
  1054→        // Only an authorized bank can submit a redeem for settlement.
  1055→        if (!bankRegistry[msg.sender].isAuthorized) revert Err90_NotAuthorizedBank();
  1056→        // Redeemer must have approved the bank to burn their MTQ.
  1057→        uint256 allowed = allowance[redeemer][msg.sender];
  1058→        if (allowed != type(uint256).max) {
  1059→            if (!(allowed >= mtqAmount)) revert Err21();
  1060→            allowance[redeemer][msg.sender] = allowed - mtqAmount;
  1061→        }
  1062→        if (!(balanceOf[redeemer] >= mtqAmount)) revert Err22();
  1063→
  1064→        _requireOracleHealthy();
  1065→        uint256 navPerToken = getNAVperToken();
  1066→        if (!(navPerToken < type(uint256).max)) revert Err26();
  1067→
  1068→        uint256 feeFraction = redeemFee(currentState);
  1069→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
  1070→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
  1071→        netUsd             = grossUsd18 - feeUsd18;
  1072→
  1073→        // LIQ-MGMT M2: per-transaction cap (audit M2). The gross USD value of
  1074→        // a single redeem is capped at MAX_REDEEM_PER_TX ($500K). This is
  1075→        // measured against grossUsd18 (NAV-per-token × mtqAmount), which is
  1076→        // the actual settlement obligation the bank incurs — NOT the net the
  1077→        // redeemer receives. Caps the largest single burn the protocol will
  1078→        // honour per tx.
  1079→        if (grossUsd18 > MAX_REDEEM_PER_TX) revert Err94_PerTxCapExceeded();
  1080→
  1081→        // LIQ-MGMT T7: daily redemption cap (audit T7). Rolls the 24h window
  1082→        // first (idempotent within the same day), then enforces that adding
  1083→        // this redeem's gross USD does not exceed dailyRedeemCap. The cap is
  1084→        // measured in gross-USD (pre-fee) since that is the reserve outflow
  1085→        // the protocol must honour. First redeem initializes the window.
  1086→        if (dailyRedeemResetAt == 0 || block.timestamp >= dailyRedeemResetAt + 1 days) {
  1087→            uint256 oldUsed = dailyRedeemedUsd;
  1088→            dailyRedeemedUsd = 0;
  1089→            dailyRedeemResetAt = block.timestamp;
  1090→            emit DailyRedeemCapReset(oldUsed, 0, block.timestamp);
  1091→        }
  1092→        if (dailyRedeemedUsd + grossUsd18 > dailyRedeemCap) revert Err95_DailyRedeemCapExceeded();
  1093→        dailyRedeemedUsd += grossUsd18;
  1094→        emit DailyRedeemCapConsumed(msg.sender, grossUsd18, dailyRedeemedUsd, dailyRedeemCap, block.timestamp);
  1095→
  1096→        _burn(redeemer, mtqAmount);
  1097→
  1098→        // Decrement the on-chain reserve mirror proportionally (basket broken).
  1099→        uint256 totalReserve = getReserveNavUsd();
  1100→        if (totalReserve > 0 && grossUsd18 > 0) {
  1101→            for (uint8 i = 0; i < 7; i++) {
  1102→                uint256 held = reserveHeldUsd[Component(i)];
  1103→                if (held == 0) continue;
  1104→                uint256 share = held * grossUsd18 / totalReserve;
  1105→                if (share > held) share = held;
  1106→                reserveHeldUsd[Component(i)] = held - share;
  1107→            }
  1108→        }
  1109→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1110→                                  reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1111→                                  reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1112→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1113→        reserveState.liquidValue = reserveState.fiatValue;
  1114→
  1115→        // Decrement the bank's outstanding MTQ counter (concentration reverse).
  1116→        if (bankMintedTotal[msg.sender] >= mtqAmount) {
  1117→            bankMintedTotal[msg.sender] -= mtqAmount;
  1118→        } else {
  1119→            bankMintedTotal[msg.sender] = 0;
  1120→        }
  1121→
  1122→        // §11.5.b T4 — Fee separation. The redeem fee (feeUsd18) is routed to
  1123→        // the dedicated feeWallet as an on-chain OBLIGATION (the bank owes the
  1124→        // fee to the feeWallet, settled off-chain). accumulatedFees is the
  1125→        // running total of unpaid fee obligations. If feeWallet has not been
  1126→        // set yet, fees are still tracked for later accounting (no revert —
  1127→        // backward compat with the LIQ-MGMT T7 tests that don't set it).
  1128→        if (feeUsd18 > 0) {
  1129→            accumulatedFees += feeUsd18;
  1130→            emit FeesCollected(0, feeUsd18, accumulatedFees, feeWallet);
  1131→        }
  1132→
  1133→        // The bank owes `netUsd` to the redeemer, settled off-chain via
  1134→        // sovereign currency transfer. Emit the settlement obligation.
  1135→        emit RedeemSettlement(redeemer, msg.sender, mtqAmount, grossUsd18, feeUsd18, netUsd, navPerToken, block.timestamp);
  1136→    }

  1212→    /// @notice LIQ-MGMT H4/O1: circuit-breaker helper. Compares the new MTQ
  1213→    ///         price (1e18 scale) against the price observed at the last
  1214→    ///         price-commit call. If the move exceeds MAX_PRICE_JUMP_PCT (10%),
  1215→    ///         the contract is auto-paused and `CircuitBreakerTriggered` is
  1216→    ///         emitted. `lastCommitPrice` is then updated to the new price.
  1217→    ///         The first commit (lastCommitPrice == 0) never triggers.
  1218→    /// @dev Uses getMTQPrice() (no guard) so the breaker can fire even when
  1219→    ///      the new price has exited the [0.50, 2.00] safety band — that is
  1220→    ///      exactly the scenario the breaker exists to catch. The pause takes
  1221→    ///      effect for all subsequent calls; the current commit call completes
  1222→    ///      (its state changes are NOT rolled back) so the offending price is
  1223→    ///      recorded on-chain for forensic review.
  1224→    function _circuitBreakerCheck(uint256 newPrice) internal {
  1225→        if (lastCommitPrice > 0 && newPrice > 0) {
  1226→            uint256 jumpPct = newPrice > lastCommitPrice
  1227→                ? ((newPrice - lastCommitPrice) * 100) / lastCommitPrice
  1228→                : ((lastCommitPrice - newPrice) * 100) / lastCommitPrice;
  1229→            if (jumpPct > MAX_PRICE_JUMP_PCT) {
  1230→                // Auto-pause on >10% price jump. The Pauser role can unpause
  1231→                // after the keeper has root-caused the move (oracle fault,
  1232→                // market shock, etc.) and re-verified the new price.
  1233→                paused = true;
  1234→                emit CircuitBreakerTriggered(newPrice, lastCommitPrice, jumpPct);
  1235→                emit Paused(msg.sender);
  1236→            }
  1237→        }
  1238→        lastCommitPrice = newPrice;
  1239→    }

  1244→    function submitTargetWeights(
  1245→        uint256[7] calldata target,
  1246→        bytes32 methodologyVersion,
  1247→        bytes32 dataVersion
  1248→    ) external onlyKeeper {
  1249→        uint256 sum = 0;
  1250→        for (uint256 i = 0; i < 7; i++) {
  1251→            if (!(target[i] > 0)) revert Err62();
  1252→            sum += target[i];
  1253→        }
  1254→        if (!(sum == 1e18)) revert Err54();
  1255→        for (uint256 i = 0; i < 7; i++) {
  1256→            if (!(target[i] >= LOWER_BOUND[i])) revert Err12();
  1257→            if (!(target[i] <= UPPER_BOUND[i])) revert Err06();
  1258→        }
  1259→        for (uint256 i = 0; i < 7; i++) {
  1260→            uint256 diff = target[i] > liveWeights.weights[i]
  1261→                ? target[i] - liveWeights.weights[i]

  1371→    function updateState(uint256 rr, uint256 lcr) external onlyKeeper {
  1372→        ProtocolState newState;
  1373→        if (rr >= RR_TARGET && lcr >= lcrTarget) {
  1374→            newState = ProtocolState.NORMAL;
  1375→        } else if (rr >= RR_FLOOR && lcr >= 0.90e18) {
  1376→            newState = ProtocolState.CAUTION;
  1377→        } else if (rr >= 1.02e18 && lcr >= 0.80e18) {
  1378→            newState = ProtocolState.STRESS;
  1379→        } else if (rr >= RR_HARD_FLOOR && lcr >= 0.70e18) {
  1380→            newState = ProtocolState.DEFENSIVE;
  1381→        } else if (rr < RR_HARD_FLOOR) {
  1382→            newState = ProtocolState.EMERGENCY;
  1383→        } else {
  1384→            newState = ProtocolState.RECOVERY;
  1385→        }
  1386→        if (newState != currentState) {
  1387→            if (_isMoreRestrictive(newState, currentState)) {
  1388→                _transitionTo(newState, true);
  1389→            } else {
  1390→                if (block.timestamp - stateEntryTime[currentState] >= RECOVERY_CONFIRMATION_PERIOD) {
  1391→                    _transitionTo(newState, false);
  1392→                }
  1393→            }
  1394→        }
  1395→    }

  1536→    function getOracleConsensus(bytes32 pair) public view returns (
  1537→        uint256 finalPrice,
  1538→        uint8  validCount,
  1539→        uint8  method,
  1540→        bool   paused_,
  1541→        uint256[3] memory prices,
  1542→        bool[3] memory valid
  1543→    ) {
  1544→        IOracleAdapter[3] memory adapters;
  1545→        adapters[0] = chainlinkAdapter;
  1546→        adapters[1] = pythAdapter;
  1547→        adapters[2] = chronicleAdapter;
  1548→
  1549→        uint256[3] memory raw;
  1550→        uint256[3] memory ts;
  1551→        uint256[3] memory conf;
  1552→
  1553→        for (uint8 i = 0; i < 3; i++) {
  1554→            valid[i] = false;
  1555→            if (address(adapters[i]) == address(0)) continue;
  1556→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
  1557→                if (p == 0) continue;
  1558→                raw[i]  = p;
  1559→                ts[i]   = t;
  1560→                conf[i] = c;
  1561→                prices[i] = p;
  1562→                if (t > block.timestamp) continue;
  1563→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
  1564→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
  1565→                valid[i] = true;
  1566→            } catch { /* leave valid[i] = false */ }
  1567→        }
  1568→
  1569→        validCount = 0;
  1570→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
  1571→
  1572→        if (validCount < 2) {
  1573→            paused_ = true; method = 0;
  1574→            return (finalPrice, validCount, method, paused_, prices, valid);
  1575→        }
  1576→
  1577→        uint256[3] memory sorted;
  1578→        uint8 n = 0;
  1579→        for (uint8 i = 0; i < 3; i++) if (valid[i]) { sorted[n] = raw[i]; n++; }
  1580→        _insertionSort(sorted, n);
  1581→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
  1582→
  1583→        for (uint8 i = 0; i < 3; i++) {
  1584→            if (valid[i]) {
  1585→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
  1586→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
  1587→                    valid[i] = false;
  1588→                    validCount--;
  1589→                }
  1590→            }
  1591→        }
  1592→
  1593→        paused_ = validCount < 3;
  1594→        if (paused_) {
  1595→            method     = 0;
  1596→            finalPrice = 0;
  1597→            return (finalPrice, validCount, method, paused_, prices, valid);
  1598→        }
  1599→
  1600→        finalPrice = med;
  1601→        method     = 2;
  1602→    }

  1757→    uint256 public constant REBALANCE_TOLERANCE    = 0.05e18;

  1766→    function executeRebalance(RebalanceTrade[] memory trades) external nonReentrant onlyKeeper whenNotPaused {
  1767→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1768→        if (!(trades.length <= 7)) revert Err48();
  1769→
  1770→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1771→            dailyTradeUsd    = 0;
  1772→            dailyTradeResetAt = block.timestamp;
  1773→        }
  1774→
  1775→        uint256 nav = getReserveNavUsd();
  1776→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1777→        uint256 rr = getReserveRatio();
  1778→        bool solvencyOverride = (rr < rrStressFloor);
  1779→
  1780→        for (uint256 i = 0; i < trades.length; i++) {
  1781→            RebalanceTrade memory t = trades[i];
  1782→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1783→            if (t.direction == 0 || t.tradeUsd == 0) continue;
  1784→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1785→
  1786→            int256 prevDir = lastDirection[t.component];
  1787→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1788→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1789→            }
  1790→
  1791→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1792→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1793→            if (t.direction == 1) {
  1794→                if (!(currentW < execW)) revert Err14();
  1795→            } else {
  1796→                if (!(currentW > execW)) revert Err46();
  1797→            }
  1798→
  1799→            uint256 newHeldUsd = (t.direction == 1)
  1800→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1801→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1802→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1803→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1804→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1805→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1806→            if (!(absDiff <= tolerance)) revert Err49();
  1807→
  1808→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1809→            dailyTradeUsd += t.tradeUsd;
  1810→
  1811→            if (t.direction == 1) {
  1812→                reserveHeldUsd[t.component] += t.tradeUsd;
  1813→            } else {
  1814→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1815→            }
  1816→            lastDirection[t.component]   = t.direction;
  1817→            lastRebalanceAt[t.component] = block.timestamp;
  1818→        }
  1819→
  1820→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1821→                                  reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1822→                                  reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1823→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1824→        reserveState.liquidValue = reserveState.fiatValue;
  1825→
  1826→        emit RebalanceExecuted(trades, block.timestamp);
  1827→    }

  1884→    // §11.5.c Stability Pool — Emergency Recovery Mechanism (T3)
  1885→    // ------------------------------------------------------------
  1886→    //  Implements the deposit / withdraw / deficit-cover surface defined
  1887→    //  in the §11.5 state declaration above. The pool is the SOLE USDC
  1888→    //  custody point in V3; every other reserve is held by the banks'
  1889→    //  qualified custodians (ABC architecture, non-custodial).
  1890→    //
  1891→    //  Reward model (MasterChef-style, single-asset):
  1892→    //    accumulatedRewardPerShare(t) = accumulatedRewardPerShare(t-1)
  1893→    //      + (elapsed_sec * stabilityRewardRate * 1e18) / totalStabilityPool
  1894→    //    pending(depositor) = deposit * accumulatedRewardPerShare / 1e18
  1895→    //      - stabilityRewardDebt[depositor]
  1896→    //  On deposit/withdraw, the pending reward is settled into the
  1897→    //  depositor's `stabilityDeposits` balance (in USDC, 6-dec) BEFORE the
  1898→    //  principal change is applied — so rewards compound automatically.
  1899→    // ============================================================
  1900→
  1901→    /// @notice Constitutional Council sets the USDC (or 6-dec stable) asset
  1902→    ///         the stability pool accepts. address(0) disables deposits.
  1903→    function setStabilityAsset(address asset) external onlyConstitutionalCouncil {
  1904→        address old = address(stabilityAsset);
  1905→        stabilityAsset = IERC20(asset);
  1906→        emit StabilityAssetSet(old, asset, msg.sender);
  1907→    }
  1908→
  1909→    /// @notice Risk Council sets the per-second reward rate (1e18 scale
  1910→    ///         fraction of totalStabilityPool). 0 = no rewards. Capped at
  1911→    ///         1e15 (0.1%/sec ≈ 100% in 10 sec — a hard ceiling to prevent
  1912→    ///         a misconfigured rate from draining the pool instantly).
  1913→    function setStabilityRewardRate(uint256 rate) external onlyRiskCouncil {
  1914→        if (rate > 1e15) revert Err88_ConcentrationLimit();
  1915→        uint256 old = stabilityRewardRate;
  1916→        _updateStabilityRewards();
  1917→        stabilityRewardRate = rate;
  1918→        emit StabilityRewardRateSet(old, rate, msg.sender);
  1919→    }
  1920→
  1921→    /// @notice Deposit USDC into the stability pool. The depositor MUST have
  1922→    ///         approved the contract to spend `usdcAmount` of `stabilityAsset`.
  1923→    ///         Any pending rewards are settled into the deposit balance first
  1924→    ///         (auto-compounding). Reverts if the asset is not set.
  1925→    function depositToStabilityPool(uint256 usdcAmount) external nonReentrant whenNotPaused {
  1926→        if (usdcAmount == 0) revert Err96_StabilityZeroAmount();
  1927→        if (address(stabilityAsset) == address(0)) revert Err99_StabilityAssetNotSet();
  1928→
  1929→        _updateStabilityRewards();
  1930→
  1931→        // Settle any pending reward into the depositor's balance BEFORE the
  1932→        // principal change (MasterChef pattern — keeps the math correct).
  1933→        uint256 pending = _pendingStabilityReward(msg.sender);
  1934→        if (pending > 0) {
  1935→            // Fund the reward from the contract's own pool balance. The
  1936→            // reward budget is provisioned off-chain by the keeper (the
  1937→            // contract holds rewards + principal in the same USDC pool).
  1938→            stabilityDeposits[msg.sender] += pending;
  1939→            // Note: we DO NOT increase totalStabilityPool here because the
  1940→            // reward was already in the pool (it just changes attribution).
  1941→        }
  1942→
  1943→        // Pull the principal USDC from the depositor.
  1944→        bool ok = stabilityAsset.transferFrom(msg.sender, address(this), usdcAmount);
  1945→        if (!ok) revert Err23();
  1946→
  1947→        stabilityDeposits[msg.sender] += usdcAmount;
  1948→        totalStabilityPool += usdcAmount;
  1949→
  1950→        // Snapshot the reward debt at the new per-share rate so future
  1951→        // accruals only credit this depositor for post-deposit rewards.
  1952→        stabilityRewardDebt[msg.sender] =
  1953→            stabilityDeposits[msg.sender] * accumulatedRewardPerShare / 1e18;
  1954→
  1955→        emit StabilityDeposit(msg.sender, usdcAmount, totalStabilityPool, block.timestamp);
  1956→    }
  1957→
  1958→    /// @notice Withdraw USDC from the stability pool. Settles pending rewards
  1959→    ///         first (auto-compounding), then transfers the requested amount
  1960→    ///         back to the depositor. Reverts if the depositor has insufficient
  1961→    ///         balance (after the reward settlement).
  1962→    function withdrawFromStabilityPool(uint256 usdcAmount) external nonReentrant {
  1963→        if (usdcAmount == 0) revert Err96_StabilityZeroAmount();
  1964→
  1965→        _updateStabilityRewards();
  1966→
  1967→        // Settle pending rewards into the balance first.
  1968→        uint256 pending = _pendingStabilityReward(msg.sender);
  1969→        if (pending > 0) {
  1970→            stabilityDeposits[msg.sender] += pending;
  1971→        }
  1972→
  1973→        uint256 bal = stabilityDeposits[msg.sender];
  1974→        if (bal < usdcAmount) revert Err97_StabilityInsufficient();
  1975→
  1976→        stabilityDeposits[msg.sender] = bal - usdcAmount;
  1977→        totalStabilityPool -= usdcAmount;
  1978→
  1979→        // Refresh the reward debt AFTER the withdrawal so future accruals
  1980→        // only credit the depositor for the remaining balance.
  1981→        stabilityRewardDebt[msg.sender] =
  1982→            stabilityDeposits[msg.sender] * accumulatedRewardPerShare / 1e18;
  1983→
  1984→        bool ok = stabilityAsset.transfer(msg.sender, usdcAmount);
  1985→        if (!ok) revert Err23();
  1986→
  1987→        emit StabilityWithdrawal(msg.sender, usdcAmount, totalStabilityPool, block.timestamp);
  1988→    }
