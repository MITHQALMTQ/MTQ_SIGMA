    94→// PRNG + distribution utilities (deterministic, reproducible)
    95→// =========================================================================
    96→
    97→function mulberry32(seed: number): () => number {
    98→  let a = seed | 0;
    99→  return () => {
   100→    a |= 0;
   101→    a = (a + 0x6D2B79F5) | 0;
   102→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
   103→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
   104→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   105→  };
   106→}
   107→
   108→function gaussian(rng: () => number, mean = 0, sd = 1): number {
   109→  const u1 = Math.max(1e-9, rng());
   110→  const u2 = rng();
   111→  return mean + sd * Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
   112→}
   113→
   114→// Cauchy distribution — heavy-tailed (no finite variance).
   115→function cauchy(rng: () => number, location = 0, scale = 1): number {
   116→  return location + scale * Math.tan(Math.PI * (rng() - 0.5));
   117→}
   118→
   119→function clamp(v: number, lo: number, hi: number): number {
   120→  return Math.max(lo, Math.min(hi, v));
   121→}
   122→
   123→function clampSym(v: number, lo: number, hi: number): number {
   124→  return Math.max(lo, Math.min(hi, v));
   125→}

   159→interface TickMetrics {
   160→  rr: number;
   161→  lcr: number;
   162→  status: string;
   163→  price: number;
   164→  nav: number;
   165→  liability: number;
   166→  inBand: boolean;
   167→  goldPrice: number;
   168→  goldNet: number;

   170→
   171→function stepEngine(
   172→  s: ReserveState,
   173→  fx: FxSnapshot,
   174→  opts: { doRebalance?: boolean; doMacro?: boolean } = {},
   175→): TickMetrics {
   176→  const doRebalance = opts.doRebalance ?? true;
   177→  const doMacro = opts.doMacro ?? true;
   178→
   179→  // Simulate "1 day has passed" so the rebalancer's daily-turnover cap and
   180→  // 24h direction lock do not permanently block trades inside a tight MC
   181→  // loop (the engine uses wall-clock time, which doesn't advance here).
   182→  // This is the standard "1 tick = 1 day" interpretation for MC stress sims.
   183→  s.dailyTurnoverUsd = 0;
   184→  s.lastTurnoverResetAt = Date.now();
   185→  s.lastTradeAt = 0;
   186→
   187→  const gfb = computeGfbIndex(fx);
   188→  const price = computeMtqPrice(gfb);
   189→  const vals = reserveAssetValues(s, fx);
   190→  const nav = vals.nav;
   191→  const liability = computeLiability(s, price);
   192→  const rr = computeReserveRatio(nav, liability);
   193→  const lcr = computeLcr(s, vals, price);
   194→  const status = determineStatus(rr, lcr);
   195→
   196→  if (doMacro) advanceMacro(s, fx.VIX, fx.DXY, 24);
   197→  updateBufferState(s, rr);
   198→
   199→  if (doRebalance && Number.isFinite(rr)) {
   200→    const decision = evaluateRebalance(s, vals, rr);
   201→    if (decision.shouldRebalance) {
   202→      applyRebalanceTrade(s, decision, vals.goldPrice);
   203→    }
   204→  }
   205→
   206→  return {
   207→    rr, lcr, status, price, nav, liability,
   208→    inBand: priceInSafetyBand(price),
   209→    goldPrice: vals.goldPrice,
   210→    goldNet: vals.goldNet,
   211→  };
   212→}
   213→
   214→// Initialise reserve + mint sufficient circulating supply to bring RR close to 1.10.
   215→// Returns the post-mint circulating supply and starting snapshot.
   216→//
   217→// CRITICAL AUDIT NOTE: After `applyMint(9M USD)`, the gold weight of NAV drops to
   218→// ~2.8% (genesis had 26% gold but the $9M user mint adds only USD). The engine's
   219→// rebalancer would restore gold to ~26% over ~7 days, but in a tight MC loop the
   220→// daily-turnover reset uses wall-clock time (so the cap is never lifted within a
   221→// sub-second run). We therefore:
   222→//   (a) Call `redistributeToStrategicPrior(s, fx)` after the initial mint — this
   223→//       preserves total NAV + total supply but re-splits the holdings across all
   224→//       7 Strategic Prior components (so gold = 26% of NAV from the start).
   225→//   (b) Reset `s.dailyTurnoverUsd`, `s.lastTurnoverResetAt`, `s.lastTradeAt` at the
   226→//       start of each tick in `stepEngine` so each tick simulates "1 day passed"
   227→//       (the rebalancer gets fresh daily turnover + lifted direction lock).
   228→function setupInitialState(initialMintUsd = 9_000_000): { state: ReserveState; fx: FxSnapshot; circ: number; startRr: number } {
   229→  const fx = makeFx();
   230→  const s = initReserveState(BASE_FIXINGS.XAU_USD);
   231→  const snap0 = computeSnapshot(s, fx);
   232→  applyMint(s, fx, snap0.status, initialMintUsd);
   233→  // Redistribute holdings to Strategic Prior composition (preserves total NAV,
   234→  // restores gold weight to 26%). This faithfully matches the engine's design
   235→  // intent (Strategic Prior is the long-term anchor), and removes the
   236→  // "post-mint USD overweight" transient that would otherwise dominate the
   237→  // early MC ticks.
   238→  redistributeToStrategicPrior(s, fx);
   239→  const vals = reserveAssetValues(s, fx);
   240→  const price = computeMtqPrice(computeGfbIndex(fx));
   241→  const liability = computeLiability(s, price);
   242→  const rr = computeReserveRatio(vals.nav, liability);
   243→  return { state: s, fx, circ: circulatingSupply(s), startRr: rr };
   244→}
   245→
   246→// Re-split the reserve holdings across the 7 Strategic Prior components.
   247→// Preserves total NAV (modulo tiny haircut rounding) and totalSupply.
   248→function redistributeToStrategicPrior(s: ReserveState, fx: FxSnapshot): void {
   249→  const vals = reserveAssetValues(s, fx);
   250→  const navNet = vals.nav;
   251→  if (navNet <= 0) return;
   252→  // Target USD-notional (net of haircut) per component
   253→  const usdNetTarget = navNet * STRATEGIC_PRIOR.USD;
   254→  const eurNetTarget = navNet * STRATEGIC_PRIOR.EUR;
   255→  const jpyNetTarget = navNet * STRATEGIC_PRIOR.JPY;
   256→  const gbpNetTarget = navNet * STRATEGIC_PRIOR.GBP;
   257→  const cnyNetTarget = navNet * STRATEGIC_PRIOR.CNY;
   258→  const chfNetTarget = navNet * STRATEGIC_PRIOR.CHF;
   259→  const goldNetTarget = navNet * STRATEGIC_PRIOR.Gold;
   260→  // Convert net → gross via haircut factors, then to native units via FX
   261→  const usdGrossTotal = usdNetTarget / (1 - HAIRCUTS.USD);
   262→  const eurGross = eurNetTarget / (1 - HAIRCUTS.EUR) / fx.EUR_USD;
   263→  const jpyGross = jpyNetTarget / (1 - HAIRCUTS.JPY) / fx.JPY_USD;
   264→  const gbpGross = gbpNetTarget / (1 - HAIRCUTS.GBP) / fx.GBP_USD;
   265→  const cnyGross = cnyNetTarget / (1 - HAIRCUTS.CNY) / fx.CNY_USD;
   266→  const chfGross = chfNetTarget / (1 - HAIRCUTS.CHF) / fx.CHF_USD;
   267→  const goldGrossTotal = goldNetTarget / (1 - HAIRCUTS.XAU) / fx.XAU_USD;
   268→  // USD split across 3 issuers (1/3 each) per §5.6 concentration policy
   269→  s.usdc = usdGrossTotal / 3;
   270→  s.usdp = usdGrossTotal / 3;
   271→  s.usdt = usdGrossTotal / 3;
   272→  s.eurc = eurGross;
   273→  s.jpy = jpyGross;
   274→  s.gbp = gbpGross;
   275→  s.cny = cnyGross;
   276→  s.chf = chfGross;
   277→  // Gold split 50/50 PAXG/XAUT (per §5.6), then split 50/50 index/reserve (per §14.1)
   278→  s.paxg = goldGrossTotal / 2;
   279→  s.xaut = goldGrossTotal / 2;
   280→  s.indexPaxg = s.paxg / 2;
   281→  s.indexXaut = s.xaut / 2;
   282→  s.reservePaxg = s.paxg / 2;
   283→  s.reserveXaut = s.xaut / 2;
   284→  s.updatedAt = Date.now();
   285→}

   291→interface TrajectoryResult {
   292→  survived: boolean;
   293→  minRR: number;
   294→  worstStatus: string;
   295→  finalStatus: string;
   296→  recoveryTick: number;     // tick at which RR first returned ≥ RR_TARGET (after being below)
   297→  pegStablePct: number;
   298→  statusTransitions: number; // count of distinct status changes
   299→}
   300→
   301→function summarizeTrajectory(
   302→  minRR: number,
   303→  worstStatus: string,
   304→  finalStatus: string,
   305→  recoveryTick: number,
   306→  pegStableTicks: number,
   307→  totalTicks: number,
   308→  statusTransitions: number,
   309→): TrajectoryResult {
   310→  return {
   311→    survived: Number.isFinite(minRR) ? minRR >= RR_HARD : true,
   312→    minRR: Number.isFinite(minRR) ? minRR : 999,
   313→    worstStatus,
   314→    finalStatus,
   315→    recoveryTick,
   316→    pegStablePct: totalTicks > 0 ? pegStableTicks / totalTicks : 1,
   317→    statusTransitions,
   318→  };
   319→}

   321→// Aggregate TrajectoryResult[] → StressResult
   322→function aggregateRuns(
   323→  scenario: string,
   324→  description: string,
   325→  runs: TrajectoryResult[],
   326→  notes: string[] = [],
   327→): StressResult {
   328→  const n = runs.length;
   329→  const survivors = runs.filter((r) => r.survived);
   330→  const breaches = runs.filter((r) => !r.survived);
   331→  const minRRs = runs.map((r) => r.minRR);
   332→  const finiteMinRRs = minRRs.filter((x) => Number.isFinite(x));
   333→  const meanMinRR = finiteMinRRs.length > 0
   334→    ? finiteMinRRs.reduce((a, b) => a + b, 0) / finiteMinRRs.length
   335→    : 999;
   336→  const worstMinRR = finiteMinRRs.length > 0 ? Math.min(...finiteMinRRs) : 0;
   337→  const recoveryTicks = runs
   338→    .filter((r) => r.recoveryTick > 0)
   339→    .map((r) => r.recoveryTick);
   340→  const meanTimeToRecover = recoveryTicks.length > 0
   341→    ? recoveryTicks.reduce((a, b) => a + b, 0) / recoveryTicks.length
   342→    : -1;
   343→  const pegStabilityPct = runs.reduce((a, r) => a + r.pegStablePct, 0) / n;
   344→  // Most common final status
   345→  const statusCounts: Record<string, number> = {};
   346→  for (const r of runs) statusCounts[r.finalStatus] = (statusCounts[r.finalStatus] ?? 0) + 1;
   347→  const finalStatus = Object.entries(statusCounts).sort((a, b) => b[1] - a[1])[0][0];
   348→  // Worst status entered across all runs (max rank)
   349→  const worstStatus = runs.reduce(
   350→    (acc, r) => (STATUS_RANK[r.worstStatus] > STATUS_RANK[acc] ? r.worstStatus : acc),
   351→    "NORMAL",
   352→  );
   353→  return {
   354→    scenario,
   355→    description,
   356→    runs: n,
   357→    survivalRate: survivors.length / n,
   358→    meanMinRR,
   359→    worstMinRR,
   360→    meanTimeToRecover,
   361→    pegStabilityPct,
   362→    finalStatus,
   363→    worstStatus,
   364→    breachCount: breaches.length,
   365→    notes,
   366→  };
   367→}

   369→// =========================================================================
   370→// Synthetic history builder (for the block-bootstrap MC family)
   371→// =========================================================================
   372→
   373→interface HistoryRow {
   374→  EUR: number; GBP: number; JPY: number; CNY: number; CHF: number; XAU: number;
   375→  VIX: number; DXY: number;
   376→}
   377→
   378→function buildSyntheticHistory(seed: number, days: number): HistoryRow[] {
   379→  const rng = mulberry32(seed);
   380→  const rows: HistoryRow[] = [];
   381→  let EUR = 1.05, GBP = 1.25, JPY = 0.0067, CNY = 0.14, CHF = 0.88, XAU = 2500, VIX = 18.5, DXY = 104.2;
   382→  for (let i = 0; i < days; i++) {
   383→    EUR *= (1 + gaussian(rng, 0, 0.005));
   384→    GBP *= (1 + gaussian(rng, 0, 0.005));
   385→    JPY *= (1 + gaussian(rng, 0, 0.006));
   386→    CNY *= (1 + gaussian(rng, 0, 0.004));
   387→    CHF *= (1 + gaussian(rng, 0, 0.005));
   388→    XAU *= (1 + gaussian(rng, 0, 0.012));
   389→    VIX = clamp(VIX + gaussian(rng, 0, 1.5), 10, 80);
   390→    DXY = clamp(DXY + gaussian(rng, 0, 0.5), 80, 120);
   391→    rows.push({ EUR, GBP, JPY, CNY, CHF, XAU, VIX, DXY });
   392→  }
   393→  return rows;
   394→}
   395→
   396→// Convert a sequence of HistoryRow absolute prices into a sequence of returns.
   397→function historyToReturns(rows: HistoryRow[]): HistoryRow[] {
   398→  const rets: HistoryRow[] = [];
   399→  for (let i = 1; i < rows.length; i++) {
   400→    const a = rows[i - 1], b = rows[i];
   401→    rets.push({
   402→      EUR: b.EUR / a.EUR - 1,
   403→      GBP: b.GBP / a.GBP - 1,
   404→      JPY: b.JPY / a.JPY - 1,
   405→      CNY: b.CNY / a.CNY - 1,
   406→      CHF: b.CHF / a.CHF - 1,
   407→      XAU: b.XAU / a.XAU - 1,
   408→      VIX: b.VIX - a.VIX,
   409→      DXY: b.DXY - a.DXY,
   410→    });
   411→  }
   412→  return rets;
   413→}

   415→// =========================================================================
   416→// MC core runner — drives the engine over `ticks` ticks with a generator
   417→// that produces FX snapshots. Each generator returns {fx, regime?}.
   418→// =========================================================================
   419→
   420→type FxGen = (rng: () => number, tick: number, prev: { EUR: number; GBP: number; JPY: number; CNY: number; CHF: number; XAU: number; VIX: number; DXY: number }) => HistoryRow;
   421→
   422→function runTrajectory(
   423→  seed: number,
   424→  ticks: number,
   425→  gen: FxGen,
   426→  opts: { initialMintUsd?: number; doRebalance?: boolean; applyShockAtTick?: (s: ReserveState, fx: FxSnapshot, tick: number) => void } = {},
   427→): TrajectoryResult {
   428→  const { state: s, fx: _fx } = setupInitialState(opts.initialMintUsd ?? 9_000_000);
   429→  void _fx;
   430→
   431→  const rng = mulberry32(seed);
   432→  // Start FX at base fixings
   433→  let cur: HistoryRow = {
   434→    EUR: BASE_FIXINGS.EUR_USD, GBP: BASE_FIXINGS.GBP_USD, JPY: BASE_FIXINGS.JPY_USD,
   435→    CNY: BASE_FIXINGS.CNY_USD, CHF: BASE_FIXINGS.CHF_USD, XAU: BASE_FIXINGS.XAU_USD,
   436→    VIX: 18.5, DXY: 104.2,
   437→  };
   438→
   439→  let minRR = Infinity;
   440→  let worstStatus = "NORMAL";
   441→  let finalStatus = "NORMAL";
   442→  let recoveryTick = -1;
   443→  let hasBeenStressed = false;
   444→  let pegStableTicks = 0;
   445→  let prevStatus = "NORMAL";
   446→  let statusTransitions = 0;
   447→
   448→  for (let t = 0; t < ticks; t++) {
   449→    const nxt = gen(rng, t, cur);
   450→    cur = nxt;
   451→    const fx = makeFx({ EUR: nxt.EUR, GBP: nxt.GBP, JPY: nxt.JPY, CNY: nxt.CNY, CHF: nxt.CHF, XAU: nxt.XAU, VIX: nxt.VIX, DXY: nxt.DXY });
   452→
   453→    if (opts.applyShockAtTick) opts.applyShockAtTick(s, fx, t);
   454→
   455→    const m = stepEngine(s, fx, { doRebalance: opts.doRebalance ?? true });
   456→    finalStatus = m.status;
   457→    if (STATUS_RANK[m.status] > STATUS_RANK[worstStatus]) worstStatus = m.status;
   458→    if (m.status !== prevStatus) {
   459→      statusTransitions++;
   460→      prevStatus = m.status;
   461→    }
   462→
   463→    if (Number.isFinite(m.rr)) {
   464→      if (m.rr < minRR) minRR = m.rr;
   465→      if (m.rr < RR_TARGET) hasBeenStressed = true;
   466→      if (recoveryTick === -1 && hasBeenStressed && m.rr >= RR_TARGET) recoveryTick = t;
   467→    }
   468→    if (m.inBand) pegStableTicks++;
   469→  }
   470→
   471→  return summarizeTrajectory(minRR, worstStatus, finalStatus, recoveryTick, pegStableTicks, ticks, statusTransitions);
   472→}

   474→// =========================================================================
   475→// Scenario 1 — §23.5 Historical block bootstrap (2000 runs, 90 days)
   476→// =========================================================================
   477→
   478→function scenario1HistoricalBootstrap(runs: number, ticks: number): StressResult {
   479→  // Build 252-day synthetic history once, derive returns, then block-bootstrap.
   480→  const hist = buildSyntheticHistory(4242, 252);
   481→  const rets = historyToReturns(hist);
   482→  const blockSize = 5; // 5-day blocks preserve short-term autocorrelation
   483→
   484→  const fxGen: FxGen = (rng, _t, prev) => {
   485→    // Sample a random 5-day block from history. Use the t-th element of that block
   486→    // (so successive ticks in the same block are correlated); if the block ends,
   487→    // draw a new one.
   488→    const idx = Math.floor(rng() * Math.max(1, rets.length - blockSize));
   489→    const offset = _t % blockSize;
   490→    const r = rets[idx + offset];
   491→    return {
   492→      EUR: Math.max(0.4, prev.EUR * (1 + r.EUR)),

   913→  // (updatePegHealth drifts it back toward 1.0; we re-assert after each call).

   937→      condition: stageMeta ? stageMeta.condition : "—",

   955→  const ladder2 = stagesReached.some((s) => s >= 2);
   956→  const ladder3 = stagesReached.some((s) => s >= 3);
   957→  const ladder4 = stagesReached.some((s) => s >= 4);

   964→    const fxNow = makeFx({ EUR: BASE_FIXINGS.EUR_USD * 0.90 });

   966→    if (price >= PRICE_SAFETY_LOWER && price <= PRICE_SAFETY_UPPER) pegStableTicks++;
   967→  }
   968→

   971→    `Eject ladder progression observed: Stage 1=${ladder1}, Stage 2=${ladder2}, Stage 3=${ladder3}, Stage 4=${ladder4}.`,

   974→    `Blueprint §21.5 ladder (per EJECT_STAGES): S1 at >12h (10%), S2 at >24h (25%), S3 at >48h (50%), S4 at >96h (100%).`,
   975→    `Note: task spec listed S1@24h / S2@48h / S3@72h / S4@96h; engine follows the blueprint (12/24/48/96).`,
   976→    `Min RR observed: ${minRR.toFixed(3)} (EUR is 20% of NAV; -10% EUR → NAV ≈ -2% → RR slightly down).`,

   979→    "Target: Stage 1 (10%) at 24h, Stage 2 (25%) at 48h, Stage 3 (50%) at 72h, Stage 4 (100%) at 96h. Verify the eject ladder kicks in correctly.",
   980→  ];

   991→    finalStatus: worstStatus,

  1144→// =========================================================================
  1145→// Scenario 11 — §23.6 Parameter perturbation
  1146→// Vary ALPHA, BETA, THETA_MAX, SMOOTHING_LAMBDA, LAMBDA_1..4 (±50%).
  1147→// For each perturbation, run 200 MC runs. Target: survival ≥ 90%.
  1148→// =========================================================================
  1149→
  1150→interface PerturbParams {
  1151→  alpha: number;
  1152→  beta: number;
  1153→  thetaMax: number;
  1154→  smoothing: number;
  1155→  lambda1: number;
  1156→  lambda2: number;
  1157→  lambda3: number;
  1158→  lambda4: number;
  1159→}
  1160→
  1161→// Self-contained engine sim with perturbed parameters (mirrors engine.ts formulas).
  1162→// We use this because the blueprint/engine constants are immutable `const` exports
  1163→// — we cannot monkey-patch them in a TypeScript-safe way. The sim below faithfully
  1164→// reproduces the engine's:
  1165→//   - GFB Index (7-component chain-linked)
  1166→//   - Reserve NAV (with haircuts)
  1167→//   - Reserve Ratio + status determination
  1168→//   - §6 raw target gold weight: θ = α·zVix + β·zDxy, clamped to ±θ_max
  1169→//   - §8.4 EMA smoothing: λ · target + (1-λ) · prev
  1170→//   - §7 rebalance cost-benefit: benefit = |dev|·NAV·λ1 + (RR<TARGET)?(1/RR)·NAV·λ3 ; cost = tradeUsd·slippage + gas
  1171→function runPerturbedSim(seed: number, ticks: number, params: PerturbParams): TrajectoryResult {
  1172→  const rng = mulberry32(seed);
  1173→  // Initial state mirrors setupInitialState(): $1.1M genesis deposit + $9M user mint,
  1174→  // with the holdings redistributed to Strategic Prior composition (gold = 26% of NAV).
  1175→  // This matches the engine's design intent and removes the post-mint USD overweight
  1176→  // transient. The total NAV is ($1.1M + $9M) = $10.1M (modulo haircuts).
  1177→  const TOTAL_DEPOSIT = 1_100_000 + 9_000_000;
  1178→  let usdUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.USD;
  1179→  let eurUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.EUR;
  1180→  let jpyUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.JPY;
  1181→  let gbpUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.GBP;
  1182→  let cnyUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.CNY;
  1183→  let chfUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.CHF;
  1184→  let goldUsd = TOTAL_DEPOSIT * STRATEGIC_PRIOR.Gold;
  1185→  let circulating = 9_000_000; // post-mint circulating (excludes 1M genesis)
  1186→  let smoothedTheta = 0;
  1187→  let vixHist: number[] = Array(90).fill(18.5);
  1188→  let dxyHist: number[] = Array(90).fill(104.2);
  1189→  let lastTradeDir: 0 | 1 | -1 = 0;
  1190→  let lastTradeAt = 0;
  1191→
  1192→  // FX state — start at base fixings
  1193→  let EUR = BASE_FIXINGS.EUR_USD, GBP = BASE_FIXINGS.GBP_USD, JPY = BASE_FIXINGS.JPY_USD;
  1194→  let CNY = BASE_FIXINGS.CNY_USD, CHF = BASE_FIXINGS.CHF_USD, XAU = BASE_FIXINGS.XAU_USD;
  1195→  let VIX = 18.5, DXY = 104.2;
  1196→
  1197→  let minRR = Infinity;
  1198→  let worstStatus = "NORMAL";
  1199→  let finalStatus = "NORMAL";
  1200→  let recoveryTick = -1;
  1201→  let hasBeenStressed = false;
  1202→  let pegStableTicks = 0;
  1203→  let prevStatus = "NORMAL";
  1204→  let statusTransitions = 0;
  1205→
  1206→  for (let t = 0; t < ticks; t++) {
  1207→    // Gaussian returns (modest stress: 1.5x normal vol to test perturbations)
  1208→    EUR = Math.max(0.4, EUR * (1 + gaussian(rng, 0, 0.0075)));
  1209→    GBP = Math.max(0.5, GBP * (1 + gaussian(rng, 0, 0.0075)));
  1210→    JPY = Math.max(0.003, JPY * (1 + gaussian(rng, 0, 0.009)));
  1211→    CNY = Math.max(0.06, CNY * (1 + gaussian(rng, 0, 0.006)));
  1212→    CHF = Math.max(0.4, CHF * (1 + gaussian(rng, 0, 0.0075)));
  1213→    XAU = Math.max(500, XAU * (1 + gaussian(rng, 0, 0.018)));
  1214→    VIX = clamp(VIX + gaussian(rng, 0, 2.0), 10, 80);
  1215→    DXY = clamp(DXY + gaussian(rng, 0, 0.7), 80, 120);
  1216→
  1217→    vixHist.push(VIX); if (vixHist.length > 90) vixHist.shift();
  1218→    dxyHist.push(DXY); if (dxyHist.length > 90) dxyHist.shift();
  1219→
  1220→    // Z-scores
  1221→    const vMean = avg(vixHist), vSd = stdev(vixHist);
  1222→    const dMean = avg(dxyHist), dSd = stdev(dxyHist);
  1223→    const zVix = vSd > 1e-9 ? (VIX - vMean) / vSd : 0;
  1224→    const zDxy = dSd > 1e-9 ? (DXY - dMean) / dSd : 0;
  1225→
  1226→    // §6 raw target gold weight with PERTURBED α, β, θ_max
  1227→    const theta = clampSym(params.alpha * zVix + params.beta * zDxy, -params.thetaMax, params.thetaMax);
  1228→    const target = clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
  1229→    // §8.4 EMA-smoothed with PERTURBED λ
  1230→    const smoothed = params.smoothing * target + (1 - params.smoothing) * (BASE_GOLD_WEIGHT + smoothedTheta);
  1231→    smoothedTheta = smoothed - BASE_GOLD_WEIGHT;
  1232→
  1233→    // GFB price (7-component chain-linked)
  1234→    const gfbNum =
  1235→      STRATEGIC_PRIOR.USD * 1.0 +
  1236→      STRATEGIC_PRIOR.EUR * EUR +
  1237→      STRATEGIC_PRIOR.JPY * JPY +
  1238→      STRATEGIC_PRIOR.GBP * GBP +
  1239→      STRATEGIC_PRIOR.CNY * CNY +
  1240→      STRATEGIC_PRIOR.CHF * CHF +
  1241→      STRATEGIC_PRIOR.Gold * XAU;
  1242→    const price = gfbNum / GFB_BASE_DENOMINATOR;
  1243→    const inBand = price >= PRICE_SAFETY_LOWER && price <= PRICE_SAFETY_UPPER;
  1244→    if (inBand) pegStableTicks++;
  1245→
  1246→    // NAV (with haircuts)
  1247→    const usdNet = usdUsd * (1 - HAIRCUTS.USD);
  1248→    const eurNet = eurUsd * (1 - HAIRCUTS.EUR);
  1249→    const jpyNet = jpyUsd * (1 - HAIRCUTS.JPY);
  1250→    const gbpNet = gbpUsd * (1 - HAIRCUTS.GBP);
  1251→    const cnyNet = cnyUsd * (1 - HAIRCUTS.CNY);
  1252→    const chfNet = chfUsd * (1 - HAIRCUTS.CHF);
  1253→    const goldNet = goldUsd * (1 - HAIRCUTS.XAU);
  1254→    const nav = usdNet + eurNet + jpyNet + gbpNet + cnyNet + chfNet + goldNet;
  1255→    const fiatNet = nav - goldNet;
  1256→
  1257→    // RR / LCR / status
  1258→    const liability = circulating * price;
  1259→    const rr = liability > 0 ? nav / liability : Infinity;
  1260→    const stressDemand = circulating * price * 0.25;
  1261→    const lcr = stressDemand > 0 ? fiatNet / stressDemand : Infinity;
  1262→    let status = "NORMAL";
  1263→    if (rr >= RR_TARGET && lcr >= 1) status = "NORMAL";
  1264→    else if (rr >= RR_STRESS && lcr >= 0.9) status = "CAUTION";
  1265→    else if (rr >= RR_HARD) status = "DEFENSIVE";
  1266→    else if (rr < RR_HARD) status = "EMERGENCY";
  1267→    finalStatus = status;
  1268→    if (STATUS_RANK[status] > STATUS_RANK[worstStatus]) worstStatus = status;
  1269→    if (status !== prevStatus) { statusTransitions++; prevStatus = status; }
  1270→
  1271→    if (Number.isFinite(rr)) {
  1272→      if (rr < minRR) minRR = rr;
  1273→      if (rr < RR_TARGET) hasBeenStressed = true;
  1274→      if (recoveryTick === -1 && hasBeenStressed && rr >= RR_TARGET) recoveryTick = t;
  1275→    }
  1276→
  1277→    // Rebalance with PERTURBED λ1, λ3
  1278→    if (Number.isFinite(rr) && nav > 0) {
  1279→      const observed = goldNet / nav;
  1280→      const deviation = observed - target;
  1281→      if (Math.abs(deviation) >= 0.005) {
  1282→        // Direction lock (24h whipsaw guard) — perturbation only changes λ's, not the lock
  1283→        const inStress = rr < RR_STRESS;
  1284→        const dir: 0 | 1 | -1 = deviation > 0 ? -1 : 1;
  1285→        const canTrade = inStress || lastTradeDir === 0 || dir === lastTradeDir || (t - lastTradeAt) >= 1;
  1286→        if (canTrade) {
  1287→          const surplusUsd = Math.abs(deviation) * nav;
  1288→          const tradeUsd = Math.min(surplusUsd, 400_000, MAX_DAILY_TURNOVER_PERTURB * nav);
  1289→          const cost = tradeUsd * 0.001 + 50;
  1290→          // §7 cost-benefit with PERTURBED λ1 (deviation) and λ3 (RR stress)
  1291→          const benefit =
  1292→            Math.abs(deviation) * nav * params.lambda1 +
  1293→            (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * params.lambda3 : 0);
  1294→          if (benefit > cost) {
  1295→            // Apply trade: gold ↔ USD
  1296→            if (dir === -1) {
  1297→              goldUsd -= tradeUsd;
  1298→              usdUsd += tradeUsd;
  1299→            } else {
  1300→              usdUsd -= tradeUsd;
  1301→              goldUsd += tradeUsd;
  1302→            }
  1303→            lastTradeDir = dir;
  1304→            lastTradeAt = t;
  1305→          }
  1306→        }
  1307→      }
  1308→    }
  1309→  }
  1310→

  1316→function avg(a: number[]): number {
  1317→  return a.reduce((x, y) => x + y, 0) / a.length;
  1318→}
  1319→function stdev(a: number[]): number {
  1320→  if (a.length === 0) return 0;
  1321→  const m = avg(a);
  1322→  return Math.sqrt(a.reduce((s, x) => s + (x - m) * (x - m), 0) / a.length);
  1323→}

  1413→  // Helper to safely run a scenario and capture errors
  1414→  async function safeRun(name: string, fn: () => StressResult): Promise<StressResult> {
  1415→    try {
  1416→      return fn();
  1417→    } catch (err) {
  1418→      const msg = err instanceof Error ? err.message : String(err);
  1419→      return {
  1420→        scenario: name,
  1421→        description: "ERROR — scenario crashed",
  1422→        runs: 0,
  1423→        survivalRate: 0,
  1424→        meanMinRR: 0,

  1428→        finalStatus: "ERROR",
  1429→        worstStatus: "ERROR",
  1430→        breachCount: 0,
  1431→        notes: [`ERROR: ${msg}`],
  1432→      };
  1433→    }
  1434→  }
  1435→

  1437→  // primary MC scenarios (1-4) and document any reduction.
  1438→  const MC_RUNS = 2000;
  1439→  const MC_TICKS = 90;
  1440→

  1442→  results.push(await safeRun("S1", () => scenario1HistoricalBootstrap(MC_RUNS, MC_TICKS)));

  1452→  results.push(await safeRun("S7", () => scenario7OracleDisagreement()));
