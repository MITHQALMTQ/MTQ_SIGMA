    44→  subtitle,
    45→  headers,
    46→  rows,

    48→  title: string;

    63→      </div>

    67→            <thead className="bg-white/[0.02] text-muted-foreground/70 sticky top-0">
    68→              <tr>
    69→                {headers.map((h, i) => (
    70→                  <th
    71→                    key={i}
    72→                    className={`text-left px-3 py-2 font-medium ${i === headers.length - 1 ? "text-right" : ""}`}

    77→              </tr>

    80→              {rows.map((row, ri) => (

    82→                  key={ri}

    84→                >

    86→                    <td
    87→                      key={ci}
    88→                      className={`px-3 py-2.5 ${ci === row.length - 1 ? "text-right" : ""}`}

   105→  if (severity === "fixed") {
   106→    return (
   107→      <Pill tone="emerald">

   109→        fixed
   110→      </Pill>
   111→    );
   112→  }
   113→  if (severity === "outstanding") {
   114→    return (
   115→      <Pill tone="amber">
   116→        <GlowDot color="amber" size="h-1.5 w-1.5" />
   117→        outstanding
   118→      </Pill>
   119→    );
   120→  }
   121→  return (
   122→    <Pill tone="muted">
   123→      <GlowDot color="gold" size="h-1.5 w-1.5" />

   125→    </Pill>
   126→  );
   127→}
