   232→function seedHistory(mean: number, sd: number, n: number): number[] {
   233→  const out: number[] = [];
   234→  let last = mean;
   235→  for (let i = 0; i < n; i++) {
   236→    // simple mean-reverting walk
   237→    const shock = (Math.random() - 0.5) * 2 * sd * 0.3;
   238→    last = last + (mean - last) * 0.1 + shock;
   239→    out.push(Math.max(0.01, last));
   240→  }
   241→  return out;
   242→}

   244→// --- §2 GFB Index (v1.0 — 7-component chain-linked) -------------------------
   245→// GFB_t = Σ_i W^Prior_i × P_{i,t}  (then normalised by GFB_BASE_DENOMINATOR so
   246→// GFB = 1.00 exactly at the base date). The 7 components are the Strategic
   247→// Prior: USD, EUR, JPY, GBP, CNY, CHF, Gold. Gold is now a first-class index
   248→// component (P_Gold = XAU_USD).
   249→export function computeGfbIndex(fx: Pick<FxRates, "EUR_USD" | "GBP_USD" | "JPY_USD" | "CNY_USD" | "CHF_USD" | "XAU_USD">): number {
   250→  // Raw USD value of the basket at time t. USD is the unit of account (price 1.0);
   251→  // every other component contributes its USD-equivalent via its FX rate.
   252→  const numerator =
   253→    STRATEGIC_PRIOR.USD  * 1.0 +
   254→    STRATEGIC_PRIOR.EUR  * fx.EUR_USD +
   255→    STRATEGIC_PRIOR.JPY  * fx.JPY_USD +
   256→    STRATEGIC_PRIOR.GBP  * fx.GBP_USD +
   257→    STRATEGIC_PRIOR.CNY  * fx.CNY_USD +
   258→    STRATEGIC_PRIOR.CHF  * fx.CHF_USD +
   259→    STRATEGIC_PRIOR.Gold * fx.XAU_USD;
   260→  // Normalised: GFB_t = numerator / GFB_BASE_DENOMINATOR  (=1.0 at base date).
   261→  // GFB_BASE_DENOMINATOR is the v1.0 chain-linked denominator (includes Gold
   262→  // and CHF at base fixings — see blueprint.ts).
   263→  return numerator / GFB_BASE_DENOMINATOR;
   264→}

   276→// --- §4 Reserve NAV / RR / LCR ----------------------------------------------
   277→// USD value of each holding (gross), then apply haircuts for NAV.
   278→export function reserveAssetValues(s: ReserveState, fx: FxRates) {
   279→  const usdGross = (s.usdc + s.usdp + s.usdt) * 1.0; // USDC + USDP + USDT combined
   280→  const eurGross = s.eurc * fx.EUR_USD;
   281→  const gbpGross = s.gbp * fx.GBP_USD;
   282→  const jpyGross = s.jpy * fx.JPY_USD;
   283→  const cnyGross = s.cny * fx.CNY_USD;
   284→  const chfGross = s.chf * fx.CHF_USD; // NEW v1.0 — CHF is a first-class reserve component
   285→  const fiatGross = usdGross + eurGross + gbpGross + jpyGross + cnyGross + chfGross;
   286→
   287→  // §4.1 Note on Gold: use the more conservative of reference vs executable.
   288→  const goldRef = fx.XAU_USD;
   289→  const goldExec = fx.XAU_USD * 0.995;
   290→  const goldPrice = Math.min(goldRef, goldExec);
   291→  const goldGross = (s.paxg + s.xaut) * goldPrice; // PAXG + XAUT combined
   292→
   293→  // Net values (after haircuts)
   294→  const usdNet = usdGross * (1 - HAIRCUTS.USD);
   295→  const eurNet = eurGross * (1 - HAIRCUTS.EUR);
   296→  const gbpNet = gbpGross * (1 - HAIRCUTS.GBP);
   297→  const jpyNet = jpyGross * (1 - HAIRCUTS.JPY);
   298→  const cnyNet = cnyGross * (1 - HAIRCUTS.CNY);
   299→  const chfNet = chfGross * (1 - HAIRCUTS.CHF); // NEW v1.0
   300→  const goldNet = goldGross * (1 - HAIRCUTS.XAU);
   301→
   302→  const nav = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet + goldNet;
   303→  const fiatNet = usdNet + eurNet + gbpNet + jpyNet + cnyNet + chfNet;
   304→
   305→  // Per-issuer breakdown (for concentration visualization)
   306→  const usdcUsd = s.usdc * 1.0 * (1 - HAIRCUTS.USD);
   307→  const usdpUsd = s.usdp * 1.0 * (1 - HAIRCUTS.USD);
   308→  const usdtUsd = s.usdt * 1.0 * (1 - HAIRCUTS.USD);
   309→  const paxgUsd = s.paxg * goldPrice * (1 - HAIRCUTS.XAU);
   310→  const xautUsd = s.xaut * goldPrice * (1 - HAIRCUTS.XAU);
   311→
   312→  return {
   313→    usdGross, eurGross, gbpGross, jpyGross, cnyGross, chfGross, fiatGross,
   314→    goldGross, goldPrice,
   315→    usdNet, eurNet, gbpNet, jpyNet, cnyNet, chfNet, goldNet,
   316→    nav, fiatNet,
   317→    usdcUsd, usdpUsd, usdtUsd, paxgUsd, xautUsd,
   318→  };
   319→}

   351→// --- §6 Adaptive Macro Engine -----------------------------------------------
   352→function rollingStats(hist: number[]): { mean: number; sd: number } {
   353→  const filled = hist.filter((x) => x > 0);
   354→  if (filled.length === 0) return { mean: 0, sd: 0 };
   355→  const mean = filled.reduce((a, b) => a + b, 0) / filled.length;
   356→  const variance = filled.reduce((a, b) => a + (b - mean) * (b - mean), 0) / filled.length;
   357→  return { mean, sd: Math.sqrt(variance) };
   358→}
   359→
   360→export function computeZScores(s: ReserveState, vix: number, dxy: number) {
   361→  const v = rollingStats(s.vixHistory);
   362→  const d = rollingStats(s.dxyHistory);
   363→  const eps = 1e-9;
   364→  const zVix = v.sd > eps ? (vix - v.mean) / (v.sd + eps) : 0;
   365→  const zDxy = d.sd > eps ? (dxy - d.mean) / (d.sd + eps) : 0;
   366→  return { zVix, zDxy, vixMean: v.mean, vixSd: v.sd, dxyMean: d.mean, dxySd: d.sd };
   367→}

   369→export function computeRawTargetTheta(zVix: number, zDxy: number): number {
   370→  const theta = ALPHA * zVix + BETA * zDxy;
   371→  return clampSym(theta, -THETA_MAX, THETA_MAX);
   372→}
   373→
   374→export function computeRawTargetGoldWeight(zVix: number, zDxy: number): number {
   375→  const theta = computeRawTargetTheta(zVix, zDxy);
   376→  return clamp(BASE_GOLD_WEIGHT + theta, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   377→}

   379→export function smoothGoldWeight(prev: number, target: number): number {
   380→  if (prev <= 0) return target;
   381→  return SMOOTHING_LAMBDA * target + (1 - SMOOTHING_LAMBDA) * prev;
   382→}

   391→export function currentBufferGoldRatio(s: ReserveState): number {
   392→  if (s.bufferState === "EMERGENCY") return BUFFER_GOLD_EMERGENCY;
   393→  if (s.rampStartAt === 0) {
   394→    return s.bufferState === "BASE" ? BUFFER_GOLD_BASE : BUFFER_GOLD_STRESS;
   395→  }
   396→  const elapsedH = (Date.now() - s.rampStartAt) / 3_600_000;
   397→  if (elapsedH >= RAMP_DURATION_HOURS) return s.rampToRatio;
   398→  const progress = elapsedH / RAMP_DURATION_HOURS;
   399→  const from = s.rampFromRatio;
   400→  const to = s.rampToRatio;
   401→  return from + (to - from) * progress;
   402→}
   403→
   404→export function bufferBaseFromRatio(rr: number): "BASE" | "STRESS" | "EMERGENCY" {
   405→  if (rr >= RR_TARGET) return "BASE";
   406→  if (rr >= RR_STRESS) return "STRESS";
   407→  return "EMERGENCY";
   408→}

   410→// Unified target gold weight (tokenomics reconciliation):
   411→// W_target = clamp( buffer_base_gold(RR) + θ_smoothed , 22%, 30% )
   412→export function computeTargetGoldWeight(s: ReserveState, rr: number): number {
   413→  const bufferBase = currentBufferGoldRatio(s); // ∈ {0.625, ~0.85, 1.0} with ramp
   414→  const baseFromBuffer = CORE_GOLD_WEIGHT + BUFFER_SIZE * bufferBase; // 0.20 + 0.10*b
   415→  const target = baseFromBuffer + s.smoothedGoldWeight - BASE_GOLD_WEIGHT; // apply θ = smoothed - base
   416→  return clamp(target, GOLD_WEIGHT_LOWER, GOLD_WEIGHT_UPPER);
   417→}

   431→  if (Date.now() - s.lastTurnoverResetAt >= 86_400_000) {

   437→export function evaluateRebalance(
   438→  s: ReserveState,
   439→  vals: { nav: number; goldNet: number; fiatNet: number },
   440→  rr: number,
   441→): RebalanceDecision {
   442→  const nav = vals.nav;
   443→  if (nav <= 0) {
   444→    return { shouldRebalance: false, direction: 0, observedGoldWeight: 0, targetGoldWeight: 0, deviation: 0, tradeUsd: 0, reason: "NAV ≤ 0" };
   445→  }
   446→  const observed = vals.goldNet / nav;
   447→  const target = computeTargetGoldWeight(s, rr);
   448→  const deviation = observed - target;
   449→
   450→  if (Math.abs(deviation) < 0.005) {
   451→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Within tolerance (<0.5%)" };
   452→  }
   453→
   454→  // Direction lock (overridden in stress)
   455→  const inStress = rr < RR_STRESS;
   456→  if (!inStress && Date.now() - s.lastTradeAt < DIRECTION_LOCK_HOURS * 3_600_000) {
   457→    if ((deviation > 0 && s.lastTradeDir === -1) || (deviation < 0 && s.lastTradeDir === 1)) {
   458→      return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: "Direction lock (24h whipsaw guard)" };
   459→    }
   460→  }
   461→
   462→  // Trade sizing
   463→  const surplusUsd = Math.abs(deviation) * nav;
   464→  const poolDepth24h = 4_000_000; // assumed PAXG/USDC 24h depth for the pilot
   465→  const maxByPool = MAX_POOL_FRACTION * poolDepth24h;
   466→  let tradeUsd = Math.min(surplusUsd, maxByPool);
   467→  maybeResetDailyTurnover(s);
   468→  const remainingDaily = MAX_DAILY_TURNOVER * nav - s.dailyTurnoverUsd;
   469→  if (tradeUsd > remainingDaily) tradeUsd = Math.max(0, remainingDaily);
   470→
   471→  // Cost-benefit (simplified): estimate cost ~ 0.1% slippage + $50 gas equiv
   472→  const estCost = tradeUsd * 0.001 + 50;
   473→  const benefit = Math.abs(deviation) * nav * LAMBDA_1 + (rr < RR_TARGET ? (1 / Math.max(rr, 0.5)) * nav * LAMBDA_3 : 0);
   474→  const direction = deviation > 0 ? -1 : 1; // excess gold → sell; deficit → buy
   475→
   476→  if (benefit <= estCost) {
   477→    return { shouldRebalance: false, direction: 0, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd: 0, reason: `Cost (${estCost.toFixed(0)}) ≥ benefit (${benefit.toFixed(0)})` };
   478→  }
   479→  return { shouldRebalance: true, direction, observedGoldWeight: observed, targetGoldWeight: target, deviation, tradeUsd, reason: "Benefit > cost; executing" };
   480→}
   481→
   482→// Apply a rebalance trade in USD terms (mutates state).
   483→export function applyRebalanceTrade(s: ReserveState, decision: RebalanceDecision, goldPrice: number): void {
   484→  if (!decision.shouldRebalance) return;
   485→  const usd = decision.tradeUsd;
   486→  if (decision.direction === -1) {
   487→    // sell gold → buy USDC (simplified: convert to USDC)
   488→    const goldUnits = usd / goldPrice;
   489→    s.paxg = Math.max(0, s.paxg - goldUnits);
   490→    s.usdc += usd;
   491→  } else {
   492→    // buy gold → spend USDC
   493→    s.usdc = Math.max(0, s.usdc - usd);
   494→    const goldUnits = usd / goldPrice;
   495→    s.paxg += goldUnits;
   496→  }
   497→  s.lastTradeDir = decision.direction;
   498→  s.lastTradeAt = Date.now();
   499→  s.dailyTurnoverUsd += usd;
   500→  s.updatedAt = Date.now();
   501→}

   503→// --- §11 Geopolitical Eject -------------------------------------------------
   504→export function updatePegHealth(s: ReserveState, dtHours: number, fx: FxRates) {
   505→  // Simulated per-asset peg drift (honest: labelled as simulated in UI).
   506→  // Stablecoins should sit at 1.0; we apply a tiny mean-reverting shock.
   507→  const drift = (cur: number, amp: number) => {
   508→    const shock = (Math.random() - 0.5) * 2 * amp;
   509→    return Math.max(0.9, Math.min(1.1, cur + (1 - cur) * 0.05 + shock));
   510→  };
   511→  s.pegHealth.USD = drift(s.pegHealth.USD, 0.0005);
   512→  s.pegHealth.EUR = drift(s.pegHealth.EUR, 0.0008);
   513→  s.pegHealth.GBP = drift(s.pegHealth.GBP, 0.0012);
   514→  s.pegHealth.JPY = drift(s.pegHealth.JPY, 0.0012);
   515→  s.pegHealth.CNY = drift(s.pegHealth.CNY, 0.0018);
   516→
   517→  const currencies: (keyof typeof s.pegHealth)[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   518→  for (const c of currencies) {
   519→    const p = s.pegHealth[c];
   520→    const outside = p < EJECT_DEPEG_BAND_LOWER || p > EJECT_DEPEG_BAND_UPPER;
   521→    if (outside) {
   522→      s.depegHours[c] += dtHours;
   523→    } else {
   524→      s.depegHours[c] = Math.max(0, s.depegHours[c] - dtHours * 0.5);
   525→    }
   526→    // Staged ladder
   527→    const h = s.depegHours[c];
   528→    let stage = 0;
   529→    if (h > 48) stage = 3;
   530→    else if (h > 24) stage = 2;
   531→    else if (h > 12) stage = 1;
   532→    if (h > 96 || p < 0.9 || p > 1.1) stage = 4;
   533→    s.ejectStage[c] = stage;
   534→  }
   535→}

   551→export function applyMint(
   552→  s: ReserveState,
   553→  fx: FxRates,
   554→  status: ProtocolStatus,
   555→  inputUsd: number,
   556→): MintResult {
   557→  const gfb = computeGfbIndex(fx);
   558→  const price = computeMtqPrice(gfb);
   559→  if (!priceInSafetyBand(price)) {
   560→    return { ok: false, reason: "MTQ price outside safety band (0.50–2.00 USD); minting paused by circuit breaker.", inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   561→  }
   562→  if (status === "DEFENSIVE" || status === "EMERGENCY") {
   563→    return { ok: false, reason: `${status}: minting paused by Risk State Machine.`, inputUsd, feeUsd: 0, netUsd: 0, mtqPrice: price, mtqMinted: 0, throttleFactor: 0, newCirculatingSupply: circulatingSupply(s), newReserveRatio: computeReserveRatio(reserveAssetValues(s, fx).nav, computeLiability(s, price)) };
   564→  }
   565→  const throttle = status === "CAUTION" ? 0.5 : status === "RECOVERY" ? 0.25 : 1.0;
   566→  const feeUsd = inputUsd * (MINT_FEE_BPS / 10_000);
   567→  const netUsd = inputUsd - feeUsd;
   568→  const minted = (netUsd / price) * throttle;
   569→
   570→  // Mutate reserve: deposit USDC (split across 3 USD issuers per concentration policy),
   571→  // mint MTQ to user (circulating supply rises). The concentration rebalancer in the
   572→  // tick loop will re-optimize the split if any issuer exceeds 25%.
   573→  const usdThird = netUsd / 3;
   574→  s.usdc += usdThird;
   575→  s.usdp += usdThird;
   576→  s.usdt += usdThird;
   577→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet (§13.2 sweep target)
   578→  s.totalSupply += minted;
   579→  s.updatedAt = Date.now();
   580→
   581→  const vals = reserveAssetValues(s, fx);
   582→  const newRr = computeReserveRatio(vals.nav, computeLiability(s, price));
   583→  return {
   584→    ok: true,
   585→    inputUsd,
   586→    feeUsd,
   587→    netUsd,
   588→    mtqPrice: price,
   589→    mtqMinted: minted,
   590→    throttleFactor: throttle,
   591→    newCirculatingSupply: circulatingSupply(s),
   592→    newReserveRatio: newRr,
   593→  };
   594→}

   628→export function applyRedeem(

   635→  const price = computeMtqPrice(gfb);

   638→  const vals0 = reserveAssetValues(s, fx);
   639→  const auditNavPerToken = circ > 0 ? vals0.nav / circ : 0;
   640→  const auditGrossUsdNav = inputMtq * auditNavPerToken;
   641→  const auditNote = "Blueprint §3.4.2 (redeem at P_MTQ) vs §12.2 (redeem at NAV_per_token=V_net/S) conflict. At RR>100% the §12.2 literal pays redeemers MORE than §3.4.2, draining the buffer surplus via arbitrage. We settle on the arbitrage-safe §3.4.2 value; the §12.2 figure is shown for audit only.";
   642→  if (inputMtq > circ) {
   643→    return { ok: false, reason: "Insufficient circulating supply for this redemption in pilot state.", inputMtq, mtqPrice: price, grossUsd: 0, feeBps: 0, feeUsd: 0, netUsd: 0, goldUsd: 0, goldPaxg: 0, basket: [], newCirculatingSupply: circ, newReserveRatio: computeReserveRatio(vals0.nav, computeLiability(s, price)), auditNavPerToken, auditGrossUsdNav, auditDeltaUsd: auditGrossUsdNav, auditNote };
   644→  }
   645→  // Fee by status (§14.1): NORMAL/CAUTION 0.15%, DEFENSIVE 0.5%, EMERGENCY 2%
   646→  let feeBps = REDEEM_FEE_BPS;
   647→  if (status === "DEFENSIVE") feeBps = 50;
   648→  if (status === "EMERGENCY") feeBps = 200;
   649→  const grossUsd = inputMtq * price; // redemption is priced against the GFB Index (§3.4.2)
   650→  const feeUsd = grossUsd * (feeBps / 10_000);
   651→  const netUsd = grossUsd - feeUsd;
   652→
   653→  // Release actual reserve composition proportionally (§12.2):
   654→  // gold portion = netUsd * W_target (legacy v1.2 buffer-derived target —
   655→  // retained until the MASE ensemble replaces it). fiat portion split per the
   656→  // 6 non-gold Strategic Prior components (USD/EUR/JPY/GBP/CNY/CHF) renormalised
   657→  // to the non-gold total.
   658→  const vals = reserveAssetValues(s, fx);
   659→  const rr = computeReserveRatio(vals.nav, computeLiability(s, price));
   660→  const wGold = computeTargetGoldWeight(s, rr);
   661→  const goldUsd = netUsd * wGold;
   662→  const fiatUsd = netUsd * (1 - wGold);
   663→  const goldPrice = vals.goldPrice;
   664→  const goldPaxg = goldUsd / goldPrice;
   665→
   666→  // v1.0: Strategic Prior weights for the 6 non-gold components, renormalised
   667→  // to the non-gold total (0.27 + 0.20 + 0.09 + 0.08 + 0.05 + 0.05 = 0.74).
   668→  const basketFiatTotal =
   669→    STRATEGIC_PRIOR.USD + STRATEGIC_PRIOR.EUR + STRATEGIC_PRIOR.JPY +
   670→    STRATEGIC_PRIOR.GBP + STRATEGIC_PRIOR.CNY + STRATEGIC_PRIOR.CHF;
   671→  const w = (k: keyof typeof STRATEGIC_PRIOR) => STRATEGIC_PRIOR[k] / basketFiatTotal;
   672→  const basket: RedeemBasket[] = [
   673→    { currency: "USD", token: "USDC", usdValue: fiatUsd * w("USD"), nativeAmount: fiatUsd * w("USD"), weight: STRATEGIC_PRIOR.USD },
   674→    { currency: "EUR", token: "EURC", usdValue: fiatUsd * w("EUR"), nativeAmount: (fiatUsd * w("EUR")) / fx.EUR_USD, weight: STRATEGIC_PRIOR.EUR },
   675→    { currency: "JPY", token: "JPY₿", usdValue: fiatUsd * w("JPY"), nativeAmount: (fiatUsd * w("JPY")) / fx.JPY_USD, weight: STRATEGIC_PRIOR.JPY },
   676→    { currency: "GBP", token: "GBP₿", usdValue: fiatUsd * w("GBP"), nativeAmount: (fiatUsd * w("GBP")) / fx.GBP_USD, weight: STRATEGIC_PRIOR.GBP },
   677→    { currency: "CNY", token: "CNY₿", usdValue: fiatUsd * w("CNY"), nativeAmount: (fiatUsd * w("CNY")) / fx.CNY_USD, weight: STRATEGIC_PRIOR.CNY },
   678→    { currency: "CHF", token: "CHF₿", usdValue: fiatUsd * w("CHF"), nativeAmount: (fiatUsd * w("CHF")) / fx.CHF_USD, weight: STRATEGIC_PRIOR.CHF }, // NEW v1.0
   679→  ];
   680→
   681→  // Mutate reserve: burn MTQ from circulating (reduce totalSupply), release assets
   682→  // USD released proportionally across USDC + USDP + USDT; gold across PAXG + XAUT
   683→  s.totalSupply -= inputMtq;
   684→  const usdRelease = basket[0].nativeAmount;
   685→  const usdThirdRelease = usdRelease / 3;
   686→  s.usdc = Math.max(0, s.usdc - usdThirdRelease);
   687→  s.usdp = Math.max(0, s.usdp - usdThirdRelease);
   688→  s.usdt = Math.max(0, s.usdt - usdThirdRelease);
   689→  s.eurc = Math.max(0, s.eurc - basket[1].nativeAmount);
   690→  s.jpy  = Math.max(0, s.jpy  - basket[2].nativeAmount);
   691→  s.gbp  = Math.max(0, s.gbp  - basket[3].nativeAmount);
   692→  s.cny  = Math.max(0, s.cny  - basket[4].nativeAmount);
   693→  s.chf  = Math.max(0, s.chf  - basket[5].nativeAmount); // NEW v1.0
   694→  // Gold released 50/50 PAXG + XAUT
   695→  const goldHalf = goldPaxg / 2;
   696→  s.paxg = Math.max(0, s.paxg - goldHalf);
   697→  s.xaut = Math.max(0, s.xaut - goldHalf);
   698→  s.treasury.hotWalletUsd += feeUsd; // fee revenue → hot wallet
   699→  s.updatedAt = Date.now();
   700→
   701→  const newVals = reserveAssetValues(s, fx);
   702→  const newRr = computeReserveRatio(newVals.nav, computeLiability(s, price));
   703→  return {
   704→    ok: true,
   705→    inputMtq,
   706→    mtqPrice: price,
   707→    grossUsd,
   708→    feeBps,
   709→    feeUsd,
   710→    netUsd,
   711→    goldUsd,
   712→    goldPaxg,
   713→    basket,
   714→    newCirculatingSupply: circulatingSupply(s),
   715→    newReserveRatio: newRr,
   716→    auditNavPerToken: circ > 0 ? vals0.nav / circ : 0,
   717→    auditGrossUsdNav,
   718→    auditDeltaUsd: auditGrossUsdNav - grossUsd,
   719→    auditNote,
   720→  };
   721→}

   900→// --- §11.3 Reintegration Score (Anti-Gaming) --------------------------------
   901→// R_score = w1·TimeInBand(48h) + w2·LiquidityDepth + w3·OracleAgreement + w4·(1−Volatility)
   902→// All inputs normalised to [0,1]; threshold >0.80 → staged re-purchase 25%→50%→100%.
   903→export function updateReintegration(
   904→  s: ReserveState,
   905→  board: OracleBoard | null,
   906→  dtHours: number,
   907→) {
   908→  const currencies: ("USD" | "EUR" | "GBP" | "JPY" | "CNY")[] = ["USD", "EUR", "GBP", "JPY", "CNY"];
   909→  for (const c of currencies) {
   910→    const r = s.reintegration[c];
   911→    // Time-in-band: increases when peg is inside [0.98,1.02]; decays when outside.
   912→    const inBand = s.pegHealth[c] >= EJECT_DEPEG_BAND_LOWER && s.pegHealth[c] <= EJECT_DEPEG_BAND_UPPER;
   913→    // dtHours is simulated elapsed; we approximate 48h window fraction.
   914→    if (inBand) {
   915→      r.timeInBand48h = Math.min(1, r.timeInBand48h + dtHours / 48);
   916→    } else {
   917→      r.timeInBand48h = Math.max(0, r.timeInBand48h - dtHours / 48);
   918→    }
   919→    // Liquidity depth: randomised around per-currency baseline (honest: simulated).
   920→    const base = c === "USD" ? 1.0 : c === "EUR" ? 0.95 : c === "CNY" ? 0.4 : 0.5;
   921→    r.liquidityDepth = clamp(base + (Math.random() - 0.5) * 0.1, 0, 1);
   922→    // Oracle agreement: valid feeds / 3 (from oracle board if present, else last known)
   923→    if (board) {
   924→      const pairMap: Record<string, string> = { USD: "—", EUR: "EUR/USD", GBP: "GBP/USD", JPY: "JPY/USD", CNY: "CNY/USD" };
   925→      const cons = board.pairs.find((p) => p.pair === pairMap[c]);
   926→      if (cons) {
   927→        r.oracleAgreement = cons.validCount / 3;
   928→        // Volatility from oracle spread (bps → fraction)
   929→        r.volatility = clamp(cons.spreadBps / 100, 0, 1);
   930→      }
   931→    }
   932→    // R_score
   933→    r.score =
   934→      REINTEGRATION_WEIGHTS.w1_TimeInBand * r.timeInBand48h +
   935→      REINTEGRATION_WEIGHTS.w2_LiquidityDepth * r.liquidityDepth +
   936→      REINTEGRATION_WEIGHTS.w3_OracleAgreement * r.oracleAgreement +
   937→      REINTEGRATION_WEIGHTS.w4_OneMinusVolatility * (1 - r.volatility);
   938→    // Repurchase staging: only for currencies that were previously ejected
   939→    if (s.ejectStage[c] > 0 && r.score > REINTEGRATION_THRESHOLD) {
   940→      r.repurchaseStage = Math.min(3, r.repurchaseStage + 1);
   941→      // As repurchase progresses, reduce eject stage
   942→      if (r.repurchaseStage >= 3) {
   943→        s.ejectStage[c] = Math.max(0, s.ejectStage[c] - 1);
   944→        if (s.ejectStage[c] === 0) r.repurchaseStage = 0;
   945→      }
   946→    } else if (r.score < REINTEGRATION_THRESHOLD * 0.6) {
   947→      r.repurchaseStage = 0;
   948→    }
   949→  }
   950→}

   969→// --- §8.5 First-Loss Waterfall ----------------------------------------------
   970→// 5 layers: (1) Operational Surplus → (2) Buffer Fiat → (3) Buffer Gold →
   971→// (4) Core Fiat → (5) Core Gold. Apply a loss; track consumed per layer.
   972→export function applyLoss(s: ReserveState, vals: { fiatNet: number; goldNet: number }, lossUsd: number) {
   973→  let remaining = lossUsd;
   974→  s.waterfall.consumed.surplus = 0; // reset for this loss event tracking is cumulative; we add
   975→  const layers = [
   976→    { name: "surplus", cap: () => s.treasury.hotWalletUsd + s.waterfall.operationalSurplus, consume: (amt: number) => {
   977→      const fromSurplus = Math.min(s.treasury.hotWalletUsd, amt);
   978→      s.treasury.hotWalletUsd -= fromSurplus;
   979→      s.waterfall.operationalSurplus -= Math.min(s.waterfall.operationalSurplus, amt - fromSurplus);
   980→      s.waterfall.consumed.surplus += amt;
   981→    } },
   982→    { name: "bufferFiat", cap: () => vals.fiatNet * 0.10 * (1 - currentBufferGoldRatio(s)), consume: (amt: number) => { s.waterfall.consumed.bufferFiat += amt; } },
   983→    { name: "bufferGold", cap: () => vals.goldNet * 0.10 * currentBufferGoldRatio(s), consume: (amt: number) => { s.waterfall.consumed.bufferGold += amt; } },
   984→    { name: "coreFiat", cap: () => vals.fiatNet * 0.80, consume: (amt: number) => { s.waterfall.consumed.coreFiat += amt; } },
   985→    { name: "coreGold", cap: () => vals.goldNet * 0.20, consume: (amt: number) => { s.waterfall.consumed.coreGold += amt; } },
   986→  ] as const;
   987→  for (let i = 0; i < layers.length && remaining > 0; i++) {
   988→    const layer = layers[i];
   989→    const cap = layer.cap();
   990→    const consumed = Math.min(cap, remaining);
   991→    layer.consume(consumed);
   992→    remaining -= consumed;
   993→    if (remaining > 0) s.waterfall.currentLayer = (i + 2) as 1 | 2 | 3 | 4 | 5;
   994→  }
   995→  if (remaining <= 0) s.waterfall.currentLayer = 1;
   996→}

   998→// --- §3.6 PriceUpdated events (0.5% change threshold) -----------------------
   999→export function maybePriceEvent(s: ReserveState, newPrice: number) {
  1000→  if (s.lastLoggedPrice <= 0) { s.lastLoggedPrice = newPrice; return; }
  1001→  const diff = Math.abs(newPrice - s.lastLoggedPrice);
  1002→  const changePct = diff / s.lastLoggedPrice;
  1003→  if (changePct >= PRICE_EVENT_THRESHOLD) {
  1004→    s.priceEvents.unshift({ ts: Date.now(), oldPrice: s.lastLoggedPrice, newPrice, changePct });
  1005→    if (s.priceEvents.length > 20) s.priceEvents.pop();
  1006→    s.lastLoggedPrice = newPrice;
  1007→  }
  1008→}

  1027→  const withinBound = isBuyGold ? quoteRate <= bound : quoteRate >= bound;

  1035→// --- FIX: §5.6 Concentration Rebalancing -------------------------------------
  1036→// Adjusts the USDC/USDP split to bring the maximum issuer share under the
  1037→// --- FIX: §5.6 Multi-Issuer Concentration Optimizer -------------------------
  1038→// 3 issuers: CIRCLE (USDC + EURC), PAXOS (USDP + PAXG), TETHER (USDT + XAUT).
  1039→// EURC is fixed to Circle (no second EUR issuer yet). GBP/JPY/CNY are TBD.
  1040→// The optimizer adjusts the splittable pools (USDC/USDP/USDT, PAXG/XAUT) to
  1041→// minimize the maximum issuer share, targeting ≤ 25% (warn threshold,
  1042→// headroom under the 30% hard limit).
  1043→export interface ConcentrationFixResult {
  1044→  adjusted: boolean;
  1045→  sharesBefore: { CIRCLE: number; PAXOS: number; TETHER: number };
  1046→  sharesAfter: { CIRCLE: number; PAXOS: number; TETHER: number };
  1047→  splits: { usdc: number; usdp: number; usdt: number; paxg: number; xaut: number };
  1048→  target: number; // 0.25
  1049→  reason: string;
  1050→}
  1051→
  1052→export function rebalanceForConcentration(
  1053→  s: ReserveState,
  1054→  vals: { usdcUsd: number; usdpUsd: number; usdtUsd: number; paxgUsd: number; xautUsd: number; nav: number; eurNet: number },
  1055→): ConcentrationFixResult {
  1056→  const nav = vals.nav;
  1057→  const TARGET = 0.25;
  1058→  const usdNetFactor = 1 - HAIRCUTS.USD;
  1059→  const goldNetFactor = 1 - HAIRCUTS.XAU;
  1060→
  1061→  const sharesBefore = {
  1062→    CIRCLE: nav > 0 ? (vals.usdcUsd + vals.eurNet) / nav : 0,
  1063→    PAXOS: nav > 0 ? (vals.usdpUsd + vals.paxgUsd) / nav : 0,
  1064→    TETHER: nav > 0 ? (vals.usdtUsd + vals.xautUsd) / nav : 0,
  1065→  };
  1066→  const maxBefore = Math.max(sharesBefore.CIRCLE, sharesBefore.PAXOS, sharesBefore.TETHER);
  1067→
  1068→  if (nav <= 0) {
  1069→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "NAV ≤ 0" };
  1070→  }
  1071→  if (maxBefore <= TARGET) {
  1072→    return { adjusted: false, sharesBefore, sharesAfter: sharesBefore, splits: { usdc: s.usdc, usdp: s.usdp, usdt: s.usdt, paxg: s.paxg, xaut: s.xaut }, target: TARGET, reason: "All issuers ≤ 25% warn threshold" };
  1073→  }
  1074→
  1075→  // Total splittable pools (in gross token units)
  1076→  const totalUsdGross = s.usdc + s.usdp + s.usdt;
  1077→  const totalGoldGross = s.paxg + s.xaut;
  1078→
  1079→  // Circle's floor = EURC (fixed). Circle's USDC budget = max(0, TARGET×NAV − EURC_usd)
  1080→  const circleUsdNetBudget = Math.max(0, TARGET * nav - vals.eurNet);
  1081→  const usdcGross = Math.min(totalUsdGross, circleUsdNetBudget / usdNetFactor);
  1082→  const remainingUsdGross = Math.max(0, totalUsdGross - usdcGross);
  1083→
  1084→  // Split remaining USD equally between USDP (Paxos) and USDT (Tether)
  1085→  const usdpGross = remainingUsdGross / 2;
  1086→  const usdtGross = remainingUsdGross / 2;
  1087→
  1088→  // Now balance gold between PAXG (Paxos) and XAUT (Tether) so both ≤ 25%.
  1089→  // Clean approach: split gold 50/50 by default; only deviate if one issuer
  1090→  // would breach the 25% cap. This guarantees a real split (the previous
  1091→  // min-budget logic could dump all gold into one issuer when budgets were
  1092→  // asymmetric or when one issuer's USD share was already small).
  1093→  const paxosUsdNet = usdpGross * usdNetFactor;
  1094→  const tetherUsdNet = usdtGross * usdNetFactor;
  1095→  const halfGoldGross = totalGoldGross / 2;
  1096→  let paxgGross = halfGoldGross;
  1097→  let xautGross = halfGoldGross;
  1098→  // Cap check 1: Paxos = USDP_net + PAXG_net ≤ TARGET×NAV
  1099→  const paxosWithHalf = paxosUsdNet + halfGoldGross * goldNetFactor;
  1100→  if (paxosWithHalf > TARGET * nav) {
  1101→    const maxPaxgNet = Math.max(0, TARGET * nav - paxosUsdNet);
  1102→    paxgGross = Math.min(halfGoldGross, maxPaxgNet / goldNetFactor);
  1103→    xautGross = totalGoldGross - paxgGross;
  1104→  }
  1105→  // Cap check 2: Tether = USDT_net + XAUT_net ≤ TARGET×NAV
  1106→  const tetherWithXaut = tetherUsdNet + xautGross * goldNetFactor;
  1107→  if (tetherWithXaut > TARGET * nav) {
  1108→    const maxXautNet = Math.max(0, TARGET * nav - tetherUsdNet);
  1109→    xautGross = Math.min(xautGross, maxXautNet / goldNetFactor);
  1110→    paxgGross = totalGoldGross - xautGross;
  1111→  }
  1112→
  1113→  s.usdc = usdcGross;
  1114→  s.usdp = usdpGross;
  1115→  s.usdt = usdtGross;
  1116→  s.paxg = paxgGross;
  1117→  s.xaut = xautGross;
  1118→
  1119→  const sharesAfter = {
  1120→    CIRCLE: (usdcGross * usdNetFactor + vals.eurNet) / nav,
  1121→    PAXOS: (usdpGross * usdNetFactor + paxgGross * goldNetFactor) / nav,
  1122→    TETHER: (usdtGross * usdNetFactor + xautGross * goldNetFactor) / nav,
  1123→  };
  1124→
  1125→  const maxAfter = Math.max(sharesAfter.CIRCLE, sharesAfter.PAXOS, sharesAfter.TETHER);
  1126→  return {
  1127→    adjusted: true,
  1128→    sharesBefore, sharesAfter,
  1129→    splits: { usdc: usdcGross, usdp: usdpGross, usdt: usdtGross, paxg: paxgGross, xaut: xautGross },
  1130→    target: TARGET,
  1131→    reason: maxAfter <= TARGET
  1132→      ? "Multi-issuer split: all 3 issuers ≤ 25% warn threshold"
  1133→      : `Optimized: max issuer ${(maxAfter * 100).toFixed(1)}% (Circle EURC floor limits further reduction)`,
  1134→  };
  1135→}
