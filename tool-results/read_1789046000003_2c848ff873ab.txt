    30→/* ---------- Regime badge ---------- */
    31→function RegimeBadge({ regime }: { regime: 0 | 1 | 2 | 3 }) {
    32→  const labels: Record<number, { label: string; tone: "emerald" | "gold" | "amber" | "rose" }> = {
    33→    0: { label: "Calm", tone: "emerald" },
    34→    1: { label: "Normal", tone: "gold" },
    35→    2: { label: "Stress", tone: "amber" },
    36→    3: { label: "Crisis", tone: "rose" },
    37→  };
    38→  const l = labels[regime] ?? labels[1];
    39→  return (
    40→    <Pill tone={l.tone}>
    41→      <GlowDot color={l.tone === "emerald" ? "emerald" : l.tone === "gold" ? "gold" : l.tone === "amber" ? "amber" : "rose"} size="h-1.5 w-1.5" />
    42→      regime: {l.label}
    43→    </Pill>
    44→  );
    45→}

    47→/* ---------- Weight heatmap cell ---------- */
    48→function WeightCell({ value, isTarget }: { value: number; isTarget?: boolean }) {
    49→  // Color intensity ∝ weight. Gold = warm gold, others = neutral amber.
    50→  const pct = Math.max(0, Math.min(100, value * 100));
    51→  const opacity = 0.06 + (pct / 100) * 0.32; // 0.06 → 0.38
    52→  return (
    53→    <td
    54→      className={`px-2 py-1.5 text-right font-mono tabular-nums ${isTarget ? "text-mtqs-gold font-semibold" : "text-white"}`}
    55→      style={{ backgroundColor: isTarget ? `rgba(232,185,100,${opacity + 0.06})` : `rgba(232,185,100,${opacity})` }}
    56→    >
    57→      {(value * 100).toFixed(2)}%
    58→    </td>
    59→  );
    60→}
