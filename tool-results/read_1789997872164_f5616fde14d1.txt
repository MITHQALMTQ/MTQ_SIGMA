    30→//     L4 monetary authorization · L5 ledger state · L6 atomic write ·

    33→//   • 130% RR target (was 110%), 80% fiat / 18% gold / 2% digital reserve
    34→//     composition targets (newly declared constants).
    35→//   • Non-custodial: no `reserveVault` USDC-holding pattern; banks hold their
    36→//     own reserves with qualified custodians. The contract only tracks
    37→//     `backingVerified[bank] = amount` as verified evidence (not custody).
    38→//   • `redeem()` burns MTQ and emits a `RedeemSettlement` event — the bank

   450→    // === V3 NEW §6 — Bank Registry (Blueprint v25.3 §6) ===
   451→    // ------------------------------------------------------------
   452→    // The Constitutional Council (7/7 multi-sig) authorizes banks. Each
   453→    // authorized bank has a per-day mint cap (USD 1e18 scale). The cap

   457→    struct BankRecord {
   458→        address bankAddress;
   459→        string  name;
   460→        string  jurisdiction;   // ISO 3166-1 alpha-2 ("US", "AE", "SA", ...)
   461→        bool    isAuthorized;
   462→        uint256 dailyMintCap;   // USD, 1e18 scale
   463→        uint256 dailyMintUsed;  // USD, 1e18 scale (resets daily)
   464→        uint256 dailyMintResetAt; // unix seconds of last reset
   465→        uint256 authorizedAt;
   466→    }
   467→
   468→    mapping(address => BankRecord) public bankRegistry;
   469→    address[] public authorizedBanks;
   470→
   471→    function authorizeBank(
   472→        address bank,
   473→        string calldata _name,
   474→        string calldata _jurisdiction,
   475→        uint256 _dailyCap
   476→    ) external onlyConstitutionalCouncil {
   477→        if (bank == address(0)) revert Err55();
   478→        if (_dailyCap == 0) revert Err87_DailyCapZero();
   479→        bool already = bankRegistry[bank].isAuthorized;
   480→        bankRegistry[bank] = BankRecord({
   481→            bankAddress: bank,
   482→            name: _name,
   483→            jurisdiction: _jurisdiction,
   484→            isAuthorized: true,
   485→            dailyMintCap: _dailyCap,
   486→            dailyMintUsed: 0,
   487→            dailyMintResetAt: block.timestamp,
   488→            authorizedAt: block.timestamp
   489→        });
   490→        if (!already) authorizedBanks.push(bank);
   491→        emit BankAuthorized(bank, _name, _jurisdiction, _dailyCap, block.timestamp);
   492→    }
   493→
   494→    function revokeBank(address bank) external onlyConstitutionalCouncil {
   495→        if (!bankRegistry[bank].isAuthorized) revert Err84_BankNotFound();
   496→        bankRegistry[bank].isAuthorized = false;
   497→        emit BankRevoked(bank, block.timestamp);
   498→    }

   500→    function getBank(address bank) external view returns (BankRecord memory) {
   501→        return bankRegistry[bank];
   502→    }

   508→    /// @dev Rolls the per-bank daily mint counter at UTC midnight.
   509→    ///      Idempotent within the same UTC day. Called internally by mint
   510→    ///      paths; also callable by anyone (no-op if already current).
   511→    function _rollDailyCap(address bank) internal {
   512→        BankRecord storage rec = bankRegistry[bank];
   513→        if (rec.dailyMintResetAt == 0) return;
   514→        uint256 today = (block.timestamp / 1 days) * 1 days;
   515→        if (rec.dailyMintResetAt < today) {
   516→            uint256 oldUsed = rec.dailyMintUsed;
   517→            rec.dailyMintUsed = 0;
   518→            rec.dailyMintResetAt = today;
   519→            emit DailyCapReset(bank, oldUsed, 0, block.timestamp);
   520→        }
   521→    }

   523→    // ============================================================
   524→    // === V3 NEW §7 — AvailableBackingCertificate (ABC) ===
   525→    // ------------------------------------------------------------
   526→    // Banks submit an ABC proving they have backing assets held by a
   527→    // qualified custodian. The keeper verifies the custodian attestation
   528→    // (production: via oracle; testnet: auto-verify via verifyBackingCertificate).
   529→    //
   530→    // MITHQAL NEVER takes custody — it only verifies evidence.
   531→    // ============================================================
   532→
   533→    struct AvailableBackingCertificate {
   534→        bytes32 certificateId;
   535→        address bank;
   536→        uint256 amountUsd;                   // 1e18 scale
   537→        bytes32 custodianAttestationHash;    // hash of custodian's attestation
   538→        uint256 custodianAttestationExpiry;  // unix seconds
   539→        bytes32 reserveAssetBreakdownHash;   // hash of the reserve composition
   540→        uint256 issuedAt;
   541→        bool    verified;
   542→    }
   543→
   544→    mapping(bytes32 => AvailableBackingCertificate) public backingCertificates;
   545→
   546→    function submitBackingCertificate(AvailableBackingCertificate calldata cert) external onlyAuthorizedBank {
   547→        // The submitting bank must match the certificate's bank field.
   548→        if (cert.bank != msg.sender) revert Err92_BackdingCertMismatch();
   549→        if (cert.amountUsd == 0) revert Err86_InvalidRequest();
   550→        if (cert.custodianAttestationExpiry <= block.timestamp) revert Err91_InvalidExpiry();
   551→        if (cert.certificateId == bytes32(0)) revert Err86_InvalidRequest();
   552→        // Reject duplicate certificate IDs.
   553→        if (backingCertificates[cert.certificateId].issuedAt != 0) revert Err69_NonceReused();
   554→
   555→        backingCertificates[cert.certificateId] = AvailableBackingCertificate({
   556→            certificateId: cert.certificateId,
   557→            bank: cert.bank,
   558→            amountUsd: cert.amountUsd,
   559→            custodianAttestationHash: cert.custodianAttestationHash,
   560→            custodianAttestationExpiry: cert.custodianAttestationExpiry,
   561→            reserveAssetBreakdownHash: cert.reserveAssetBreakdownHash,
   562→            issuedAt: block.timestamp,
   563→            verified: false
   564→        });
   565→        emit BackingCertificateSubmitted(cert.certificateId, cert.bank, cert.amountUsd, block.timestamp);
   566→    }
   567→
   568→    /// @notice Verify an ABC's custodian attestation. Production: oracle call.
   569→    ///         Testnet/pilot: keeper auto-verifies (the keeper is trusted).
   570→    function verifyBackingCertificate(bytes32 certId) external onlyKeeper returns (bool) {
   571→        AvailableBackingCertificate storage c = backingCertificates[certId];
   572→        if (c.issuedAt == 0) revert Err85_CertNotFound();
   573→        if (c.custodianAttestationExpiry <= block.timestamp) revert Err72_BackdingCertExpired();
   574→        c.verified = true;
   575→        // Credit the bank's verified-backing pool (evidence only — not custody).
   576→        backingVerified[c.bank] += c.amountUsd;
   577→        emit BackingCertificateVerified(certId, msg.sender, true, block.timestamp);
   578→        return true;
   579→    }

   581→    function getCertificate(bytes32 certId) external view returns (AvailableBackingCertificate memory) {
   582→        return backingCertificates[certId];
   583→    }

   585→    // ============================================================
   586→    // === V3 NEW §8 — 16-Step Bank Minting Workflow ===
   587→    // ------------------------------------------------------------
   588→    // BM-01 PENDING        — bank submits requestMint()
   589→    // BM-02 RECEIVED       — MBG gateway ack (keeper: API auth + schema OK)
   590→    // BM-03 KYC            — KYC passed (keeper: off-chain evidence)
   591→    // BM-04 AML            — AML/sanctions passed (keeper)
   592→    // BM-05 BACKING        — ABC submitted & in registry
   593→    // BM-06 EVIDENCE       — ABC verified (custodian attestation valid)
   594→    // BM-07 REQUESTED      — MBG has translated request to on-chain
   595→    // BM-08 TRANSLATED     — request payload validated against policy
   596→    // BM-09 ELIGIBLE       — bank + jurisdiction eligible
   597→    // BM-10 JURISDICTION   — jurisdiction checks (sanctions/exchange control)
   598→    // BM-11 BACKING_VERIFIED — backing ≥ mint amount
   599→    // BM-12 BANK_RISK      — bank-level risk (cap, concentration, credit)
   600→    // BM-13 SYSTEM_RISK    — system-level risk (RR/LCR/state machine)
   601→    // BM-14 DMCE           — Daily Mint Cap / Concentration / Exposure check
   602→    // BM-15 AUTHORIZED     — MITHQAL Monetary Control authorization (L4)
   603→    // BM-16 MINTED         — executeMint() final atomic write (L6+L7)
   604→    // REJECTED             — terminal failure state
   605→    // ============================================================
   606→
   607→    enum WorkflowState {
   608→        NONE,
   609→        BM01_PENDING,
   610→        BM02_RECEIVED,
   611→        BM03_KYC,
   612→        BM04_AML,
   613→        BM05_BACKING,
   614→        BM06_EVIDENCE,
   615→        BM07_REQUESTED,
   616→        BM08_TRANSLATED,
   617→        BM09_ELIGIBLE,
   618→        BM10_JURISDICTION,
   619→        BM11_BACKING_VERIFIED,
   620→        BM12_BANK_RISK,
   621→        BM13_SYSTEM_RISK,
   622→        BM14_DMCE,
   623→        BM15_AUTHORIZED,
   624→        BM16_MINTED,
   625→        REJECTED
   626→    }

   628→    struct BankMintRequest {
   629→        address bank;
   630→        bytes32 corporateCustomerHash;     // hash of the corporate customer identity
   631→        uint256 amountUsd;                  // 1e18 scale
   632→        bytes32 backingCertificateId;
   633→        string  jurisdiction;               // ISO 3166-1 alpha-2
   634→        uint256 nonce;                      // bank-unique anti-replay
   635→        uint256 expiry;                     // unix seconds; request expires after
   636→    }
   637→
   638→    struct MintRequest {
   639→        bytes32 requestId;
   640→        BankMintRequest request;
   641→        WorkflowState state;
   642→        uint256 createdAt;
   643→        uint256 authorizedAt;
   644→        uint256 mintedAt;
   645→        uint256 mintedAmount;
   646→        string  rejectReason;
   647→    }
   648→
   649→    mapping(bytes32 => MintRequest) public mintRequests;
   650→    /// @notice Anti-replay: bank → nonce → consumed.
   651→    mapping(address => mapping(uint256 => bool)) public usedNonces;
   652→    /// @notice Per-bank concentration: total outstanding MTQ minted via this bank.
   653→    mapping(address => uint256) public bankMintedTotal;
   654→    /// @notice DMCE: Daily Mint Cap & Concentration Exposure — single-bank cap
   655→    ///         as a fraction of total supply (1e18 scale). Default 5%.
   656→    uint256 public dmceBankConcentrationCap = 0.05e18;

   658→    /// @notice Request a mint — only an authorized bank. Creates state=BM01_PENDING.
   659→    function requestMint(BankMintRequest calldata req) external onlyAuthorizedBank whenNotPaused returns (bytes32 requestId) {
   660→        if (req.bank != msg.sender) revert Err70_JurisdictionMismatch();
   661→        if (req.amountUsd == 0) revert Err86_InvalidRequest();
   662→        if (req.expiry <= block.timestamp) revert Err91_InvalidExpiry();
   663→        if (usedNonces[msg.sender][req.nonce]) revert Err69_NonceReused();
   664→        usedNonces[msg.sender][req.nonce] = true;
   665→
   666→        // L3 POLICY (early): per-bank daily cap precheck (full enforcement
   667→        // happens at BM-12/BM-14, but we reject early to fail fast).
   668→        _rollDailyCap(msg.sender);
   669→        BankRecord storage bank = bankRegistry[msg.sender];
   670→        if (bank.dailyMintUsed + req.amountUsd > bank.dailyMintCap) revert Err67_DailyCapExceeded();
   671→
   672→        // The certificate must already exist (BM-05 will verify it is in the
   673→        // registry; BM-06 will verify it is verified).
   674→        if (backingCertificates[req.backingCertificateId].issuedAt == 0) revert Err85_CertNotFound();
   675→
   676→        requestId = keccak256(abi.encodePacked(
   677→            msg.sender,
   678→            req.corporateCustomerHash,
   679→            req.amountUsd,
   680→            req.backingCertificateId,
   681→            req.nonce,
   682→            block.chainid,
   683→            block.timestamp
   684→        ));
   685→        if (mintRequests[requestId].createdAt != 0) revert Err69_NonceReused();
   686→
   687→        mintRequests[requestId] = MintRequest({
   688→            requestId: requestId,
   689→            request: req,
   690→            state: WorkflowState.BM01_PENDING,
   691→            createdAt: block.timestamp,
   692→            authorizedAt: 0,
   693→            mintedAt: 0,
   694→            mintedAmount: 0,
   695→            rejectReason: ""
   696→        });
   697→        emit MintRequested(requestId, msg.sender, req.amountUsd, req.backingCertificateId, req.jurisdiction, block.timestamp);
   698→        emit WorkflowAdvanced(requestId, WorkflowState.NONE, WorkflowState.BM01_PENDING, msg.sender, block.timestamp);
   699→    }

   701→    /// @notice Advance a request through the workflow. Only KEEPER (BM-02..BM-14)
   702→    ///         or MONETARY_CONTROL (BM-15) may advance. State transitions must
   703→    ///         be strictly sequential (no skipping). The keeper performs the
   704→    ///         L3 policy checks (KYC/AML/eligibility/jurisdiction/bank-risk/
   705→    ///         system-risk/DMCE) off-chain and only calls advanceWorkflow when
   706→    ///         the corresponding gate has passed. The L4 monetary authorization
   707→    ///         gate (BM-15) requires the MONETARY_CONTROL_ROLE signer.
   708→    function advanceWorkflow(bytes32 requestId, WorkflowState toState) external whenNotPaused returns (bool) {
   709→        MintRequest storage r = mintRequests[requestId];
   710→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   711→        if (r.state == WorkflowState.BM16_MINTED) revert Err76_WorkflowAlreadyMinted();
   712→        if (r.state == WorkflowState.REJECTED) revert Err77_WorkflowAlreadyRejected();
   713→        if (r.request.expiry <= block.timestamp) revert Err68_RequestExpired();
   714→
   715→        // L2 WORKFLOW GATE — strict sequential progression.
   716→        // toState must equal (current + 1).
   717→        if (uint256(toState) != uint256(r.state) + 1) revert Err74_WorkflowNotSequential();
   718→
   719→        // Role gate per state:
   720→        //   BM-02..BM-14 → KEEPER_ROLE
   721→        //   BM-15        → MONETARY_CONTROL_ROLE (L4)
   722→        if (toState == WorkflowState.BM15_AUTHORIZED) {
   723→            if (!(_roles[MONETARY_CONTROL_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err80_L4MonetaryGate();
   724→        } else {
   725→            if (!(_roles[KEEPER_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err31();
   726→        }
   727→
   728→        // L3 POLICY GATE — keeper has run the off-chain policy engine and
   729→        // attests that all checks for `toState` have passed. We re-verify
   730→        // a small number of on-chain invariants per state for defense in depth.
   731→        if (toState == WorkflowState.BM05_BACKING) {
   732→            // The ABC must exist.
   733→            if (backingCertificates[r.request.backingCertificateId].issuedAt == 0) revert Err85_CertNotFound();
   734→        } else if (toState == WorkflowState.BM06_EVIDENCE) {
   735→            // The ABC must be verified.
   736→            AvailableBackingCertificate storage c = backingCertificates[r.request.backingCertificateId];
   737→            if (!c.verified) revert Err71_BackdingCertNotVerified();
   738→            if (c.custodianAttestationExpiry <= block.timestamp) revert Err72_BackdingCertExpired();
   739→        } else if (toState == WorkflowState.BM09_ELIGIBLE) {
   740→            // Bank must still be authorized.
   741→            if (!bankRegistry[r.request.bank].isAuthorized) revert Err66_BankNotAuthorized();
   742→        } else if (toState == WorkflowState.BM10_JURISDICTION) {
   743→            // Jurisdiction on request must match bank's registered jurisdiction.
   744→            BankRecord storage b = bankRegistry[r.request.bank];
   745→            if (keccak256(bytes(b.jurisdiction)) != keccak256(bytes(r.request.jurisdiction))) revert Err70_JurisdictionMismatch();
   746→        } else if (toState == WorkflowState.BM11_BACKING_VERIFIED) {
   747→            // Verified backing pool for the bank must be ≥ requested amount.
   748→            if (backingVerified[r.request.bank] < r.request.amountUsd) revert Err73_BackdingCertInsufficient();
   749→        } else if (toState == WorkflowState.BM12_BANK_RISK) {
   750→            // Per-bank daily cap precheck (rolls the day first).
   751→            _rollDailyCap(r.request.bank);
   752→            BankRecord storage br = bankRegistry[r.request.bank];
   753→            if (br.dailyMintUsed + r.request.amountUsd > br.dailyMintCap) revert Err67_DailyCapExceeded();
   754→        } else if (toState == WorkflowState.BM13_SYSTEM_RISK) {
   755→            // Minting must be allowed by the 6-state risk machine.
   756→            if (!mintingAllowed(currentState)) revert Err25();
   757→            // Oracle must be healthy (strict 3-source quorum on all 6 pairs).
   758→            _requireOracleHealthy();
   759→        } else if (toState == WorkflowState.BM14_DMCE) {
   760→            // DMCE: single-bank concentration cap.
   761→            uint256 concentration = (totalSupply == 0)
   762→                ? type(uint256).max
   763→                : bankMintedTotal[r.request.bank] * 1e18 / totalSupply;
   764→            // Reject if adding this mint would breach the concentration cap.
   765→            uint256 newMintedMTQ = r.request.amountUsd * 1e18 / getMTQPriceWithGuard();
   766→            uint256 newConcentration = (totalSupply + newMintedMTQ == 0)
   767→                ? 0
   768→                : (bankMintedTotal[r.request.bank] + newMintedMTQ) * 1e18 / (totalSupply + newMintedMTQ);
   769→            if (newConcentration > dmceBankConcentrationCap && concentration > dmceBankConcentrationCap) revert Err88_ConcentrationLimit();
   770→        } else if (toState == WorkflowState.BM15_AUTHORIZED) {
   771→            // L4 MONETARY CONTROL — the modifier already gated this.
   772→            // Mark the authorization timestamp.
   773→            r.authorizedAt = block.timestamp;
   774→        }
   775→
   776→        WorkflowState fromState = r.state;
   777→        r.state = toState;
   778→        emit WorkflowAdvanced(requestId, fromState, toState, msg.sender, block.timestamp);
   779→        return true;
   780→    }

   782→    /// @notice Reject a request (terminal). Keeper or Monetary Control only.
   783→    function rejectMint(bytes32 requestId, string calldata reason) external whenNotPaused returns (bool) {
   784→        MintRequest storage r = mintRequests[requestId];
   785→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   786→        if (r.state == WorkflowState.BM16_MINTED) revert Err76_WorkflowAlreadyMinted();
   787→        if (r.state == WorkflowState.REJECTED) revert Err77_WorkflowAlreadyRejected();
   788→        if (!(_roles[KEEPER_ROLE][msg.sender] || _roles[MONETARY_CONTROL_ROLE][msg.sender] || _roles[DEFAULT_ADMIN_ROLE][msg.sender])) revert Err31();
   789→        WorkflowState fromState = r.state;
   790→        r.state = WorkflowState.REJECTED;
   791→        r.rejectReason = reason;
   792→        emit WorkflowAdvanced(requestId, fromState, WorkflowState.REJECTED, msg.sender, block.timestamp);
   793→        emit MintRejected(requestId, msg.sender, reason, block.timestamp);
   794→        return true;
   795→    }

   797→    /// @notice L5+L6+L7 FINALITY GATE — execute the mint atomically.
   798→    ///         Requires state == BM15_AUTHORIZED (L2/L5), oracle healthy (L3
   799→    ///         recheck), keeper signature (L7). Single transaction = atomic
   800→    ///         write (L6). Mints MTQ to the bank; the bank distributes to the
   801→    ///         corporate customer off-chain (sovereign currency settlement).
   802→    function executeMint(bytes32 requestId) external onlyKeeper whenNotPaused nonReentrant returns (uint256 minted) {
   803→        MintRequest storage r = mintRequests[requestId];
   804→        if (r.createdAt == 0) revert Err86_InvalidRequest();
   805→        // L2 WORKFLOW GATE — must be at BM15_AUTHORIZED.
   806→        if (r.state != WorkflowState.BM15_AUTHORIZED) revert Err78_L2WorkflowGate();
   807→        // L5 LEDGER GATE — must not already be minted.
   808→        if (r.mintedAt != 0) revert Err76_WorkflowAlreadyMinted();
   809→        // L3 POLICY GATE (recheck) — minting allowed, oracle healthy, not expired.
   810→        if (!mintingAllowed(currentState)) revert Err25();
   811→        _requireOracleHealthy();
   812→        if (r.request.expiry <= block.timestamp) revert Err68_RequestExpired();
   813→        // Re-check backing is still verified and sufficient (defence in depth).
   814→        if (backingVerified[r.request.bank] < r.request.amountUsd) revert Err73_BackdingCertInsufficient();
   815→
   816→        // L4 MONETARY AUTHORIZATION — already enforced at advanceWorkflow(BM15)
   817→        // by the onlyMonetaryControl modifier (we re-verify authorizedAt > 0).
   818→        if (r.authorizedAt == 0) revert Err80_L4MonetaryGate();
   819→
   820→        // L7 FINALITY ORACLE — the keeper signature (msg.sender = keeper) is
   821→        // the on-chain finality proof. Without it, no mint can land.
   822→
   823→        // L6 ATOMIC WRITE — all state changes below are in a single tx; if any
   824→        // reverts, the whole mint reverts (atomic, all-or-nothing).
   825→
   826→        // Compute mint amount: amountUsd (1e18) × 1e18 / P_MTQ (1e18).
   827→        uint256 price = getMTQPriceWithGuard();
   828→        uint256 grossMint = r.request.amountUsd * 1e18 / price;
   829→
   830→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25).
   831→        uint256 throttle = mintThrottle(currentState);
   832→        minted = grossMint * throttle / 1e18;
   833→
   834→        // State transitions + accounting.
   835→        r.state = WorkflowState.BM16_MINTED;
   836→        r.mintedAt = block.timestamp;
   837→        r.mintedAmount = minted;
   838→
   839→        // Decrement the bank's verified-backing pool by the minted USD amount
   840→        // (the backing is now "consumed" by the mint — the bank's custodian
   841→        // will release the corresponding reserves to the redeemer at redeem
   842→        // time, off-chain).
   843→        if (backingVerified[r.request.bank] >= r.request.amountUsd) {
   844→            backingVerified[r.request.bank] -= r.request.amountUsd;
   845→        } else {
   846→            // Should never hit thanks to the BM-11 check; defense in depth.
   847→            backingVerified[r.request.bank] = 0;
   848→        }
   849→
   850→        // Roll + charge the per-bank daily mint cap.
   851→        _rollDailyCap(r.request.bank);
   852→        bankRegistry[r.request.bank].dailyMintUsed += r.request.amountUsd;
   853→        bankMintedTotal[r.request.bank] += minted;
   854→
   855→        // L7 finality: mint to the bank (NOT the corporate customer — the bank
   856→        // settles with the customer off-chain per Blueprint v25.3 §6).
   857→        _mint(r.request.bank, minted);
   858→
   859→        emit WorkflowAdvanced(requestId, WorkflowState.BM15_AUTHORIZED, WorkflowState.BM16_MINTED, msg.sender, block.timestamp);
   860→        emit MintExecuted(requestId, r.request.bank, r.request.amountUsd, minted, price, block.timestamp);
   861→    }

   863→    /// @notice V3 non-custodial redeem — burns MTQ from the redeemer and emits
   864→    ///         a `RedeemSettlement` event. The bank settles with the redeemer
   865→    ///         off-chain (sovereign currency transfer). The contract itself
   866→    ///         does NOT push USDC: there is no reserveVault in V3.
   867→    ///         Only an authorized bank can call this on behalf of a redeemer
   868→    ///         (the redeemer must have approved the bank to burn their MTQ via
   869→    ///         the standard ERC-20 allowance).
   870→    function redeem(uint256 mtqAmount, address redeemer) external nonReentrant whenNotPaused returns (uint256 netUsd) {
   871→        if (!(mtqAmount > 0)) revert Err56();
   872→        if (!(redemptionAllowed(currentState))) revert Err43();
   873→        // Only an authorized bank can submit a redeem for settlement.
   874→        if (!bankRegistry[msg.sender].isAuthorized) revert Err90_NotAuthorizedBank();
   875→        // Redeemer must have approved the bank to burn their MTQ.
   876→        uint256 allowed = allowance[redeemer][msg.sender];
   877→        if (allowed != type(uint256).max) {
   878→            if (!(allowed >= mtqAmount)) revert Err21();
   879→            allowance[redeemer][msg.sender] = allowed - mtqAmount;
   880→        }
   881→        if (!(balanceOf[redeemer] >= mtqAmount)) revert Err22();
   882→
   883→        _requireOracleHealthy();
   884→        uint256 navPerToken = getNAVperToken();
   885→        if (!(navPerToken < type(uint256).max)) revert Err26();
   886→
   887→        uint256 feeFraction = redeemFee(currentState);
   888→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
   889→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
   890→        netUsd             = grossUsd18 - feeUsd18;
   891→
   892→        _burn(redeemer, mtqAmount);
   893→
   894→        // Decrement the on-chain reserve mirror proportionally (basket broken).
   895→        uint256 totalReserve = getReserveNavUsd();
   896→        if (totalReserve > 0 && grossUsd18 > 0) {
   897→            for (uint8 i = 0; i < 7; i++) {
   898→                uint256 held = reserveHeldUsd[Component(i)];
   899→                if (held == 0) continue;
   900→                uint256 share = held * grossUsd18 / totalReserve;
   901→                if (share > held) share = held;
   902→                reserveHeldUsd[Component(i)] = held - share;
   903→            }
   904→        }
   905→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
   906→                                  reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
   907→                                  reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
   908→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
   909→        reserveState.liquidValue = reserveState.fiatValue;
   910→
   911→        // Decrement the bank's outstanding MTQ counter (concentration reverse).
   912→        if (bankMintedTotal[msg.sender] >= mtqAmount) {
   913→            bankMintedTotal[msg.sender] -= mtqAmount;
   914→        } else {
   915→            bankMintedTotal[msg.sender] = 0;
   916→        }
   917→
   918→        // The bank owes `netUsd` to the redeemer, settled off-chain via
   919→        // sovereign currency transfer. Emit the settlement obligation.
   920→        emit RedeemSettlement(redeemer, msg.sender, mtqAmount, grossUsd18, feeUsd18, netUsd, navPerToken, block.timestamp);
   921→    }

  1111→        ProtocolState newState;

  1121→            newState = ProtocolState.EMERGENCY;

  1127→                _transitionTo(newState, true);

  1132→            }
  1133→        }

  1495→    function executeRebalance(RebalanceTrade[] memory trades) external nonReentrant onlyKeeper whenNotPaused {
  1496→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1497→        if (!(trades.length <= 7)) revert Err48();
  1498→
  1499→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1500→            dailyTradeUsd    = 0;
  1501→            dailyTradeResetAt = block.timestamp;
  1502→        }
  1503→
  1504→        uint256 nav = getReserveNavUsd();
  1505→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1506→        uint256 rr = getReserveRatio();
  1507→        bool solvencyOverride = (rr < rrStressFloor);
  1508→
  1509→        for (uint256 i = 0; i < trades.length; i++) {
  1510→            RebalanceTrade memory t = trades[i];
  1511→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1512→            if (t.direction == 0 || t.tradeUsd == 0) continue;
  1513→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1514→
  1515→            int256 prevDir = lastDirection[t.component];
  1516→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1517→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1518→            }
  1519→
  1520→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1521→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1522→            if (t.direction == 1) {
  1523→                if (!(currentW < execW)) revert Err14();
  1524→            } else {
  1525→                if (!(currentW > execW)) revert Err46();
  1526→            }
  1527→
  1528→            uint256 newHeldUsd = (t.direction == 1)
  1529→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1530→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1531→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1532→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1533→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1534→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1535→            if (!(absDiff <= tolerance)) revert Err49();
  1536→
  1537→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1538→            dailyTradeUsd += t.tradeUsd;
  1539→
  1540→            if (t.direction == 1) {
  1541→                reserveHeldUsd[t.component] += t.tradeUsd;
  1542→            } else {
  1543→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1544→            }
  1545→            lastDirection[t.component]   = t.direction;
  1546→            lastRebalanceAt[t.component] = block.timestamp;

  1550→                                  reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
