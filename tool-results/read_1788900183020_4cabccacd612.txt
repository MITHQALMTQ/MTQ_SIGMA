   359→    /// @notice Listing 2 (§7.7) submitTargetWeights — MASE verification + smoothing.
   360→    /// @dev   Only the MASE submitter (KEEPER_ROLE). Enforces sum=1, positivity,
   361→    ///        admissibility envelopes, per-component velocity, then applies
   362→    ///        stress-adaptive smoothing rho 0.50 (normal) / 0.75 (stress).
   363→    function submitTargetWeights(
   364→        uint256[7] calldata target,
   365→        bytes32 methodologyVersion,
   366→        bytes32 dataVersion
   367→    ) external onlyKeeper {
   368→        // 1. Sum-to-one and non-negativity
   369→        uint256 sum = 0;
   370→        for (uint256 i = 0; i < 7; i++) {
   371→            if (!(target[i] > 0)) revert Err62();
   372→            sum += target[i];
   373→        }
   374→        if (!(sum == 1e18)) revert Err54();
   375→        // 2. Constitutional admissibility envelopes
   376→        for (uint256 i = 0; i < 7; i++) {
   377→            if (!(target[i] >= LOWER_BOUND[i])) revert Err12();
   378→            if (!(target[i] <= UPPER_BOUND[i])) revert Err06();
   379→        }
   380→        // 3. Weight-velocity constraint vs previous live weights
   381→        for (uint256 i = 0; i < 7; i++) {
   382→            uint256 diff = target[i] > liveWeights.weights[i]
   383→                ? target[i] - liveWeights.weights[i]
   384→                : liveWeights.weights[i] - target[i];
   385→            if (!(diff <= MAX_VELOCITY[i])) revert Err52();
   386→        }
   387→        // 4. Stress-adaptive smoothing: W_smooth = rho * W_prev + (1 - rho) * W_target
   388→        uint256 rho = crisisFlag ? smoothingRhoStress : smoothingRhoNormal;
   389→        uint256[7] memory smoothed;
   390→        for (uint256 i = 0; i < 7; i++) {
   391→            smoothed[i] = (rho * liveWeights.weights[i] + (1e18 - rho) * target[i]) / 1e18;
   392→        }
   393→        // 5. Commit registry state
   394→        for (uint256 i = 0; i < 7; i++) liveWeights.weights[i] = smoothed[i];
   395→        liveWeights.targetWeights    = target;
   396→        liveWeights.methodologyVersion = methodologyVersion;
   397→        liveWeights.dataVersion       = dataVersion;
   398→        liveWeights.updatedAt         = block.timestamp;
   399→        emit WeightsAccepted(smoothed, target, methodologyVersion, dataVersion, block.timestamp);
   400→    }

   438→    function _basePrice18(uint256 i) internal pure returns (uint256) {
   439→        if (i == 0) return 1e18;             // USD = 1 USD
   440→        if (i == 1) return BASE_EUR_USD;     // 1.05
   441→        if (i == 2) return BASE_JPY_USD;     // 0.0067
   442→        if (i == 3) return BASE_GBP_USD;     // 1.25
   443→        if (i == 4) return BASE_CNY_USD;     // 0.14
   444→        if (i == 5) return BASE_CHF_USD;     // 1.13
   445→        return BASE_GOLD_USD;                 // 2500
   446→    }

   448→    /// @notice Listing 3 advanceIndex — §9.2 authoritative recursion.
   449→    ///         I_t = I_{t-1} × Σ_i W_{i,t-1} × (P_{i,t} / P_{i,t-1})
   450→    ///         Prices are passed in by the keeper (1e18 scale, USD-quoted).
   451→    function advanceIndex(uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   452→        if (!(indexValue > 0)) revert Err20();
   453→
   454→        // 1. Period return weighted by W_{i,t-1} in force over the period (COO-16).
   455→        uint256 growth = 0;
   456→        for (uint256 i = 0; i < 7; i++) {
   457→            if (!(currentPrices[i] > 0 && lastPrices[i] > 0)) revert Err09();
   458→            growth += (lastWeights[i] * currentPrices[i]) / lastPrices[i];
   459→        }
   460→        indexValue = (indexValue * growth) / 1e18;
   461→
   462→        // 2. Commit the period end-state (prices).
   463→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   464→        lastIndexUpdate = block.timestamp;
   465→        emit IndexAdvanced(indexValue, block.timestamp);
   466→    }
   467→
   468→    /// @notice Listing 3 commitWeights — chain-link adjustment at weight change.
   469→    ///         Computes D_t = B_t^- / B_t^+ (Master §9.3), updates G_t, updates
   470→    ///         lastPrices and lastWeights. I_t is UNCHANGED by the weight change
   471→    ///         itself — ZERO artificial return (the recursion is continuous).
   472→    function commitWeights(uint256[7] calldata newWeights, uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   473→        if (!(indexValue > 0)) revert Err20();
   474→
   475→        // 1. B_t^- and B_t^+ are base-relative aggregates measured against
   476→        //    the genesis fixings (§9.3, MS §75).
   477→        uint256 bMinus = 0;
   478→        uint256 bPlus  = 0;
   479→        bool    changed = false;
   480→        for (uint256 i = 0; i < 7; i++) {
   481→            if (!(currentPrices[i] > 0)) revert Err09();
   482→            uint256 rel = (currentPrices[i] * 1e18) / _basePrice18(i); // P_{i,t} / P_{i,0}
   483→            bMinus += (lastWeights[i] * rel) / 1e18;
   484→            bPlus  += (newWeights[i]    * rel) / 1e18;
   485→            if (newWeights[i] != lastWeights[i]) changed = true;
   486→        }
   487→
   488→        // 2. Apply divisor D_t; record into G_t (cumulative). I_t unchanged.
   489→        if (changed) {
   490→            if (!(bPlus > 0)) revert Err57();
   491→            uint256 newD = (bMinus * 1e18) / bPlus;
   492→            chainLinkDivisor = (chainLinkDivisor * newD) / 1e18;
   493→            for (uint256 i = 0; i < 7; i++) lastWeights[i] = newWeights[i];
   494→            emit ChainLinkAdjusted(newD, bMinus, bPlus, block.timestamp);
   495→        }
   496→
   497→        // 3. Commit the period end-state prices too.
   498→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   499→        emit WeightsCommitted(newWeights, chainLinkDivisor, block.timestamp);
   500→    }
   501→
   502→    /// @notice Listing 3 genesis — one-time seed of I_0 = 1.0000 (ADMIN_ROLE).
   503→    ///         Prices are seeded from immutable base fixings, so I_0 = 1.0000
   504→    ///         exactly independent of feed noise.
   505→    function genesisIndex() external onlyAdmin {
   506→        if (!(indexValue == 1e18 && lastIndexUpdate == 0)) revert Err19();
   507→        if (!(liveWeights.updatedAt > 0)) revert Err01();
   508→        // Seed lastWeights from MASE registry
   509→        uint256 sum = 0;
   510→        for (uint256 i = 0; i < 7; i++) {
   511→            lastWeights[i] = liveWeights.weights[i];
   512→            lastPrices[i]  = _basePrice18(i);
   513→            sum += liveWeights.weights[i];
   514→        }
   515→        if (!(sum == 1e18)) revert Err44();
   516→        indexValue = 1e18;
   517→        chainLinkDivisor = 1e18;
   518→        lastIndexUpdate = block.timestamp;
   519→        emit GenesisVerified(INDEX_BASE_DENOMINATOR, block.timestamp);
   520→    }

   523→    function getIndex() external view returns (uint256) { return indexValue; }

   525→    /// @notice Listing 3 view: returns P_i,t / P_i,0 for each component.
   526→    function getRelatives(uint256[7] calldata currentPrices) external pure returns (uint256[7] memory rel) {
   527→        for (uint256 i = 0; i < 7; i++) {
   528→            rel[i] = (currentPrices[i] * 1e18) / _basePrice18(i);

   530→    }

   555→    /// @notice Listing 13 updateState — the CANONICAL state function (§14.6.2).
   556→    ///         Worse applicable condition binds; RECOVERY requires 48h
   557→    ///         confirmation; transitions to a more restrictive state are
   558→    ///         immediate, transitions to a less restrictive state require
   559→    ///         the RECOVERY_CONFIRMATION_PERIOD to elapse.
   560→    function updateState(uint256 rr, uint256 lcr) external onlyKeeper {
   561→        ProtocolState newState;
   562→        // Determine the target state based on RR and LCR
   563→        if (rr >= rrTarget && lcr >= lcrTarget) {
   564→            newState = ProtocolState.NORMAL;
   565→        } else if (rr >= rrStressFloor && lcr >= 0.90e18) {
   566→            newState = ProtocolState.CAUTION;
   567→        } else if (rr >= 1.02e18 && lcr >= 0.80e18) {
   568→            newState = ProtocolState.STRESS;
   569→        } else if (rr >= RR_HARD_FLOOR && lcr >= 0.70e18) {
   570→            newState = ProtocolState.DEFENSIVE;
   571→        } else if (rr < RR_HARD_FLOOR) {
   572→            newState = ProtocolState.EMERGENCY;
   573→        } else {
   574→            // RECOVERY: solvency ≥ hard floor but ratios not yet back to NORMAL.
   575→            newState = ProtocolState.RECOVERY;
   576→        }
   577→
   578→        if (newState != currentState) {
   579→            if (_isMoreRestrictive(newState, currentState)) {
   580→                _transitionTo(newState, true);
   581→            } else {
   582→                // Less restrictive → 48h confirmation
   583→                if (block.timestamp - stateEntryTime[currentState] >= RECOVERY_CONFIRMATION_PERIOD) {
   584→                    _transitionTo(newState, false);
   585→                }
   586→                // else: stay (newState discarded — not yet confirmed)
   587→            }
   588→        }
   589→    }
   590→
   591→    function _transitionTo(ProtocolState newState, bool immediate) internal {
   592→        ProtocolState oldState = currentState;
   593→        currentState = newState;
   594→        stateEntryTime[newState] = block.timestamp;
   595→        emit StateChanged(oldState, newState, immediate, block.timestamp);
   596→    }

   598→    /// @dev §21.2 Restrictiveness rank (RECOVERY sits between CAUTION and STRESS).
   599→    function _restrictiveness(ProtocolState s) internal pure returns (uint256) {
   600→        if (s == ProtocolState.NORMAL)    return 0; // S1
   601→        if (s == ProtocolState.CAUTION)   return 1; // S2
   602→        if (s == ProtocolState.RECOVERY)  return 2; // S6
   603→        if (s == ProtocolState.STRESS)    return 3; // S3
   604→        if (s == ProtocolState.DEFENSIVE) return 4; // S4
   605→        return 5;                                       // S5 EMERGENCY
   606→    }
   607→    function _isMoreRestrictive(ProtocolState a, ProtocolState b) internal pure returns (bool) {
   608→        return _restrictiveness(a) > _restrictiveness(b);
   609→    }

   703→    /// @notice Listing 14 registerParameter — Constitutional action (7/7 + 90d off-chain).
   704→    function registerParameter(
   705→        bytes32 parameterId,
   706→        ParameterLayer layer,
   707→        uint256 value,
   708→        uint256 minValue,
   709→        uint256 maxValue
   710→    ) external onlyConstitutionalCouncil {
   711→        if (!(minValue <= value && value <= maxValue)) revert Err51();
   712→        parameterRegistry[parameterId] = ParameterRecord({
   713→            layer: layer, value: value, minValue: minValue, maxValue: maxValue, registered: true
   714→        });

   718→    /// @notice Listing 14 proposeChange — only the authorized caller per layer:
   719→    ///         DAO for MONETARY, riskCouncil for RISK, emergencyCouncil for
   720→    ///         EMERGENCY, constitutionalCouncil for CONSTITUTIONAL.
   721→    function proposeChange(bytes32 parameterId, uint256 newValue) external {
   722→        ParameterRecord storage rec = parameterRegistry[parameterId];
   723→        if (!(rec.registered)) revert Err50();
   724→        if (!(newValue >= rec.minValue)) revert Err11();
   725→        if (!(newValue <= rec.maxValue)) revert Err05();
   726→        if (rec.layer == ParameterLayer.MONETARY)         if (!(msg.sender == dao)) revert Err36();
   727→        else if (rec.layer == ParameterLayer.RISK)        if (!(msg.sender == riskCouncil)) revert Err38();
   728→        else if (rec.layer == ParameterLayer.EMERGENCY)   if (!(msg.sender == emergencyCouncil)) revert Err37();
   729→        else /* CONSTITUTIONAL */                          if (!(msg.sender == constitutionalCouncil)) revert Err35();
   730→        proposals[parameterId] = Proposal({
   731→            value: newValue, proposer: msg.sender, timestamp: block.timestamp, executed: false
   732→        });
   733→        emit ParameterProposed(parameterId, newValue, block.timestamp);
   734→    }

   736→    /// @notice Listing 14 executeChange — anyone can call after the timelock
   737→    ///         elapses; the timelock + role check on propose are the protection.
   738→    function executeChange(bytes32 parameterId) external {
   739→        Proposal storage p = proposals[parameterId];
   740→        if (!(p.timestamp > 0)) revert Err27();
   741→        if (!(!p.executed)) revert Err07();
   742→        ParameterRecord storage rec = parameterRegistry[parameterId];
   743→        uint256 delay;
   744→        if (rec.layer == ParameterLayer.MONETARY)         delay = TIMELOCK_MONETARY;
   745→        else if (rec.layer == ParameterLayer.RISK)        delay = TIMELOCK_RISK;
   746→        else if (rec.layer == ParameterLayer.EMERGENCY)  delay = TIMELOCK_EMERGENCY;
   747→        else                                              delay = TIMELOCK_CONSTITUTIONAL;
   748→        if (!(block.timestamp >= p.timestamp + delay)) revert Err47();
   749→        if (!(p.value >= rec.minValue && p.value <= rec.maxValue)) revert Err40();
   750→        p.executed = true;
   751→        rec.value = p.value;
   752→        _applyParam(parameterId, p.value);
   753→        emit ParameterExecuted(parameterId, p.value, block.timestamp);
   754→    }

   756→    /// @notice Listing 14 cancelChange — only the proposer role (within timelock).
   757→    function cancelChange(bytes32 parameterId) external {
   758→        Proposal storage p = proposals[parameterId];
   759→        if (!(p.timestamp > 0 && !p.executed)) revert Err34();
   760→        ParameterRecord storage rec = parameterRegistry[parameterId];
   761→        if (rec.layer == ParameterLayer.MONETARY)        if (!(msg.sender == dao)) revert Err36();
   762→        else if (rec.layer == ParameterLayer.RISK)       if (!(msg.sender == riskCouncil)) revert Err38();
   763→        else if (rec.layer == ParameterLayer.EMERGENCY)  if (!(msg.sender == emergencyCouncil)) revert Err37();
   764→        else                                             if (!(msg.sender == constitutionalCouncil)) revert Err35();
   765→        delete proposals[parameterId];
   766→        emit ParameterVetoed(parameterId, block.timestamp);
   767→    }

   775→        else if (parameterId == PARAM_REDEEM_FEE_STRESS) redeemFeeStress = value;

   800→    uint256 public constant ORACLE_CONFIDENCE_MAX    = 0.01e18;  // §9.2.3 <1% of price
   801→    uint256 public constant ORACLE_DEVIATION_MAX_BPS = 250;      // §9.2.4 <2.5% (250 bps)

   803→    /// @notice Set oracle adapter for a source (0=Chainlink, 1=Pyth, 2=Chronicle).
   804→    function setOracleAdapter(uint8 source, address adapter) external onlyRole(ORACLE_ROLE) {
   805→        if (!(source <= 2)) revert Err10();
   806→        if (!(adapter != address(0))) revert Err55(); // H8 source-independence
   807→        if (source == 0)      chainlinkAdapter  = IOracleAdapter(adapter);
   808→        else if (source == 1) pythAdapter       = IOracleAdapter(adapter);
   809→        else                  chronicleAdapter = IOracleAdapter(adapter);

   813→    /// @notice §9.1-9.3 Consensus across all 3 adapters for a single pair.
   814→    /// @return finalPrice  USD-per-unit, 1e18 scale (0 if paused)
   815→    /// @return validCount  Number of valid feeds after all filters
   816→    /// @return method      0=paused, 1=average(2), 2=median(3)
   817→    /// @return paused_     True if fewer than 2 feeds are valid
   818→    /// @return prices      Raw prices per adapter (1e18) — index 0/1/2 = CL/Pyth/Cro
   819→    /// @return valid       Per-adapter validity flag after all filters
   820→    function getOracleConsensus(bytes32 pair) public view returns (
   821→        uint256 finalPrice,
   822→        uint8  validCount,
   823→        uint8  method,
   824→        bool   paused_,
   825→        uint256[3] memory prices,
   826→        bool[3] memory valid
   827→    ) {
   828→        IOracleAdapter[3] memory adapters;
   829→        adapters[0] = chainlinkAdapter;
   830→        adapters[1] = pythAdapter;
   831→        adapters[2] = chronicleAdapter;
   832→
   833→        uint256[3] memory raw;
   834→        uint256[3] memory ts;
   835→        uint256[3] memory conf;
   836→
   837→        for (uint8 i = 0; i < 3; i++) {
   838→            valid[i] = false;
   839→            if (address(adapters[i]) == address(0)) continue;
   840→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
   841→                if (p == 0) continue; // C6: reject zero-price feeds
   842→                raw[i]  = p;
   843→                ts[i]   = t;
   844→                conf[i] = c;
   845→                prices[i] = p;
   846→                // §9.2.1 Staleness ≤60s (also reject future timestamps)
   847→                if (t > block.timestamp) continue;
   848→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
   849→                // §9.2.3 Confidence < 1% of price
   850→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
   851→                valid[i] = true;
   852→            } catch { /* leave valid[i] = false */ }
   853→        }
   854→
   855→        validCount = 0;
   856→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
   857→
   858→        if (validCount < 2) {
   859→            paused_ = true; method = 0;
   860→            return (finalPrice, validCount, method, paused_, prices, valid);
   861→        }
   862→
   863→        // Sort the valid prices and pick median (n=3) or average (n=2)
   864→        uint256[3] memory sorted;
   865→        uint8 n = 0;
   866→        for (uint8 i = 0; i < 3; i++) if (valid[i]) { sorted[n] = raw[i]; n++; }
   867→        _insertionSort(sorted, n);
   868→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
   869→
   870→        // §9.2.4 Deviation < 2.5% from median → discard
   871→        for (uint8 i = 0; i < 3; i++) {
   872→            if (valid[i]) {
   873→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
   874→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
   875→                    valid[i] = false;
   876→                    validCount--;
   877→                }
   878→            }
   879→        }
   880→
   881→        if (validCount < 2) {
   882→            paused_ = true; method = 0;
   883→            return (finalPrice, validCount, method, paused_, prices, valid);
   884→        }
   885→
   886→        paused_ = false;
   887→        if (validCount == 3) {
   888→            finalPrice = med;
   889→            method     = 2; // median(3)
   890→        } else {
   891→            uint256 sum = 0; uint8 cnt = 0;
   892→            for (uint8 i = 0; i < 3; i++) if (valid[i]) { sum += raw[i]; cnt++; }
   893→            finalPrice = sum / cnt;
   894→            method     = 1; // average(2)
   895→        }
   896→    }

   954→    /// @notice §5 + §19.3.2 Reserve NAV (USD, 1e18) — applies per-asset
   955→    ///         haircuts pulled from the registry. Frozen/delisted assets
   956→    ///         contribute 0. If the registry is unset, falls back to gross.
   957→    function getNAV() public view returns (uint256) {
   958→        if (address(assetRegistry) == address(0)) return getReserveNavUsd();
   959→        bytes32[7] memory codes;
   960→        codes[0] = bytes32("USD"); codes[1] = bytes32("EUR");
   961→        codes[2] = bytes32("JPY"); codes[3] = bytes32("GBP");
   962→        codes[4] = bytes32("CNY"); codes[5] = bytes32("CHF");
   963→        codes[6] = bytes32("XAU");
   964→        uint256 nav = 0;
   965→        for (uint8 i = 0; i < 7; i++) {
   966→            uint256 held = reserveHeldUsd[Component(i)];
   967→            if (held == 0) continue;
   968→            try IAssetRegistry(assetRegistry).getAsset(codes[i]) returns (
   969→                address, uint256 haircut, uint8 state, bytes32
   970→            ) {
   971→                if (state == 2 || state == 3) continue; // frozen / delisted → 0 NAV
   972→                nav += held * (1e18 - haircut) / 1e18;
   973→            } catch { nav += held; }
   974→        }
   975→        return nav;
   976→    }

  1037→    // §3.4.2 + §12.1  Mint — priced against P_MTQ (I5) with canonical throttle
  1038→    //   Pull USDC (6 dec), fee = mintFee, MTQ minted = netUsd18 × 1e18 / P_MTQ,
  1039→    //   throttled by mintThrottle(currentState). REVERT if !mintingAllowed.
  1040→    // ============================================================
  1041→    function mint(uint256 usdcAmount) external nonReentrant whenNotPaused returns (uint256 minted) {
  1042→        if (!(usdcAmount > 0)) revert Err56();
  1043→        if (!(mintingAllowed(currentState))) revert Err25();
  1044→
  1045→        uint256 price = getMTQPriceWithGuard();
  1046→
  1047→        // Pull USDC from minter to reserve vault (6 decimals).
  1048→        if (!(usdc.transferFrom(msg.sender, reserveVault, usdcAmount))) revert Err03();
  1049→
  1050→        // §12.1 fee = mintFee (1e18 fraction). E.g. 0.001e18 = 0.10%.
  1051→        uint256 feeUsd = usdcAmount * mintFee / 1e18;
  1052→        uint256 netUsd = usdcAmount - feeUsd;
  1053→        // Scale 6-dec USDC → 18-dec USD value, then divide by 18-dec price.
  1054→        uint256 netUsd18  = netUsd * 1e12;
  1055→        uint256 grossMint = netUsd18 * 1e18 / price;
  1056→
  1057→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25)
  1058→        uint256 throttle = mintThrottle(currentState);
  1059→        minted = grossMint * throttle / 1e18;
  1060→
  1061→        // Checks-Effects-Interactions (H1): state updated BEFORE any external call.
  1062→        _mint(msg.sender, minted);
  1063→        emit Mint(msg.sender, usdcAmount, feeUsd, minted, price);
  1064→    }

  1067→    // §19.3.2 + I6  Redeem — NAV-based (NOT P_MTQ)
  1068→    //   Burn MTQ, grossUsd = mtqAmount × NAVperToken (NOT × P_MTQ),
  1069→    //   fee = redeemFee(currentState), net → USDC 6 dec from reserve vault,
  1070→    //   decrement reserveHeldUsd proportionally (basket broken).
  1071→    //   REVERT if !redemptionAllowed(currentState). Apply price guard (H4).
  1072→    // ============================================================
  1073→    function redeem(uint256 mtqAmount) external nonReentrant whenNotPaused returns (uint256 usdcOut) {
  1074→        if (!(mtqAmount > 0)) revert Err56();
  1075→        if (!(balanceOf[msg.sender] >= mtqAmount)) revert Err22();
  1076→        if (!(redemptionAllowed(currentState))) revert Err43();
  1077→
  1078→        // H4: apply price guard even on redeem (rejects if outside [0.50, 2.00]).
  1079→        uint256 price = getMTQPriceWithGuard();
  1080→        // §19.3.2 NAV-per-token (NOT P_MTQ)
  1081→        uint256 navPerToken = getNAVperToken();
  1082→        if (!(navPerToken < type(uint256).max)) revert Err26();
  1083→
  1084→        // §14.6.3 fee by state (1e18 fraction)
  1085→        uint256 feeFraction = redeemFee(currentState);
  1086→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
  1087→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
  1088→        uint256 netUsd18   = grossUsd18 - feeUsd18;
  1089→        usdcOut            = netUsd18 / 1e12; // 18-dec → 6-dec USDC
  1090→
  1091→        // Checks-Effects-Interactions (H2): burn + decrement BEFORE external call.
  1092→        _burn(msg.sender, mtqAmount);
  1093→        // Decrement reserveHeldUsd proportionally (basket broken on redeem).
  1094→        // Use gross USD value across all components so the share is exact.
  1095→        uint256 totalReserve = getReserveNavUsd();
  1096→        if (totalReserve > 0 && grossUsd18 > 0) {
  1097→            for (uint8 i = 0; i < 7; i++) {
  1098→                uint256 held = reserveHeldUsd[Component(i)];
  1099→                if (held == 0) continue;
  1100→                uint256 share = held * grossUsd18 / totalReserve;
  1101→                if (share > held) share = held; // safety
  1102→                reserveHeldUsd[Component(i)] = held - share;
  1103→            }
  1104→        }
  1105→        // Update reserveState mirror
  1106→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1107→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1108→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1109→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1110→        reserveState.liquidValue = reserveState.fiatValue;
  1111→
  1112→        // External call LAST (after all state updates).
  1113→        if (!(usdc.transferFrom(reserveVault, msg.sender, usdcOut))) revert Err04();
  1114→        emit Redeem(msg.sender, mtqAmount, feeUsd18 / 1e12, usdcOut, navPerToken);
  1115→        // silence unused var warning
  1116→        if (price == 0) revert Err59();
  1117→    }

  1120→    // §10  MARP Rebalancing Execution — keeper-callable
  1121→    //   For each trade: validates it moves toward the latest committed
  1122→    //   lastWeights (executionWeight) within a 5% tolerance, enforces 24h
  1123→    //   direction lock (with RR<1.05 solvency override — C4 fix), validates
  1124→    //   total daily trade USD ≤ 5% of NAV, caps trades.length at 7 (H3 fix),
  1125→    //   updates reserve holdings (on-chain mirror).
  1126→    // ============================================================
  1127→
  1128→    struct RebalanceTrade {
  1129→        Component component;
  1130→        int256    direction; // +1 buy, -1 sell, 0 hold
  1131→        uint256   tradeUsd;  // USD value, 1e18 scale
  1132→        uint256   level;     // 1-6 (MARP decision level)
  1133→    }
  1134→
  1135→    uint256 public constant DIRECTION_LOCK_HOURS  = 24 hours;
  1136→    uint256 public constant MAX_DAILY_TURNOVER    = 0.05e18;
  1137→    uint256 public constant REBALANCE_TOLERANCE    = 0.05e18;
  1138→
  1139→    mapping(Component => int256)  public lastDirection;
  1140→    mapping(Component => uint256) public lastRebalanceAt;
  1141→    uint256 public dailyTradeUsd;
  1142→    uint256 public dailyTradeResetAt;
  1143→
  1144→    event RebalanceExecuted(RebalanceTrade[] trades, uint256 timestamp);
  1145→
  1146→    function executeRebalance(RebalanceTrade[] calldata trades) external nonReentrant onlyKeeper whenNotPaused {
  1147→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1148→        if (!(trades.length <= 7)) revert Err48(); // H3 cap
  1149→
  1150→        // Reset 24h daily turnover window if elapsed
  1151→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1152→            dailyTradeUsd    = 0;
  1153→            dailyTradeResetAt = block.timestamp;
  1154→        }
  1155→
  1156→        uint256 nav = getReserveNavUsd();
  1157→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1158→        uint256 rr = getReserveRatio();
  1159→        bool solvencyOverride = (rr < rrStressFloor); // §11.5.3 / C4 fix
  1160→
  1161→        for (uint256 i = 0; i < trades.length; i++) {
  1162→            RebalanceTrade calldata t = trades[i];
  1163→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1164→            if (t.direction == 0 || t.tradeUsd == 0) continue; // hold
  1165→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1166→
  1167→            // §10 24h direction lock — opposite direction blocked within 24h,
  1168→            // UNLESS RR<1.05 (solvency override — C4 fix).
  1169→            int256 prevDir = lastDirection[t.component];
  1170→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1171→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1172→            }
  1173→
  1174→            // Validate trade moves toward execution weight (within 5% tolerance)
  1175→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1176→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1177→            if (t.direction == 1) {
  1178→                if (!(currentW < execW)) revert Err14();
  1179→            } else {
  1180→                if (!(currentW > execW)) revert Err46();
  1181→            }
  1182→
  1183→            // New weight after trade must be within 5pp (absolute) of execW
  1184→            uint256 newHeldUsd = (t.direction == 1)
  1185→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1186→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1187→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1188→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1189→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1190→            // Solvency override relaxes overshoot tolerance by 2x during stress
  1191→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1192→            if (!(absDiff <= tolerance)) revert Err49();
  1193→
  1194→            // §10 daily turnover cap (5% of NAV)
  1195→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1196→            dailyTradeUsd += t.tradeUsd;
  1197→
  1198→            // Update reserve mirror (checks-effects-interactions)
  1199→            if (t.direction == 1) {
  1200→                reserveHeldUsd[t.component] += t.tradeUsd;
  1201→            } else {
  1202→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1203→            }
  1204→            lastDirection[t.component]   = t.direction;
  1205→            lastRebalanceAt[t.component] = block.timestamp;
  1206→        }
  1207→
  1208→        // Refresh reserveState aggregate mirror after rebalancing
  1209→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1210→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1211→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1212→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1213→        reserveState.liquidValue = reserveState.fiatValue;
  1214→
  1215→        emit RebalanceExecuted(trades, block.timestamp);
  1216→    }

  1377→    /// @dev In-place insertion sort of sorted[0..n), ascending.
  1378→    function _insertionSort(uint256[3] memory arr, uint8 n) internal pure {
  1379→        for (uint8 i = 1; i < n; i++) {
  1380→            uint256 key = arr[i];
  1381→            uint8 j = i;
  1382→            while (j > 0 && arr[j - 1] > key) {
  1383→                arr[j] = arr[j - 1];
  1384→                j--;
  1385→            }
  1386→            arr[j] = key;
  1387→        }
  1388→    }
