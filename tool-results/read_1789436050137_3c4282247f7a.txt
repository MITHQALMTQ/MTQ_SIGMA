   465→    /// @notice Listing 2 (§7.7) submitTargetWeights — MASE verification + smoothing.
   466→    /// @dev   Only the MASE submitter (KEEPER_ROLE). Enforces sum=1, positivity,
   467→    ///        admissibility envelopes, per-component velocity, then applies
   468→    ///        stress-adaptive smoothing rho 0.50 (normal) / 0.75 (stress).
   469→    function submitTargetWeights(
   470→        uint256[7] calldata target,
   471→        bytes32 methodologyVersion,
   472→        bytes32 dataVersion
   473→    ) external onlyKeeper {
   474→        // 1. Sum-to-one and non-negativity
   475→        uint256 sum = 0;
   476→        for (uint256 i = 0; i < 7; i++) {
   477→            if (!(target[i] > 0)) revert Err62();
   478→            sum += target[i];
   479→        }
   480→        if (!(sum == 1e18)) revert Err54();
   481→        // 2. Constitutional admissibility envelopes
   482→        for (uint256 i = 0; i < 7; i++) {
   483→            if (!(target[i] >= LOWER_BOUND[i])) revert Err12();
   484→            if (!(target[i] <= UPPER_BOUND[i])) revert Err06();
   485→        }
   486→        // 3. Weight-velocity constraint vs previous live weights
   487→        for (uint256 i = 0; i < 7; i++) {
   488→            uint256 diff = target[i] > liveWeights.weights[i]
   489→                ? target[i] - liveWeights.weights[i]
   490→                : liveWeights.weights[i] - target[i];
   491→            if (!(diff <= MAX_VELOCITY[i])) revert Err52();
   492→        }
   493→        // 4. Stress-adaptive smoothing: W_smooth = rho * W_prev + (1 - rho) * W_target
   494→        uint256 rho = crisisFlag ? smoothingRhoStress : smoothingRhoNormal;
   495→        uint256[7] memory smoothed;
   496→        for (uint256 i = 0; i < 7; i++) {
   497→            smoothed[i] = (rho * liveWeights.weights[i] + (1e18 - rho) * target[i]) / 1e18;
   498→        }
   499→        // 5. Commit registry state
   500→        for (uint256 i = 0; i < 7; i++) liveWeights.weights[i] = smoothed[i];
   501→        liveWeights.targetWeights    = target;
   502→        liveWeights.methodologyVersion = methodologyVersion;
   503→        liveWeights.dataVersion       = dataVersion;
   504→        liveWeights.updatedAt         = block.timestamp;
   505→        emit WeightsAccepted(smoothed, target, methodologyVersion, dataVersion, block.timestamp);
   506→    }

   541→    /// @dev §9.8 returns the base-fixing price for component i (1e18 scale).
   542→    ///      USD → 1.0; EUR → 1.05; JPY → 0.0067; GBP → 1.25;
   543→    ///      CNY → 0.14; CHF → 1.13 (FIXED); Gold → 2500.
   544→    function _basePrice18(uint256 i) internal pure returns (uint256) {
   545→        if (i == 0) return 1e18;             // USD = 1 USD
   546→        if (i == 1) return BASE_EUR_USD;     // 1.05
   547→        if (i == 2) return BASE_JPY_USD;     // 0.0067
   548→        if (i == 3) return BASE_GBP_USD;     // 1.25
   549→        if (i == 4) return BASE_CNY_USD;     // 0.14
   550→        if (i == 5) return BASE_CHF_USD;     // 1.13
   551→        return BASE_GOLD_USD;                 // 2500
   552→    }

   554→    /// @notice Listing 3 advanceIndex — §9.2 authoritative recursion.
   555→    ///         I_t = I_{t-1} × Σ_i W_{i,t-1} × (P_{i,t} / P_{i,t-1})
   556→    ///         Prices are passed in by the keeper (1e18 scale, USD-quoted).
   557→    function advanceIndex(uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   558→        if (!(indexValue > 0)) revert Err20();
   559→
   560→        // 1. Period return weighted by W_{i,t-1} in force over the period (COO-16).
   561→        uint256 growth = 0;
   562→        for (uint256 i = 0; i < 7; i++) {
   563→            if (!(currentPrices[i] > 0 && lastPrices[i] > 0)) revert Err09();
   564→            growth += (lastWeights[i] * currentPrices[i]) / lastPrices[i];
   565→        }
   566→        indexValue = (indexValue * growth) / 1e18;
   567→
   568→        // 2. Commit the period end-state (prices).
   569→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   570→        lastIndexUpdate = block.timestamp;
   571→        emit IndexAdvanced(indexValue, block.timestamp);
   572→    }

   574→    /// @notice Listing 3 commitWeights — chain-link adjustment at weight change.
   575→    ///         Computes D_t = B_t^- / B_t^+ (Master §9.3), updates G_t, updates
   576→    ///         lastPrices and lastWeights. I_t is UNCHANGED by the weight change
   577→    ///         itself — ZERO artificial return (the recursion is continuous).
   578→    function commitWeights(uint256[7] calldata newWeights, uint256[7] calldata currentPrices) external onlyKeeper whenNotPaused {
   579→        if (!(indexValue > 0)) revert Err20();
   580→
   581→        // 1. B_t^- and B_t^+ are base-relative aggregates measured against
   582→        //    the genesis fixings (§9.3, MS §75).
   583→        uint256 bMinus = 0;
   584→        uint256 bPlus  = 0;
   585→        bool    changed = false;
   586→        for (uint256 i = 0; i < 7; i++) {
   587→            if (!(currentPrices[i] > 0)) revert Err09();
   588→            uint256 rel = (currentPrices[i] * 1e18) / _basePrice18(i); // P_{i,t} / P_{i,0}
   589→            bMinus += (lastWeights[i] * rel) / 1e18;
   590→            bPlus  += (newWeights[i]    * rel) / 1e18;
   591→            if (newWeights[i] != lastWeights[i]) changed = true;
   592→        }
   593→
   594→        // 2. Apply divisor D_t; record into G_t (cumulative). I_t unchanged.
   595→        if (changed) {
   596→            if (!(bPlus > 0)) revert Err57();
   597→            uint256 newD = (bMinus * 1e18) / bPlus;
   598→            chainLinkDivisor = (chainLinkDivisor * newD) / 1e18;
   599→            for (uint256 i = 0; i < 7; i++) lastWeights[i] = newWeights[i];
   600→            emit ChainLinkAdjusted(newD, bMinus, bPlus, block.timestamp);
   601→        }
   602→
   603→        // 3. Commit the period end-state prices too.
   604→        for (uint256 i = 0; i < 7; i++) lastPrices[i] = currentPrices[i];
   605→        emit WeightsCommitted(newWeights, chainLinkDivisor, block.timestamp);
   606→    }

   608→    /// @notice Listing 3 genesis — one-time seed of I_0 = 1.0000 (ADMIN_ROLE).
   609→    ///         Prices are seeded from immutable base fixings, so I_0 = 1.0000
   610→    ///         exactly independent of feed noise.
   611→    function genesisIndex() external onlyAdmin {
   612→        if (!(indexValue == 1e18 && lastIndexUpdate == 0)) revert Err19();
   613→        if (!(liveWeights.updatedAt > 0)) revert Err01();
   614→        // Seed lastWeights from MASE registry
   615→        uint256 sum = 0;
   616→        for (uint256 i = 0; i < 7; i++) {
   617→            lastWeights[i] = liveWeights.weights[i];
   618→            lastPrices[i]  = _basePrice18(i);
   619→            sum += liveWeights.weights[i];
   620→        }
   621→        if (!(sum == 1e18)) revert Err44();
   622→        indexValue = 1e18;
   623→        chainLinkDivisor = 1e18;
   624→        lastIndexUpdate = block.timestamp;
   625→        emit GenesisVerified(INDEX_BASE_DENOMINATOR, block.timestamp);
   626→    }

   661→    /// @notice Listing 13 updateState — the CANONICAL state function (§14.6.2).
   662→    ///         Worse applicable condition binds; RECOVERY requires 48h
   663→    ///         confirmation; transitions to a more restrictive state are
   664→    ///         immediate, transitions to a less restrictive state require
   665→    ///         the RECOVERY_CONFIRMATION_PERIOD to elapse.
   666→    function updateState(uint256 rr, uint256 lcr) external onlyKeeper {
   667→        ProtocolState newState;
   668→        // Determine the target state based on RR and LCR
   669→        if (rr >= rrTarget && lcr >= lcrTarget) {
   670→            newState = ProtocolState.NORMAL;
   671→        } else if (rr >= rrStressFloor && lcr >= 0.90e18) {
   672→            newState = ProtocolState.CAUTION;
   673→        } else if (rr >= 1.02e18 && lcr >= 0.80e18) {
   674→            newState = ProtocolState.STRESS;
   675→        } else if (rr >= RR_HARD_FLOOR && lcr >= 0.70e18) {
   676→            newState = ProtocolState.DEFENSIVE;
   677→        } else if (rr < RR_HARD_FLOOR) {
   678→            newState = ProtocolState.EMERGENCY;
   679→        } else {
   680→            // RECOVERY: solvency ≥ hard floor but ratios not yet back to NORMAL.
   681→            newState = ProtocolState.RECOVERY;
   682→        }
   683→
   684→        if (newState != currentState) {
   685→            if (_isMoreRestrictive(newState, currentState)) {
   686→                _transitionTo(newState, true);
   687→            } else {
   688→                // Less restrictive → 48h confirmation
   689→                if (block.timestamp - stateEntryTime[currentState] >= RECOVERY_CONFIRMATION_PERIOD) {
   690→                    _transitionTo(newState, false);
   691→                }
   692→                // else: stay (newState discarded — not yet confirmed)
   693→            }
   694→        }
   695→    }

   697→    function _transitionTo(ProtocolState newState, bool immediate) internal {
   698→        ProtocolState oldState = currentState;
   699→        currentState = newState;
   700→        stateEntryTime[newState] = block.timestamp;
   701→        emit StateChanged(oldState, newState, immediate, block.timestamp);
   702→    }

   707→        if (s == ProtocolState.CAUTION)   return 1; // S2

   711→        return 5;                                       // S5 EMERGENCY

   838→        });
   839→        emit ParameterProposed(parameterId, newValue, block.timestamp);
   840→    }

   842→    /// @notice Listing 14 executeChange — anyone can call after the timelock
   843→    ///         elapses; the timelock + role check on propose are the protection.
   844→    function executeChange(bytes32 parameterId) external {
   845→        Proposal storage p = proposals[parameterId];
   846→        if (!(p.timestamp > 0)) revert Err27();
   847→        if (!(!p.executed)) revert Err07();
   848→        ParameterRecord storage rec = parameterRegistry[parameterId];
   849→        uint256 delay;
   850→        if (rec.layer == ParameterLayer.MONETARY)         delay = TIMELOCK_MONETARY;
   851→        else if (rec.layer == ParameterLayer.RISK)        delay = TIMELOCK_RISK;
   852→        else if (rec.layer == ParameterLayer.EMERGENCY)  delay = TIMELOCK_EMERGENCY;
   853→        else                                              delay = TIMELOCK_CONSTITUTIONAL;
   854→        if (!(block.timestamp >= p.timestamp + delay)) revert Err47();
   855→        if (!(p.value >= rec.minValue && p.value <= rec.maxValue)) revert Err40();
   856→        p.executed = true;
   857→        rec.value = p.value;
   858→        _applyParam(parameterId, p.value);
   859→        emit ParameterExecuted(parameterId, p.value, block.timestamp);
   860→    }

   862→    /// @notice Listing 14 cancelChange — only the proposer role (within timelock).
   863→    function cancelChange(bytes32 parameterId) external {
   864→        Proposal storage p = proposals[parameterId];
   865→        if (!(p.timestamp > 0 && !p.executed)) revert Err34();
   866→        ParameterRecord storage rec = parameterRegistry[parameterId];
   867→        if (rec.layer == ParameterLayer.MONETARY)        if (!(msg.sender == dao)) revert Err36();
   868→        else if (rec.layer == ParameterLayer.RISK)       if (!(msg.sender == riskCouncil)) revert Err38();
   869→        else if (rec.layer == ParameterLayer.EMERGENCY)  if (!(msg.sender == emergencyCouncil)) revert Err37();
   870→        else                                             if (!(msg.sender == constitutionalCouncil)) revert Err35();
   871→        delete proposals[parameterId];
   872→        emit ParameterVetoed(parameterId, block.timestamp);
   873→    }

   919→    /// @notice §9.1-9.3 Consensus across all 3 adapters for a single pair.
   920→    /// @return finalPrice  USD-per-unit, 1e18 scale (0 if paused)
   921→    /// @return validCount  Number of valid feeds after all filters
   922→    /// @return method      0=paused, 1=average(2), 2=median(3)
   923→    /// @return paused_     True if fewer than 2 feeds are valid
   924→    /// @return prices      Raw prices per adapter (1e18) — index 0/1/2 = CL/Pyth/Cro
   925→    /// @return valid       Per-adapter validity flag after all filters
   926→    function getOracleConsensus(bytes32 pair) public view returns (
   927→        uint256 finalPrice,
   928→        uint8  validCount,
   929→        uint8  method,
   930→        bool   paused_,
   931→        uint256[3] memory prices,
   932→        bool[3] memory valid
   933→    ) {
   934→        IOracleAdapter[3] memory adapters;
   935→        adapters[0] = chainlinkAdapter;
   936→        adapters[1] = pythAdapter;
   937→        adapters[2] = chronicleAdapter;
   938→
   939→        uint256[3] memory raw;
   940→        uint256[3] memory ts;
   941→        uint256[3] memory conf;
   942→
   943→        for (uint8 i = 0; i < 3; i++) {
   944→            valid[i] = false;
   945→            if (address(adapters[i]) == address(0)) continue;
   946→            try adapters[i].getPrice(pair) returns (uint256 p, uint256 t, uint256 c) {
   947→                if (p == 0) continue; // C6: reject zero-price feeds
   948→                raw[i]  = p;
   949→                ts[i]   = t;
   950→                conf[i] = c;
   951→                prices[i] = p;
   952→                // §9.2.1 Staleness ≤60s (also reject future timestamps)
   953→                if (t > block.timestamp) continue;
   954→                if (block.timestamp - t > ORACLE_STALENESS_SEC) continue;
   955→                // §9.2.3 Confidence < 1% of price
   956→                if (conf[i] >= ORACLE_CONFIDENCE_MAX) continue;
   957→                valid[i] = true;
   958→            } catch { /* leave valid[i] = false */ }
   959→        }
   960→
   961→        validCount = 0;
   962→        for (uint8 i = 0; i < 3; i++) if (valid[i]) validCount++;
   963→
   964→        if (validCount < 2) {
   965→            paused_ = true; method = 0;
   966→            return (finalPrice, validCount, method, paused_, prices, valid);
   967→        }
   968→
   969→        // Sort the valid prices and pick median (n=3) or average (n=2)
   970→        uint256[3] memory sorted;
   971→        uint8 n = 0;
   972→        for (uint8 i = 0; i < 3; i++) if (valid[i]) { sorted[n] = raw[i]; n++; }
   973→        _insertionSort(sorted, n);
   974→        uint256 med = (n == 3) ? sorted[1] : (n == 2 ? (sorted[0] + sorted[1]) / 2 : sorted[0]);
   975→
   976→        // §9.2.4 Deviation < 2.5% from median → discard
   977→        for (uint8 i = 0; i < 3; i++) {
   978→            if (valid[i]) {
   979→                uint256 d = (raw[i] > med) ? raw[i] - med : med - raw[i];
   980→                if (d * 10000 > med * ORACLE_DEVIATION_MAX_BPS) {
   981→                    valid[i] = false;
   982→                    validCount--;
   983→                }
   984→            }
   985→        }
   986→
   987→        if (validCount < 2) {
   988→            paused_ = true; method = 0;
   989→            return (finalPrice, validCount, method, paused_, prices, valid);
   990→        }
   991→
   992→        paused_ = false;
   993→        if (validCount == 3) {
   994→            finalPrice = med;
   995→            method     = 2; // median(3)
   996→        } else {
   997→            uint256 sum = 0; uint8 cnt = 0;
   998→            for (uint8 i = 0; i < 3; i++) if (valid[i]) { sum += raw[i]; cnt++; }
   999→            finalPrice = sum / cnt;
  1000→            method     = 1; // average(2)
  1001→        }
  1002→    }

  1060→    /// @notice §5 + §19.3.2 Reserve NAV (USD, 1e18) — applies per-asset
  1061→    ///         haircuts pulled from the registry. Frozen/delisted assets
  1062→    ///         contribute 0. If the registry is unset, falls back to gross.
  1063→    function getNAV() public view returns (uint256) {
  1064→        if (address(assetRegistry) == address(0)) return getReserveNavUsd();
  1065→        bytes32[7] memory codes;
  1066→        codes[0] = bytes32("USD"); codes[1] = bytes32("EUR");
  1067→        codes[2] = bytes32("JPY"); codes[3] = bytes32("GBP");
  1068→        codes[4] = bytes32("CNY"); codes[5] = bytes32("CHF");
  1069→        codes[6] = bytes32("XAU");
  1070→        uint256 nav = 0;
  1071→        for (uint8 i = 0; i < 7; i++) {
  1072→            uint256 held = reserveHeldUsd[Component(i)];
  1073→            if (held == 0) continue;
  1074→            try IAssetRegistry(assetRegistry).getAsset(codes[i]) returns (
  1075→                address, uint256 haircut, uint8 state, bytes32
  1076→            ) {
  1077→                if (state == 2 || state == 3) continue; // frozen / delisted → 0 NAV
  1078→                nav += held * (1e18 - haircut) / 1e18;
  1079→            } catch { nav += held; }
  1080→        }
  1081→        return nav;
  1082→    }

  1143→    // §3.4.2 + §12.1  Mint — priced against P_MTQ (I5) with canonical throttle
  1144→    //   Pull USDC (6 dec), fee = mintFee, MTQ minted = netUsd18 × 1e18 / P_MTQ,
  1145→    //   throttled by mintThrottle(currentState). REVERT if !mintingAllowed.
  1146→    // ============================================================
  1147→    function mint(uint256 usdcAmount) external nonReentrant whenNotPaused returns (uint256 minted) {
  1148→        if (!(usdcAmount > 0)) revert Err56();
  1149→        if (!(mintingAllowed(currentState))) revert Err25();
  1150→
  1151→        uint256 price = getMTQPriceWithGuard();
  1152→
  1153→        // Pull USDC from minter to reserve vault (6 decimals).
  1154→        if (!(usdc.transferFrom(msg.sender, reserveVault, usdcAmount))) revert Err03();
  1155→
  1156→        // §12.1 fee = mintFee (1e18 fraction). E.g. 0.001e18 = 0.10%.
  1157→        uint256 feeUsd = usdcAmount * mintFee / 1e18;
  1158→        uint256 netUsd = usdcAmount - feeUsd;
  1159→        // Scale 6-dec USDC → 18-dec USD value, then divide by 18-dec price.
  1160→        uint256 netUsd18  = netUsd * 1e12;
  1161→        uint256 grossMint = netUsd18 * 1e18 / price;
  1162→
  1163→        // §21.4 canonical throttle (NORMAL 1.0, CAUTION 0.5, RECOVERY 0.25)
  1164→        uint256 throttle = mintThrottle(currentState);
  1165→        minted = grossMint * throttle / 1e18;
  1166→
  1167→        // Checks-Effects-Interactions (H1): state updated BEFORE any external call.
  1168→        _mint(msg.sender, minted);
  1169→        emit Mint(msg.sender, usdcAmount, feeUsd, minted, price);
  1170→    }

  1173→    // §19.3.2 + I6  Redeem — NAV-based (NOT P_MTQ)
  1174→    //   Burn MTQ, grossUsd = mtqAmount × NAVperToken (NOT × P_MTQ),
  1175→    //   fee = redeemFee(currentState), net → USDC 6 dec from reserve vault,
  1176→    //   decrement reserveHeldUsd proportionally (basket broken).
  1177→    //   REVERT if !redemptionAllowed(currentState). Apply price guard (H4).
  1178→    // ============================================================
  1179→    function redeem(uint256 mtqAmount) external nonReentrant whenNotPaused returns (uint256 usdcOut) {
  1180→        if (!(mtqAmount > 0)) revert Err56();
  1181→        if (!(balanceOf[msg.sender] >= mtqAmount)) revert Err22();
  1182→        if (!(redemptionAllowed(currentState))) revert Err43();
  1183→
  1184→        // H4: apply price guard even on redeem (rejects if outside [0.50, 2.00]).
  1185→        uint256 price = getMTQPriceWithGuard();
  1186→        // §19.3.2 NAV-per-token (NOT P_MTQ)
  1187→        uint256 navPerToken = getNAVperToken();
  1188→        if (!(navPerToken < type(uint256).max)) revert Err26();
  1189→
  1190→        // §14.6.3 fee by state (1e18 fraction)
  1191→        uint256 feeFraction = redeemFee(currentState);
  1192→        uint256 grossUsd18 = mtqAmount * navPerToken / 1e18;
  1193→        uint256 feeUsd18   = grossUsd18 * feeFraction / 1e18;
  1194→        uint256 netUsd18   = grossUsd18 - feeUsd18;
  1195→        usdcOut            = netUsd18 / 1e12; // 18-dec → 6-dec USDC
  1196→
  1197→        // Checks-Effects-Interactions (H2): burn + decrement BEFORE external call.
  1198→        _burn(msg.sender, mtqAmount);
  1199→        // Decrement reserveHeldUsd proportionally (basket broken on redeem).
  1200→        // Use gross USD value across all components so the share is exact.
  1201→        uint256 totalReserve = getReserveNavUsd();
  1202→        if (totalReserve > 0 && grossUsd18 > 0) {
  1203→            for (uint8 i = 0; i < 7; i++) {
  1204→                uint256 held = reserveHeldUsd[Component(i)];
  1205→                if (held == 0) continue;
  1206→                uint256 share = held * grossUsd18 / totalReserve;
  1207→                if (share > held) share = held; // safety
  1208→                reserveHeldUsd[Component(i)] = held - share;
  1209→            }
  1210→        }
  1211→        // Update reserveState mirror
  1212→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1213→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1214→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1215→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1216→        reserveState.liquidValue = reserveState.fiatValue;
  1217→
  1218→        // External call LAST (after all state updates).
  1219→        if (!(usdc.transferFrom(reserveVault, msg.sender, usdcOut))) revert Err04();
  1220→        emit Redeem(msg.sender, mtqAmount, feeUsd18 / 1e12, usdcOut, navPerToken);
  1221→        // silence unused var warning
  1222→        if (price == 0) revert Err59();
  1223→    }

  1241→    uint256 public constant DIRECTION_LOCK_HOURS  = 24 hours;
  1242→    uint256 public constant MAX_DAILY_TURNOVER    = 0.05e18;
  1243→    uint256 public constant REBALANCE_TOLERANCE    = 0.05e18;
  1244→
  1245→    mapping(Component => int256)  public lastDirection;
  1246→    mapping(Component => uint256) public lastRebalanceAt;
  1247→    uint256 public dailyTradeUsd;
  1248→    uint256 public dailyTradeResetAt;

  1250→    event RebalanceExecuted(RebalanceTrade[] trades, uint256 timestamp);

  1252→    function executeRebalance(RebalanceTrade[] memory trades) external nonReentrant onlyKeeper whenNotPaused {
  1253→        if (!(liveWeights.updatedAt > 0)) revert Err28();
  1254→        if (!(trades.length <= 7)) revert Err48(); // H3 cap
  1255→
  1256→        // Reset 24h daily turnover window if elapsed
  1257→        if (block.timestamp >= dailyTradeResetAt + DIRECTION_LOCK_HOURS) {
  1258→            dailyTradeUsd    = 0;
  1259→            dailyTradeResetAt = block.timestamp;
  1260→        }
  1261→
  1262→        uint256 nav = getReserveNavUsd();
  1263→        uint256 maxDaily = nav * MAX_DAILY_TURNOVER / 1e18;
  1264→        uint256 rr = getReserveRatio();
  1265→        bool solvencyOverride = (rr < rrStressFloor); // §11.5.3 / C4 fix
  1266→
  1267→        for (uint256 i = 0; i < trades.length; i++) {
  1268→            RebalanceTrade memory t = trades[i];
  1269→            if (!(t.level >= 1 && t.level <= 6)) revert Err24();
  1270→            if (t.direction == 0 || t.tradeUsd == 0) continue; // hold
  1271→            if (!(t.direction == 1 || t.direction == -1)) revert Err08();
  1272→
  1273→            // §10 24h direction lock — opposite direction blocked within 24h,
  1274→            // UNLESS RR<1.05 (solvency override — C4 fix).
  1275→            int256 prevDir = lastDirection[t.component];
  1276→            if (prevDir != 0 && prevDir != t.direction && !solvencyOverride) {
  1277→                if (!(block.timestamp >= lastRebalanceAt[t.component] + DIRECTION_LOCK_HOURS)) revert Err16();
  1278→            }
  1279→
  1280→            // Validate trade moves toward execution weight (within 5% tolerance)
  1281→            uint256 execW    = liveWeights.weights[uint256(t.component)];
  1282→            uint256 currentW = (nav == 0) ? 0 : reserveHeldUsd[t.component] * 1e18 / nav;
  1283→            if (t.direction == 1) {
  1284→                if (!(currentW < execW)) revert Err14();
  1285→            } else {
  1286→                if (!(currentW > execW)) revert Err46();
  1287→            }
  1288→
  1289→            // New weight after trade must be within 5pp (absolute) of execW
  1290→            uint256 newHeldUsd = (t.direction == 1)
  1291→                ? reserveHeldUsd[t.component] + t.tradeUsd
  1292→                : reserveHeldUsd[t.component] - t.tradeUsd;
  1293→            if (!(reserveHeldUsd[t.component] >= t.tradeUsd || t.direction == 1)) revert Err23();
  1294→            uint256 newW = (nav == 0) ? 0 : newHeldUsd * 1e18 / nav;
  1295→            uint256 absDiff = (newW > execW) ? newW - execW : execW - newW;
  1296→            // Solvency override relaxes overshoot tolerance by 2x during stress
  1297→            uint256 tolerance = solvencyOverride ? REBALANCE_TOLERANCE * 2 : REBALANCE_TOLERANCE;
  1298→            if (!(absDiff <= tolerance)) revert Err49();
  1299→
  1300→            // §10 daily turnover cap (5% of NAV)
  1301→            if (!(dailyTradeUsd + t.tradeUsd <= maxDaily)) revert Err15();
  1302→            dailyTradeUsd += t.tradeUsd;
  1303→
  1304→            // Update reserve mirror (checks-effects-interactions)
  1305→            if (t.direction == 1) {
  1306→                reserveHeldUsd[t.component] += t.tradeUsd;
  1307→            } else {
  1308→                reserveHeldUsd[t.component] -= t.tradeUsd;
  1309→            }
  1310→            lastDirection[t.component]   = t.direction;
  1311→            lastRebalanceAt[t.component] = block.timestamp;
  1312→        }
  1313→
  1314→        // Refresh reserveState aggregate mirror after rebalancing
  1315→        reserveState.fiatValue  = reserveHeldUsd[Component.USD] + reserveHeldUsd[Component.EUR] +
  1316→                                   reserveHeldUsd[Component.JPY] + reserveHeldUsd[Component.GBP] +
  1317→                                   reserveHeldUsd[Component.CNY] + reserveHeldUsd[Component.CHF];
  1318→        reserveState.goldValue   = reserveHeldUsd[Component.Gold];
  1319→        reserveState.liquidValue = reserveState.fiatValue;
  1320→
  1321→        emit RebalanceExecuted(trades, block.timestamp);
  1322→    }

  1483→    /// @dev In-place insertion sort of sorted[0..n), ascending.
  1484→    function _insertionSort(uint256[3] memory arr, uint8 n) internal pure {
  1485→        for (uint8 i = 1; i < n; i++) {
  1486→            uint256 key = arr[i];
  1487→            uint8 j = i;
  1488→            while (j > 0 && arr[j - 1] > key) {
  1489→                arr[j] = arr[j - 1];
  1490→                j--;
  1491→            }
  1492→            arr[j] = key;
  1493→        }
  1494→    }
